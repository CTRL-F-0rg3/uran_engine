1//! Boss: zdrowie, fazy i zestaw ataków.
2//!
3//! Świadomie **bez silnika fizyki**: ataki są analityczne, a cała
4//! walka mieści się w kilkunastu wzorcach. Podejście „maszyna stanów"
5//! zamiast AI opartego na heurystyce daje **czytelność** — gracz musi
6//! rozpoznać, że nadciąga „tsunami", bo to jedyna metoda, żeby się
7//! ratować.
8//!
9//! # Fazy
10//!
11//! Boss ma trzy fazy, progi zależą od procenta zdrowia. Każda faza
12//! **dodaje** ataki i skraca przerwy, ale nie usuwa poprzednich —
13//! gracz musi pamiętać wszystko, czego się nauczył. Zmiana fazy daje
14//! krótką przerwę i wyraźny sygnał dźwiękowy/wizualny, żeby gracz
15//! zdążył zareagować.
16

21/// Aktywny atak bossa.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum Attack {
24    /// Nic — boss się przygotowuje.
25    Idle,
26    /// Salwa pocisków wachlarzem w stronę gracza.
27    Fan,
28    /// Deszcz pocisków z góry w losowych miejscach.
29    Rain,
30    /// Fala uderzająca po ziemi — trzeba przeskoczyć.
31    GroundSlam,
32    /// Skok do gracza ze zderzeniem.
33    Charge,
34    /// Spiralny strzał — pociski rozchodzą się z dwóch ramion.
35    Spiral,
36}
37
38impl Attack {
39    /// Czy atak trwa wystarczająco długo, by go zobaczyć.
40    pub fn is_telegraphed(self) -> bool {
41        matches!(self, Attack::GroundSlam | Attack::Charge)
42    }
43}

45/// Prosta maszyna losująca (xorshift32) — bez zewnętrznej zależności.
46#[derive(Debug, Clone, Copy)]
47pub struct Rng(u32);
48
49impl Rng {
50    pub fn new(seed: u32) -> Self {
51        // `0` to punkt stały xorshifta, więc wpychamy na niego 1.
52        Self(if seed == 0 { 0x1234_5678 } else { seed })
53    }
54
55    /// Liczba z przedziału `[0, 1)`.
56    pub fn f32(&mut self) -> f32 {
57        let mut x = self.0;
58        x ^= x << 13;
59        x ^= x >> 17;
60        x ^= x << 5;
61        self.0 = x;
62        (x & 0x00FF_FFFF) as f32 / 0x00FF_FFFF as f32
63    }
64
65    /// Liczba z przedziału `[min, max)`.
66    pub fn range(&mut self, min: f32, max: f32) -> f32 {
67        min + self.f32() * (max - min)
68    }
69
70    /// Liczba całkowita z przedziału `[0, n)`.
71    pub fn below(&mut self, n: u32) -> u32 {
72        (self.f32() * n as f32) as u32
73    }
74}

79/// Parametry bossa.
80#[derive(Debug, Clone, Copy)]
81pub struct BossConfig {
82    pub max_hp: f32,
83    /// Proporcja zdrowia, poniżej której wchodzimy w fazę 2 i 3.
84    pub phase2_at: f32,
85    pub phase3_at: f32,
86    /// Czas „namysłu" między atakami w fazie 1 (s).
87    pub think_time: [f32; PHASE_COUNT as usize],
88    /// Szerokość i wysokość ciała bossa.
89    pub size: Vec2,
90}
91
92impl Default for BossConfig {
93    fn default() -> Self {
94        Self {
95            max_hp: 1200.0,
96            phase2_at: 0.66,
97            phase3_at: 0.33,
98            // Faza 1 daje graczowi czas na naukę, faza 3 prawie go
99            // nie zostawia. To kluczowa krzywa trudności.
100            think_time: [2.6, 1.9, 1.25],
101            size: Vec2::new(150.0, 190.0),
102        }
103    }

106/// Stan bossa w klatce.
107#[derive(Debug, Clone, Copy)]
108pub struct Boss {
109    /// Środek ciała.
110    pub pos: Vec2,
111    /// Środek ciała w poprzedniej klatce — do testów „wejścia".
112    pub prev_pos: Vec2,
113    pub hp: f32,
114    /// 1..=3 — zmienia się przy progu zdrowia.
115    pub phase: u8,
116    /// Aktualny atak.
117    pub attack: Attack,
118    /// Ile sekund do następnego ataku.
119    pub timer: f32,
120    /// Ile seksekund trwa bieżący atak.
121    pub attack_time: f32,
122    /// Wychył: ile zostało do wychyłu (s). Zerowe = brak wychyłu.
123    pub windup: f32,
124    /// Czy boss zmierza do gracza (charge).
125    pub charging: bool,
126    /// Ziarno losowania — powtarzalne walki ułatwiają testy.
127    pub rng: Rng,
128    /// Czy boss martwy.
129    pub dead: bool,
130    /// Animacja śmierci (s).
131    pub death_time: f32,
132}
133
134impl Boss {
135    /// Czy boss właśnie się wychyla przed atakiem.
136    ///
137    /// Osobny bool trzymany w strukturze rozjeżdżałby się z
138    /// `windup` w dwa miejsca; stan czytamy z jednego źródła.
139    pub fn winding(&self) -> bool {
140        self.windup > 0.0
141    }
142
143    pub fn new(pos: Vec2, cfg: &BossConfig) -> Self {
144        Self {
145            pos,
146            prev_pos: pos,
147            hp: cfg.max_hp,
148            phase: 1,
149            attack: Attack::Idle,
150            timer: 1.5,
151            attack_time: 0.0,
152            windup: 0.0,
153            charging: false,
154            rng: Rng::new(0xC0FFEE),
155            dead: false,
156            death_time: 0.0,
157        }
158    }
159
160    /// Prostokąt ciała bossa.
161    pub fn rect(&self, cfg: &BossConfig) -> Rect {
162        Rect::from_center(self.pos, cfg.size)
163    }
164
165    /// Zadaje bossowi obrażenia. Zdrowie nie spada poniżej zera.
166    pub fn take_damage(&mut self, amount: f32, cfg: &BossConfig) {
167        if self.dead {
168            return;
169        }
170        self.hp = (self.hp - amount).max(0.0);
171        if self.hp <= 0.0 {
172            self.hp = 0.0;
173            self.dead = true;
174            self.attack = Attack::Idle;
175            self.charging = false;
176            self.windup = 0.0;
177            self.death_time = 0.0;
178            let _ = cfg;
179        }
180    }
181
182    /// Proporcja zdrowia 0..1.
183    pub fn hp_ratio(&self, cfg: &BossConfig) -> f32 {
184        (self.hp / cfg.max_hp).clamp(0.0, 1.0)
185    }
186
187    /// Czy boss jest martwy.
188    pub fn is_dead(&self) -> bool {
189        self.dead
190    }
191
192    /// Wybiera następny atak — dobór zależy od fazy.
193    ///
194    /// Lista dozwolonych ataków **rośnie** z fazą, a losowanie
195    /// ważyone (`Fan` jest częstszy w fazie 1, `Spiral` pojawia się
196    /// dopiero w fazie 3). Dzięki temu gracz poznaje ataki stopniowo
197    /// zamiast uczyć się pięciu wzorców naraz.
198    fn choose_attack(&mut self, phase: u8) -> Attack {
199        // Wagi: (atak, waga). Suma wag w danej fazie bywa różna —
200        // to zamierzone, względne wagi wystarczą.
201        let pool: &[(Attack, u32)] = match phase {
202            1 => &[(Attack::Fan, 5), (Attack::GroundSlam, 3), (Attack::Rain, 2)],
203            2 => &[
204                (Attack::Fan, 4),
205                (Attack::GroundSlam, 3),
206                (Attack::Rain, 3),
207                (Attack::Charge, 3),
208            ],
209            _ => &[
210                (Attack::Fan, 3),
211                (Attack::GroundSlam, 2),
212                (Attack::Rain, 2),
213                (Attack::Charge, 3),
214                (Attack::Spiral, 2),
215            ],
216        };
217        let total: u32 = pool.iter().map(|(_, w)| w).sum();
218        let mut pick = self.rng.below(total);
219        for (a, w) in pool {
220            if pick < *w {
221                return *a;
222            }
223            pick -= *w;
224        }
225        pool[0].0
226    }
227
228    /// Krok symulacji bossa. Zwraca pociski, które boss wystrzelił.
229    ///
230    /// Podejście „funkcja zwraca pociski" zamiast „boss trzyma listę"
231    /// jest celowe: [`Boss`] zostaje czystym typem skopiowalnym
232    /// (`Copy`), a stan świata — pociski — żyje w `Game`. Dzięki temu
233    /// testy bossa nie potrzebują świata ani renderer.
234    pub fn step(&mut self, dt: f32, player_pos: Vec2, cfg: &BossConfig, out: &mut Vec<Projectile>) {
235        if self.dead {
236            self.death_time += dt;
237            return;
238        }
239        self.prev_pos = self.pos;
240        let dt = dt.clamp(0.0, 0.05);
241
242        // --- zmiana fazy po przekroczeniu progu zdrowia ---
243        let ratio = self.hp_ratio(cfg);
244        let new_phase = if ratio <= cfg.phase3_at {
245            3
246        } else if ratio <= cfg.phase2_at {
247            2
248        } else {
249            1
250        };
251        if new_phase > self.phase {
252            self.phase = new_phase;
253            // Krótka przerwa na zmianę fazy: natychmiastowy atak po
254            // progu zdrowia byłby nieczytelny, bo gracz nie zdąży
255            // zobaczyć, co się stało.
256            self.attack = Attack::Idle;
257            self.timer = 0.9;
258            self.charging = false;
259            self.windup = 0.0;
260            return;
261        }
262
263        // --- wychył (telegraph) ---
264        //
265        // Przed atakami, które trzeba **zobaczyć** (`GroundSlam`,
266        // `Charge`) boss stoi nieruchomo i „nabiera". Bez tego
267        // gracz nie ma szans odgadnąć, że nadciąga fala.
268        if self.windup > 0.0 {
269            self.windup -= dt;
270            if self.windup <= 0.0 {
271                self.fire(self.attack, player_pos, cfg, out);
272                self.attack_time = 0.45;
273            }
274            return;
275        }
276
277        // --- trwający atak ---
278        if self.attack != Attack::Idle && self.attack_time > 0.0 {
279            self.attack_time -= dt;
280            // Niektóre ataki „strzelają" w trakcie, a nie jednym
281            // strzałem — stąd osobny strzał w każdej klatce.
282            match self.attack {
283                Attack::Spiral | Attack::Rain => {
284                    self.fire(self.attack, player_pos, cfg, out);
285                }
286                _ => {}
287            }
288            if self.attack_time <= 0.0 {
289                self.attack = Attack::Idle;
290                self.timer = cfg.think_time[(self.phase - 1) as usize];
291            }
292            return;
293        }
294
295        // --- charge: ruch w stronę gracza ---
296        if self.charging {
297            let dir = (player_pos - self.pos).normalize_or_zero();
298            self.pos += dir * CHARGE_SPEED * dt;
299            if self.attack_time <= 0.0 {
300                self.charging = false;
301                self.attack = Attack::Idle;
302                self.timer = cfg.think_time[(self.phase - 1) as usize];
303            }
304            return;
305        }
306
307        // --- odliczanie do następnego ataku ---
308        self.timer -= dt;
309        if self.timer <= 0.0 {
310            self.attack = self.choose_attack(self.phase);
311            if self.attack.is_telegraphed() {
312                // Atak z wychyłem: zapowiadamy i czekamy.
313                self.windup = WINDUP_TIME;
314            } else if self.attack == Attack::Charge {
315                self.charging = true;
316                self.attack_time = 0.9;
317            } else {
318                self.fire(self.attack, player_pos, cfg, out);
319                self.attack_time = 0.45;
320            }
321        }
322    }
323}

325/// Prędkość bossa w ataku `Charge`.
326pub const CHARGE_SPEED: f32 = 330.0;
327/// Czas wychyłu przed atakiem (s) — tyle gracz ma na reakcję.
328pub const WINDUP_TIME: f32 = 0.65;

330/// Buduje pocisk bossa wychodzący z jego środka.
331fn spawn(out: &mut Vec<Projectile>, from: Vec2, dir: Vec2, speed: f32, damage: f32, size: f32) {
332    out.push(Projectile {
333        pos: from,
334        vel: dir.normalize_or_zero() * speed,
335        owner: Owner::Boss,
336        damage,
337        size: Vec2::splat(size),
338        life: BULLET_LIFE,
339        piercing: false,
340    });
341}

343impl Boss {
344    /// Wystrzała pociski zależnie od ataku.
345    ///
346    /// Rozmieszczenie pocisków **jest** rozgrywką: gracz musi znaleźć
347    /// lukę, więc kąty i prędkości są dobrane tak, by przesunięcie o
348    /// kafel wystarczało — za wolne pociski dają przewagę, za szybkie
349    /// zabijają bez szansy na reakcję.
350    fn fire(
351        &mut self,
352        attack: Attack,
353        player_pos: Vec2,
354        _cfg: &BossConfig,
355        out: &mut Vec<Projectile>,
356    ) {
357        // Punkt startowy: środek bossa, przesunięty lekko w stronę
358        // gracza, żeby pocisk nie wychodził z wnętrza ciała.
359        let origin = self.pos;
360        match attack {
361            Attack::Idle | Attack::Charge => {}
362
363            Attack::Fan => {
364                // Wachlarz 5 pocisków skierowanych na gracza, rozłożonych
365                // na 50°. Węższy wachlarz = łatwiej przeczytać.
366                let aim = (player_pos - origin).normalize_or_zero();
367                let base = aim.x.atan2(aim.y);
368                for i in 0..5 {
369                    let a = base + (i as f32 - 2.0) * 0.22;
370                    spawn(out, origin, Vec2::new(a.cos(), a.sin()), 210.0, 8.0, 11.0);
371                }
372            }
373
374            Attack::Rain => {
375                // Deszcz: pociski padają w losowych miejscach w okolicy
376                // gracza. Generator jest seeded, więc deszcz jest
377                // powtarzalny — gracz może się nauczyć wzorca.
378                for _ in 0..3 {
379                    let x = player_pos.x + self.rng.range(-260.0, 260.0);
380                    let y = player_pos.y + self.rng.range(220.0, 380.0);
381                    spawn(out, Vec2::new(x, y), Vec2::new(0.0, -1.0), 300.0, 9.0, 10.0);
382                }
383            }
384
385            Attack::GroundSlam => {
386                // Fala po ziemi: pociski jadą w obie strony od bossa,
387                // po **ziemi** (wysyłamy je lekko w dół, bo gracz
388                // musi przeskoczyć). To jedyny atak wymuszający skok.
389                for s in [-1.0f32, 1.0] {
390                    spawn(
391                        out,
392                        origin,
393                        Vec2::new(s, -0.16).normalize_or_zero(),
394                        250.0,
395                        10.0,
396                        16.0,
397                    );
398                    spawn(
399                        out,
400                        origin,
401                        Vec2::new(s, -0.22).normalize_or_zero(),
402                        250.0,
403                        10.0,
404                        16.0,
405                    );
406                }
407            }
408
409            Attack::Spiral => {
410                // Dwa ramiona wirujące w przeciwnych fazach. Dzięki
411                // temu w fuzie fazy 3 tworzą się luki, a w szczycie
412                // gęsty mur — gracz musi znaleźć rytm, nie szczelinę.
413                let t = self.death_time.max(self.attack_time);
414                let base = (self.rng.f32() * 6.0) as f32;
415                for k in 0..2 {
416                    let a = base + t * 5.0 + k as f32 * std::f32::consts::PI;
417                    spawn(out, origin, Vec2::new(a.cos(), a.sin()), 190.0, 7.0, 9.0);
418                }
419            }
420        }
421    }
422}

430    fn boss() -> (Boss, BossConfig) {
431        (
432            Boss::new(Vec2::new(900.0, 200.0), &BossConfig::default()),
433            BossConfig::default(),

437    /// Przechodzi `n` klatek i zbiera pociski.
438    fn run(b: &mut Boss, n: usize, player: Vec2, cfg: &BossConfig) -> Vec<Projectile> {
439        let mut out = Vec::new();
440        for _ in 0..n {
441            b.step(DT, player, cfg, &mut out);
442        }
443        out
444    }

474        let (mut b, cfg) = boss();
