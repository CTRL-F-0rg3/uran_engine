1//! Boss Fight — walka z jednym potężnym bossem na tilemapie 2D.
2//!
3//! Uruchomienie: `cargo run -p boss-fight`
4//! Sterowanie: `A/D` — ruch, `spacja` — skok, `W` — atak mieczem,
5//! `S` — tarcza, `R` — restart, `Esc` — wyjście.
6//!
7//! # Architektura
8//!
9//! * [`tilemap`] — mapa kafelków i kolizje ze światem,
10//! * [`player`] — ruch, skok (z coyote time i jump bufferem),
11//! * [`boss`] — ataki, fazy i zdrowie bossa,
12//! * [`projectile`] — pociski gracza i bossa,
13//! * `main.rs` — wejście, rysowanie, HUD.

30/// Świat gry w jednostkach (kafle × [`TILE`]).
31const WORLD: Rect = Rect::new(
32    Vec2::new(0.0, 0.0),
33    Vec2::new(MAP_W as f32 * TILE, MAP_H as f32 * TILE),
34);
35
36/// Maksymalna liczba pocisków naraz. Bez limitu deszcz w fazie 3
37/// potrafiłby wygenerować tysiące obiektów i zabić grę samą pamięcią.
38const MAX_BULLETS: usize = 400;
39
40/// Jak długo gracz jest nietykalny po trafieniu (s).
41const HURT_COOLDOWN: f32 = 1.1;
42/// Ile zdrowia gracz ma na start.
43const PLAYER_MAX_HP: f32 = 100.0;
44/// Czas między strzałami gracza (s).
45const ATTACK_COOLDOWN: f32 = 0.22;
46/// Jak mocno strzał gracza rani bossa.
47const PLAYER_DAMAGE: f32 = 26.0;
48
49/// Buduje arenę: podłoga, ściany boczne i platformy.
50///
51/// Rysujemy ją **ręcznie**, a nie generatorem, bo walka z bossem
52/// potrzebuje czytelnej, zaplanowanej areny, a nie losowego labiryntu.
53/// Kolejność platform jest celowa: wysokie po lewej (do unikania),
54/// niskie po prawej (do zejścia), a podłoga w środku zostawia miejsce
55/// na ground slam.
56fn build_arena() -> TileMap {
57    let mut grid: Vec<Vec<Tile>> = vec![vec![Tile::Empty; MAP_W]; MAP_H];
58
59    // Podłoga na dole i sufit na górze — bez sufitu gracz mógłby
60    // wyskoczyć z mapy podczas deszczu.
61    grid[0] = vec![Tile::Solid; MAP_W];
62    grid[MAP_H - 1] = vec![Tile::Solid; MAP_W];
63    // Ściany boczne: gracz nie wychodzi poza mapę.
64    for row in grid.iter_mut() {
65        row[0] = Tile::Solid;
66        row[MAP_W - 1] = Tile::Solid;
67    }
68
69    // Platformy: cztery poziomy schodkowo, żeby dać się unikać
70    // ataków w powietrzu i wchodzić na nie skokiem. Wszystkie
71    // mieszczą się w 40 kolumnach (0 i 39 to ściany).
72    let platforms: &[(usize, usize, usize)] = &[
73        (4, 2, 6),   // niskie, lewe — miejsce na rozbieg
74        (7, 9, 7),   // środkowe
75        (10, 18, 6), // wysokie, prawe
76        (5, 27, 8),  // najdalsze, nad spawnem bossa
77    ];
78    for &(y, x0, len) in platforms {
79        for x in x0..(x0 + len).min(MAP_W - 1) {
80            grid[y][x] = Tile::Platform;
81        }
82    }
83
84    // Kolce na podłodze — groźba, która uczy pilnowania podłoża
85    // zamiast tylko reagowania na pociski. Zostawiamy szczelinę
86    // szeroką na skok, żeby nie były nie do przejścia.
87    for &(x0, len) in &[(15usize, 3usize), (30, 3)] {
88        for x in x0..(x0 + len).min(MAP_W - 1) {
89            grid[1][x] = Tile::Spike;
90        }
91    }
92
93    let refs: Vec<&[Tile]> = grid.iter().map(|r| r.as_slice()).collect();
94    TileMap::from_rows(&refs)
95}

97/// Wszystko, czego gra potrzebuje, w jednym stanie.
98struct Game {
99    map: TileMap,
100    player: Player,
101    player_cfg: PlayerConfig,
102    player_hp: f32,
103    /// Ile zostało nietykalności (s) — własny zegar, bo `Player`
104    /// trzyma go dla efektów, a HUD potrzebuje wartości globalnej.
105    hurt_cd: f32,
106    /// Czy gracz właśnie przyjął cios (do animacji).
107    flash: f32,
108
109    boss: Boss,
110    boss_cfg: BossConfig,
111
112    bullets: Vec<Projectile>,
113    /// Ile zostało do następnego strzału gracza (s).
114    attack_cd: f32,
115    /// Czy gracz trzyma przycisk ataku.
116    attacking: bool,
117
118    /// Trzęsienie kamery.
119    shake: f32,
120    /// Czas świata (animacje).
121    time: f32,
122    /// Czy walka zakończona (wygrana albo porażka).
123    over: bool,
124    won: bool,
125    /// Licznik klatek od startu — do zrzutu ekranu.
126    frames: u32,
127    font: Option<Font>,
128}
129
130impl Game {
131    fn new() -> Self {
132        let boss_cfg = BossConfig::default();
133        let boss_pos = Vec2::new(WORLD.center().x + 320.0, TILE * 6.0);
134        let mut player = Player::default();
135        player.pos = Vec2::new(TILE * 3.0, TILE);
136        player.grounded = true;
137        Self {
138            map: build_arena(),
139            player,
140            player_cfg: PlayerConfig::default(),
141            player_hp: PLAYER_MAX_HP,
142            hurt_cd: 0.0,
143            flash: 0.0,
144            boss: Boss::new(boss_pos, &boss_cfg),
145            boss_cfg,
146            bullets: Vec::new(),
147            attack_cd: 0.0,
148            attacking: false,
149            shake: 0.0,
150            time: 0.0,
151            over: false,
152            won: false,
153            frames: 0,
154            font: None,
155        }
156    }
157
158    fn reset(&mut self) {
159        let font = self.font;
160        *self = Self::new();
161        self.font = font;
162    }
163}

165/// Wejście + krok symulacji.
166fn update(ctx: &mut Ctx, game: &mut Game) {
167    let dt = ctx.dt();
168    game.frames = game.frames.wrapping_add(1);
169    if !game.over {
170        game.time += dt;
171    }
172
173    if ctx.input.just_pressed(Key::Escape) {
174        std::process::exit(0);
175    }
176    if ctx.input.just_pressed(Key::KeyR) {
177        game.reset();
178    }
179
180    let frozen = game.over;
181    let dt = if frozen { 0.0 } else { dt };
182
183    // --- wejście gracza ---
184    //
185    // `just_pressed` to **wciśnięcie**, `pressed` to trzymanie. Skok
186    // potrzebuje obu: wciśnięcia, żeby wystartować, i trzymania, żeby
187    // sięgnąć pełnej wysokości.
188    let move_x =
189        ctx.input.axis(Key::KeyA, Key::KeyD) + ctx.input.axis(Key::ArrowLeft, Key::ArrowRight);
190    let jump_pressed = ctx.input.just_pressed(Key::Space);
191    let jump_held = ctx.input.pressed(Key::Space);
192    let attack_pressed = ctx.input.pressed(Key::KeyJ) || ctx.input.pressed(Key::KeyW);
193    game.attacking = attack_pressed;
194
195    game.player.step(
196        dt,
197        move_x.clamp(-1.0, 1.0),
198        jump_pressed,
199        jump_held,
200        &game.map,
201        &game.player_cfg,
202    );
203
204    // --- strzał gracza ---
205    //
206    // Pocisk leci **poziomo** w stronę, w którą gracz patrzy. Skok
207    // pod kątem w górę dzięki temu nie jest możliwy, a to utrudnia
208    // trafianie bossa szybciej niż chodzenie — w zamian gracz musi
209    // podchodzić, co odsłania go na pociski. Taki kompromis daje
210    // rozgrywkę zamiast bezmyślnego stawiania pocisków z dystansu.
211    if attack_pressed && !frozen {
212        game.attack_cd -= dt;
213        if game.attack_cd <= 0.0 {
214            game.attack_cd = ATTACK_COOLDOWN;
215            let c = game.player.center();
216            let dir = game.player.facing;
217            game.bullets.push(Projectile {
218                pos: c + Vec2::new(dir * 16.0, 0.0),
219                vel: Vec2::new(dir * 620.0, 0.0),
220                owner: Owner::Player,
221                damage: PLAYER_DAMAGE,
222                size: Vec2::new(18.0, 7.0),
223                life: BULLET_LIFE,
224                piercing: false,
225            });
226        }
227    }
228
229    // --- boss ---
230    game.boss
231        .step(dt, game.player.center(), &game.boss_cfg, &mut game.bullets);
232
233    // --- ruch pocisków ---
234    //
235    // Osobny krok przed kolizjami: `boss::step` tylko **spawnuje**
236    // pociski, samo ich poruszanie jest wspólne dla obu stron. Bez
237    // tej pętli pociski stałyby w miejscu, a gracz nie mógłby nic
238    // trafić ani zostać trafiony.
239    for b in &mut game.bullets {
240        b.step(dt);
241    }
242
243    // --- pociski i kolizje ---
244    let boss_rect = game.boss.rect(&game.boss_cfg);
245    let player_rect = game.player.rect();
246    let mut player_damage = 0.0f32;
247
248    // `retain` zamiast `remove_if`: usuwamy w tej samej klatce, w
249    // której pocisk zadał obrażenia, żeby nie dało się „strzelić
250    // w klatkę po śmierci".
251    game.bullets.retain(|b| {
252        let mut keep = true;
253        if b.owner == Owner::Player {
254            if b.rect().intersects(&boss_rect) {
255                game.boss.take_damage(b.damage, &game.boss_cfg);
256                game.shake = (game.shake + 2.0).min(9.0);
257                if !b.piercing {
258                    keep = false;
259                }
260            }
261        } else if b.rect().intersects(&player_rect) {
262            // Nietykalność: kilka pocisków w tej samej klatce nie może
263            // zabić gracza cztery razy.
264            if game.hurt_cd <= 0.0 {
265                player_damage = player_damage.max(b.damage);
266                game.hurt_cd = HURT_COOLDOWN;
267                game.flash = 0.28;
268                game.shake = (game.shake + 7.0).min(16.0);
269            }
270            keep = false;
271        }
272        // Pocisk poza mapą znika — inaczej kumulowałby się w pamięci.
273        keep && !b.is_dead() && b.pos.y > -TILE && b.pos.y < WORLD.top() + TILE * 4.0
274    });
275
276    // Limit pocisków: gdy boss przesyła, odrzucamy najstarsze.
277    if game.bullets.len() > MAX_BULLETS {
278        let excess = game.bullets.len() - MAX_BULLETS;
279        game.bullets.drain(0..excess);
280    }
281
282    if player_damage > 0.0 {
283        game.player_hp = (game.player_hp - player_damage).max(0.0);
284        game.player.take_hit();
285        if game.player_hp <= 0.0 {
286            game.over = true;
287        }
288    }
289
290    // Kolce ranią niezależnie od pocisków — to groźba środowiskowa.
291    if game.hurt_cd <= 0.0 && game.map.touches_spike(game.player.rect()) {
292        game.player_hp = (game.player_hp - 18.0).max(0.0);
293        game.hurt_cd = HURT_COOLDOWN;
294        game.flash = 0.28;
295        game.player.take_hit();
296        if game.player_hp <= 0.0 {
297            game.over = true;
298        }
299    }
300
301    // --- timery i koniec walki ---
302    game.hurt_cd = (game.hurt_cd - dt).max(0.0);
303    game.flash = (game.flash - dt).max(0.0);
304    game.shake = (game.shake - dt * 22.0).max(0.0);
305    if game.boss.is_dead() {
306        game.over = true;
307        game.won = true;
308    }
309    // Spadnięcie z areny też znaczy porażkę — nie ma „darmowego” resetu
310    // przez skok w otchłań.
311    if game.map.is_below_world(game.player.pos) {
312        game.over = true;
313    }
314}

316/// Rysowanie całej sceny.
317fn draw(ctx: &mut Ctx, game: &Game) {
318    // Trzęsienie kamery: losowe przesunięcie zanikające w czasie.
319    let (ox, oy) = shake_offset(game);
320
321    // --- tło i kafle ---
322    ctx.gfx.layer(-20).color(Color::from_hex(0x0B0D14));
323    ctx.gfx.draw_rect(WORLD);
324    draw_tiles(ctx, game);
325
326    // --- boss ---
327    draw_boss(ctx, game, ox, oy);
328
329    // --- pociski ---
330    ctx.gfx.layer(20);
331    for b in &game.bullets {
332        let p = b.pos + Vec2::new(ox, oy);
333        let c = match b.owner {
334            Owner::Player => Color::from_hex(0x7FE3FF),
335            Owner::Boss => Color::from_hex(0xFF4D6D),
336        };
337        // Poświata addytywna: pocisk musi być widoczny na tle trawy
338        // i w chmurze pocisków bossa.
339        ctx.gfx.blend(BlendMode::Additive);
340        ctx.gfx
341            .color(c.with_alpha(0.22))
342            .draw_circle(p, b.size.x * 1.6);
343        ctx.gfx.blend(BlendMode::Alpha);
344        ctx.gfx
345            .rotate_about(p, b.vel.x.atan2(b.vel.y))
346            .color(c)
347            .draw_rect(Rect::from_size(b.size));
348    }
349
350    // --- gracz ---
351    draw_player(ctx, game, ox, oy);
352
353    draw_hud(ctx, game);
354}

356/// Przesunięcie kamery przy trzęsieniu.
357fn shake_offset(game: &Game) -> (f32, f32) {
358    if game.shake <= 0.01 {
359        return (0.0, 0.0);
360    }
361    // Deterministyczne „losowanie" z czasu gry: dwie sinusoidy o
362    // różnych częstotliwościach. `sin` zamiast RNG, żeby zrzut
363    // ekranu był powtarzalny przy tych samych klatkach.
364    let t = game.time * 60.0;
365    ((t * 2.7).sin() * game.shake, (t * 3.9).cos() * game.shake)
366}

368/// Rysuje widoczne kafle: pełne bloki, platformy i kolce.
369///
370/// Kafle rysujemy wektorowo, bo silnik nie ma tilemapa ani atlasu
371/// (patrz [`tilemap`]). Warstwy: tło → kafle → akcje, żeby boss
372/// i gracz zawsze byli na wierzchu.
373fn draw_tiles(ctx: &mut Ctx, game: &Game) {
374    let view = ctx.camera.visible_rect(ctx.window.size);
375    let tiles = game.map.visible_tiles(view);
376    for (cx, cy, kind) in tiles {
377        let r = TileMap::tile_rect(cx, cy);
378        match kind {
379            // Solidny blok: ciemny kamień z jaśniejszą obwódką na
380            // górze, żeby platforma była czytelna jako „stojak”.
381            Tile::Solid => {
382                ctx.gfx.layer(-15).color(Color::from_hex(0x2A2F3E));
383                ctx.gfx.draw_rect(r);
384                ctx.gfx
385                    .color(Color::from_hex(0x3E4557))
386                    .draw_rect(Rect::from_xywh(r.min.x, r.top() - 3.0, r.width(), 3.0));
387            }
388            // Platforma: cienka listwa z zaostrzonymi końcami — czytelny
389            // sygnał „można tu stanąć, ale nie z dołu".
390            Tile::Platform => {
391                ctx.gfx.layer(-14).color(Color::from_hex(0x6B4F2A));
392                ctx.gfx
393                    .draw_rect(Rect::from_xywh(r.min.x, r.top() - 8.0, r.width(), 8.0));
394                ctx.gfx
395                    .color(Color::from_hex(0x8A6836))
396                    .draw_rect(Rect::from_xywh(r.min.x, r.top() - 8.0, r.width(), 3.0));
397            }
398            // Kolec: trójkąt wskazujący w górę — kierunek groźby
399            // musi być czytelny bez słowa.
400            Tile::Spike => {
401                ctx.gfx.layer(-13).color(Color::from_hex(0x8A2B3C));
402                let h = r.height() * 0.8;
403                ctx.gfx.draw_polygon(&[
404                    Vec2::new(r.min.x, r.min.y),
405                    Vec2::new(r.center().x, r.min.y + h),
406                    Vec2::new(r.max.x, r.min.y),
407                ]);
408            }
409            Tile::Empty => {}
410        }
411    }
412}

414/// Rysuje bossa: ciało, fazę, wychył i atak.
415///
416/// **Wychył to element rozgrywki**, nie ozdoba: boss zmienia kolor i
417/// zaczyna pulsować, gdy szykuje `GroundSlam` albo `Charge`. Bez tego
418/// sygnału gracz nie ma szans odgadnąć ataku wymagającego skoku.
419fn draw_boss(ctx: &mut Ctx, game: &Game, ox: f32, oy: f32) {
420    let b = &game.boss;
421    if b.is_dead() && b.death_time > 2.5 {
422        return;
423    }
424    let cfg = &game.boss_cfg;
425    let rect = b.rect(cfg);
426    let center = b.pos + Vec2::new(ox, oy);
427    let winding = b.winding();
428    // Śmierć: boss blednie i rozsypuje się w dół.
429    let fade = if b.is_dead() {
430        (1.0 - b.death_time / 2.5).clamp(0.0, 1.0)
431    } else {
432        1.0
433    };
434
435    // Kolor zależy od fazy: ciepły w fazie 1, ostrzejszy w fazie 3 —
436    // gracz widzi, że robi się niebezpiecznie, bez czytania HUD-a.
437    let body = match b.phase {
438        1 => Color::from_hex(0x8E3B5A),
439        2 => Color::from_hex(0xA3325F),
440        _ => Color::from_hex(0xC42A6E),
441    }
442    .with_alpha(fade);
443
444    // Poświata — boss musi być rozpoznawalny w gęstwie pocisków.
445    ctx.gfx.layer(10).blend(BlendMode::Additive);
446    ctx.gfx
447        .color(body.with_alpha(0.16 * fade))
448        .draw_circle(center, rect.width() * 0.85);
449    ctx.gfx.blend(BlendMode::Alpha);
450
451    // Wychył: pulsujący pierścień narastający do ataku. Zmiana
452    // rozmiaru to informacja „zaraz uderzy".
453    if winding {
454        let t = 1.0 - (b.windup / crate::boss::WINDUP_TIME).clamp(0.0, 1.0);
455        ctx.gfx.color(Color::from_hex(0xFFD166).with_alpha(0.85));
456        ctx.gfx
457            .draw_ring(center, rect.width() * (0.7 + t * 0.35), 5.0);
458    }
459
460    // Korpus.
461    ctx.gfx.color(body).draw_rect(rect);
462    // Rdzeń: jasne „oko" w środku, pulsuje z czasem świata.
463    let pulse = 1.0 + (game.time * 3.0).sin() * 0.12;
464    ctx.gfx
465        .color(Color::from_hex(0xFFE9F0).with_alpha(0.9 * fade));
466    ctx.gfx.draw_circle(center, rect.width() * 0.17 * pulse);
467    // „Ucho" w narożniku pokazuje, gdzie boss patrzy.
468    let look = (game.player.center() - b.pos).normalize_or_zero();
469    ctx.gfx
470        .color(Color::from_hex(0xFFD166).with_alpha(fade))
471        .draw_circle(center + look * rect.width() * 0.32, rect.width() * 0.09);
472
473    // Pasek zdrowia bossa nad głową — liczba, którą gracz musi
474    // śledzi, bo fazy zmieniają się na jej progu.
475    let hp_w = rect.width() * 1.1;
476    let hp_y = rect.top() + 18.0 + oy;
477    let hp_x = center.x - hp_w * 0.5;
478    ctx.gfx
479        .color(Color::from_hex(0x1A1D28))
480        .draw_rect(Rect::from_xywh(hp_x, hp_y, hp_w, 8.0));
481    ctx.gfx.color(Color::from_hex(0xE8467C).with_alpha(fade));
482    ctx.gfx
483        .draw_rect(Rect::from_xywh(hp_x, hp_y, hp_w * b.hp_ratio(cfg), 8.0));
484    // Podziałki faz: dwa białe kreski pokazują progi 66% i 33%.
485    ctx.gfx.color(Color::from_hex(0xFFFFFF).with_alpha(0.8));
486    for p in [cfg.phase2_at, cfg.phase3_at] {
487        ctx.gfx
488            .draw_rect(Rect::from_xywh(hp_x + hp_w * p, hp_y - 2.0, 2.0, 12.0));
489    }
490}

492/// Rysuje gracza: ciało, oczy, animacja chodu i nietykalność.
493fn draw_player(ctx: &mut Ctx, game: &Game, ox: f32, oy: f32) {
494    let p = &game.player;
495    let center = p.center() + Vec2::new(ox, oy);
496    let rect = p.rect().translate(Vec2::new(ox, oy));
497    // Migotanie przy nietykalności: sygnał dla gracza, że cios
498    // właśnie przeszedł i że teraz jest bezpieczny.
499    let blinking = game.hurt_cd > 0.0;
500    let visible = !blinking || (game.time * 22.0).fract() < 0.55;
501    if !visible {
502        return;
503    }
504
505    // Poświata: gracz musi być wyróżniony spośród pocisków.
506    ctx.gfx.layer(28).blend(BlendMode::Additive);
507    ctx.gfx
508        .color(Color::from_hex(0x4FD0FF).with_alpha(0.18))
509        .draw_circle(center, rect.width() * 1.1);
510    ctx.gfx.blend(BlendMode::Alpha);
511
512    // Widoczność postaci. Gracz ma 20×34 px na 1280×720 — to **mało**,
513    // bo arena jest dobrana tak, by 1 kafel = 1 px (patrz `MAP_W`).
514    // Dlatego rysujemy go 1,5× i z ciemną obwódką: sylwetka musi
515    // być rozpoznawalna wśród pocisków, a nie ginąć w tle.
516    let s = PLAYER_DRAW_SCALE;
517    let body = Rect::from_center(center, rect.size() * s);
518    ctx.gfx
519        .color(Color::from_hex(0x08131A))
520        .draw_rect(Rect::from_center(center, body.size() + Vec2::splat(3.0)));
521
522    // Kolor zależny od stanu: błękit = zdrowy, czerwony = ranny.
523    let c = if game.flash > 0.0 {
524        Color::from_hex(0xFF6B7A)
525    } else {
526        Color::from_hex(0x3DB8E8)
527    };
528    ctx.gfx.layer(30).color(c);
529    ctx.gfx.draw_rect(body);
530
531    // Oczy: przesunięte w stronę patrzenia. To jedyny element, który
532    // mówi graczowi (i obserwatorowi), gdzie gracz patrzy.
533    let f = p.facing;
534    ctx.gfx.color(Color::from_hex(0xE8F6FF));
535    for eye_x in [-5.0f32, 5.0] {
536        ctx.gfx.draw_circle(
537            center + Vec2::new(eye_x * s + f * 2.0, body.height() * 0.5 - 7.0),
538            2.6,
539        );
540    }
541
542    // Lufa przy strzale: krótka linia w kierunku patrzenia.
543    if game.attacking {
544        ctx.gfx.color(Color::from_hex(0x7FE3FF));
545        ctx.gfx.draw_line(
546            center + Vec2::new(f * 14.0, 0.0),
547            center + Vec2::new(f * 30.0, 0.0),
548            4.0,
549        );
550    }
551}

553/// Przeskalowanie postaci przy rysowaniu — patrz [`draw_player`].
554///
555/// Kolizje używają 20×34 px (`player::BODY_W/H`), ale rysunek jest
556/// 1,5× większy. Rozjazd jest tu celowy: hitbox musi być **mniejszy**
557/// niż sylwetka, inaczej gracze biiliby się o coś, co wygląda, jak
558/// jest obok.
559const PLAYER_DRAW_SCALE: f32 = 1.5;
560
561/// HUD w przestrzeni ekranu: paski, nazwa ataku, podpowiedzi, koniec.
562///
563/// UWAGA: tylko ASCII. Polskie diakrytyczne wywołują błąd atlasu glifów
564/// w `uran-render` (patrz `draw_hud` w `uran-game`).
565fn draw_hud(ctx: &mut Ctx, game: &Game) {
566    let Some(font) = game.font else { return };
567    let size = ctx.window.logical_size();
568    ctx.gfx.screen_space();
569
570    // --- pasek zdrowia gracza (lewy górny róg) ---
571    let bar = Rect::from_xywh(24.0, 24.0, 240.0, 18.0);
572    ctx.gfx.color(Color::from_hex(0x1A1D28)).draw_rect(bar);
573    let ratio = (game.player_hp / PLAYER_MAX_HP).clamp(0.0, 1.0);
574    ctx.gfx
575        .color(if game.hurt_cd > 0.0 {
576            Color::from_hex(0xE8467C)
577        } else {
578            Color::from_hex(0x3DB8E8)
579        })
580        .draw_rect(Rect::from_xywh(
581            bar.min.x,
582            bar.min.y,
583            bar.width() * ratio,
584            bar.height(),
585        ));
586    ctx.gfx.color(Color::from_hex(0xE8EEF8)).draw_text(
587        font,
588        &format!("HP {}", game.player_hp.ceil() as i32),
589        Vec2::new(32.0, 27.0),
590        14.0,
591        TextAlign::Left,
592    );
593
594    // --- nazwa ataku bossa (środek góry) ---
595    //
596    // Pokazujemy **w czasie wychyłu**, żeby gracz miał czas zaplanować
597    // ruch. Sam tekst nie wystarczy — obok niego pulsuje pierścień.
598    if game.boss.winding() {
599        let label = match game.boss.attack {
600            Attack::GroundSlam => "FALA ZIEMI",
601            Attack::Charge => "SZARZA",
602            other => {
603                let _ = other;
604                ""
605            }
606        };
607        ctx.gfx.color(Color::from_hex(0xFFD166)).draw_text(
608            font,
609            label,
610            Vec2::new(size.x * 0.5, 26.0),
611            26.0,
612            TextAlign::Center,
613        );
614    }
615
616    // --- pasek bossa u góry (środek) ---
617    let bw = 460.0;
618    let bbar = Rect::from_xywh(size.x * 0.5 - bw * 0.5, 56.0, bw, 14.0);
619    ctx.gfx.color(Color::from_hex(0x1A1D28)).draw_rect(bbar);
620    ctx.gfx
621        .color(Color::from_hex(0xE8467C))
622        .draw_rect(Rect::from_xywh(
623            bbar.min.x,
624            bbar.min.y,
625            bbar.width() * game.boss.hp_ratio(&game.boss_cfg),
626            bbar.height(),
627        ));
628    ctx.gfx.color(Color::from_hex(0xFFFFFF).with_alpha(0.85));
629    for p in [game.boss_cfg.phase2_at, game.boss_cfg.phase3_at] {
630        ctx.gfx.draw_rect(Rect::from_xywh(
631            bbar.min.x + bbar.width() * p,
632            bbar.min.y - 3.0,
633            2.0,
634            bbar.height() + 6.0,
635        ));
636    }
637    ctx.gfx.color(Color::from_hex(0xB9C4D4)).draw_text(
638        font,
639        &format!("FAZA {}", game.boss.phase),
640        Vec2::new(size.x * 0.5, 76.0),
641        16.0,
642        TextAlign::Center,
643    );
644
645    // --- podpowiedź (prawy dolny róg) ---
646    //
647    // **Prawy**, nie lewy: gracz startuje w lewej części areny i
648    // podpowiedź w lewym rogu zasłaniała mu sylwetkę. Po jednej
649    // stronie jest pole, po drugiej sterowanie.
650    ctx.gfx.color(Color::from_hex(0x8FA6C4)).draw_text(
651        font,
652        "AD ruch   spacja skok   J atak   R restart",
653        Vec2::new(size.x - 24.0, size.y - 32.0),
654        16.0,
655        TextAlign::Right,
656    );
657
658    // --- koniec walki ---
659    if game.over {
660        // Przyciemnienie: sygnalizuje, że scena przestała być żywa.
661        ctx.gfx.color(Color::from_hex(0x000000).with_alpha(0.55));
662        ctx.gfx.draw_rect(Rect::from_size(size));
663        let (msg, col) = if game.won {
664            ("BOSS POKONANY", Color::from_hex(0x7FE3A0))
665        } else {
666            ("PORAZKA  -  R aby sprobowac", Color::from_hex(0xE8467C))
667        };
668        ctx.gfx.color(col).draw_text(
669            font,
670            msg,
671            Vec2::new(size.x * 0.5, size.y * 0.44),
672            40.0,
673            TextAlign::Center,
674        );
675    }
676}

678/// Ścieżki czcionek próbowane po kolei.
679///
680/// `UranSans.ttf` **nie ma w repozytorium** — to nazwa projektowa,
681/// nie istniejący plik. Dlatego lista kończy się na czcionkach
682/// systemowych (`/usr/share/fonts/...`), które są na każdej Linuksie
683/// z pakietem `fonts-dejavu`. Bez tej listy HUD jest pusty i wygląda
684/// jak błąd renderowania.
685///
686/// Ten sam zestaw co w `uran-game` — jedna lista w całym repo
687/// oznacza jedno miejsce do poprawki.
688const FONT_FALLBACKS: &[&str] = &[
689    "assets/fonts/UranSans.ttf",
690    "/usr/share/fonts/TTF/DejaVuSans.ttf",
691    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
692    "/usr/share/fonts/TTF/LiberationSans-Regular.ttf",
693    "/usr/share/fonts/liberation/LiberationSans-Regular.ttf",
694    "/usr/share/fonts/TTF/Hack-Regular.ttf",
695    "/usr/share/fonts/TTF/MesloLGS-NF-Regular.ttf",
696];
697
698/// Jednorazowa konfiguracja: czcionka i kamera.
699fn setup(ctx: &mut Ctx, game: &mut Game) {
700    if game.font.is_none() {
701        for path in FONT_FALLBACKS {
702            if let Ok(h) = ctx.assets.load_font(path) {
703                game.font = Some(h);
704                break;
705            }
706        }
707    }
708    if game.font.is_none() {
709        eprintln!("⚠️  nie znaleziono czcionki TTF — HUD bedzie pusty");
710    }
711    ctx.clear_color = Color::from_hex(0x05070C);
712    // Kamera pokazuje całą arenę: walka z bossem musi być czytelna
713    // w całości, a nie śledzić gracza jak w platformówce.
714    ctx.camera.fit_world(WORLD, ctx.window.size);
715}
716
717fn main() {
718    let game = Arc::new(Mutex::new(Game::new()));
719
720    let setup_system = {
721        let game = Arc::clone(&game);
722        move |ctx: &mut Ctx| {
723            let mut guard = game.lock().expect("lock gry");
724            setup(ctx, &mut guard);
725        }
726    };
727    let update_system = {
728        let game = Arc::clone(&game);
729        move |ctx: &mut Ctx| {
730            let mut guard = game.lock().expect("lock gry");
731            update(ctx, &mut guard);
732        }
733    };
734    let draw_system = {
735        let game = Arc::clone(&game);
736        move |ctx: &mut Ctx| {
737            let guard = game.lock().expect("lock gry");
738            draw(ctx, &guard);
739        }
740    };
741
742    App::new()
743        // Zrzut po 90 klatkach (~1,5 s walki) — używany w testach
744        // regresji wizualnej (`--screenshot /tmp/shot.png`).
745        .screenshot(screenshot_path(), 90)
746        .window(
747            windowed(1280, 720)
748                .title("Boss Fight")
749                .background(0x05070C)
750                .vsync(true)
751                .samples(1),
752        )
753        .add_startup_system(setup_system)
754        .add_system(update_system)
755        .add_system(draw_system)
756        .run();
757}

759/// Ścieżka zrzutu z linii poleceń (albo `None`).
760fn screenshot_path() -> Option<std::path::PathBuf> {
761    let args: Vec<String> = std::env::args().collect();
762    args.iter()
763        .position(|a| a == "--screenshot")
764        .and_then(|i| args.get(i + 1))
765        .map(std::path::PathBuf::from)
766}
