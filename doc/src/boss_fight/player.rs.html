1//! Gracz: ruch poziomy, skok i kolizje z tilemapą.
2//!
3//! Świadomie **bez silnika fizyki** (jak [`junak-rider`]): ruch jest
4//! analityczny, a świat to prostokątne kafle. Rapier byłby o rząd
5//! wielkości cięższy i wniósłby dokładnie ten sam błąd.
6//!
7//! ## Dwa okna na skok
8//!
9//! `coyote` i `jump_buffer` to dwa najczęstsze źródła „dziwnego
10//! sterowania" w platformówkach:
11//!
12//! * **Coyote time** — gracz może skoczyć przez ~0,1 s po zejściu
13//!   z krawędzi. Bez tego skok „nie działa", bo gracz wyciska spację
14//!   odrobinę za późno.
15//! * **Jump buffer** — naciśnięcie spacji przed lądowaniem (~0,12 s)
16//!   zapamiętujujemy i wykonujemy skok natychmiast po dotknięciu ziemi.
17//!
18//! Bez obu okien gra jest „śmierdząca", a testerzy zgłaszają to jako
19//! „skok nie działa", nie rozumiejąc przyczyny.
20
21use uran_math::{Rect, Vec2};

23use crate::tilemap::{TILE, TileMap};

25/// Szerokość ciała gracza.
26pub const BODY_W: f32 = 20.0;
27/// Wysokość ciała gracza.
28pub const BODY_H: f32 = 34.0;
29
30/// Parametry gracza — jeden zestaw, bo to jeden model.
31#[derive(Debug, Clone, Copy)]
32pub struct PlayerConfig {
33    /// Prędkość biegu (jednostki świata na sekundę).
34    pub run_speed: f32,
35    /// Przyspieszenie poziome — jak szybko gracz osiąga pełną prędkość.
36    pub accel: f32,
37    /// Tarcie podłoża. Zero oznacza „ślizganie", zbyt dużo „klej".
38    pub friction: f32,
39    /// Prędkość skoku w chwili startu.
40    pub jump_speed: f32,
41    /// Grawitacja (ujemna = w dół).
42    pub gravity: f32,
43    /// Mnożnik grawitacji przy spadaniu (większy 1 = szybszy, „ciężki").
44    pub fall_multiplier: f32,
45    /// Jak długo po zejściu z krawędzi można jeszcze skoczyć (s).
46    pub coyote: f32,
47    /// Jak długo naciśnięcie skoku czeka na lądowanie (s).
48    pub jump_buffer: f32,
49    /// Najwyższa prędkość spadania — chroni przed tunelowaniem.
50    pub terminal_velocity: f32,
51}
52
53impl Default for PlayerConfig {
54    fn default() -> Self {
55        Self {
56            // Prędkość 260 i kafel 32 dają ~8 kafli na sekundę: szybko,
57            // ale gracz wciąż zdąży zobaczyć nadciągający pocisk.
58            run_speed: 260.0,
59            accel: 2000.0,
60            friction: 2400.0,
61            // Skok ~3 kafle wysoko: wystarczy na unik w powietrzu,
62            // a nie na przeskoczenie całej mapy.
63            jump_speed: 560.0,
64            gravity: -1500.0,
65            fall_multiplier: 1.35,
66            coyote: 0.10,
67            jump_buffer: 0.12,
68            terminal_velocity: 900.0,
69        }
70    }
71}

73/// Stan gracza: prostokąt ciała, prędkość i okna skoku.
74#[derive(Debug, Clone, Copy)]
75pub struct Player {
76    /// **Dół** ciała (lewy dolny róg) — jak w całym silniku.
77    pub pos: Vec2,
78    /// Prędkość w jednostkach na sekundę.
79    pub vel: Vec2,
80    /// Czy stoi na ziemi.
81    pub grounded: bool,
82    /// Ile zostało z okna coyote.
83    coyote_left: f32,
84    /// Ile zostało z bufora skoku.
85    buffer_left: f32,
86    /// Czy trzyma klawisz skoku (dla skoku zmiennej wysokości).
87    jump_held: bool,
88    /// Kierunek patrzenia: 1 = w prawo, -1 = w lewo.
89    pub facing: f32,
90    /// Czy gracz jest ranny (blokada na chwilę).
91    pub hurt: bool,
92    /// Ile zostało nietykalności po trafieniu.
93    invuln: f32,
94}
95
96impl Default for Player {
97    fn default() -> Self {
98        Self {
99            pos: Vec2::ZERO,
100            vel: Vec2::ZERO,
101            grounded: false,
102            coyote_left: 0.0,
103            buffer_left: 0.0,
104            jump_held: false,
105            facing: 1.0,
106            hurt: false,
107            invuln: 0.0,
108        }
109    }
110}
111
112impl Player {
113    /// Prostokąt ciała gracza.
114    pub fn rect(&self) -> Rect {
115        Rect::from_xywh(self.pos.x, self.pos.y, BODY_W, BODY_H)
116    }
117
118    /// Środek ciała — do celowania i efektów.
119    pub fn center(&self) -> Vec2 {
120        self.pos + Vec2::new(BODY_W * 0.5, BODY_H * 0.5)
121    }
122
123    /// Zadaje graczowi obrażenia i daje nietykalność.
124    ///
125    /// Nietykalność chroni przed „umieraniem" w klatce, w której
126    /// trafienie zgłosiło kilka pocisków naraz — to bardzo frustrujące,
127    /// bo gracz nie ma jak uciec.
128    pub fn take_hit(&mut self) {
129        if self.invuln > 0.0 {
130            return;
131        }
132        self.hurt = true;
133        self.invuln = INVULN_TIME;
134    }
135
136    /// Krok symulacji: wejście, grawitacja, skok, kolizje.
137    ///
138    /// `move_x` i `jump_pressed` to wejście w klatce (`-1/0/1`,
139    /// `true/false`). Kolizje rozwiązuje [`TileMap`], więc ten moduł
140    /// zna się z mapą, a nie z konkretnymi kaflami.
141    pub fn step(
142        &mut self,
143        dt: f32,
144        move_x: f32,
145        jump_pressed: bool,
146        jump_held: bool,
147        map: &TileMap,
148        cfg: &PlayerConfig,
149    ) {
150        // `dt` z debuggera albo przeciągnięte okno potrafi być ogromne.
151        // Bez klamrowania skok wyrzuciłby gracza w kosmos, a pozycja
152        // dostałaby NaN, co psuje renderer na zawsze.
153        let dt = dt.clamp(0.0, 0.05);
154
155        // --- ruch poziomy: przyspieszenie albo tarcie ---
156        if move_x.abs() > 0.01 {
157            self.facing = move_x.signum();
158            self.vel.x += move_x * cfg.accel * dt;
159            self.vel.x = self.vel.x.clamp(-cfg.run_speed, cfg.run_speed);
160        } else {
161            // Tarcie nie może przewrócić znaku — inaczej gracz
162            // „drży" na miejscu zamiast się zatrzymać.
163            let drop = cfg.friction * dt;
164            self.vel.x = if self.vel.x.abs() <= drop {
165                0.0
166            } else {
167                self.vel.x - self.vel.x.signum() * drop
168            };
169        }
170
171        // --- okna skoku ---
172        if jump_pressed {
173            self.buffer_left = cfg.jump_buffer;
174        } else {
175            self.buffer_left = (self.buffer_left - dt).max(0.0);
176        }
177        self.jump_held = jump_held;
178        if self.grounded {
179            self.coyote_left = cfg.coyote;
180        } else {
181            self.coyote_left = (self.coyote_left - dt).max(0.0);
182        }
183
184        // --- skok ---
185        //
186        // Wypadający skok gwarantujemy przez `coyote_left > 0`, a skok
187        // naciśnięty przed lądowaniem przez `buffer_left > 0`. Obie
188        // warunki razem dają standardowe, „niedziwne" sterowanie.
189        if self.buffer_left > 0.0 && self.coyote_left > 0.0 {
190            self.vel.y = cfg.jump_speed;
191            self.grounded = false;
192            self.coyote_left = 0.0;
193            self.buffer_left = 0.0;
194        }
195
196        // --- grawitacja ---
197        //
198        // Szybciej w dół niż w górę (`fall_multiplier`) — klasyczny
199        // „asymetryczny skok", dzięki któremu opadanie nie trwa tyle
200        // co wzbijanie. Bez tego unikanie ataków w powietrzu byłoby
201        // męczące.
202        if !self.grounded {
203            let g = if self.vel.y < 0.0 {
204                cfg.gravity * cfg.fall_multiplier
205            } else {
206                cfg.gravity
207            };
208            self.vel.y += g * dt;
209            if self.vel.y < -cfg.terminal_velocity {
210                self.vel.y = -cfg.terminal_velocity;
211            }
212        }
213
214        // Puścienie skoku ucina wznoszenie — skok zmiennej wysokości.
215        if !self.jump_held && self.vel.y > 0.0 {
216            self.vel.y *= 0.45;
217        }
218
219        if self.invuln > 0.0 {
220            self.invuln = (self.invuln - dt).max(0.0);
221            if self.invuln == 0.0 {
222                self.hurt = false;
223            }
224        }
225
226        self.integrate(dt, map);
227    }
228
229    /// Ruch z rozwiązywaniem kolizji: najpierw X, potem Y.
230    ///
231    /// Kolejność jest celowa. Po ruchu po X **testujemy** kolizję i
232    /// cofamy pozycję, a dopiero potem ruszamy Y. Bez cofania gracz
233    /// wchodziłby w ścianę i dopiero przy następnym kaflu wypychałby
234    /// się — wyglądałoby to jak „przyklejenie do ściany".
235    fn integrate(&mut self, dt: f32, map: &TileMap) {
236        // --- oś X ---
237        let step_x = self.vel.x * dt;
238        self.pos.x += step_x;
239        if map.overlaps_solid(self.rect()) {
240            self.pos.x -= step_x;
241            self.vel.x = 0.0;
242        }
243
244        // --- oś Y ---
245        //
246        // Zapamiętujemy, czy **przed** ruchem gracz spadał. Bez tego
247        // nie odróżnilibyśmy lądowania od wznoszenia: po skoku stopi
248        // są w powietrzu, a pod nimi wciąż jest podłoga, więc test
249        // „czy coś jest pod stopami" dawałby fałszywe trafienie
250        // i gracz przyklejałby się do podłogi w miejscu skoku.
251        let falling = self.vel.y <= 0.0;
252        let step_y = self.vel.y * dt;
253        self.pos.y += step_y;
254
255        if !falling {
256            // Wznoszenie: jedyny warunek to sufit (pełny blok).
257            // Platformy **nie** blokują od dołu — to ich sens.
258            if map.overlaps_solid(self.rect()) {
259                self.pos.y -= step_y;
260                self.vel.y = 0.0;
261            }
262            self.grounded = false;
263            return;
264        }
265
266        // Spadanie: lądujemy, gdy pod stopami pojawił się kafel.
267        if map.ground_below(self.rect(), 2.0) || map.overlaps_solid(self.rect()) {
268            self.land(map);
269        } else {
270            self.grounded = false;
271        }
272    }
273
274    /// Lądowanie: stopy na dachu kafla, prędkość zerowa.
275    ///
276    /// Wyrównujemy do **najwyższego dachu pod stopami**, a nie do
277    /// pierwszego znalezionego — inaczej przy schodkach gracz
278    /// lądowałby „w powietrzu" na najniższym z kafli.
279    fn land(&mut self, map: &TileMap) {
280        let cx0 = (self.pos.x / TILE).floor() as i32;
281        let cx1 = ((self.pos.x + BODY_W) / TILE).floor() as i32;
282        let cy_foot = (self.pos.y / TILE).floor() as i32;
283        let mut best: Option<f32> = None;
284        for cy in (cy_foot - 2)..=(cy_foot + 1) {
285            for cx in cx0..=cx1 {
286                if map.at(cx, cy).blocks_from_above() {
287                    // Dach kafla to górna krawędź: `y = (cy + 1) * TILE`.
288                    let top = (cy + 1) as f32 * TILE;
289                    // Bierzemy tylko dachy nie wyższe niż pół kafla nad
290                    // stopami — wyższe to sufit, w który gracz wpadł.
291                    if top <= self.pos.y + TILE * 0.5 {
292                        best = Some(best.map_or(top, |b: f32| b.max(top)));
293                    }
294                }
295            }
296        }
297        match best {
298            Some(top) => {
299                self.pos.y = top;
300                self.grounded = true;
301            }
302            None => self.grounded = false,
303        }
304        self.vel.y = 0.0;
305    }
306}

313    const DT: f32 = 1.0 / 60.0;

315    /// Mapa z wierszem podłogi na dole, reszta pusta.
316    fn map_with(rows: &[(usize, usize, Tile)]) -> TileMap {
317        let mut grid: Vec<Vec<Tile>> = vec![vec![Tile::Empty; MAP_W]; MAP_H];
318        grid[0] = vec![Tile::Solid; MAP_W];
319        for &(y, x, t) in rows {
320            grid[y][x] = t;
321        }
322        let refs: Vec<&[Tile]> = grid.iter().map(|r| r.as_slice()).collect();
323        TileMap::from_rows(&refs)
324    }
325
326    /// Płaska mapa: sama podłoga.
327    fn flat_map() -> TileMap {
328        map_with(&[])
329    }
330
331    /// Gracz stoi na podłodze (`y = TILE`, dach kafla rzędu 0).
332    fn standing() -> (Player, PlayerConfig) {
333        let mut p = Player::default();
334        p.pos = Vec2::new(100.0, TILE);
335        p.grounded = true;
336        (p, PlayerConfig::default())
337    }
338
339    /// Wykonuje `n` klatek z danym wejściem.
340    ///
341    /// `jump` to **wciśnięcie** (edge). Przytrzymanie ustawiamy
342    /// osobno przez `held`, bo skok zmiennej wysokości ucina wznoszenie
343    /// przy puszczeniu — test z samym edge'em widziałby skok o połowę
344    /// niższy niż w prawdziwej grze.
345    fn step_n(
346        p: &mut Player,
347        n: usize,
348        move_x: f32,
349        jump: bool,
350        map: &TileMap,
351        cfg: &PlayerConfig,
352    ) {
353        for _ in 0..n {
354            p.step(DT, move_x, jump, false, map, cfg);
355        }
356    }
357
358    #[test]
359    fn standing_still_does_not_move_the_player() {
360        let (p, cfg) = standing();
361        let map = flat_map();
362        let before = p.pos;
363        let mut p2 = p;
364        step_n(&mut p2, 30, 0.0, false, &map, &cfg);
365        assert!((p2.pos - before).length() < 0.01, "gracz drży w miejscu");
366    }
367
368    #[test]
369    fn gravity_pulls_the_player_down_to_the_floor() {
370        let (mut p, cfg) = standing();
371        let map = flat_map();
372        p.pos.y = TILE + 100.0;
373        p.grounded = false;
374        step_n(&mut p, 120, 0.0, false, &map, &cfg);
375        assert!(p.grounded, "gracz nie wylądował");
376        assert!((p.pos.y - TILE).abs() < 0.5, "stopy {} != {TILE}", p.pos.y);
377    }
378
379    #[test]
380    fn running_accelerates_to_the_configured_speed() {
381        let (mut p, cfg) = standing();
382        let map = flat_map();
383        step_n(&mut p, 120, 1.0, false, &map, &cfg);
384        assert!(
385            (p.vel.x - cfg.run_speed).abs() < 1.0,
386            "prędkość {} != {}",
387            p.vel.x,
388            cfg.run_speed
389        );
390        assert!(p.pos.x > 100.0, "gracz nie poszedł");
391    }

393    #[test]
394    fn friction_stops_the_player_fully() {
395        let (mut p, cfg) = standing();
396        let map = flat_map();
397        step_n(&mut p, 60, 1.0, false, &map, &cfg);
398        step_n(&mut p, 300, 0.0, false, &map, &cfg);
399        assert_eq!(p.vel.x, 0.0, "gracz musi się zatrzymać");
400    }

402    #[test]
403    fn jump_leaves_the_ground_and_comes_back() {
404        let (mut p, cfg) = standing();
405        let map = flat_map();
406        p.step(DT, 0.0, true, true, &map, &cfg);
407        assert!(p.vel.y > 0.0, "skok nie nadał prędkości");
408        assert!(!p.grounded, "gracz jest w powietrzu");
409        let start = p.pos.y;
410        let mut peak = start;
411        // Przytrzymujemy klawisz — inaczej skok zmiennej wysokości
412        // ucina wznoszenie i test widziałby za niski szczyt. Wciśnięcie
413        // (`i == 0`) to **edge**: powtarzane co klatkę odnawiałoby skok
414        // w locie, co jest błędem, a nie cechą skoku.
415        for i in 0..120 {
416            p.step(DT, 0.0, i == 0, true, &map, &cfg);
417            peak = peak.max(p.pos.y);
418        }
419        // `jump_speed² / 2 * gravity` ≈ 104 px ≈ 3,2 kafla.
420        assert!(
421            peak - start > TILE * 2.5,
422            "skok za niski: {start} -> {peak}"
423        );
424        assert!(p.grounded, "gracz nie wylądował");
425    }

427    #[test]
428    fn releasing_jump_early_makes_a_shorter_hop() {
429        let (mut p, cfg) = standing();
430        let map = flat_map();
431        let start = p.pos.y;
432        // Naciskamy i **puszczamy** natychmiast.
433        p.step(DT, 0.0, true, false, &map, &cfg);
434        let mut peak = p.pos.y;
435        for _ in 0..120 {
436            p.step(DT, 0.0, false, false, &map, &cfg);
437            peak = peak.max(p.pos.y);
438        }
439        // Krótki skok jest wyraźnie niższy niż pełny (sprawdzamy
440        // niżej) — tu pilnujemy tylko, że w ogóle wznosi.
441        assert!(peak > start + 2.0, "puszczony skok to nie skok");
442    }

444    #[test]
445    fn no_double_jump_in_mid_air() {
446        let (mut p, cfg) = standing();
447        let map = flat_map();

449        let first = p.vel.y;

452    }
