1//! Pociski: zarówno atak gracza, jak i pociski bossa.
2//!
3//! Jeden typ struktury obsługuje obie strony walki — różni je
4//! `owner`. To świadomy wybór: gdyby były dwie osobne listy, kod
5//! rysowania, kolizji i sprzątania musiałby się powtarzać, a
6//! najbardziej prawdopodobny błąd („pocisk gracza zadał obrażenia
7//! graczowi") byłby niemal niemożliwy do zauważenia.
8//!
9//! Wszystko w jednostkach świata i sekundach.
10
11use uran_math::{Rect, Vec2};
12
13/// Kto wystrzelił pocisk.
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub enum Owner {
16    Player,
17    Boss,
18}
19
20/// Pocisk: prostokąt, prędkość, obrażenia i czas życia.
21#[derive(Debug, Clone, Copy)]
22pub struct Projectile {
23    /// Środek pocisku.
24    pub pos: Vec2,
25    /// Prędkość w jednostkach na sekundę.
26    pub vel: Vec2,
27    /// Kto wystrzelił.
28    pub owner: Owner,
29    /// Ile obrażeń zadaje.
30    pub damage: f32,
31    /// Prostokąt pocisku.
32    pub size: Vec2,
33    /// Ile sekund zostało do zniknięcia.
34    pub life: f32,
35    /// Czy pocisk jest ciężki (przebija, nie znika po trafieniu).
36    pub piercing: bool,
37}
38
39impl Projectile {
40    /// Prostokąt kolizji pocisku.
41    pub fn rect(&self) -> Rect {
42        Rect::from_center(self.pos, self.size)
43    }
44
45    /// Krok: ruch po linii prostej i starzenie.
46    ///
47    /// Pociski nie reagują na grawitację ani kafle — celowo. Boss
48    /// strzela prostymi, a gracz może strzelać pod górę; gdyby pocisk
49    /// spadał, jego tor przestałby być przewidywalny, a unikanie
50    /// ataków stałoby się nieuczciwe.
51    pub fn step(&mut self, dt: f32) {
52        self.pos += self.vel * dt;
53        self.life -= dt;
54    }
55
56    /// Czy pocisk wygasł.
57    pub fn is_dead(&self) -> bool {
58        self.life <= 0.0
59    }
60}
61
62/// Ile sekund żyje pocisk — bez tego pociski bossa leciałyby w
63/// nieskończoność i kumulowały się w pamięci.
64pub const BULLET_LIFE: f32 = 6.0;
65
66/// Czy pocisk gracza trafia bossa.
67///
68/// Osobna funkcja, a nie test w `main.rs`, bo `main.rs` ma GPU i okno
69/// — testy logiki walki trzymamy w modułach, które działają bez okna.
70#[cfg(test)]
71pub fn player_hits_boss(bullet: &Projectile, boss_rect: Rect) -> bool {
72    bullet.owner == Owner::Player && bullet.rect().intersects(&boss_rect)
73}
74
75/// Czy pocisk bossa trafia gracza.
76#[cfg(test)]
77pub fn boss_hits_player(bullet: &Projectile, player_rect: Rect) -> bool {
78    bullet.owner == Owner::Boss && bullet.rect().intersects(&player_rect)
79}
80
81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    fn bullet(owner: Owner) -> Projectile {
86        Projectile {
87            pos: Vec2::ZERO,
88            vel: Vec2::new(200.0, 0.0),
89            owner,
90            damage: 10.0,
91            size: Vec2::splat(8.0),
92            life: BULLET_LIFE,
93            piercing: false,
94        }
95    }
96
97    #[test]
98    fn moves_along_its_velocity() {
99        let mut b = bullet(Owner::Player);
100        b.step(0.5);
101        assert!((b.pos.x - 100.0).abs() < 1e-3, "x = {}", b.pos.x);
102        assert_eq!(b.pos.y, 0.0);
103    }
104
105    #[test]
106    fn dies_after_its_lifetime() {
107        let mut b = bullet(Owner::Player);
108        assert!(!b.is_dead());
109        for _ in 0..(BULLET_LIFE * 100.0) as u32 {
110            b.step(0.01);
111        }
112        assert!(b.is_dead(), "pocisk nie zniknął");
113    }
114
115    #[test]
116    fn owner_prevents_friendly_fire() {
117        let p = bullet(Owner::Player);
118        let boss = Rect::from_xywh(-10.0, -10.0, 20.0, 20.0);
119        // Pocisk gracza trafia bossa…
120        assert!(player_hits_boss(&p, boss));
121        // …ale nie gracza, nawet gdy stoi w tym samym miejscu.
122        assert!(!boss_hits_player(&p, boss));
123    }
124
125    #[test]
126    fn boss_bullet_only_hits_the_player() {
127        let b = bullet(Owner::Boss);
128        let r = Rect::from_xywh(-10.0, -10.0, 20.0, 20.0);
129        assert!(boss_hits_player(&b, r));
130        assert!(!player_hits_boss(&b, r));
131    }
132
133    #[test]
134    fn miss_when_far_away() {
135        let b = bullet(Owner::Player);
136        assert!(!player_hits_boss(
137            &b,
138            Rect::from_xywh(500.0, 0.0, 10.0, 10.0)
139        ));
140    }
141}
