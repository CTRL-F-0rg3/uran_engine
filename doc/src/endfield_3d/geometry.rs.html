1//! Proste bryły 3D budowane w kodzie — bez plików `.obj`.
2//!
3//! Cała scena powstaje z `Mesh` składanych z trójkątów. Dzięki temu
4//! demo nie zależy od żadnego assetu i uruchamia się z pustego
5//! katalogu, a kształt każdego obiektu da się zmienić w kodzie.
6//!
7//! ## Wiatr (winding)
8//!
9//! Wszystkie ściany budujemy w kolejności dającej **normalną na
10//! zewnątrz**, bo potok ma `cull_mode: Back`. Trójkąt z podwórną
11//! orientacją znika razem z całą bryłą — objaw jest mylący, bo reszta
12//! sceny renderuje się poprawnie.
13
14use uran_math::Vec3;
15use uran_render3d::{Mesh, Vertex};

17/// Prostopadłościan: 6 ścian, każda w 2 trójkąty.
18///
19/// `half` to połowa rozmiaru, więc bryła jest symetryczna wokół
20/// `center` — wygodniej niż podawanie dwóch rogów.
21pub fn box_mesh(center: Vec3, half: Vec3, color: [f32; 3]) -> Mesh {
22    let mut m = Mesh::new();
23    let h = half;
24    let c = center;
25
26    // Każda ściana: 4 narożniki w kolejności dającej CCW widziany
27    // z zewnątrz.
28    let faces: [([Vec3; 4], Vec3); 6] = [
29        // górna (+Y)
30        (
31            [
32                c + Vec3::new(-h.x, h.y, -h.z),
33                c + Vec3::new(-h.x, h.y, h.z),
34                c + Vec3::new(h.x, h.y, h.z),
35                c + Vec3::new(h.x, h.y, -h.z),
36            ],
37            Vec3::Y,
38        ),
39        // dolna (-Y)
40        (
41            [
42                c + Vec3::new(-h.x, -h.y, h.z),
43                c + Vec3::new(-h.x, -h.y, -h.z),
44                c + Vec3::new(h.x, -h.y, -h.z),
45                c + Vec3::new(h.x, -h.y, h.z),
46            ],
47            Vec3::NEG_Y,
48        ),
49        // przednia (+Z)
50        (
51            [
52                c + Vec3::new(-h.x, -h.y, h.z),
53                c + Vec3::new(h.x, -h.y, h.z),
54                c + Vec3::new(h.x, h.y, h.z),
55                c + Vec3::new(-h.x, h.y, h.z),
56            ],
57            Vec3::Z,
58        ),
59        // tylna (-Z)
60        (
61            [
62                c + Vec3::new(h.x, -h.y, -h.z),
63                c + Vec3::new(-h.x, -h.y, -h.z),
64                c + Vec3::new(-h.x, h.y, -h.z),
65                c + Vec3::new(h.x, h.y, -h.z),
66            ],
67            Vec3::NEG_Z,
68        ),
69        // prawa (+X)
70        (
71            [
72                c + Vec3::new(h.x, -h.y, h.z),
73                c + Vec3::new(h.x, -h.y, -h.z),
74                c + Vec3::new(h.x, h.y, -h.z),
75                c + Vec3::new(h.x, h.y, h.z),
76            ],
77            Vec3::X,
78        ),
79        // lewa (-X)
80        (
81            [
82                c + Vec3::new(-h.x, -h.y, -h.z),
83                c + Vec3::new(-h.x, -h.y, h.z),
84                c + Vec3::new(-h.x, h.y, h.z),
85                c + Vec3::new(-h.x, h.y, -h.z),
86            ],
87            Vec3::NEG_X,
88        ),
89    ];
90
91    for (corners, normal) in faces {
92        // UV: dolna krawędź V=0, górna V=1. Przy braku map nie ma to
93        // znaczenia, ale układ jest spójny na wypadek dodania tekstury.
94        let uvs = [[0.0, 0.0], [1.0, 0.0], [1.0, 1.0], [0.0, 1.0]];
95        for (i, corner) in corners.iter().enumerate() {
96            m.push(Vertex::new_uv(*corner, normal, uvs[i], color));
97        }
98        m.triangle(0, 1, 2);
99        m.triangle(0, 2, 3);
100    }
101    m
102}

104/// Kula (UV-sphere) o podziale `rings` × `slices`.
105///
106/// Używamy jej dla głowy postaci i dla emitujących elementów — kuliste
107/// bryły łapią światło w sposób nieosiągalny prostopadłościanem.
108pub fn sphere(center: Vec3, radius: f32, rings: u32, slices: u32, color: [f32; 3]) -> Mesh {
109    let mut m = Mesh::new();
110    let (ri, si) = (rings.max(2), slices.max(3));
111
112    for r in 0..=ri {
113        // `v` od 0 (biegun północny) do 1 (południowy).
114        let v = r as f32 / ri as f32;
115        let phi = v * std::f32::consts::PI;
116        let (sin_phi, cos_phi) = phi.sin_cos();
117        for s in 0..=si {
118            let u = s as f32 / si as f32;
119            let theta = u * std::f32::consts::TAU;
120            let (sin_t, cos_t) = theta.sin_cos();
121            let dir = Vec3::new(sin_phi * cos_t, cos_phi, sin_phi * sin_t);
122            let p = center + dir * radius;
123            // Normalna = kierunek promienia. Dla kuli to dokładnie ta
124            // sama definicja co pozycja znormalizowana.
125            m.push(Vertex::new_uv(p, dir, [u, 1.0 - v], color));
126        }
127    }
128
129    // `si + 1` wierzchołków na pierścień — ostatni powtarza pierwszy,
130    // żeby UV nie miało szwu.
131    let stride = (si + 1) as u32;
132    for r in 0..ri {
133        for s in 0..si {
134            let a = r * stride + s;
135            let b = a + 1;
136            let c = a + stride;
137            let d = c + 1;
138            m.triangle(a, c, b);
139            m.triangle(b, c, d);
140        }
141    }
142    m
143}

145/// Walec (boki + dwie pokrywy) o osi Y.
146///
147/// Dla kolumn, rur i beczułek. `sides` to liczba ścian bocznych —
148/// 12 wystarczy, żeby przy tej skali nie było widocznych krawędzi.
149pub fn cylinder(center: Vec3, radius: f32, half_height: f32, sides: u32, color: [f32; 3]) -> Mesh {
150    let mut m = Mesh::new();
151    let n = sides.max(3);
152    let h = half_height;
153
154    // --- boki
155    for i in 0..n {
156        let a0 = i as f32 / n as f32 * std::f32::consts::TAU;
157        let a1 = (i + 1) as f32 / n as f32 * std::f32::consts::TAU;
158        let (s0, c0) = a0.sin_cos();
159        let (s1, c1) = a1.sin_cos();
160        let i0 = m.vertices.len() as u32;
161        // Normalna ściany bocznej to kierunek radialny w połowie łuku
162        // — dzięki temu ściana nie ma widocznego załamania w miejscu
163        // styku sąsiednich segmentów.
164        let mid = (a0 + a1) * 0.5;
165        let nrm = Vec3::new(mid.cos(), 0.0, mid.sin());
166        m.push(Vertex::new_uv(
167            center + Vec3::new(c0 * radius, -h, s0 * radius),
168            nrm,
169            [i as f32 / n as f32, 0.0],
170            color,
171        ));
172        m.push(Vertex::new_uv(
173            center + Vec3::new(c1 * radius, -h, s1 * radius),
174            nrm,
175            [(i + 1) as f32 / n as f32, 0.0],
176            color,
177        ));
178        m.push(Vertex::new_uv(
179            center + Vec3::new(c1 * radius, h, s1 * radius),
180            nrm,
181            [(i + 1) as f32 / n as f32, 1.0],
182            color,
183        ));
184        m.push(Vertex::new_uv(
185            center + Vec3::new(c0 * radius, h, s0 * radius),
186            nrm,
187            [i as f32 / n as f32, 1.0],
188            color,
189        ));
190        m.triangle(i0, i0 + 1, i0 + 2);
191        m.triangle(i0, i0 + 2, i0 + 3);
192    }
193
194    // --- pokrywy
195    //
196    // Dwie osobne pętle, bo poprawny wiatr wymaga odwrotnej kolejności
197    // na górze i na dole. Próba współdzielenia pętli z `if` wygląda
198    // krócej, ale przy debugowaniu cullinga jest nieczytelna.
199    for (y, nrm, forward) in [(h, Vec3::Y, true), (-h, Vec3::NEG_Y, false)] {
200        let c_idx = m.vertices.len() as u32;
201        m.push(Vertex::new_uv(
202            center + Vec3::new(0.0, y, 0.0),
203            nrm,
204            [0.5, 0.5],
205            color,
206        ));
207        for i in 0..n {
208            let a = i as f32 / n as f32 * std::f32::consts::TAU;
209            let (s, c) = a.sin_cos();
210            m.push(Vertex::new_uv(
211                center + Vec3::new(c * radius, y, s * radius),
212                nrm,
213                [c * 0.5 + 0.5, s * 0.5 + 0.5],
214                color,
215            ));
216        }
217        for i in 0..n {
218            let a = c_idx + 1 + i;
219            let b = c_idx + 1 + (i + 1) % n;
220            if forward {
221                m.triangle(c_idx, a, b);
222            } else {
223                m.triangle(c_idx, b, a);
224            }
225        }
226    }
227    m
228}

230/// Płaska płyta z podziałem na segmenty — podłoga i ściany.
231///
232/// Podział jest potrzebny, żeby cień i odbicia SSR miały co liczyć:
233/// jedna wielka płyta dałaby płaszczyznę o stałej normalnej. Pozwala
234/// też zafalować podłogę bez dokładania osobnych siatek.
235pub fn plate(
236    center: Vec3,
237    half: Vec3,
238    seg_x: u32,
239    seg_z: u32,
240    color_fn: impl Fn(f32, f32) -> [f32; 3],
241) -> Mesh {
242    let mut m = Mesh::new();
243    let (nx, nz) = (seg_x.max(1), seg_z.max(1));
244
245    for gz in 0..nz {
246        for gx in 0..nx {
247            let x0 = center.x - half.x + gx as f32 * (half.x * 2.0 / nx as f32);
248            let x1 = center.x - half.x + (gx + 1) as f32 * (half.x * 2.0 / nx as f32);
249            let z0 = center.z - half.z + gz as f32 * (half.z * 2.0 / nz as f32);
250            let z1 = center.z - half.z + (gz + 1) as f32 * (half.z * 2.0 / nz as f32);
251            let col = color_fn((x0 + x1) * 0.5, (z0 + z1) * 0.5);
252
253            let p00 = Vec3::new(x0, center.y, z0);
254            let p10 = Vec3::new(x1, center.y, z0);
255            let p11 = Vec3::new(x1, center.y, z1);
256            let p01 = Vec3::new(x0, center.y, z1);
257            // Naprzemienne przekątne co wiersz — inaczej siatka daje
258            // widoczny wzór schodków pod światłem.
259            if (gx + gz) % 2 == 0 {
260                m.tri(
261                    Vertex::new(p00, Vec3::Y, col),
262                    Vertex::new(p11, Vec3::Y, col),
263                    Vertex::new(p10, Vec3::Y, col),
264                );
265                m.tri(
266                    Vertex::new(p00, Vec3::Y, col),
267                    Vertex::new(p01, Vec3::Y, col),
268                    Vertex::new(p11, Vec3::Y, col),
269                );
270            } else {
271                m.tri(
272                    Vertex::new(p00, Vec3::Y, col),
273                    Vertex::new(p01, Vec3::Y, col),
274                    Vertex::new(p10, Vec3::Y, col),
275                );
276                m.tri(
277                    Vertex::new(p10, Vec3::Y, col),
278                    Vertex::new(p01, Vec3::Y, col),
279                    Vertex::new(p11, Vec3::Y, col),
280                );
281            }
282        }
283    }
284    m
285}

287/// Rura (cylindryczna) między dwoma punktami — przewody i kable.
288///
289/// Osobno od `cylinder`, bo rura bywa długa, cienka i idzie w dowolnym
290/// kierunku. Budujemy ją z dwóch wektorów prostopadłych do osi, zamiast
291/// z pełnej macierzy obrotu — to dwa mnożenia zamiast dziewięciu.
292pub fn pipe(from: Vec3, to: Vec3, radius: f32, color: [f32; 3], sides: u32) -> Mesh {
293    let dir = to - from;
294    let len = dir.length();
295    if len < 1e-5 {
296        // Zerowa długość dałaby `normalize(0)` = NaN, a NaN w pozycji
297        // wierzchołka znika w cullingu. Pusta siatka jest bezpieczna.
298        return Mesh::new();
299    }
300    let axis = dir / len;
301    // Gdy rura idzie pionowo, `cross` z osią Y zeruje się — wybieramy
302    // wtedy inną oś pomocniczą.
303    let helper = if axis.dot(Vec3::Y).abs() > 0.99 {
304        Vec3::X
305    } else {
306        Vec3::Y
307    };
308    let side = axis.cross(helper).normalize_or_zero();
309    let other = axis.cross(side).normalize_or_zero();
310
311    let mut m = Mesh::new();
312    let n = sides.max(3);
313    for i in 0..n {
314        let a0 = i as f32 / n as f32 * std::f32::consts::TAU;
315        let a1 = (i + 1) as f32 / n as f32 * std::f32::consts::TAU;
316        let d0 = side * a0.cos() + other * a0.sin();
317        let d1 = side * a1.cos() + other * a1.sin();
318        let mid = (a0 + a1) * 0.5;
319        let nrm = side * mid.cos() + other * mid.sin();
320        let i0 = m.vertices.len() as u32;
321        m.push(Vertex::new_uv(
322            from + d0 * radius,
323            nrm,
324            [i as f32 / n as f32, 0.0],
325            color,
326        ));
327        m.push(Vertex::new_uv(
328            to + d0 * radius,
329            nrm,
330            [i as f32 / n as f32, 1.0],
331            color,
332        ));
333        m.push(Vertex::new_uv(
334            to + d1 * radius,
335            nrm,
336            [(i + 1) as f32 / n as f32, 1.0],
337            color,
338        ));
339        m.push(Vertex::new_uv(
340            from + d1 * radius,
341            nrm,
342            [(i + 1) as f32 / n as f32, 0.0],
343            color,
344        ));
345        m.triangle(i0, i0 + 2, i0 + 1);
346        m.triangle(i0, i0 + 3, i0 + 2);
347    }
348    m
349}

355    /// Normalna trójkąta `a,b,c` z mnożenia wektorów.
356    fn tri_normal(a: Vec3, b: Vec3, c: Vec3) -> Vec3 {
357        (b - a).cross(c - a).normalize_or_zero()
358    }
359
360    /// Wszystkie ściany prostopadłościanu zwracają się na zewnątrz.
361    ///
362    /// To najczęstszy błąd tej warstwy i najdroższy w naprawie:
363    /// trójkąt z podwórną orientacją znika po cichu, a objaw
364    /// („dziury w podłodze") wskazuje zupełnie inne miejsce.
365    #[test]
366    fn box_faces_point_outwards() {
367        let m = box_mesh(Vec3::ZERO, Vec3::splat(1.0), [1.0; 3]);
368        for (i, ind) in m.indices.chunks(3).enumerate() {
369            let a = m.vertices[ind[0] as usize].pos();
370            let b = m.vertices[ind[1] as usize].pos();
371            let c = m.vertices[ind[2] as usize].pos();
372            let n = tri_normal(a, b, c);
373            let centroid = (a + b + c) / 3.0;
374            assert!(
375                centroid.dot(n) > 0.0,
376                "trójkąt {i} zwrócony do wewnątrz: n={n:?}, centroid={centroid:?}"
377            );
378        }
379    }
380
381    #[test]
382    fn sphere_vertices_lie_on_the_sphere() {
383        let c = Vec3::new(1.0, 2.0, 3.0);
384        let r = 0.5;
385        let m = sphere(c, r, 8, 12, [1.0; 3]);
386        for v in &m.vertices {
387            let dir = v.pos() - c;
388            assert!(
389                (dir.length() - r).abs() < 0.01,
390                "wierzchołek {} leży {} od środka, a nie {r}",
391                v.pos(),
392                dir.length()
393            );
394            assert!(
395                dir.normalize_or_zero().dot(v.normal()) > 0.99,
396                "normalna {:?} nie zgadza się z pozycją {:?}",
397                v.normal(),
398                v.pos()
399            );
400        }
401    }

403    /// Ściany boczne walca mają normalne radialne.
404    ///
405    /// Uwaga na kryterium: JEDEN segment ma jedną normalną, wypośrodkowaną
406    /// na kącie środkowym, a nie cztery różne. Dzięki temu walec ma
407    /// płaskie „boki" jak cylinder, a nie gładki walec. Test porównuje
408    /// więc normalną z KIERUNKIEM radialnym, sprawdzając znak iloczynu,
409    /// a nie równość — równość wymagałaby czterech normalnych na
410    /// segment, czyli gładkiego cieniowania, a to zabija krawędzie, o
411    /// których tu chodzi.
412    #[test]
413    fn cylinder_side_normals_are_radial() {
414        let m = cylinder(Vec3::ZERO, 1.0, 2.0, 8, [1.0; 3]);
415        // 8 segmentów × 4 wierzchołki = 32 wierzchołki boków.
416        for v in m.vertices.iter().take(32) {
417            assert!(
418                v.normal().y.abs() < 1e-5,
419                "boczna normalna {:?} ma składową Y",
420                v.normal()
421            );
422            let radial = Vec3::new(v.pos().x, 0.0, v.pos().z).normalize_or_zero();
423            // Dodatni iloczyn = normalna wskazuje NA ZEWNĄTRZ. Dokładna
424            // równość nie zachodzi dla wierzchołków narożnych segmentu,
425            // bo ich normalna celuje w środek łuku.
426            assert!(
427                radial.dot(v.normal()) > 0.7,
428                "normalna {:?} nie wskazuje na zewnątrz w punkcie {:?}",
429                v.normal(),
430                v.pos()
431            );
432        }
433    }

435    #[test]
436    fn pipe_normals_are_perpendicular_to_axis() {
437        let a = Vec3::new(0.0, 0.0, 0.0);
438        let b = Vec3::new(0.0, 5.0, 0.0);
439        let m = pipe(a, b, 0.3, [1.0; 3], 8);
440        assert!(!m.vertices.is_empty(), "rura jest pusta");
441        let axis = (b - a).normalize_or_zero();
442        for v in &m.vertices {
443            let dot = v.normal().dot(axis);
444            assert!(
445                dot.abs() < 1e-3,
446                "normalna {:?} nie jest prostopadła do osi (dot={dot})",
447                v.normal()
448            );
449        }
450    }

452    #[test]

454        let m = plate(Vec3::ZERO, Vec3::new(4.0, 0.0, 4.0), 4, 4, |_, _| [1.0; 3]);

463        }
