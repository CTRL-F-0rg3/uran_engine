1//! Scena: industrialna platforma badawcza pod otwartym niebem.
2//!
3//! # Dlaczego ta scena
4//!
5//! Żeby ocenić styl Endfield, potrzeba otoczenia, które daje materiałowi
6//! czym oddychać. Sama postać na pustej płaszczyźnie pokaże rysunkową
7//! rampę, ale nie pokaże odbic SSR, mgły ani promieni słonecznych.
8//!
9//! Scena jest więc celowo zbudowana tak, żeby każdy z tych efektów
10//! miał gdzie działać:
11//!
12//! | element sceny | co pokazuje |
13//! |---|---|
14//! | wypolerowana podłoga z panelami | odbicia SSR, anamorficzna poświata |
15//! | słupy i belki | kontury ekranowe na krawędziach |
16//! | neonowe listwy | bloom, flara, promienie słoneczne |
17//! | otwarta przestrzeń nad platformą | fizyczne niebo Rayleigh + Mie |
18//! | dystans do przeciwległej krawędzi | mgła eksponencjalna, DoF |
19//! | jasny żółty rdzeń reaktora | ciepły odbłysk na otoczeniu |

26/// Połowa boku platformy. Scena jest kwadratowa.
27pub const ARENA_HALF: f32 = 26.0;

29/// Wysokość podłogi. Wszystko inne pozycjonujemy względem niej.
30///
31/// Osobna stała, a nie `0.0` w kodzie, bo podłoga jest PUNKTEM
32/// odniesienia dla całej sceny: słupy, postacie i pociski liczą
33/// wysokość od niej. Podniesienie podłogi o 2 m przesunęłoby wtedy
34/// reaktor, a nie całą scenę razem z nim.
35pub const FLOOR_Y: f32 = 0.0;

37/// Paleta barw — spójna w całej scenie.

40/// grzebania w kilku plikach. Kolory są podane w sRGB, bo tak je

44/// # UWAGA: to są kolory MATERIAŁÓW, nie wierzchołków

132/// Czy (x, z) leży na linii szwu paneli.
133///
134/// Szwy to nie tekstura, tylko kolor wierzchołków, więc pozycje
135/// siatki dzielimy tu. Dzięki temu podłoga ma czytelną siatkę
136/// perspektywiczną, a nie jednolitą plamę — i jest coś, czego
137/// oczy się złapią przy ruchu kamery.
138fn on_seam(x: f32, z: f32) -> bool {
139    const PANEL: f32 = 4.0;
140    let fx = (x / PANEL).fract().abs();
141    let fz = (z / PANEL).fract().abs();
142    // 0.06 to szerokość szwu w jednostkach świata. Węższy znika
143    // w aliasingu, szerszy wygląda jak mapa drogowa.
144    fx < 0.06 || fz < 0.06
145}

147/// Buduje podłogę platformy.
148///
149/// Podział na 40×40 segmentów daje wystarczająco gęstą siatkę:
150/// mapa cieni ma 2048² na całą scenę, więc jedna kratka podłogi
151/// dostaje kilka texeli. Rzadsza siatka daje widoczne „schodki"
152/// na granicach cienia.
153pub fn build_floor() -> Mesh {
154    geometry::plate(
155        Vec3::new(0.0, FLOOR_Y, 0.0),
156        Vec3::new(ARENA_HALF, 0.0, ARENA_HALF),
157        40,
158        40,
159        |x, z| {
160            // Podłoga JEST wyjątkiem: szew i gradient muszą zmieniać się
161            // na jednej bryle, więc tu kolor siedzi w wierzchołkach
162            // (materiał podłogi ma `vertex_color: true`).
163            if on_seam(x, z) {
164                palette::FLOOR_SEAM
165            } else {
166                // Delikatne zaciemnienie do środka platformy: środek
167                // jest dalej od krawędzi, więc traci więcej światła
168                // otoczenia. Daje wrażenie głębi bez geometrii.
169                let d = (x * x + z * z).sqrt() / ARENA_HALF;
170                let k = 1.0 - 0.18 * d;
171                [
172                    palette::FLOOR[0] * k,
173                    palette::FLOOR[1] * k,
174                    palette::FLOOR[2] * k,
175                ]
176            }
177        },
178    )
179}

181/// Wkłada bryłę `src` do `dst`, przesuwając ją do `at` i obracając
182/// wokół osi Y o `yaw`.
183///
184/// Robimy to przez transformację pozycji i normalnych — ten sam
185/// wzorzec, którego wymaga wczytanie `.obj`, tyle że napisany
186/// ręcznie. Alternatywą była macierz, ale dla kilkunastu żeber
187/// płacimy tylko za jedno mnożenie.
188fn merge_into(dst: &mut Mesh, src: &Mesh, at: Vec3, yaw: f32) {
189    let (s, c) = yaw.sin_cos();
190    let rot = |p: Vec3| Vec3::new(p.x * c - p.z * s, p.y, p.x * s + p.z * c);
191    let base = dst.vertices.len() as u32;
192    for v in &src.vertices {
193        dst.vertices.push(uran_render3d::Vertex::new_uv(
194            at + rot(v.pos()),
195            rot(v.normal()),
196            v.uv,
197            v.color,
198        ));
199    }
200    for idx in &src.indices {
201        dst.indices.push(idx + base);
202    }
203}

205/// Buduje wszystkie siatki sceny.
206pub fn build_meshes() -> SceneMeshes {
207    use geometry::{box_mesh, cylinder, pipe, sphere};
208    // Kolor wierzchołka dla całej geometrii poza podłogą: biel, bo
209    // barwę niesie materiał (`Surface::base`).
210    let w = VERTEX_WHITE;
211
212    // Podest centralny: niski walec pod reaktorem.
213    let dais = cylinder(Vec3::new(0.0, 0.25, 0.0), 7.0, 0.25, 24, w);
214
215    // --- rdzeń: świecąca kula w metalowej obudowie
216    //
217    // Dwie bryły, nie jedna: świecąca kula jest WIDOCZNA przez
218    // szczeliny obudowy. Dlatego obudowa ma 8 pionowych żeber
219    // zamiast pełnego cylindra — światło rdzenia wychodzi
220    // między nimi i daje scenie główne źródło poświaty.
221    let core_inner = sphere(Vec3::new(0.0, 3.4, 0.0), 1.5, 20, 24, w);
222
223    // Żebra obudowy: 8 pionowych słupków wokół rdzenia.
224    let mut core_shell = Mesh::new();
225    for i in 0..8 {
226        let a = i as f32 / 8.0 * std::f32::consts::TAU;
227        let (s, c) = a.sin_cos();
228        let pos = Vec3::new(c * 1.9, 2.4, s * 1.9);
229        // Bryła jest budowana W ZERZE, a `pos` ją tylko ustawia —
230        // inaczej przesunięcie byłoby policzone dwa razy.
231        let rib = box_mesh(Vec3::ZERO, Vec3::new(0.16, 2.4, 0.30), w);
232        merge_into(&mut core_shell, &rib, pos, a);
233    }
234    // Pierścienie na górze i na dole — spinają żebra w jedną bryłę.
235    for y in [0.3f32, 4.6] {
236        let ring = cylinder(Vec3::new(0.0, y, 0.0), 2.1, 0.14, 16, w);
237        merge_into(&mut core_shell, &ring, Vec3::ZERO, 0.0);
238    }
239
240    // Słup: kwadratowy przekrój ostrzejszy niż walec.
241    //
242    // Kwadrat daje czytelne narożniki dla konturów ekranowych —
243    // okrągły słup dałby krawędź, której głębokość zmienia się
244    // płynnie i kontur by pulsował.
245    let pillar = box_mesh(Vec3::ZERO, Vec3::new(0.55, 4.0, 0.55), w);
246    // Belka stropowa dłuższa — jedna siatka, inna skala przy rysowaniu.
247    let beam = box_mesh(Vec3::ZERO, Vec3::new(9.0, 0.35, 0.45), w);
248    let crate_mesh = box_mesh(Vec3::ZERO, Vec3::splat(1.0), w);
249    // Listwa emisyjna: wąski, długi prostopadłościan.
250    let strip = box_mesh(Vec3::ZERO, Vec3::new(0.14, 0.05, 4.0), w);
251    // Rura biegnie wzdłuż osi X.
252    let conduit = pipe(
253        Vec3::new(-1.0, 0.0, 0.0),
254        Vec3::new(1.0, 0.0, 0.0),
255        0.16,
256        w,
257        10,
258    );
259
260    // --- przeciwnik: skorupa + rdzeń
261    //
262    // Skorupa to ośmiokąt „pancerza", a rdzeń wystaje spomiędzy
263    // segmentów i świeci — czytelny cel dla gracza.
264    let drone_core = sphere(Vec3::ZERO, 0.42, 12, 16, w);
265    let mut drone_shell = Mesh::new();
266    for i in 0..8 {
267        let a = i as f32 / 8.0 * std::f32::consts::TAU;
268        let pos = Vec3::new(a.cos() * 0.62, 0.0, a.sin() * 0.62);
269        // Jak wyżej: geometria w zerze, pozycję dokłada `merge_into`.
270        let plate_mesh = box_mesh(Vec3::ZERO, Vec3::new(0.10, 0.34, 0.20), w);
271        merge_into(&mut drone_shell, &plate_mesh, pos, a);
272    }
273
274    // Pocisk: wydłużona bryła. Wydłużenie sprawia, że przy ruchu
275    // widać kierunek lotu — kulka dawałaby kropkę bez informacji.
276    let bolt = box_mesh(Vec3::ZERO, Vec3::new(0.10, 0.10, 0.55), w);
277
278    SceneMeshes {
279        floor: build_floor(),
280        dais,
281        core_inner,
282        core_shell,
283        pillar,
284        beam,
285        crate_mesh,
286        strip,
287        conduit,
288        drone_core,
289        drone_shell,
290        bolt,
291    }
