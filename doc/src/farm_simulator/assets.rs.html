1//! Wczytywanie modeli budynków z katalogu `assets/`.
2//!
3//! ## Konwencja osi: te pliki są Y-up
4//!
5//! Wbrew pierwszemu wrażeniu (`# Blender v2.79` w nagłówku) pliki
6//! Quaternius **nie są** Z-up. Dowód z samych wymiarów:
7//!
8//! | model | x | y | z | wniosek |
9//! |-------|---|---|---|---------|
10//! | `Fence` | 5,89 | **1,10** | 0,17 | płot jest długi w x, niski w y |
11//! | `Silo` | 3,67 | **9,07** | 3,51 | silos jest wysoki w y, okrągły w x/z |
12//!
13//! Gdyby to był Z-up, `Fence` miałby 5,89 m **grubości** i 0,17 m
14//! wysokości, a `Silo` byłby płaskim krążkiem. Żaden z tych modeli
15//! nie pasowałby do opisu.
16//!
17//! Dlatego ładujemy `AxisUp::Y` (to układ docelowy, importer nic nie
18//! obraca). Przy `AxisUp::Z` wszystkie budynki lądują na boku — co
19//! widać na zrzucie ekranu jako „kupa belek" zamiast wsi.
20//!
21//! ## Dlaczego osobna funkcja szukająca pliku
22//!
23//! `cargo run -p farm-simulator` uruchamia grę z katalogu repozytorium,
24//! a `assets/` leży w katalogu crate'a. Sama ścieżka względna by nie
25//! zadziałała i dostalibyśmy pusty świat zamiast farmy.
26
27use std::path::{Path, PathBuf};

29use uran_render3d::import::{self, AxisUp, ImportedModel};

31/// Katalog z modelami w repozytorium.
32const ASSET_SUBDIR: &str = "Farm Buildings by Quaternius/OBJ";
33
34/// Wybrane budynki: nazwa pliku + docelowa wysokość w metrach.
35///
36/// Wysokość jest jedyną skalą, jakiej potrzebujemy — budynki stawiamy
37/// na ziemi (import już to robi) i skalujemy jednolicie, żeby zachować
38/// proporcje.
39pub struct BuildingSpec {
40    pub file: &'static str,
41    /// Wysokość docelowa w metrach.
42    pub height: f32,
43}
44
45/// Lista budynków użytych w grze.
46///
47/// Wybór jest celowo mały: stodoło, silos, młyn i studnia. Każdy
48/// budynek to kilkanaście siatek GPU i kilkadziesiąt draw calli, a gra
49/// ma być czytelna — sześć różnych budynków rozrzuconych po polu
50/// tylko przeszkadza w rozpoznaniu, co jest uprawą.
51pub const BUILDINGS: &[BuildingSpec] = &[
52    BuildingSpec {
53        file: "BigBarn",
54        // Model ma 7,89 m — zostawiamy naturalną skalę, żeby proporcje
55        // dachu i ścian odpowiadały oryginałowi.
56        height: 7.89,
57    },
58    BuildingSpec {
59        file: "Silo",
60        // Naturalne 9,07 m: silos ma być wyraźnie wyższy od stodoła,
61        // bo to on zaznacza pion i pomaga zlokalizować się na mapie.
62        height: 9.07,
63    },
64    BuildingSpec {
65        file: "Windmill",
66        // 11,20 m — najwyższy element, dobrze widoczny z całej farmy.
67        height: 11.20,
68    },
69    BuildingSpec {
70        file: "Well",
71        height: 2.15,
72    },
73];

75/// Wysokość płotu w metrach.
76///
77/// Model ma 1,10 m i 5,89 m długości. Nie skalujemy — naturalna
78/// wysokość jest poprawna, a proporcja długość/wysokość (5,4:1)
79/// daje dokładnie taki płot, jakiego oczekujemy.
80pub const FENCE_HEIGHT: f32 = 1.10;

82/// Ścieżka katalogu z modelami, znaleziona w kilku miejscach.
83pub fn asset_dir() -> Option<PathBuf> {
84    let rel = Path::new("assets").join(ASSET_SUBDIR);
85    if rel.is_dir() {
86        return Some(rel);
87    }
88    let manifest = Path::new(env!("CARGO_MANIFEST_DIR"))
89        .join("assets")
90        .join(ASSET_SUBDIR);
91    manifest.is_dir().then_some(manifest)
92}
93
94/// Ścieżka jednego pliku `.obj`.
95pub fn model_path(file: &str) -> Option<PathBuf> {
96    let dir = asset_dir()?;
97    let p = dir.join(format!("{file}.obj"));
98    p.is_file().then_some(p)
99}
100
101/// Wczytuje model budynku i skaluje go do zadanej wysokości.
102///
103/// Zwraca `None` z komunikatem na `stderr`, gdy pliku nie ma — gra ma
104/// działać dalej (sama farma), bo brak stodoła to brak dekoracji,
105/// a nie powód do pustego kadru.
106pub fn load_building(file: &str, height: f32) -> Option<ImportedModel> {
107    let path = model_path(file)?;
108    // `AxisUp::Y` — te pliki SĄ Y-up (dowody w dokumentacji modułu).
109    match import::load_from_file_with_axes(&path, AxisUp::Y) {
110        Ok(mut model) => {
111            model.scale_to_height(height);
112            Some(model)
113        }
114        Err(e) => {
115            eprintln!("⚠️  nie udało się wczytać `{}`: {e}", path.display());
116            None
117        }
118    }
119}

121/// Ile budynków udało się wczytać (do diagnostyki na starcie).
122pub fn available_buildings() -> usize {
123    BUILDINGS
124        .iter()
125        .filter(|b| model_path(b.file).is_some())
126        .count()
127}
