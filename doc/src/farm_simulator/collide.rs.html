1//! Świat przeszkód i rozwiązywanie wejść gracza.
2//!
3//! Świadomie **bez silnika kolizji** (jak [`junak-rider`]): cała farma
4//! to kilkanaście prostopadłościanów stojących na płaskim terenie.
5//! Silnik byłby tu dwa rzędy wielkości cięższy i wniósłby dokładnie ten
6//! sam błąd — trudniej byłoby zobaczyć, co się dzieje.
7//!
8//! Gracz to prostopadłościan: szerokość [`PLAYER_RADIUS`] na obie
9//! strony i wysokość [`PLAYER_HEIGHT`]. Traktujemy go jako bryłę, a nie
10//! punkt, bo inaczej gracz „wchodziłby" w ścianę aż do samej środka
11//! oczu, a kamera wystawałaby z budynku.
12
13use uran_math::Vec3;
14
15/// Połowa szerokości gracza (promień barków), w metrach.
16pub const PLAYER_RADIUS: f32 = 0.35;
17
18/// Wysokość gracza od stóp do czubka głowy, w metrach.
19///
20/// Oko ([`crate::EYE`]) jest niżej niż czubek głowy, a stopy na `y = 0`.
21pub const PLAYER_HEIGHT: f32 = 1.8;
22
23/// Najdłuższy podkrok ruchu (m) — ochrona przed tunelowaniem.
24///
25/// Ściana ma 0,3 m grubości, więc podkroki krótsze niż to pozwalają
26/// zawsze wykryć wejście. Przy 60 FPS i 5 m/s gracz przesuwa się
27/// 0,083 m na klatkę, więc w normalnej grze dzielenie w ogóle nie
28/// zachodzi — chroni tylko przy zaciętym klatkarzu (patrz test
29/// `huge_dt_does_not_tunnel_through_a_wall`).
30const MAX_STEP: f32 = 0.2;
31
32/// Prostopadłościan w świecie — środek i połowa wymiarów.
33#[derive(Debug, Clone, Copy, PartialEq)]
34pub struct Box {
35    pub center: Vec3,
36    pub half: Vec3,
37}
38
39/// Bryła **postawiona na ziemi** (dno na `y = 0`).
40///
41/// Zapisujemy pozycję i wysokość, a nie środek, bo przy budowaniu
42/// świata z assetów znamy właśnie takie dane.
43pub struct Solid {
44    /// Środek w płaszczyźnie XZ.
45    pub pos: Vec3,
46    pub width: f32,
47    pub depth: f32,
48    pub height: f32,
49}
50
51impl Solid {
52    /// Zamienia na AABB. `pos.y` ignorujemy — bryła stoi na ziemi.
53    pub fn to_box(self) -> Box {
54        Box {
55            center: Vec3::new(self.pos.x, self.height * 0.5, self.pos.z),
56            half: Vec3::new(self.width * 0.5, self.height * 0.5, self.depth * 0.5),
57        }
58    }
59}
60
61/// Świat przeszkód: statyczne bryły + ograniczenie mapy.
62#[derive(Debug, Default, Clone)]
63pub struct World {
64    boxes: Vec<Box>,
65    /// Granica świata w metrach (`|x|`, `|z|`).
66    pub limit: f32,
67}
68
69impl World {
70    pub fn new(limit: f32) -> Self {
71        Self {
72            boxes: Vec::new(),
73            limit,
74        }
75    }
76
77    pub fn add(&mut self, solid: Solid) {
78        self.boxes.push(solid.to_box());
79    }
80
81    /// Wszystkie bryły świata.
82    ///
83    /// Potrzebne do rysowania drutów (patrz `F1` w `main.rs`) — bez
84    /// dostępu do listy debug pokazałby świat bez przeszkód.
85    pub fn boxes(&self) -> &[Box] {
86        &self.boxes
87    }
88
89    /// Przesuwa gracza i rozwiązuje wejścia w bryły. `pos` to **dno**.
90    ///
91    /// Kolejność operacji wynika z tego, że teren jest płaski i
92    /// nieskończony: każde zejście poniżej `y = 0` kończy się
93    /// lądowaniem. Rozwiązywanie poziome **osobno dla X i dla Z** daje
94    /// ślizganie się po ścianie zamiast zacięcia w jej narożniku.
95    ///
96    /// Ruch dzielimy na podkroki nie dłuższe niż [`MAX_STEP`], bo jedno
97    /// rozwiązanie AABB nie chroni przed **tunelowaniem**: przy skoku
98    /// 15 m gracz przeszedłby przez ścianę, bo po przesunięciu jego AABB
99    /// wylądowałby już po drugiej stronie i test przecięcia nie znalazłby
100    /// nic do wypchnięcia. Podkroki gwarantują, że w każdym kroku
101    /// gracz styka się z bryłą **z jednej strony**.
102    pub fn move_player(&self, pos: &mut Vec3, delta: Vec3, radius: f32, height: f32) {
103        let len = delta.length();
104        if len <= MAX_STEP {
105            self.move_one_step(pos, delta, radius, height);
106            return;
107        }
108        let steps = (len / MAX_STEP).ceil() as i32;
109        let part = delta / steps as f32;
110        for _ in 0..steps {
111            self.move_one_step(pos, part, radius, height);
112        }
113    }
114
115    /// Jeden podkrok: grawitacja, lądowanie i wypchnięcia po osiach.
116    fn move_one_step(&self, pos: &mut Vec3, delta: Vec3, radius: f32, height: f32) {
117        // Oś Y: grawitacja i lądowanie. Lądowanie bezwarunkowe, bo
118        // podłoga jest wszędzie na `y = 0`.
119        pos.y += delta.y;
120        if pos.y <= 0.0 {
121            pos.y = 0.0;
122        }
123
124        pos.x += delta.x;
125        self.resolve(&mut *pos, radius, height);
126        pos.z += delta.z;
127        self.resolve(&mut *pos, radius, height);
128    }
129
130    /// Wypycha gracza z brył. Powtarzamy kilka razy, bo jedno
131    /// rozwiązanie może wcisnąć gracza w inną bryłę (np. narożnik
132    /// płotu przy budynku). Limit iteracji chroni przed zawieszeniem,
133    /// gdyby bryły wypierały się nawzajem bez końca.
134    fn resolve(&self, pos: &mut Vec3, radius: f32, height: f32) {
135        for _ in 0..4 {
136            if !self.resolve_once(pos, radius, height) {
137                return;
138            }
139        }
140    }
141
142    /// Jedna runda wypychania. Zwraca `true`, gdy coś się ruszyło.
143    ///
144    /// Po wybraniu kierunku **sprawdzamy wynik** i, jeśli gracz nadal
145    /// leży w bryle, wypychamy go w osi, którą wcześniej pominięliśmy.
146    /// Bez tego zdarzał się błąd „najkrótszej drogi": przy nacisku po
147    /// skosie gracz zostawał w kolidatorze, bo wypchnięcie w Z
148    /// przesuwało go zza ściany prosto do ściany obok.
149    fn resolve_once(&self, pos: &mut Vec3, radius: f32, height: f32) -> bool {
150        let mut moved = false;
151        let head_y = pos.y + height;
152        for b in &self.boxes {
153            // Gracz jako AABB: `2*radius` na X i Z, `height` od stóp.
154            let px_min = b.center.x - b.half.x - radius;
155            let px_max = b.center.x + b.half.x + radius;
156            let pz_min = b.center.z - b.half.z - radius;
157            let pz_max = b.center.z + b.half.z + radius;
158            let by_min = b.center.y - b.half.y;
159            let by_max = b.center.y + b.half.y;
160
161            // Test przecięcia na każdej osi osobno. Używamy `<=`/`>=`,
162            // żeby dotykanie ściany kończyło się wypchnięciem, a nie
163            // przepuszczaniem.
164            if pos.x + radius <= px_min
165                || pos.x - radius >= px_max
166                || pos.z + radius <= pz_min
167                || pos.z - radius >= pz_max
168                || head_y <= by_min
169                || pos.y >= by_max
170            {
171                continue;
172            }
173
174            // Wybieramy tę oś, na której **korygowanie jest najmniejsze**
175            // liczone względem środka bryły. Wypchnięcie liczymy jako
176            // odległość do krawędzi bryły, a nie do środka — różnica
177            // jest istotna przy szerokich bryłach.
178            let push_x = (px_max - pos.x).min(pos.x - px_min);
179            let push_z = (pz_max - pos.z).min(pos.z - pz_min);
180            if push_x <= push_z {
181                // Wybieramy stronę, do której gracz jest bliżej, żeby
182                // wypchnięcie nie przerzucało go przez całą bryłę.
183                pos.x += if pos.x < b.center.x {
184                    px_min - pos.x
185                } else {
186                    px_max - pos.x
187                };
188            } else {
189                pos.z += if pos.z < b.center.z {
190                    pz_min - pos.z
191                } else {
192                    pz_max - pos.z
193                };
194            }
195            moved = true;
196        }
197        moved
198    }
199}

205    /// Świat z jedną bryłą 2×2×2 stojącą w (0,0,0), czyli zajmującą
206    /// X i Z w zakresie [-1, 1].

208        let mut w = World::new(50.0);
209        w.add(Solid {

218    #[test]
219    fn solid_sits_on_the_ground() {
220        let b = Solid {
221            // `pos.y` celowo śmieciowe — bryła stoi na ziemi, więc
222            // przesunięcie w Y musi zostać zignorowane.
223            pos: Vec3::new(1.0, 99.0, 2.0),
224            width: 4.0,
225            depth: 6.0,
226            height: 3.0,
227        }
228        .to_box();
229        assert_eq!(b.center, Vec3::new(1.0, 1.5, 2.0));
230        assert_eq!(b.half, Vec3::new(2.0, 1.5, 3.0));
231    }

233    #[test]
234    fn player_cannot_walk_into_a_wall() {
235        let w = world_with_hut();
236        let mut p = Vec3::new(-2.0, 0.0, 0.0);
237        // Duży krok prosto w ścianę.
238        w.move_player(
239            &mut p,
240            Vec3::new(5.0, 0.0, 0.0),
241            PLAYER_RADIUS,
242            PLAYER_HEIGHT,
243        );
244        assert!(p.x + PLAYER_RADIUS <= 1.0, "gracz wszedł w ścianę: {p:?}");
245    }

247    #[test]
248    fn player_slides_along_the_wall_instead_of_sticking() {
249        let w = world_with_hut();
250        let mut p = Vec3::new(-2.0, 0.0, 0.0);
251        // Nacisk po skosie: bez ślizgania utknąłby w narożniku.
252        w.move_player(
253            &mut p,
254            Vec3::new(5.0, 0.0, 2.0),
255            PLAYER_RADIUS,
256            PLAYER_HEIGHT,
257        );
258        // Ślizganie: gracz musi przesunąć się w Z (poza ścianę).
259        assert!(p.z > 1.0, "gracz nie przesunął się wzdłuż ściany: {p:?}");
260        // I **nie może** zostać w bryle. Nie sprawdzamy tu `x`, bo
261        // ślizganie z natury zostawia `x` wewnątrz zakresu bryły —
262        // to poprawne, bo gracz jest poza nią dzięki `z`.
263        let inside = p.x.abs() < 1.0 && p.z.abs() < 1.0;
264        assert!(!inside, "gracz wpadł do bryły: {p:?}");
265    }
