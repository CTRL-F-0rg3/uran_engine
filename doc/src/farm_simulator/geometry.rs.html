1//! Geometria budowana w kodzie: teren, roślina, postać gracza.
2//!
3//! Modele budynków bierzemy z plików `.obj` (patrz `assets.rs`), ale
4//! rośliny i postać rysujemy sami. Powód: roślina musi zmieniać
5//! wygląd zależnie od wzrostu, a pojedynczy model `.obj` to jedna
6//! bryła — animowałibyśmy ją skalą, co psuje proporcje liści.
7//!
8//! Wszystkie bryły trzymamy w metrach i stawiamy na `y = 0`, bo
9//! `model_matrix` tylko przesuwa — nie chcemy podnosić modeli ręcznie
10//! w kodzie gry.
11
12use uran_math::Vec3;
13use uran_render3d::{Mesh, Vertex};

15/// Barwy gruntu. Trzymane jako osobne stałe, bo shader mnoży przez nie
16/// albedo wierzchołka — kolor to jedyny „materiał" tych brył.
17pub mod palette {
18    /// Trawa. Ciepły, lekko żółtawy zielony — bliżej polskiej łąki
19    /// niż fluorowa zieleń z gier.
20    pub const GRASS: [f32; 3] = [0.36, 0.54, 0.22];
21    /// Ciemniejsza trawa na dalszych pasach (daje wrażenie pola).
22    pub const GRASS_DARK: [f32; 3] = [0.29, 0.45, 0.19];
23    /// Ziemia uprawna. Ciepła brązowa, wyraźnie ciemniejsza od trawy,
24    /// żeby pole było rozpoznawalne z góry.
25    pub const SOIL: [f32; 3] = [0.42, 0.29, 0.18];
26    /// Złota dojrzała roślina.
27    pub const RIPE: [f32; 3] = [0.85, 0.68, 0.22];
28    /// Zielona niedojrzała roślina.
29    pub const GREEN: [f32; 3] = [0.30, 0.62, 0.24];
30    /// Pień i liście — ciemniejszy zielony dla kontrastu.
31    pub const STEM: [f32; 3] = [0.24, 0.48, 0.20];
32    /// Koszulka gracza — ciepły pomarańcz, odróżnia od zieleni pola.
33    pub const SHIRT: [f32; 3] = [0.90, 0.48, 0.18];
34    /// Spodnie gracza.
35    pub const PANTS: [f32; 3] = [0.24, 0.30, 0.45];
36    /// Skóra.
37    pub const SKIN: [f32; 3] = [0.93, 0.76, 0.60];
38}

40/// Kwadratowa płyta z normalną w górę.
41///
42/// Kolejność wierzchołków `(p00, p11, p10)` daje skręt w prawo
43/// (normalna +Y). Odwrócenie daje -Y i cała płyta znika przy
44/// back-face culling — dlatego kolejność tu jest jawna.
45pub fn quad_y(cx: f32, cz: f32, half: f32, y: f32, color: [f32; 3]) -> Mesh {
46    let p00 = Vec3::new(cx - half, y, cz - half);
47    let p10 = Vec3::new(cx + half, y, cz - half);
48    let p11 = Vec3::new(cx + half, y, cz + half);
49    let p01 = Vec3::new(cx - half, y, cz + half);
50    let n = Vec3::Y;
51    let mut m = Mesh::new();
52    m.tri(
53        Vertex::new(p00, n, color),
54        Vertex::new(p11, n, color),
55        Vertex::new(p10, n, color),
56    );
57    m.tri(
58        Vertex::new(p00, n, color),
59        Vertex::new(p01, n, color),
60        Vertex::new(p11, n, color),
61    );
62    m
63}

65/// Równa plansza trawy `size` × `size` m, podzielona na kwadraty `step`.
66///
67/// Naprzemienne odcienie wierszy dają perspektywę: płaska jednokolorowa
68/// zielona kwadratowa płaszczyzna wygląda jak plik graficzny, a nie
69/// jak łąka. Zmiana odcienia na co drugi kwadrat jest najtańszym
70/// sposobem na pokazanie skali świata.
71///
72/// Zwiększamy każdy kwadrat o `+0.01` na boku, żeby sąsiadujące
73/// kwadraty lekko na siebie nachodziły. Bez tego przez pory w bicie
74/// widać by czarną siatkę.
75pub fn ground(size: f32, step: f32) -> Mesh {
76    let n = (size / step).round().max(1.0) as i32;
77    let half = step * 0.5 + 0.01;
78    let mut mesh = Mesh::new();
79    for gz in 0..n {
80        for gx in 0..n {
81            let cx = -size * 0.5 + (gx as f32 + 0.5) * step;
82            let cz = -size * 0.5 + (gz as f32 + 0.5) * step;
83            let c = if (gx + gz) % 2 == 0 {
84                palette::GRASS
85            } else {
86                palette::GRASS_DARK
87            };
88            let p00 = Vec3::new(cx - half, 0.0, cz - half);
89            let p10 = Vec3::new(cx + half, 0.0, cz - half);
90            let p11 = Vec3::new(cx + half, 0.0, cz + half);
91            let p01 = Vec3::new(cx - half, 0.0, cz + half);
92            // Kolejność daje skręt w prawo => normalna +Y (patrz `quad_y`).
93            mesh.tri(
94                Vertex::new(p00, Vec3::Y, c),
95                Vertex::new(p11, Vec3::Y, c),
96                Vertex::new(p10, Vec3::Y, c),
97            );
98            mesh.tri(
99                Vertex::new(p00, Vec3::Y, c),
100                Vertex::new(p01, Vec3::Y, c),
101                Vertex::new(p11, Vec3::Y, c),
102            );
103        }
104    }
105    mesh
106}

108/// Ostrosłup: prostokątna podstawa na `y = base_y`, wierzchołek
109/// na `y = base_y + height`.
110///
111/// Wariant z podstawą przesuniętą (zamiast zawsze `y = 0`) jest
112/// potrzebny dla liści, które wyrastają z pędu na pewnej wysokości.
113/// Bez tego musielibyśmy przesuwać już dodane wierzchołki, a to
114/// łatwo zepsuć: przesuwa się wtedy też środek podstawy.
115fn pyramid_at(
116    mesh: &mut Mesh,
117    lo: [f32; 2],
118    hi: [f32; 2],
119    base_y: f32,
120    height: f32,
121    color: [f32; 3],
122) {
123    let (cx, cz) = ((lo[0] + hi[0]) * 0.5, (lo[1] + hi[1]) * 0.5);
124    let p00 = Vec3::new(lo[0], base_y, lo[1]);
125    let p10 = Vec3::new(hi[0], base_y, lo[1]);
126    let p11 = Vec3::new(hi[0], base_y, hi[1]);
127    let p01 = Vec3::new(lo[0], base_y, hi[1]);
128    let top = Vec3::new(cx, base_y + height, cz);
129    // Każda ściana ma WŁASNĄ normalną: przy wspólnej cała bryła
130    // wyglądałaby płasko, bez cieni, jak naklejka.
131    let mut face = |a: Vec3, b: Vec3| {
132        let n = (b - a).cross(top - a).normalize_or_zero();
133        mesh.tri(
134            Vertex::new(a, n, color),
135            Vertex::new(b, n, color),
136            Vertex::new(top, n, color),
137        );
138    };
139    face(p00, p10);
140    face(p10, p11);
141    face(p11, p01);
142    face(p01, p00);
143}

145/// Ostrosłup stojący na `y = 0`.
146fn pyramid(mesh: &mut Mesh, lo: [f32; 2], hi: [f32; 2], height: f32, color: [f32; 3]) {
147    pyramid_at(mesh, lo, hi, 0.0, height, color);
148}

150/// Roślina na jednej działce.
151///
152/// `growth`: 0.0 = świeżo zasadzony pęd, 1.0 = dojrzała roślina.
153///
154/// Wzrost zmienia **wysokość pędu i kształt liści**, a nie tylko skalę.
155/// Skalowanie całej bryły dałoby kiełbki cienkie na górze i grube na
156/// dole. Tutaj pęd wydłuża się, a liście wyrastają stopniowo — dzięki
157/// temu wzrost widać z odległości kamery.
158pub fn plant(growth: f32) -> Mesh {
159    let g = growth.clamp(0.0, 1.0);
160    let mut mesh = Mesh::new();
161
162    // Kolor przechodzi zielony -> złoty. `smoothstep` daje łagodne
163    // przejście w drugiej połowie wzrostu, gdy zieleń „dojrzewa".
164    let r = ((g - 0.45) / 0.55).clamp(0.0, 1.0);
165    let t = r * r * (3.0 - 2.0 * r);
166    let leaf = [
167        palette::GREEN[0] + (palette::RIPE[0] - palette::GREEN[0]) * t,
168        palette::GREEN[1] + (palette::RIPE[1] - palette::GREEN[1]) * t,
169        palette::GREEN[2] + (palette::RIPE[2] - palette::GREEN[2]) * t,
170    ];
171
172    // Wysokość pędu: 0,25 m (pęd) -> 1,35 m (dojrzała roślina).
173    let stem_h = 0.25 + g * 1.10;
174    // Grubość pędu maleje, żeby nie był jednym klocem.
175    let stem_w = 0.085 - g * 0.040;
176    pyramid(
177        &mut mesh,
178        [-stem_w, -stem_w],
179        [stem_w, stem_w],
180        stem_h,
181        palette::STEM,
182    );
183
184    // Liście pojawiają się od 40% wzrostu. Dwa liście wystarczą:
185    // na działce 2 m więcej zlewa się w zieloną plamę.
186    if g > 0.4 {
187        let a = ((g - 0.4) / 0.6).clamp(0.0, 1.0);
188        let base_y = stem_h * (0.35 + 0.40 * a);
189        let lw = 0.34 * a;
190        let lh = 0.16 * a;
191        // Dwa liście na wprost, w osi X — czytelne z każdej strony.
192        pyramid_at(
193            &mut mesh,
194            [0.0, -lw * 0.30],
195            [lw * 2.0, lw * 0.30],
196            base_y,
197            lh,
198            leaf,
199        );
200        pyramid_at(
201            &mut mesh,
202            [-lw * 2.0, -lw * 0.30],
203            [0.0, lw * 0.30],
204            base_y,
205            lh,
206            leaf,
207        );
208    }
209
210    // Dojrzały kłos: złoty stożek na szczycie pędu. Bez niego nie
211    // widać, że roślina nadaje się do zbioru — a to cel gry.
212    if g >= 1.0 {
213        let w = 0.10;
214        pyramid_at(
215            &mut mesh,
216            [-w, -w],
217            [w, w],
218            stem_h - 0.10,
219            0.42,
220            palette::RIPE,
221        );
222    }
223
224    mesh
225}

227/// Postać gracza: nogi, tors, głowa.
228///
229/// Proporta „chibi" (głowa duża wobec ciała) daje sylwetkę
230/// rozpoznawalną z typowej odległości kamery (~14 m).
231pub fn player() -> Mesh {
232    let mut m = Mesh::new();
233    // Nogi: dwa krótkie słupki.
234    for sx in [-0.13f32, 0.13] {
235        pyramid(
236            &mut m,
237            [sx - 0.09, -0.10],
238            [sx + 0.09, 0.10],
239            0.75,
240            palette::PANTS,
241        );
242    }
243    // Tors.
244    pyramid(&mut m, [-0.19, -0.13], [0.19, 0.13], 1.30, palette::SHIRT);
245    // Głowa: ostrosłup w górę — w niskiej perspektywie wygląda jak
246    // głowa z daszkiem, a nie jak sześcian.
247    pyramid(&mut m, [-0.15, -0.13], [0.15, 0.13], 1.62, palette::SKIN);
248    m
249}

259/// Grube krawędzie prostopadłościana — bryła `collide::Box` widoczna
260/// jako drunek.
261///
262/// Rysujemy **krawędzie** (12 prostych), nie ściany: ściany by były
263/// półprzezroczyste, wymagały sortowania i zasłaniałyby świat.
264/// Drunek nie przeszkadza patrzeć i jest czytelny z każdego kąta.
265///
266/// Grubość to nie prawdziwa szerokość linii — po prostu mały kwadrat
267/// wokół każdej krawędzi, bo rasteryzator nie zna linii. `t` dobrane
268/// tak, żeby 2 cm było widoczne z kilku metrów, a nie zasłaniało
269/// sceny.
270pub fn collider_box(center: Vec3, half: Vec3, color: [f32; 3], t: f32) -> Mesh {
271    // Osiem narożników, w stałej kolejności bitów (x,y,z):
272    // indeks to bit0=x, bit1=y, bit2=z; minus = 0, plus = 1.
273    let mut c = [Vec3::ZERO; 8];
274    for (i, p) in c.iter_mut().enumerate() {
275        let s = |b: u32| if i & (1 << b) != 0 { 1.0 } else { -1.0 };
276        *p = Vec3::new(
277            center.x + s(0) * half.x,
278            center.y + s(1) * half.y,
279            center.z + s(2) * half.z,
280        );
281    }
282    // Dwanaście krawędzi: pary narożników różniące się **jednym**
283    // bitem. To kompletna lista — jej pominięcie albo duplikat to
284    // najczęstszy błąd przy ręcznym rysowaniu prostopadłościanu.
285    const EDGES: [(usize, usize); 12] = [
286        (0, 1),
287        (2, 3),
288        (4, 5),
289        (6, 7),
290        (0, 2),
291        (1, 3),
292        (4, 6),
293        (5, 7),
294        (0, 4),
295        (1, 5),
296        (2, 6),
297        (3, 7),
298    ];
299    let mut m = Mesh::new();
300    for (a, b) in EDGES {
301        bar(&mut m, c[a], c[b], t, color);
302    }
303    m
304}

306/// Kwadratowy pręt między dwoma punktami — jeden odcinek drutu.
307fn bar(m: &mut Mesh, a: Vec3, b: Vec3, t: f32, color: [f32; 3]) {
308    let d = (b - a).normalize();
309    // Dwa wektory prostopadłe do osi pręta. Wybieramy dowolny wektor
310    // nie równoległy do `d` i krzyżujemy — unikamy w ten sposób
311    // przypadku degeneracji przy osi X/Y/Z.
312    let helper = if d.x.abs() < 0.9 { Vec3::X } else { Vec3::Y };
313    let u = d.cross(helper).normalize() * t;
314    let v = d.cross(u).normalize() * t;
315    // Cztery ściany kwadratu wokół osi.
316    let corners = [a - u - v, a + u - v, a + u + v, a - u + v];
317    for i in 0..4 {
318        let p0 = corners[i];
319        let p1 = corners[(i + 1) % 4];
320        // Normalna ściany: na zewnątrz pręta, liczona z osi.
321        let n = (p0 + p1) * 0.5 - (a + b) * 0.5;
322        let n = if n.length_squared() > 1e-12 {
323            n.normalize()
324        } else {
325            u
326        };
327        m.tri(
328            Vertex::new(p0, n, color),
329            Vertex::new(p1, n, color),
330            Vertex::new(b, n, color),
331        );
332        m.tri(
333            Vertex::new(p0, n, color),
334            Vertex::new(b, n, color),
335            Vertex::new(a, n, color),
336        );
337    }
338    // Końce kwadratu — bez nich pręt byłby otwarty i przy pewnych
339    // kątach patrzenia znikałyby jego krawędzie.
340    for &p in &[a, b] {
341        for i in 0..4 {
342            let p0 = corners[i];
343            let p1 = corners[(i + 1) % 4];
344            m.tri(
345                Vertex::new(p0, d, color),
346                Vertex::new(p1, d, color),
347                Vertex::new(p, d, color),
348            );
349        }
350    }
351}

357    #[test]
358    fn ground_is_closed_and_faces_up() {
359        let g = ground(40.0, 4.0);
360        assert!(g.is_closed(), "teren musi mieć komplet trójkątów");
361        assert!(g.vertices.len() > 100);
362        // Każdy wierzchołek ma normalną w górę — inaczej teren zniknie
363        // przy back-face culling.
364        for v in &g.vertices {
365            assert!(v.normal().y > 0.9, "normalna terenu nie w górę: {v:?}");
366        }

373        for v in &q.vertices {

378    #[test]
379    fn plant_grows_taller_with_growth() {
380        // Wysokość bryły to najwyższy punkt wierzchołka.
381        let height = |g: f32| {
382            plant(g)
383                .vertices
384                .iter()
385                .map(|v| v.pos().y)
386                .fold(0.0f32, f32::max)
387        };
388        let small = height(0.0);
389        let mid = height(0.5);
390        let big = height(1.0);
391        assert!(small < mid, "{small} !< {mid}");
392        assert!(mid < big, "{mid} !< {big}");
393        // Pęd musi być widoczny od razu po zasadzeniu.
394        assert!(small > 0.1, "świeżo zasadzona roślina to {} m", small);
395    }
