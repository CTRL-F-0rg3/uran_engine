1//! Kierunki w pierwszej osobie: yaw/pitch na wektory bazowe.
2//!
3//! # Dlaczego ten moduł istnieje
4//!
5//! Kamera pierwszoosobowa potrzebuje trzech wektorów: dokąd patrzymy,
6//! w którą stronę idziemy przy `W` i w którą przy `D`. Błąd w tej
7//! matematyce jest bardzo łatwy do popełnienia i bardzo trudny do
8//! zauważenia na własnym kodzie — stąd osobny plik z testami.
9//!
10//! ## Skąd bierze się odwrócone A/D
11//!
12//! „W prawo na ekranie" to oś X macierzy widzenia, czyli
13//! `Mat4::look_at_rh` — dokładnie ta, której używa renderer. Dla
14//! patrzenia poziomego na `+Z` dostajemy:
15//!
16//! ```text
17//!   forward = +Z,  up = +Y
18//!   right   = +Z × +Y = -X
19//! ```
20//!
21//! Czyli `-X`. To wygląda sprzecznie z intuicją („patrząc na `+Z`,
22//! widzę `+X` po prawej"), ale tak wynika z prawoskrętnej konwencji.
23//!
24//! **Uwaga na historię tego miejsca.** Pierwotna wersja dawała tu
25//! `+X`, przez co A i D były zamienione. Poprawka na `-X` była
26//! prawidłowa, ale została wycofana pod wpływem błędnego testu, który
27//! porównywał `right` ze **składową** macierzy zamiast z **projekcją**
28//! punktu na ekran. Iloczyn wektorów jest antysymetryczny, więc oba
29//! zapisy (`forward × up` i `up × forward`) różnią się znakiem i tylko
30//! jeden pasuje do tego, co naprawdę robi renderer.
31//!
32//! Dlatego znak `right` ustalamy wyłącznie testem, nie rachunkiem na
33//! kartce: [`tests::right_matches_the_view_matrix`] rzutuje punkt
34//! przesunięty o `right` przez prawdziwą macierz i wymaga dodatniego
35//! `x` w NDC, czyli prawej połowy kadru. Zmiana konwencji w silniku
36//! wywali ten test zamiast po cichu odwrócić sterowanie.
37//!
38//! Uwaga na drugą pułapkę: skoro `right` zmienia znak, zmienia się
39//! też kolejność iloczynu w [`up`] — `right × forward`, nie
40//! `forward × right`.
41//!
42//! ## Konwencja
43//!
44//! * `yaw = 0` — patrzymy w `+Z`, zgodnie z `Mat4::look_at_rh`,
45//! * dodatni `yaw` obraca w lewo (konwencja matematyczna),
46//! * `forward` jest zawsze znormalizowany i **nigdy** równoległy do Y
47//!   (`pitch` jest przycięty w [`clamp_pitch`]), więc [`flat`] zawsze
48//!   zwraca wektor jednostkowy.
49
50use uran_math::Vec3;
51
52/// Ograniczenie kąta patrzenia w górę/dół, w radianach (85°).
53///
54/// Prawie 90°. Pełne 90° oznaczałoby `forward` równoległy do osi
55/// `look_at`, a wtedy wektor „w prawo" degeneruje do zera i chód
56/// na wschód przestaje działać. 5° zostaje jako zapas.
57pub const PITCH_LIMIT: f32 = std::f32::consts::FRAC_PI_2 - 0.0873;
58
59/// Przycięcie kąta do bezpiecznego zakresu.
60pub fn clamp_pitch(pitch: f32) -> f32 {
61    pitch.clamp(-PITCH_LIMIT, PITCH_LIMIT)
62}
63
64/// Kierunek patrzenia z kątów `yaw` (poziom) i `pitch` (pion).
65///
66/// Znak `pitch` jest ujemny w górę, bo oś Y wskazuje w górę, a
67/// dodatni obrót wokół X odchyla wektor w dół.
68pub fn forward(yaw: f32, pitch: f32) -> Vec3 {
69    let (sy, cy) = yaw.sin_cos();
70    let (sp, cp) = clamp_pitch(pitch).sin_cos();
71    Vec3::new(sy * cp, sp, cy * cp)
72}
73
74/// Wektor „w prawo" — poziomy, prostopadły do kierunku patrzenia.
75///
76/// Znak ustalony testem `right_matches_the_view_matrix` przeciw
77/// prawdziwej macierzy `Mat4::look_at_rh`. Dla `yaw = 0` (patrzymy na
78/// `+Z`) daje `-X` — patrz opis modułu, dlaczego to nie jest `+X`.
79pub fn right(yaw: f32) -> Vec3 {
80    let (sy, cy) = yaw.sin_cos();
81    Vec3::new(-cy, 0.0, sy)
82}
83
84/// Wektor „w górę" ekranu, prostopadły do patrzenia i do „w prawo".
85///
86/// Kolejność iloczynu jest tu istotna i łatwo ją pomylić: dla
87/// `yaw = 0` mamy `forward = +Z` i `right = +X`, a `Z × X = +Y` —
88/// poprawnie. Zapis odwrotny (`right × forward`) dałby `-Y`, czyli
89/// wektor skierowany w dół.
90///
91/// Przy okazji wynik jest zawsze w górę — składowa Y to `cos(pitch)`,
92/// czyli dodatnia dla każdego dopuszczalnego pitchu. Test
93/// `up_points_up_not_down` pilnuje tego znaku.
94pub fn up(yaw: f32, pitch: f32) -> Vec3 {
95    right(yaw).cross(forward(yaw, pitch))
96}
97
98/// Kierunek patrzenia **bez składowej pionowej** — po nim chodzimy.
99///
100/// Rzut na płaszczyznę `XZ`, potem normalizacja. Normalizacja jest
101/// konieczna: rzut na płaszczyznę sam w sobie nie jest jednostkowy.
102pub fn flat(yaw: f32) -> Vec3 {
103    let (sy, cy) = yaw.sin_cos();
104    Vec3::new(sy, 0.0, cy)
105}
106
107/// Kierunek chodu na płaszczyźnie z dwóch składowych wejścia.
108///
109/// `forward = 1` odpowiada `W`, `strafe = 1` odpowiada `D`; obie
110/// wartości są w zakresie `[-1, 1]`, bo liczy je [`crate::update`]
111/// jako różnicę stanu dwóch klawiszy, a nie z prędkości.
112pub fn move_dir(yaw: f32, forward: f32, strafe: f32) -> Vec3 {
113    (flat(yaw) * forward + right(yaw) * strafe).normalize_or_zero()
114}

120    /// Odwrócone A/D to najczęstszy błąd w FPS-ie, więc testujemy go
121    /// wprost: przy patrzeniu w `+Z` wektor w prawo to `-X`.
122    ///
123    /// Znak bierzemy z `right_matches_the_view_matrix`, który sprawdza
124    /// to rzutem przez prawdziwą macierz `look_at_rh` — ten test sam
125    /// w sobie tylko powtarza jego wynik.
126    #[test]
127    fn strafe_right_points_where_the_screen_right_is() {
128        let r = right(0.0);
129        assert!((r.x + 1.0).abs() < 1e-5, "prawo musi być -X, jest {r:?}");
130        assert!(r.y.abs() < 1e-6, "prawo nie ma składowej pionowej: {r:?}");
131    }

133    /// Prostopadłość osi pod kątem — regresja na `forward × Y`.
134    #[test]
135    fn right_is_orthogonal_to_flat_forward() {
136        for k in 0..16 {
137            let yaw = k as f32 * std::f32::consts::TAU / 16.0;
138            let f = flat(yaw);
139            let r = right(yaw);
140            let dot = f.x * r.x + f.y * r.y + f.z * r.z;
141            assert!(
142                dot.abs() < 1e-5,
143                "kąty nie są prostopadłe przy {yaw}: {dot}"
144            );
145        }
146    }

148    /// Obrót o 90°: patrzymy w `+X`, a w prawo jest `-Z`.
149    ///
150    /// Sprawdzamy spójność obu wektorów, bo `right` sam w sobie jest
151    /// tylko względem „w prawo" — z definicji nie wiadomo, czy to `+Z`
152    /// czy `-Z`, dopóki nie porównamy go z kierunkiem patrzenia.
153    #[test]
154    fn yaw_rotates_the_right_vector_consistently() {
155        let r = right(std::f32::consts::FRAC_PI_2);
156        let f = flat(std::f32::consts::FRAC_PI_2);
157        assert!((f.x - 1.0).abs() < 1e-5, "yaw=90° patrzy w +X: {f:?}");
158        // Obrót o 90°: patrzymy w `+X`, a prawa strona kadru to `+Z`.
159        // Znak potwierdza `right_matches_the_view_matrix`.
160        assert!((r.z - 1.0).abs() < 1e-5, "prawo przy yaw=90° to +Z: {r:?}");
161    }
162    /// Regres: `right` musi wskazywać tam, gdzie naprawdę jest prawa
163    /// strona kadru.
164    ///
165    /// To najważniejszy test w tym pliku i jedyny, który naprawdę
166    /// chroni A/D. Wcześniejsze wersje sprawdzały znak wektora
167    /// „na oko" (`d.x > 0`) — taka kontrola jest zielona dla obu znaków,
168    /// bo nie odwołuje się do tego, jak znak ląduje na ekranie.
169    ///
170    /// Zamiast tego rzutujemy punkt przesunięty o `right` przez tę
171    /// samą macierz `look_at_rh`, której używa renderer, i sprawdzamy,
172    /// że ląduje po **prawej** stronie (dodatnie X w przestrzeni
173    /// kamery). Tak znak jest jednoznaczny — wynika z macierzy, nie
174    /// z rachunku „na kartce".
175    #[test]
176    fn right_matches_the_view_matrix() {
177        use uran_math::Mat4;
178        for k in 0..8 {
179            let yaw = k as f32 * std::f32::consts::FRAC_PI_4;
180            let eye = Vec3::ZERO;
181            let fwd = forward(yaw, 0.0);
182            let view = Mat4::look_at_rh(eye, eye + fwd, Vec3::Y);
183
184            // Punkt 10 m przed graczem i 1 m w stronę `right(yaw)`.
185            let probe = eye + fwd * 10.0 + right(yaw);
186            let p = view * probe.extend(1.0);
187            // Dzielimy przez `w`, bo dopiero to daje NDC — tak jak
188            // zrobi to rasteryzacja. Bez tego porównywalibyśmy
189            // współrzędne oczne, a nie to, co widać na ekranie.
190            let ndc_x = p.x / p.w;
191            assert!(
192                ndc_x > 0.5,
193                "right({yaw}) ląduje po LEWEJ stronie kadru (ndc.x={ndc_x:.3}): \
194                 prawa={:?}, macierz daje dla +X ndc.x={:.3}",
195                right(yaw),
196                {
197                    let other = view * (eye + fwd * 10.0 + Vec3::X).extend(1.0);
198                    other.x / other.w
199                }
200            );
201        }
202    }

204    /// `W` ma iść dokładnie tam, gdzie patrzymy — to właśnie ta
205    /// własność rozjechała się przy kamerze trzecioosobowej.
206    #[test]
207    fn w_walks_where_we_look() {
208        for k in 0..16 {
209            let yaw = k as f32 * std::f32::consts::TAU / 16.0;
210            let d = move_dir(yaw, 1.0, 0.0);
211            let f = flat(yaw);
212            let dot = d.x * f.x + d.z * f.z;
213            assert!(dot > 0.999, "W nie idzie do przodu przy yaw={yaw}: {dot}");
214        }
215    }

217    /// `D` musi iść w prawo ekranu, czyli **nie** w lewo świata.
218    ///
219    /// Oczekiwane kierunki biorą się z `Mat4::look_at_rh` (patrz
220    /// `right_matches_the_view_matrix`), a nie z intuicji: przy
221    /// patrzeniu na `+Z`, czyli `yaw = 0`, prawa strona ekranu to
222    /// `-X`. Stąd `D` daje `-X`, a `A` — `+X`.
223    #[test]
224    fn d_walks_right_not_left() {
225        let d = move_dir(0.0, 0.0, 1.0);
226        assert!(d.x < -0.9, "D musi iść w -X przy yaw=0, jest {d:?}");
227        let d = move_dir(0.0, 0.0, -1.0);
228        assert!(d.x > 0.9, "A musi iść w +X przy yaw=0, jest {d:?}");
229    }

231    /// Regres: ruch w bok musi być zgodny z osią X kamery **po
232    /// obrocie** — nie tylko przy `yaw = 0`. Obracać się i chodzić
233    /// w bok to dokładnie to, o co prosiłeś myszką.
234    #[test]
235    fn strafing_follows_the_camera_after_turning() {
236        for k in 0..8 {
237            let yaw = k as f32 * std::f32::consts::FRAC_PI_4;
238            let d = move_dir(yaw, 0.0, 1.0);
239            let r = right(yaw);
240            let dot = d.x * r.x + d.z * r.z;
241            assert!(
242                dot > 0.999,
243                "D nie idzie w prawo ekranu przy yaw={yaw}: {d:?} vs {r:?}"
244            );
245        }
246    }

248    /// Ruch po skosie nie może być szybszy niż po prostej.
249    #[test]
250    fn diagonal_is_not_faster_than_straight() {
251        let a = move_dir(0.0, 1.0, 0.0).length();
252        let d = move_dir(0.0, 1.0, 1.0).length();
253        assert!((a - 1.0).abs() < 1e-5, "ruch po prostej to 1, jest {a}");
254        assert!((d - 1.0).abs() < 1e-5, "ruch po skosie to 1, jest {d}");
255    }

257    /// Kąt nie wychodzi poza 90°, więc `right` nigdy nie znika.
258    #[test]
259    fn pitch_is_clamped_below_vertical() {
260        assert!(clamp_pitch(10.0) < std::f32::consts::FRAC_PI_2);
261        assert!(clamp_pitch(-10.0) > -std::f32::consts::FRAC_PI_2);
262        for p in [-5.0f32, -1.0, 0.0, 1.0, 5.0] {
263            assert!(up(0.0, p).length() > 0.5, "`up` degeneruje przy pitch={p}");
264            assert!(right(0.0).length() > 0.99);
265        }
266    }

268    /// Wektor patrzenia musi być jednostkowy — długość `flat` zależy
269    /// od `pitch` i przy zerowym kącie padłaby do zera.
270    #[test]
271    fn forward_is_unit_length() {
272        for k in 0..17 {
273            let pitch = k as f32 * 1.0 - 8.0;
274            let f = forward(0.3, pitch);
275            assert!(
276                (f.length() - 1.0).abs() < 1e-4,
277                "forward nie jest jednostkowy przy pitch={pitch}: {}",
278                f.length()
279            );
280        }

283    #[test]
284    fn up_points_up_not_down() {
285        // Sam test prostopadłości nie wykryłby odwrócenia znaku — `-Y`
286        // też jest prostopadły do patrzenia. Sprawdzamy więc sam znak.
287        let u = up(0.0, 0.0);
288        assert!(u.y > 0.99, "`up` musi wskazywać w górę, jest {u:?}");
289        for k in 0..16 {
290            let yaw = k as f32 * std::f32::consts::TAU / 16.0;
291            // Przy patrzeniu poziomym wektor „w górę ekranu" ma zawsze
292            // dodatnią składową Y, niezależnie od obrotu wokół Y.
293            assert!(up(yaw, 0.0).y > 0.99, "up skierowany w dół przy yaw={yaw}");
294        }
295    }

297    /// `up` jest prostopadły do patrzenia — obrót głową nie wywraca
298    /// kamery na bok.
299    #[test]
300    fn up_is_orthogonal_to_forward() {
301        let f = forward(0.7, 0.4);
302        let u = up(0.7, 0.4);
303        let dot = f.x * u.x + f.y * u.y + f.z * u.z;
