1//! Farm Simulator — pole, sadzenie, czekanie, zbiór.
2//!
3//! # Architektura
4//!
5//! Gra stoi na sześciu modułach:
6//!
7//! * [`farm`] — mechanika: siatka działek i pętla sadz → rośnie → zbierz,
8//! * [`geometry`] — bryły budowane w kodzie (trawa, roślina, postać),
9//! * [`look`] — kierunki w pierwszej osobie (yaw/pitch → wektory),
10//! * [`player`] — chód, grawitacja, skok i kolizje,
11//! * [`collide`] — świat przeszkód i rozwiązywanie wejść gracza,
12//! * [`assets`] — budynki wczytywane z plików `.obj` Quaternius.
13//!
14//! `main.rs` trzyma tylko warstwę gry: wejście, kamerę, listę
15//! rysowania i HUD. Cała logika rozgrywki jest testowana w `farm.rs`,
16//! geometria w `geometry.rs`, kierunki w `look.rs`, a fizyka w
17//! `player.rs` i `collide.rs` — wszystkie działają bez GPU.
18//!
19//! Świadomie **bez silnika kolizji** (jak [`junak-rider`]): świat to
20//! kilkanaście prostopadłościanów, więc własne AABB jest tańsze o
21//! rząd wielkości i łatwiejsze do zrozumienia.
22//!
23//! # Sterowanie
24//!
25//! `W/A/S/D` — chodzenie **wzdłuż kierunku patrzenia** (obrót myszą
26//! obraca też kierunek chodzenia), `mysz` — obrót kamery,
27//! `Spacja` — skok, `E` — sadz lub zbierz, `R` — nowa farma,
28//! `F1` — druty kolizji (pomarańczowe bryły świata, błękitny sześcian
29//! gracza), `Esc` — wyjście.

48/// Rozmiar świata w metrach. Kwadratowy, bo mgła i kadr cienia
49/// zakładają się na jeden wymiar — przy prostokącie trzeba by liczyć
50/// dwa różne promienie.
51const WORLD: f32 = 90.0;
52
53/// Wysokość oka nad ziemią (typowa dla postaci ~1,7 m).
54const EYE: f32 = 1.66;
55/// Największa odległość gracza od środka mapy.
56const PLAY_LIMIT: f32 = WORLD * 0.5 - 2.0;
57/// Czułość myszy w radianach na piksel.
58///
59/// 0,0035 rad/px ≈ 0,2°/px, czyli pełne 360° wymaga ~1800 px ruchu
60/// w poziomie. Poprzednie 0,0022 było zbyt „lepkie" w pierwszej
61/// osobie — obrót głowy przy 75° FOV musi być szybki, inaczej
62/// przeciwnik (albo krawędź pola) ucieka z kadru, zanim gracz
63/// zdąży zareagować. Przy tej wartości krótki ruch nadgarstka
64/// daje pełny obrót o 90°, a ramiona nadal kontrolują drobne
65/// korekty.
66const LOOK_SENS: f32 = 0.0035;
67
68/// Kierunek startowego patrzenia: `yaw = PI` = na `-Z`.
69///
70/// Gracz startuje **przed** polem (patrz [`Game::new`]), czyli po
71/// stronie `+Z` osi, a samo pole leży w środku świata, czyli w `0`.
72/// Żeby widzieć pole, trzeba patrzeć na `-Z`. `yaw = 0` patrzyłoby
73/// w przeciwną stronę, na pustą trawę — stąd stała zamiast `0.0`.
74const START_YAW: f32 = std::f32::consts::PI;
75/// Startowy pitch: lekko w dół, na pole przed sobą.
76///
77/// -0,22 rad (≈12,6°) to tyle, żeby ziemia zajmowała dolne ~2/3
78/// kadru, a horyzont i budynki górne 1/3. Zerowy pitch kazałby
79/// patrzeć w niebo i pole ledwo by było widać.
80const START_PITCH: f32 = -0.22;
81/// Odległość startu od **płotu**, nie od pola.
82///
83/// Płot stoi na `fd/2 + 0.5` i ma 1,1 m wysokości. Oko gracza jest na
84/// wysokości 1,66 m, więc stojąc tuż za płotem, gracz patrzyłby
85/// przez jego szczebelnicę — dolna połowa kadru to były drewno.
86/// 7 m odsuwa gracza na tyle, żeby płot znalazł się u dołu ekranu
87/// jako ramka, a nie jako przeszkoda.
88const START_OFFSET: f32 = 7.0;
89
90/// Najwyższe położenie stóp gracza (m) — sufit dla skoku.
91///
92/// Zabezpieczenie przed sytuacją, w której kamera ucieka w górę: gdyby
93/// kolidator kiedyś wypchnął gracza w osi Y (dziś tego nie robi —
94/// [`collide::World`] rozwiązuje tylko X i Z), pozycja stóp i tak nie
95/// przekroczyłaby tej wartości. Skok osiąga ~1,4 m, więc zapas jest
96/// duży, a błąd w kodzie nie objawiłby się jako „latanie po mapie".
97const MAX_FOOT_Y: f32 = 6.0;

101/// Jedno źródło prawdy dla rysowania ([`build_props`]) i dla kolizji

188    /// naciśnięciu i puszczeniu klawisza.

190    /// Czas świata (mgła i animacje).

192    /// Czy wszystkie działki są dojrzałe.

194    /// Czy rysować druty kolizji (`F1`).

196    font: Option<Font>,

199impl Game {
200    /// Buduje świat i renderer 3D na gotowym urządzeniu.
201    fn new(gpu: &uran_render::GpuContext, format: wgpu::TextureFormat, samples: u32) -> Self {
202        let mut scene = Renderer3d::new(gpu, format, samples);
203        // Niebo: jasny błękit. Kolor podajemy w sRGB, bo shader sceny
204        // miesza z nim mgłę i musi zgadzać się z gradientem nieba.
205        scene.set_clear_color([0.58, 0.74, 0.90, 1.0]);
206        {
207            let l = scene.lighting_mut();
208            // Słońce ~48° nad horyzontem i wyraźnie z boku. Niski kąt
209            // daje długie cienie, które pokazują, gdzie jest pole
210            // uprawne, a gdzie zwykła trawa.
211            l.light_dir = Vec3::new(0.42, 0.66, 0.62).normalize();
212            // Ciepłe, lekko złotawe — popołudniowe słońce nad polem.
213            l.light_color = [1.0, 0.95, 0.84];
214            // 0.36/0.45/0.60 → 0.13/0.16/0.21.
215            //
216            // Stara wartość była 3-4× większa od natężenia słońca w
217            // sensie proporcji i **zalewała każdą powierzchnię
218            // niebieskim rozświetleniem**: na zrzucie widać było to jako
219            // trawa w jednolitym, „plastikowym" odcieniu i budynki
220            // przesunięte w fiolet. Przy słońcu 4.0 i takim ambientcie
221            // kadr nie miał żadnego kontrastu — cień i strona na słońcu
222            // wyglądały tak samo, tylko inaczej zabarwione.
223            //
224            // 0.13/0.16/0.21 nadal „niesie niebo" (nie zerowe), więc
225            // cień jest chłodny i czytelny, ale realnie ciemniejszy
226            // od strony naświetlonej. Zgodne z globalnym domyślnym
227            // `Lighting::default()`, które ma dokładnie te wartości.
228            l.ambient = [0.13, 0.16, 0.21];
229        }
230        {
231            let cam = scene.camera_mut();
232            // 75°: pierwsza osoba potrzebuje szerszego pola widzenia niż
233            // kamera trzecioosobowa. Przy 60° budynki przy krawędziach
234            // kadru były mocno zniekształcone, a horyzont ucinał się
235            // zbyt blisko — przy skręcie myszą było „wąsko".
236            cam.fov_y = 75.0_f32.to_radians();
237            cam.near = 0.1;
238            // Dalej płaszczyzna: przy oknie 90 m i budynku 11 m wysokości
239            // 200 m w zupełności wystarcza, a mniejsza wartość pozwala
240            // precyzyjniej dzielić głębokość w buforze cieni.
241            cam.far = 200.0;
242        }
243
244        // --- teren i pole ---
245        let ground = geometry::ground(WORLD, 6.0);
246        let ground = scene.add_mesh(gpu, &ground, "trawa");
247
248        // Gleba: jedna płyta pod całym polem. Działki różnią się tym,
249        // co na nich rośnie, a nie kolorem gruntu — inaczej zamiast
250        // uprawy widać by szachownicę.
251        let fw = farm::FIELD_W as f32 * TILE;
252        let fd = farm::FIELD_D as f32 * TILE;
253        // Podnosimy glebę o 2 cm nad trawę: inaczej z poziomu kamery
254        // obie powierzchnie zlewają się w jedną płaszczyznę.
255        let soil = scene.add_mesh(
256            gpu,
257            &geometry::quad_y(0.0, 0.0, fw * 0.5, 0.02, palette::SOIL),
258            "gleba",
259        );
260
261        // --- roślina w trzech fazach wzrostu ---
262        let plant = [
263            scene.add_mesh(gpu, &geometry::plant(0.12), "roslina 0"),
264            scene.add_mesh(gpu, &geometry::plant(0.55), "roslina 1"),
265            scene.add_mesh(gpu, &geometry::plant(1.0), "roslina 2"),
266        ];
267
268        // --- druty kolizji (F1) ---
269        //
270        // **Kubit jednostkowy** wokół środka: skalą do `half * 2`
271        // dajemy dowolny rozmiar, więc jedna siatka obsługuje wszystkie
272        // kolidatory. Grubość drutu jest w metrach i nie skaluje się —
273        // przy skali kolidatora `half` w metrach, podział trójek jest
274        // mniej mylący niż ułamek proporcji.
275        //
276        // Kolidator mnożymy przez `2 * half` w [`build_draw_list`],
277        // a stąd wewnątrz siatki zostaje grubość 6 cm na krawędzi kubita
278        // jednostkowego — po skalowaniu daje 3 cm na ścianie płotu
279        // i kilkanaście centymetrów na stodole.
280        let collider = scene.add_mesh(
281            gpu,
282            &geometry::collider_box(
283                Vec3::ZERO,
284                Vec3::new(0.5, 0.5, 0.5),
285                geometry::COLLIDER,
286                0.03,
287            ),
288            "kolidator",
289        );
290        let player_box = scene.add_mesh(
291            gpu,
292            &geometry::collider_box(
293                Vec3::ZERO,
294                Vec3::new(0.5, 0.5, 0.5),
295                geometry::PLAYER_BOX,
296                0.06,
297            ),
298            "gracz-box",
299        );
300
301        // Siatki postaci nie ma: w pierwszej osobie oko gracza siedzi
302        // wewnątrz jego głowy, więc narysowana postać zasłoniłaby cały
303        // ekran. Zamiast niej HUD rysuje celownik.
304        let _ = &geometry::player;
305
306        // --- płot i budynki wokół farmy ---
307        let buildings = build_props(&mut scene, gpu);
308
309        println!(
310            "🌾  farma: {}×{} działek, {}×{:.0} m, {}/{} budynków",
311            farm::FIELD_W,
312            farm::FIELD_D,
313            fw,
314            fd,
315            assets::available_buildings(),
316            assets::BUILDINGS.len()
317        );
318
319        Self {
320            scene: Arc::new(Mutex::new(scene)),
321            meshes: Meshes {
322                ground,
323                soil,
324                plant,
325                buildings,
326                collider,
327                player_box,
328            },
329            farm: Farm::new(),
330            // Startujemy przed polem i patrzymy **na** pole: pozycja
331            // jest po stronie `+Z`, więc kierunek patrzenia to `-Z`
332            // (`START_YAW`). `W` od razu prowadzi na działki.
333            player: {
334                let mut p = player::Player::default();
335                p.teleport(0.0, farm::FIELD_D as f32 * TILE * 0.5 + START_OFFSET);
336                p
337            },
338            world: build_collision_world(),
339            player_cfg: player::PlayerConfig::default(),
340            player_pos: Vec3::new(0.0, 0.0, fd * 0.5 + START_OFFSET),
341            yaw: START_YAW,
342            pitch: START_PITCH,
343            eye: Vec3::new(0.0, EYE, fd * 0.5 + START_OFFSET),
344            view_dir: Vec3::new(0.0, 0.0, -1.0),
345            walking: 0.0,
346            walk_blend: 0.0,
347            time: 0.0,
348            won: false,
349            show_colliders: false,
350            font: None,
351        }
352    }
353}

355/// Rejestruje wczytany model: część na część, materiał na materiał.
356///
357/// Materiały deduplikujemy po nazwie: ten sam kolor z `.mtl` występuje
358/// w kilku częściach modelu, a każda kopia to osobny bind group
359/// i osobne tekstury na karcie.
360fn register_model(
361    scene: &mut Renderer3d,
362    gpu: &uran_render::GpuContext,
363    model: uran_render3d::ImportedModel,
364) -> Vec<(MeshId, MaterialId)> {
365    let mut seen: Vec<(String, MaterialId)> = Vec::new();
366    let mut out = Vec::with_capacity(model.parts.len());
367    for part in &model.parts {
368        let mat = match seen.iter().find(|(n, _)| *n == part.material.name) {
369            Some((_, id)) => *id,
370            None => {
371                let id = scene.add_material(gpu, &part.material);
372                seen.push((part.material.name.clone(), id));
373                id
374            }
375        };
376        let id = scene.add_mesh(gpu, &part.mesh, &part.name);
377        out.push((id, mat));
378    }
379    out
380}

382/// Rozstawienie płotu i budynków wokół pola.
383///
384/// Pozycje dobrane tak, żeby z dowolnego miejsca pola widać było
385/// co najmniej jeden budynek — puste niebo wygląda jak brakujący świat,
386/// a nie łąka otoczona lasem.
387fn build_props(scene: &mut Renderer3d, gpu: &uran_render::GpuContext) -> Vec<Prop> {
388    let mut out: Vec<Prop> = Vec::new();
389    let fw = farm::FIELD_W as f32 * TILE;
390    let fd = farm::FIELD_D as f32 * TILE;
391
392    // --- płot wokół pola ---
393    //
394    // Segment `Fence.obj` ma 5,89 m długości przy naturalnej skali
395    // (1,10 m wysokości), więc rozstawiamy go co ~5,5 m z lekkim
396    // zakładką. Wcześniejszy krok 2,2 m kładł segmenty JEDEN NA
397    // DRUGIM — na zrzucie widać by „kupę belek" zamiast płotu.
398    if let Some(model) = assets::load_building("Fence", assets::FENCE_HEIGHT) {
399        let fence = register_model(scene, gpu, model);
400        // 5,5 m: tyle ma segment pomniejszony o zakładkę, żeby słupki
401        // stykały się wizualnie bez szczeliny.
402        let step = 5.5f32;
403        let half_w = fw * 0.5 + 0.5;
404        let half_d = fd * 0.5 + 0.5;
405        // `ceil` w górę + `min 2` — pole musi mieć choćby dwa segmenty
406        // na bok, inaczej płot wygląda jak jedna belka.
407        let nx = (fw / step).ceil().max(2.0) as i32;
408        let nz = (fd / step).ceil().max(2.0) as i32;
409        // `false` = wiersz wzdłuż X, `true` = wzdłuż Z.
410        for (along_z, z_edge, x_edge, yaw, n) in [
411            (false, -half_d, 0.0f32, 0.0f32, nx),
412            (false, half_d, 0.0, std::f32::consts::PI, nx),
413            (true, 0.0, -half_w, std::f32::consts::FRAC_PI_2, nz),
414            (true, 0.0, half_w, -std::f32::consts::FRAC_PI_2, nz),
415        ] {
416            for k in 0..n {
417                // `-0.5 .. +0.5` to środek pierwszego i ostatniego
418                // segmentu, dzięki czemu płot jest symetryczny.
419                let t = (k as f32 + 0.5) / n as f32 - 0.5;
420                let pos = if along_z {
421                    Vec3::new(x_edge, 0.0, t * n as f32 * step)
422                } else {
423                    Vec3::new(t * n as f32 * step, 0.0, z_edge)
424                };
425                out.push(Prop {
426                    parts: fence.clone(),
427                    pos,
428                    yaw,
429                });
430            }
431        }
432    }
433
434    // --- budynki ---
435    //
436    // Każdy ma własny obrót: stodoło prostopadle do młyna czyta się
437    // jako wieś, a wszystko równolegle — jako parking.
438    // Pole ma 12×12 m, więc budynki stawiamy od 9 m od jego środka
439    // w bok — inaczej stodoło (7,7 m szerokości) zasłaniałoby uprawę.
440    for (file, pos, yaw, _w, _d, _h) in PROP_PLACES {
441        let Some(spec) = assets::BUILDINGS.iter().find(|b| b.file == *file) else {
442            continue;
443        };
444        let Some(model) = assets::load_building(file, spec.height) else {
445            continue;
446        };
447        out.push(Prop {
448            parts: register_model(scene, gpu, model),
449            pos: *pos,
450            yaw: *yaw,
451        });
452    }
453
454    out
455}

457/// Buduje świat kolizji z **tych samych** danych co [`build_props`].
458///
459/// Pozycje budynków żyją w jednej stałej [`PROP_PLACES`], z której
460/// korzystają obie funkcje. Duplikowanie ich w dwóch miejscach to
461/// zaproszenie do błędu: wystarczy przesunąć stodoło w rysowaniu
462/// i gracz przechodzi przez jego środek.
463fn build_collision_world() -> collide::World {
464    let fw = farm::FIELD_W as f32 * TILE;
465    let fd = farm::FIELD_D as f32 * TILE;
466    let mut w = collide::World::new(PLAY_LIMIT);
467
468    // --- płot wokół pola ---
469    //
470    // Cztery **ciągłe** ściany, nie po jednym segmencie: gracz nie
471    // powinien wciskać się w szczelinę między słupkami. Ściana ma
472    // 0,3 m grubości i pełną wysokość płotu, więc przeskoczyć go
473    // można, ale przejść obok — nie da się wejść na pole inaczej
474    // niż przez bramę.
475    let half_w = fw * 0.5 + 0.5;
476    let half_d = fd * 0.5 + 0.5;
477    let t = 0.3;
478    for (x, z, width, depth) in [
479        // północna i południowa
480        (0.0f32, -half_d, fw + 1.0, t),
481        (0.0, half_d, fw + 1.0, t),
482        // wschodnia i zachodnia
483        (-half_w, 0.0, t, fd + 1.0),
484        (half_w, 0.0, t, fd + 1.0),
485    ] {
486        w.add(collide::Solid {
487            pos: Vec3::new(x, 0.0, z),
488            width,
489            depth,
490            height: assets::FENCE_HEIGHT,
491        });
492    }
493
494    // --- budynki ---
495    //
496    // Rozmiary bierzemy z naturalnych proporcji modeli, a nie z AABB
497    // modelu — budynki stoją na trawie, więc gracz nie wchodzi w ich
498    // wnętrze przez dach.
499    for (_, pos, yaw, width, depth, height) in PROP_PLACES {
500        // Obrót o `yaw` zmienia obrys w XZ, więc bierzemy obrys
501        // prostokąta **po obrocie**. Inaczej gracz wchodziłby w róg
502        // obróconej bryły albo blokował się o powietrze przy jej
503        // krawędzi.
504        // `abs()` na wyniku metody, a nie na niej samej: `x.sin().abs()`
505        // daje `f32`, a `x.abs().sin()` już nie.
506        let (s, c) = (yaw.sin().abs(), yaw.cos().abs());
507        w.add(collide::Solid {
508            pos: *pos,
509            width: width * c + depth * s,
510            depth: width * s + depth * c,
511            height: *height,
512        });
513    }
514
515    w
516}

518/// Macierz postaci.
519///
520/// Postać rysowana jest tylko wtedy, gdy kamera pierwszoosobowa
521/// patrzy w dół wystarczająco (patrz [`build_draw_list`]) — wtedy
522/// obrót postaci ma znaczenie, bo widzimy jej ramiona i tors. Przy
523/// patrzeniu prosto przed siebie sylwetki nie widać, więc `yaw`
524/// jest skądś „z definicji" zerowy i nie próbujemy zgadywać, którą
525/// stronę ma bark.
526fn player_matrix(p: Vec3) -> Mat4 {
527    model_matrix(p, 0.0, 0.0, 0.0, Vec3::ONE)
528}
529
530/// Kamera pierwszoosobowa: oko gracza, cel w kierunku patrzenia.
531///
532/// Pozycja oka to `player_pos` podniesiony o [`EYE`] — wysokość
533/// oczu człowieka, a nie „wysokość głowy modelu". Kierunek celu to
534/// `eye + view_dir`, czyli wektor jednostkowy [`look::forward`].
535///
536/// Wcześniej kamera była trzecioosobowa (`+CAM_BACK` w Z, `+CAM_UP`
537/// w Y) i patrzyła na gracza. Przy pierwszej osobie takie ustawienie
538/// oznaczałoby, że gracz patrzy **odwrotnie** niż idzie.
539fn update_camera(game: &mut Game) {
540    if let Ok(mut scene) = game.scene.lock() {
541        let cam = scene.camera_mut();
542        cam.position = game.eye;
543        // Cel wyliczamy z kierunku, nie z kątów — dzięki temu kamera
544        // zawsze jest spójna z tym, co narysował `look::forward`, nawet
545        // jeśli kąty zostaną kiedyś przycięte inaczej.
546        cam.target = game.eye + game.view_dir;
547    }
548}

550/// Drobne przesunięcie rośliny w obrębie działki, żeby nie stały
551/// w idealnym szyku.
552///
553/// Używamy liczb „współczynników złotego ciągu" zamiast `rand`, bo:
554/// * ten sam `(ix, iz)` daje zawsze to samo przesunięcie — rośliny
555///   nie skaczą między klatkami,
556/// * nie potrzebujemy stanu generatora w strukturze gry.
557fn jitter(ix: usize, iz: usize) -> (f32, f32) {
558    let a = 0.55 * (0.618_034 * (ix as f32 * 7.0 + iz as f32 * 3.0).sin());
559    let b = 0.55 * (0.618_034 * (ix as f32 * 3.0 + iz as f32 * 7.0).cos());
560    (a, b)
561}
562
563/// Składa listę obiektów 3D na bieżącą klatkę.
564///
565/// Kolejność ma znaczenie tylko dla czytelności — głębią zajmuje się
566/// depth buffer, a potok nie ma mieszania alfa.
567fn build_draw_list(game: &mut Game) {
568    let mut cmds: Vec<DrawCmd> = Vec::with_capacity(160);
569    let one = model_matrix(Vec3::ZERO, 0.0, 0.0, 0.0, Vec3::ONE);
570    let m = &game.meshes;
571
572    // Trawa i gleba leżą nieruchomo.
573    cmds.push(DrawCmd::new(m.ground, one));
574    cmds.push(DrawCmd::new(m.soil, one));
575
576    // Płot i budynki.
577    for prop in &m.buildings {
578        let mat = model_matrix(prop.pos, prop.yaw, 0.0, 0.0, Vec3::ONE);
579        for (id, mat_id) in &prop.parts {
580            cmds.push(DrawCmd::new(*id, mat).with_material(*mat_id));
581        }
582    }
583
584    // Rośliny: dla każdej zasadzonej działki dobieramy siatkę wg
585    // wzrostu. Trzy siatki zamiast skaliowania jednej, bo skalowanie
586    // zniekształcałoby proporcje łodygi.
587    for iz in 0..farm::FIELD_D {
588        for ix in 0..farm::FIELD_W {
589            let tile = game.farm.get(ix, iz);
590            if tile == farm::Tile::Empty {
591                continue;
592            }
593            let g = tile.growth();
594            let stage = if tile.is_ripe() {
595                2
596            } else if g < 0.5 {
597                0
598            } else {
599                1
600            };
601            let (jx, jz) = jitter(ix, iz);
602            let pos = tile_center(ix, iz) + Vec3::new(jx, 0.0, jz);
603            cmds.push(DrawCmd::new(m.plant[stage], player_matrix(pos)));
604        }
605    }
606
607    // Druty kolizji (`F1`) — na końcu, żeby nie mieszały się z
608    // roślinami przy sortowaniu i były zawsze na wierzchu.
609    push_collider_debug(game, &mut cmds);
610
611    // Gracz.
612    if let Ok(mut scene) = game.scene.lock() {
613        scene.set_commands(cmds);
614    }
615}

617/// Druty kolizji: pomarańczowe bryły świata + błękitny sześcian gracza.
618///
619/// Rysujemy **krawędzie** (patrz [`geometry::collider_box`]), więc nic
620/// nie zasłania i widać, gdzie naprawdę kończy się przeszkoda.
621/// Sześcian gracza pokazuje promień kolizji, który jest niewidoczny —
622/// bez niego gracz wydaje się wchodzić w ścianę „do połowy".
623///
624/// Debug bez GPU: `update` przełącza tryb klawiszem `F1`, a stan
625/// pokazuje `draw_hud`.
626fn push_collider_debug(game: &Game, cmds: &mut Vec<DrawCmd>) {
627    if !game.show_colliders {
628        return;
629    }
630    let m = &game.meshes;
631    for b in game.world.boxes() {
632        // Siatka to kubit jednostkowy, więc skalujemy do pełnej bryły:
633        // `2 * half` daje szerokość 2 m przy `half = 1`.
634        let scale = Mat4::from_scale(b.half * 2.0);
635        cmds.push(DrawCmd::new(
636            m.collider,
637            Mat4::from_translation(b.center) * scale,
638        ));
639    }
640    // Sześcian gracza: dolna krawędź na wysokości stóp, więc przesuwamy
641    // o pół wysokości. Środek `pos` to stopy, a nie środek ciała.
642    let feet = game.player.pos;
643    let half = Vec3::new(
644        collide::PLAYER_RADIUS,
645        collide::PLAYER_HEIGHT * 0.5,
646        collide::PLAYER_RADIUS,
647    );
648    cmds.push(DrawCmd::new(
649        m.player_box,
650        Mat4::from_translation(feet + Vec3::new(0.0, collide::PLAYER_HEIGHT * 0.5, 0.0))
651            * Mat4::from_scale(half * 2.0),
652    ));
653}

655/// Krok symulacji + wejście.
656fn update(ctx: &mut Ctx, game: &mut Game) {
657    let dt = ctx.dt();
658    game.time += dt;
659    if let Ok(mut scene) = game.scene.lock() {
660        scene.advance(dt);
661    }
662
663    if ctx.input.just_pressed(Key::Escape) {
664        std::process::exit(0);
665    }
666    // `F1` przełącza druty kolizji. Bez podpowiedzi w HUD-u tryb
667    // wyglądałby jak błąd renderowania, więc stan pokazujemy w
668    // lewym dolnym rogu (patrz `draw_hud`).
669    if ctx.input.just_pressed(Key::F1) {
670        game.show_colliders = !game.show_colliders;
671    }
672
673    if ctx.input.just_pressed(Key::KeyR) {
674        game.farm = Farm::new();
675        game.won = false;
676        // Reset przywraca dokładnie stan startowy — także pozycję
677        // i kierunek patrzenia. Bez tego gracz stałby tyłem do pola
678        // albo na środku uprawy.
679        game.player
680            .teleport(0.0, farm::FIELD_D as f32 * TILE * 0.5 + START_OFFSET);
681        game.player_pos = game.player.pos;
682        game.yaw = START_YAW;
683        game.pitch = START_PITCH;
684        game.walking = 0.0;
685        game.walk_blend = 0.0;
686    }
687
688    // --- obrót kamery myszą ---
689    //
690    // `mouse_delta()` to ruch względny (nie pozycja kursora), więc
691    // obrót nie skacze, gdy kursor znalazł się w rogu okna. Silnik
692    // limituje deltę do 200 px, co chroni przed skokiem po powrocie
693    // z alt-tab.
694    //
695    // Znak `-=` dla `yaw`: przesunięcie myszy w prawo ma obrócić kamerę
696    // w prawo, czyli zmniejszyć kąt, bo dodatni `yaw` skręca w lewo
697    // (konwencja matematyczna).
698    let m = ctx.input.mouse_delta();
699    if m.x != 0.0 || m.y != 0.0 {
700        game.yaw -= m.x * LOOK_SENS;
701        game.pitch = clamp_pitch(game.pitch - m.y * LOOK_SENS);
702    }
703    // `yaw` rośnie w nieskończoność — przy 100 obrotach myszy
704    // wychodzi już tysiąc radianów i `sin_cos` traci precyzję.
705    // Składamy do `-PI..PI`.
706    if game.yaw > std::f32::consts::PI || game.yaw < -std::f32::consts::PI {
707        game.yaw = game.yaw.rem_euclid(std::f32::consts::TAU);
708        if game.yaw > std::f32::consts::PI {
709            game.yaw -= std::f32::consts::TAU;
710        }
711    }
712
713    // --- ruch gracza ---
714    //
715    // Dwie składowe wejścia: `W/S` na osi „do przodu / do tyłu" względem
716    // kierunku patrzenia i `A/D` na „w lewo / w prawo". Oś pionowa jest
717    // zawsze `0` — gracz nie lata i nie chodzi po ścianach.
718    //
719    // Wszystko liczy [`look::move_dir`], który **normalizuje** wynik.
720    // Bez tego `W` + `D` dawałoby √2 × WALK_SPEED, czyli chód po skosie
721    // byłby szybszy niż po prostej.
722    let fwd_in = ctx.input.axis(Key::KeyS, Key::KeyW);
723    let strafe_in = ctx.input.axis(Key::KeyA, Key::KeyD);
724    // `move_dir` bierze `yaw`, więc obrót myszą obraca też kierunek
725    // chodzenia — to właśnie ta własność, o którą prosiłeś.
726    let dir = move_dir(game.yaw, fwd_in, strafe_in);
727    // Ruch: `Player::step` liczy chód w kierunku `dir`, grawitację
728    // i skok, a `collide::World` rozwiązuje wejścia w budynki i płot.
729    //
730    // Krok wykonujemy **zawsze**, także gdy `dir` jest zerowe:
731    // inaczej gracz zastyłby w powietrzu po wypchnięciu ze skoku
732    // i nigdy by nie wylądował.
733    // Skakamy **wciśnięciem** Spacji (`just_pressed`), nie jej
734    // trzymaniem — inaczej gracz wskakiwałby w miejscu przy każdym
735    // lądowaniu, bo `pressed` jest prawdziwe przez cały czas wciśnięcia.
736    let jump = ctx.input.just_pressed(Key::Space);
737    game.player.step(
738        dt,
739        dir,
740        jump,
741        &game.world,
742        collide::PLAYER_RADIUS,
743        collide::PLAYER_HEIGHT,
744        &game.player_cfg,
745    );
746
747    // Ograniczenie do kwadratu świata: bez niego gracz wychodzi
748    // poza teren i kamera pokazuje pustą mgłę. `Player::step` pilnuje
749    // przeszkód, więc granicę mapy trzymamy osobno.
750    let limit = game.world.limit;
751    game.player.pos.x = game.player.pos.x.clamp(-limit, limit);
752    game.player.pos.z = game.player.pos.z.clamp(-limit, limit);
753
754    if dir.length_squared() > 1e-6 {
755        // Faza rośnie o **przebytą** drogę (ok. 2 kroki na metr), a nie
756        // o czas — dzięki temu rytm kołysania nie zależy od `dt`
757        // i przy 30 FPS wygląda tak samo jak przy 144.
758        let step = dir * (game.player_cfg.walk_speed * dt);
759        game.walking += step.length() * 6.0;
760    }
761    // `walk_blend` wygładza start i stop. `1 - exp(-k*dt)` to
762    // wygładzanie niezależne od częstotliwości klatek: przy małym `dt`
763    // zbliża się do `k*dt`, przy dużym nie "przeskakuje" do 1.
764    // Stać w miejscu ma wygaszać szybciej niż ruszać.
765    let target = f32::from(dir.length_squared() > 1e-6);
766    let k = if target > game.walk_blend { 11.0 } else { 16.0 };
767    game.walk_blend += (target - game.walk_blend) * (1.0 - (-k * dt).exp());
768
769    // Oko trzymamy w jednym miejscu, żeby rysowanie nie musiało
770    // zgadywać, co widzi gracz. `eye` unosi się do wysokości oczu,
771    // `view_dir` to pełny kierunek z pitchem.
772    game.view_dir = forward(game.yaw, game.pitch);
773
774    // Pozycja stóp jest przycięta do `0..=MAX_FOOT_Y`.
775    //
776    // Dolna granica to podłoga: bez niej oko schodziłoby pod ziemię
777    // przy pierwszym kroku w dół. Górna to **zabezpieczenie**, nie
778    // mechanika skoku — `Player::step` sam pilnuje grawitacji, a skok
779    // osiąga ~1,4 m przy suficie 6 m. Bez tej górnej granicy każdy
780    // błąd w fizyce objawiałby się jako „kamera ucieka w górę" zamiast
781    // zrzutu błędu, a taki objaw jest bardzo trudny do zlokalizowania.
782    let feet = game.player.pos;
783    let feet = Vec3::new(feet.x, feet.y.clamp(0.0, MAX_FOOT_Y), feet.z);
784    game.player.pos.y = feet.y;
785    game.player_pos = feet;
786
787    // Lekkie kołysanie przy chodzeniu. Przesuwamy oko wzdłuż osi
788    // „w górę ekranu" (`look::up`), a nie wzdłuż globalnego Y — dzięki
789    // temu głowa chodzi w górę i w dół także wtedy, gdy gracz patrzy
790    // pod nogi, i kołysanie nie zmienia kierunku patrzenia.
791    let bob = game.walking.sin() * 0.045 * game.walk_blend;
792    game.eye = feet + Vec3::new(0.0, EYE, 0.0) + up(game.yaw, game.pitch) * bob;
793
794    // --- interakcja z polem ---
795    //
796    // W pierwszej osobie gracz nie stoi na działce, tylko obok niej,
797    // więc `E` musi działać na tę działkę, **którą wskazuje**, nie
798    // tę, na której stoi. Bierzemy więc punkt przed graczem.
799    if ctx.input.just_pressed(Key::KeyE) {
800        let target = game.eye + game.view_dir * farm::REACH;
801        match tile_at(target) {
802            Some((ix, iz)) => {
803                game.farm.interact(ix, iz);
804            }
805            None => {
806                game.farm.message = "Nie ma tu pola".into();
807                game.farm.message_time = 1.2;
808            }
809        }
810    }
811
812    // Rośliny dojrzewają niezależnie od tego, czy gracz patrzy.
813    game.farm.step(dt);
814    if !game.won && game.farm.is_won() {
815        game.won = true;
816    }
817
818    update_camera(game);
819    build_draw_list(game);
820}

822/// HUD: stan pola, liczniki i podpowiedź sterowania.
823///
824/// Rysujemy w przestrzeni EKRANU (renderer 2D), tak jak w `uran-tanks`
825/// i `junak-rider`, i z tej samego powodu: 2D nie przechodzi przez
826/// post-processing 3D, więc tekst nie zostaje zniekształcony przez ACES
827/// ani rozmyty przez bloom.
828fn draw_hud(ctx: &mut Ctx, game: &Game) {
829    let size = ctx.window.logical_size();
830    let (empty, growing, ready) = game.farm.counts();
831
832    // --- pasek stanu pola (u góry ekranu) ---
833    ctx.gfx
834        .screen_space()
835        .layer(100)
836        .color(Color::from_hex(0x1B2430))
837        .draw_rect(Rect::new(
838            Vec2::new(size.x * 0.5 - 190.0, 14.0),
839            Vec2::new(size.x * 0.5 + 190.0, 44.0),
840        ));
841    // Proporcja paska to `ready / wszystkie`: to stan, który gracz
842    // realnie zmienia — sadzenie samo w sobie nie kończy pracy.
843    let total: f32 = (empty + growing + ready).max(1) as f32;
844    let t = ready as f32 / total;
845    ctx.gfx
846        .color(Color::from_hex(0xE8C547))
847        .draw_rect(Rect::new(
848            Vec2::new(size.x * 0.5 - 188.0, 38.0),
849            Vec2::new(size.x * 0.5 - 188.0 + 376.0 * t, 42.0),
850        ));
851
852    // --- celownik ---
853    //
854    // Kursor systemowy jest ukryty (`cursor_visible(false)` na oknie),
855    // bo w pierwszej osobie mysz steruje kamerą, a nie wskazuje. Bez
856    // własnego celownika gracz traci orientację, gdzie patrzy.
857    //
858    // Cztery krótkie „ramiona" zamiast pełnego krzyża: nie zasłaniają
859    // tego, co jest pod nimi, a mimo to jednoznacznie wskazują środek
860    // ekranu. Każde ramię rysujemy dwa razy — ciemnym prostokątem
861    // o 1 px większym pod spodem, potem białym — żeby celownik
862    // był widoczny i na jasnym trawniku, i na ciemnym budynku.
863    {
864        let c = Vec2::new(size.x * 0.5, size.y * 0.5);
865        const ARM: f32 = 7.0;
866        const GAP: f32 = 3.0;
867        let arms = [
868            // poziome: lewe i prawe
869            (Vec2::new(c.x - GAP - ARM, c.y), Vec2::new(c.x - GAP, c.y)),
870            (Vec2::new(c.x + GAP, c.y), Vec2::new(c.x + GAP + ARM, c.y)),
871            // pionowe: górne i dolne
872            (Vec2::new(c.x, c.y - GAP - ARM), Vec2::new(c.x, c.y - GAP)),
873            (Vec2::new(c.x, c.y + GAP), Vec2::new(c.x, c.y + GAP + ARM)),
874        ];
875        for (a, b) in arms {
876            let outline = Rect::new(
877                Vec2::new(a.x - 1.0, a.y - 1.0),
878                Vec2::new(b.x + 1.0, b.y + 1.0),
879            );
880            ctx.gfx.color(Color::from_hex(0x1B2430)).draw_rect(outline);
881            ctx.gfx
882                .color(Color::from_hex(0xF4F7FB))
883                .draw_rect(Rect::new(a, b));
884        }
885    }
886
887    // Wskaźnik działki przed graczem: pokazuje, co stanie się po `E`.
888    // Wskazujemy tam, gdzie patrzymy (`eye + view_dir * REACH`), bo w
889    // pierwszej osobie gracz nie stoi na polu, tylko obok niego.
890    if let Some((ix, iz)) = tile_at(game.eye + game.view_dir * farm::REACH) {
891        let (hint, c) = match game.farm.get(ix, iz) {
892            farm::Tile::Empty => ("[E] Zasadz", Color::from_hex(0x9BE07A)),
893            farm::Tile::Ready => ("[E] Zbierz", Color::from_hex(0xF2C14E)),
894            farm::Tile::Growing { .. } => ("Rosnie...", Color::from_hex(0xB9C4D4)),
895        };
896        if let Some(font) = game.font {
897            ctx.gfx.color(c).draw_text(
898                font,
899                hint,
900                Vec2::new(size.x * 0.5, size.y * 0.5 - 90.0),
901                26.0,
902                TextAlign::Center,
903            );
904        }
905    }
906
907    // --- komunikat akcji ---
908    if game.farm.message_time > 0.0 {
909        if let Some(font) = game.font {
910            ctx.gfx.color(Color::from_hex(0xF4F7FB)).draw_text(
911                font,
912                &game.farm.message,
913                Vec2::new(size.x * 0.5, size.y * 0.5 - 130.0),
914                22.0,
915                TextAlign::Center,
916            );
917        }
918    }
919
920    let Some(font) = game.font else { return };
921
922    // --- liczniki (lewy górny róg) ---
923    //
924    // UWAGA: tylko znaki ASCII. Polskie diakrytyczne wywołują błąd
925    // atlasu glifów w `uran-render` przy przepełnieniu (patrz
926    // `draw_hud` w uran-tanks), więc HUD zostaje czytelny bez nich.
927    ctx.gfx.color(Color::from_hex(0xE8EEF8)).draw_text(
928        font,
929        &format!("Zebrano: {}", game.farm.harvested),
930        Vec2::new(24.0, 24.0),
931        24.0,
932        TextAlign::Left,
933    );
934    ctx.gfx.color(Color::from_hex(0xB9C4D4)).draw_text(
935        font,
936        &format!("Puste {empty}  Rosnie {growing}  Gotowe {ready}"),
937        Vec2::new(24.0, 54.0),
938        18.0,
939        TextAlign::Left,
940    );
941
942    // --- sterowanie (lewy dolny róg) ---
943    ctx.gfx.color(Color::from_hex(0x8FA6C4)).draw_text(
944        font,
945        "WSAD chodzenie   mysz obrot   spacja skok   E sadz/zbierz   R nowa farma   F1 kolizje",
946        Vec2::new(24.0, size.y - 32.0),
947        16.0,
948        TextAlign::Left,
949    );
950
951    // --- znacznik trybu debug (prawy dolny róg) ---
952    //
953    // Bez tego druty wyglądałyby jak błąd renderowania, a nie jak
954    // włączone narzędzie. Kolor pasuje do pomarańczu kolidatorów.
955    if game.show_colliders {
956        ctx.gfx.color(Color::from_hex(0xFF7333)).draw_text(
957            font,
958            &format!("KOLIZJE ON  ({} brył)", game.world.boxes().len()),
959            Vec2::new(size.x - 24.0, size.y - 32.0),
960            16.0,
961            TextAlign::Right,
962        );
963    }
964
965    // --- wygrana ---
966    if game.won {
967        ctx.gfx
968            .screen_space()
969            .layer(101)
970            .color(Color::from_hex(0x141A24))
971            .draw_rect(Rect::new(
972                Vec2::new(0.0, size.y * 0.5 - 60.0),
973                Vec2::new(size.x, size.y * 0.5 + 60.0),
974            ));
975        ctx.gfx.color(Color::from_hex(0xF2C14E)).draw_text(
976            font,
977            "POLE DOJRZALE - ZBIERASZ!",
978            Vec2::new(size.x * 0.5, size.y * 0.5 - 12.0),
979            40.0,
980            TextAlign::Center,
981        );
982        ctx.gfx.color(Color::from_hex(0xE8EEF8)).draw_text(
983            font,
984            "R - nowa farma",
985            Vec2::new(size.x * 0.5, size.y * 0.5 + 26.0),
986            20.0,
987            TextAlign::Center,
988        );
989    }
990}

992/// Proxy renderera 3D podpinany do renderera 2D (patrz `uran-render`).
993struct SceneProxy(Arc<Mutex<Renderer3d>>);
994
995impl uran_render::Scene3d for SceneProxy {
996    fn draw(&mut self, target: &uran_render::Scene3dTarget<'_>) {
997        if let Ok(mut scene) = self.0.lock() {
998            uran_render::Scene3d::draw(&mut *scene, target);
999        }
1000    }
1001
1002    fn resize(&mut self, w: u32, h: u32) {
1003        if let Ok(mut scene) = self.0.lock() {
1004            uran_render::Scene3d::resize(&mut *scene, w, h);
1005        }
1006    }
1007}

1009/// Punkt wejścia.
1010fn main() {
1011    let game = Arc::new(Mutex::new(None::<Game>));
1012
1013    let setup_system = {
1014        let game = Arc::clone(&game);
1015        move |ctx: &mut Ctx| {
1016            let mut guard = game.lock().expect("lock gry");
1017            if guard.is_some() {
1018                return;
1019            }
1020            // JEDEN `Device` obsługuje oba renderery, a `samples` MUSI
1021            // się zgadzać — inaczej hdr-owa tekstura post-processingu
1022            // nie przejdzie walidacji wgpu.
1023            let built = {
1024                let r = ctx.renderer.as_ref().expect("renderer gotowy");
1025                Game::new(r.gpu(), r.format(), r.samples())
1026            };
1027            *guard = Some(built);
1028            let g = guard.as_mut().expect("właśnie utworzone");
1029            g.font = ctx.assets.load_font("assets/fonts/UranSans.ttf").ok();
1030            let proxy = Box::new(SceneProxy(Arc::clone(&g.scene)));
1031            if let Some(r) = ctx.renderer.as_mut() {
1032                r.set_scene3d(proxy);
1033            }
1034        }
1035    };
1036
1037    let update_system = {
1038        let game = Arc::clone(&game);
1039        move |ctx: &mut Ctx| {
1040            let mut guard = game.lock().expect("lock gry");
1041            if let Some(g) = guard.as_mut() {
1042                update(ctx, g);
1043            }
1044        }
1045    };
1046
1047    let draw_system = {
1048        let game = Arc::clone(&game);
1049        move |ctx: &mut Ctx| {
1050            let guard = game.lock().expect("lock gry");
1051            if let Some(g) = guard.as_ref() {
1052                draw_hud(ctx, g);
1053            }
1054        }
1055    };
1056
1057    App::new()
1058        // Zrzut po 40 klatkach: tyle trzeba, żeby world mgły i cieni
1059        // się ustabilizowały. Używany w testach regresji wizualnej
1060        // (`FARM_SHOT=1 cargo run -p farm-simulator`).
1061        .screenshot(
1062            std::env::var("FARM_SHOT")
1063                .ok()
1064                .map(std::path::PathBuf::from),
1065            40,
1066        )
1067        .window(
1068            windowed(1280, 720)
1069                .title("Farm Simulator")
1070                .background(0x94BDE6)
1071                .vsync(true)
1072                .samples(1)
1073                // Kursor ukryty: w pierwszej osobie mysz obraca kamerę,
1074                // a wskazywanie w grze robi celownik rysowany w HUD.
1075                .cursor_visible(false)
1076                // ...i zablokowany w centrum okna. Bez `Locked` kursor
1077                // ucieka do rogów — przy 75° FOV i czułości 0,0035
1078                // kilka obrotów głowy wystarczy, żeby doszedł do
1079                // ściany okna i sterowanie umarło.
1080                .cursor_locked(true)
1081                // Pełny ekran, ale NIE przy zrzucie: `FARM_SHOT` potrzebuje
1082                // stałych 1280x720 do porównania piksel po pikselu, a
1083                // fullscreen wziąłby rozmiar monitora. Dlatego zrzut
1084                // świadomie rezygnuje z trybu pełnoekranowego.
1085                .fullscreen(std::env::var_os("FARM_SHOT").is_none()),
1086        )
1087        .add_startup_system(setup_system)
1088        .add_system(update_system)
1089        .add_system(draw_system)
1090        .run();
1091}
