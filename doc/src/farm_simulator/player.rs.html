1//! Chód, grawitacja i skok gracza.
2//!
3//! Świadomie **bez silnika fizyki** (jak [`junak-rider`]): ruch jest
4//! w pełni analityczny, a świat to płaska ziemia z kilkunastoma
5//! prostopadłościanami. Rapier byłby dwa rzędy wielkości cięższy i
6//! wniósłby dokładnie ten sam błąd, tylko trudniej widoczny.
7//!
8//! Wszystko w metrach i sekundach, zgodnie z resztą silnika.
9
10use uran_math::Vec3;
11
12use crate::collide::World;
13
14/// Parametry gracza — jeden zestaw, bo to jeden model.
15#[derive(Debug, Clone, Copy)]
16pub struct PlayerConfig {
17    /// Prędkość chodu (m/s).
18    pub walk_speed: f32,
19    /// Prędkość skoku w chwili startu (m/s, w górę).
20    pub jump_speed: f32,
21    /// Grawitacja (m/s²). Ujemna.
22    pub gravity: f32,
23    /// Najwyższa prędkość spadania (m/s) — ogranicza tunelowanie
24    /// przy krótkich klatkach.
25    pub terminal_velocity: f32,
26}
27
28impl Default for PlayerConfig {
29    fn default() -> Self {
30        Self {
31            walk_speed: 5.0,
32            // 7,4 m/s przy grawitacji 19,6 daje szczyt `v²/2g` ≈ 1,40 m.
33            // To najmniej, co przeskakuje płot (1,1 m) z zapasem, a
34            // jednocześnie tyle, ile potrzeba, żeby skok nie był
35            // „balonem" unoszącym gracza na półtora metra.
36            jump_speed: 7.4,
37            gravity: -19.6,
38            terminal_velocity: 45.0,
39        }
40    }
41}
42
43/// Stan gracza: pozycja dna ciała i prędkość pionowa.
44#[derive(Debug, Clone, Copy)]
45pub struct Player {
46    /// **Dno** gracza (stopy). Nie środek — patrz [`crate::collide`].
47    pub pos: Vec3,
48    /// Prędkość pionowa (m/s). Ujemna = spadanie.
49    pub vy: f32,
50    /// Czy gracz stoi na ziemi.
51    pub grounded: bool,
52}
53
54impl Default for Player {
55    fn default() -> Self {
56        Self {
57            pos: Vec3::ZERO,
58            vy: 0.0,
59            grounded: true,
60        }
61    }
62}
63
64impl Player {
65    /// Krok symulacji: ruch poziomy ze świata, grawitacja i skok.
66    ///
67    /// `wish_dir` to kierunek ruchu **bez** składowej Y (liczony przez
68    /// [`crate::look::move_dir`], który normalizuje wynik), a `world`
69    /// rozwiązuje wejścia w przeszkody.
70    pub fn step(
71        &mut self,
72        dt: f32,
73        wish_dir: Vec3,
74        want_jump: bool,
75        world: &World,
76        radius: f32,
77        height: f32,
78        cfg: &PlayerConfig,
79    ) {
80        // `dt` z debuggera albo przeciągnięte okno potrafi być ogromne.
81        // Bez tego skok wyrzuciłby gracza w kosmos, a pozycja dostałaby
82        // NaN, co psuje renderer na zawsze (macierze z NaN nie wracają
83        // do poprawnego stanu).
84        let dt = dt.clamp(0.0, 0.1);
85
86        // --- skok ---
87        //
88        // `grounded` z **poprzedniej** klatki: dzięki temu skok działa
89        // także wtedy, gdy gracz właśnie wylądował (w klatce lądowania
90        // `pos.y` jest już 0, ale grawitacja jeszcze nie zdążyła przyjąć
91        // ujemnej prędkości).
92        if want_jump && self.grounded {
93            self.vy = cfg.jump_speed;
94            self.grounded = false;
95        }
96
97        // --- grawitacja ---
98        if !self.grounded {
99            self.vy += cfg.gravity * dt;
100            if self.vy < -cfg.terminal_velocity {
101                self.vy = -cfg.terminal_velocity;
102            }
103        }
104
105        // --- ruch ---
106        let delta = wish_dir * (cfg.walk_speed * dt) + Vec3::new(0.0, self.vy * dt, 0.0);
107        world.move_player(&mut self.pos, delta, radius, height);
108
109        // --- lądowanie ---
110        //
111        // Podłoga jest wszędzie na `y = 0` (płaski teren), więc
112        // lądowanie sprowadza się do testu wysokości.
113        //
114        // Warunek to `pos.y <= 0.0`, a nie `pos.y <= ground_epsilon`.
115        // Tolerancja wygląda na sensowną (gracz stojący „nad" ziemią
116        // powinien być na ziemi), ale **zjada skok**: pierwsza klatka
117        // po wciśnięciu unosi gracza tylko o `v*dt` ≈ 0,09 m, więc
118        // przy tolerancji 0,35 m warunek byłby prawdziwy od razu i
119        // `vy` zerowałby się w tej samej klatce, w której skok się
120        // zaczął. Gracz nigdy by nie oderwał się od ziemi.
121        if self.pos.y <= 0.0 {
122            self.pos.y = 0.0;
123            self.vy = 0.0;
124            self.grounded = true;
125        } else {
126            self.grounded = false;
127        }
128    }
129
130    /// Resetuje gracza do stanu stojącego w danym miejscu.
131    pub fn teleport(&mut self, x: f32, z: f32) {
132        self.pos = Vec3::new(x, 0.0, z);
133        self.vy = 0.0;
134        self.grounded = true;
135    }
136}

143    const DT: f32 = 1.0 / 60.0;

145    /// Świat pusty — do testów samej grawitacji i skoku.
146    fn open_world() -> World {
147        World::new(50.0)
148    }
149
150    /// Wykonuje `n` klatek chodu w kierunku `dir`.
151    fn walk(p: &mut Player, n: usize, dir: Vec3, w: &World, c: &PlayerConfig) {
152        for _ in 0..n {
153            p.step(DT, dir, false, w, PLAYER_RADIUS, PLAYER_HEIGHT, c);
154        }
155    }

157    #[test]
158    fn starts_on_the_ground() {
159        let p = Player::default();
160        assert!(p.grounded);
161        assert_eq!(p.pos.y, 0.0);
162        assert_eq!(p.vy, 0.0);
163    }

165    #[test]
166    fn walking_moves_along_the_requested_direction() {
167        let w = open_world();
168        let c = PlayerConfig::default();
169        let mut p = Player::default();
170        walk(&mut p, 60, Vec3::new(1.0, 0.0, 0.0), &w, &c);
171        // 1 s chodu z prędkością 5 m/s = ~5 m.
172        assert!((p.pos.x - 5.0).abs() < 0.1, "poszedł {} m", p.pos.x);
173        assert!(p.pos.z.abs() < 1e-3, "uciekł w bok: {:?}", p.pos);
174    }

176    #[test]
177    fn jump_rises_then_lands_back_grounded() {
178        let w = open_world();
179        let c = PlayerConfig::default();
180        let mut p = Player::default();
181
182        p.step(DT, Vec3::ZERO, true, &w, PLAYER_RADIUS, PLAYER_HEIGHT, &c);
183        assert!(p.vy > 0.0, "skok nie nadał prędkości w górę: {}", p.vy);
184        assert!(!p.grounded, "w trakcie skoku gracz jest w powietrzu");
185
186        // Szczyt: prędkość przechodzi przez zero.
187        let mut peak = 0.0f32;
188        for _ in 0..200 {
189            p.step(DT, Vec3::ZERO, false, &w, PLAYER_RADIUS, PLAYER_HEIGHT, &c);
190            peak = peak.max(p.pos.y);
191        }
192        assert!(peak > 0.8, "szczyt za niski: {peak}");
193        assert!(peak < 2.0, "skok za wysoki: {peak}");
194        assert!(p.grounded, "nie wylądował z powrotem");
195        assert_eq!(p.pos.y, 0.0);
196        assert_eq!(p.vy, 0.0, "po lądowaniu prędkość musi się wyzerować");
197    }
