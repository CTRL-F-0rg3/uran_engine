2//!
3//! Model 3D wczytujemy z pliku `.obj` prosto z Blendera (patrz

22/// Ścieżka modelu, szukana w kilku miejscach.
23///
24/// Ścieżka względna sama w sobie działa tylko z katalogu `junak-rider/`,
25/// a `cargo run -p junak-rider` uruchamia grę z katalogu repozytorium —
26/// bez fallbacku model by się nie wczytał i dostalibyśmy klocek
27/// awaryjny zamiast motocykla.
28const MODEL_FILE: &str = "junak m10.obj";
29
30/// Znajduje plik modelu: najpierw obok programu, potem w katalogu crate'a.
31fn find_model() -> Option<std::path::PathBuf> {
32    let mut tried = Vec::new();
33
34    // 1. ścieżka podana jawnie
35    if let Ok(p) = std::env::var("JUNAK_MODEL") {
36        let p = std::path::PathBuf::from(p);
37        if p.is_file() {
38            return Some(p);
39        }
40        tried.push(p);
41    }
42
43    // 2. zwykła ścieżka względna (uruchomienie z katalogu crate'a)
44    let rel = std::path::PathBuf::from("assets").join(MODEL_FILE);
45    if rel.is_file() {
46        return Some(rel);
47    }
48    tried.push(rel);
49
50    // 3. katalog źródłowy crate'a — niezawodne przy `cargo run` z roota
51    let manifest = std::path::PathBuf::from(env!("CARGO_MANIFEST_DIR"))
52        .join("assets")
53        .join(MODEL_FILE);
54    if manifest.is_file() {
55        return Some(manifest);
56    }
57    tried.push(manifest);
58
59    eprintln!("⚠️  nie znaleziono modelu, próbowałem:");
60    for p in &tried {
61        eprintln!("      {}", p.display());
62    }
63    None
64}

66/// Model po imporcie w dzielnikach osi to 2,49 × 1,20 × 1,27 m, czyli
67/// realny motocykl. Skalujemy do wysokości 1,2 m, żeby niezależnie od
68/// skali eksportu z Blendera model zawsze miał sensowne wymiary.
69const BIKE_HEIGHT: f32 = 1.2;
70
71/// Przesunięcie modelu względem początku siatki.
72///
73/// Import stawia model NA ziemi (min Y = 0) i wyśrodkowuje go w XZ,
74/// ale oś obrotu silnika to środek kadła, a nie środek całej bryły —
75/// dlatego podnościmy o połowę wysokości siedzenia i cofnęciem w osi Z.
76const BIKE_LIFT: f32 = 0.0;
77
78/// Kamera: jak daleko za motocyklem i jak wysoko.
79///
80/// Wysokość celowo tuż NAD siedzeniem (ok. 1,5 m), a nie wysoko
81/// nad głową. Kamera patrząca z góry zgniata motocykl do płaskiej
82/// plamy — nie widać ani kierownicy, ani nadwozia, ani tego, że
83/// przechyla się w zakręcie.
84const CAM_BACK: f32 = 4.3;
85const CAM_UP: f32 = 1.52;
86/// Jak bardzo kamera pochyla się podczas skrętu (radiany).
87const CAM_SIDE: f32 = 1.1;
88
89/// Identyfikatory siatek w rejestrze renderera.
90struct Meshes {
91    /// (siatka, materiał) dla każdej części modelu
92    bike: Vec<(MeshId, MaterialId)>,
93    asphalt: MeshId,
94    verge: MeshId,
95    paint: MeshId,
96}

98struct Game {
99    scene: Arc<Mutex<Renderer3d>>,
100    meshes: Meshes,
101    bike: Bike,
102    cfg: BikeConfig,
103    /// Przebyta droga (m) — do HUD, nigdy nie rośnie w nieskończoność
104    /// w jednej sesji, więc `f32` wystarcza.
105    distance: f32,
106    top_speed: f32,
107    time: f32,
108    font: Option<Font>,
109}

111impl Game {
112    fn new(gpu: &uran_render::GpuContext, format: wgpu::TextureFormat, samples: u32) -> Self {
113        let mut scene = Renderer3d::new(gpu, format, samples);
114        // Niskie słońce długo rysuje cień motocykla i podkreśla
115        // kształt nadwozia. Bez cienia bryła wygląda jak płaska naklejka.
116        scene.set_clear_color([0.45, 0.60, 0.82, 1.0]);
117        {
118            let l = scene.lighting_mut();
119            l.light_dir = Vec3::new(0.35, 0.42, 0.84);
120            l.light_color = [1.0, 0.95, 0.86];
121            // 0.34/0.44/0.62 → 0.13/0.16/0.21 (jak w `farm-simulator`
122            // i w globalnym `Lighting::default()`).
123            //
124            // Stara wartość była 3-4× większa w sensie proporcji do
125            // słońca 4.0, więc zacierała kontrast: strona na słońcu i
126            // cień wyglądały jednak jasno, tyle że w innym odcieniu.
127            // Motocykl wtedy czytał się jak płaska naklejka, mimo że
128            // cień rzutowany był poprawnie.
129            l.ambient = [0.13, 0.16, 0.21];
130        }
131        {
132            let cam = scene.camera_mut();
133            cam.fov_y = std::f32::consts::FRAC_PI_2 * 0.82;
134            cam.near = 0.15;
135            cam.far = 900.0;
136        }
137
138        // --- model z pliku .obj
139        //
140        // Konwencja osi jest jawna, bo ten plik jest Y-up. Dowód
141        // z samego pliku: część `wheel_f` to cztery łuki przy
142        // `z = ±0.227` rozciągnięte w x i y — czyli dysk w
143        // płaszczyźnie XY, cienki w Z. Oś koła to Z, a oś koła
144        // motocykla jest zawsze pozioma, więc Z to SZEROKOŚĆ.
145        // Skoro koło przednie leży na skrajnym ujemnym X, to X jest
146        // długością, a pozostała oś Y — wysokością. Przy Z-up
147        // (`AxisUp::Z`) bryła wylądowałaby na boku.
148        let mut bike_parts = Vec::new();
149        let loaded = find_model().and_then(|p| {
150            match import::obj::load_from_file_with_axes(&p, import::AxisUp::Y) {
151                Ok(m) => Some(m),
152                Err(e) => {
153                    eprintln!("⚠️  nie udało się wczytać `{}`: {e}", p.display());
154                    None
155                }
156            }
157        });
158
159        match loaded {
160            Some(mut model) => {
161                model.scale_to_height(BIKE_HEIGHT);
162                let s = model.bounds_size();
163                println!(
164                    "🛵  model: {} części, {:.2} × {:.2} × {:.2} m",
165                    model.parts.len(),
166                    s.x,
167                    s.y,
168                    s.z
169                );
170                for part in &model.parts {
171                    // Material z pliku `.mtl` (albo zapasowy, gdy pliku
172                    // nie ma) rejestrujemy w banku i wiążemy z siatką.
173                    // Dzięki temu dodanie tekstur to PODMIANA plików
174                    // w katalogu `assets/`, a nie zmiana w kodzie.
175                    let mat = scene.add_material(gpu, &part.material);
176                    let id = scene.add_mesh(gpu, &part.mesh, &part.name);
177                    bike_parts.push((id, mat));
178                }
179            }
180            None => {
181                // Gra ma działać nawet bez pliku — pokazujemy wtedy
182                // czerwony klocek, żeby było widać, że czegoś brakuje,
183                // zamiast dostawać pusty kadr.
184                let mut m = uran_render3d::Mesh::new();
185                m.tri(
186                    uran_render3d::Vertex::new(
187                        Vec3::new(-0.4, 0.0, -0.4),
188                        Vec3::X,
189                        [0.9, 0.2, 0.2],
190                    ),
191                    uran_render3d::Vertex::new(Vec3::new(0.4, 0.0, -0.4), Vec3::X, [0.9, 0.2, 0.2]),
192                    uran_render3d::Vertex::new(Vec3::new(0.0, 1.2, -0.4), Vec3::X, [0.9, 0.2, 0.2]),
193                );
194                m.tri(
195                    uran_render3d::Vertex::new(Vec3::new(-0.4, 0.0, 0.4), Vec3::X, [0.9, 0.2, 0.2]),
196                    uran_render3d::Vertex::new(Vec3::new(0.4, 0.0, 0.4), Vec3::X, [0.9, 0.2, 0.2]),
197                    uran_render3d::Vertex::new(Vec3::new(0.0, 1.2, 0.4), Vec3::X, [0.9, 0.2, 0.2]),
198                );
199                let id = scene.add_mesh(gpu, &m, "fallback tetra");
200                bike_parts.push((id, MaterialId(0)));
201            }
202        }
203
204        // --- droga
205        let r = road::build_road();
206        let asphalt = scene.add_mesh(gpu, &r.asphalt, "asfalt");
207        let verge = scene.add_mesh(gpu, &r.verge, "pobocze");
208        let paint = scene.add_mesh(gpu, &r.paint, "oznakowanie");
209
210        Self {
211            scene: Arc::new(Mutex::new(scene)),
212            meshes: Meshes {
213                bike: bike_parts,
214                asphalt,
215                verge,
216                paint,
217            },
218            bike: Bike::default(),
219            cfg: BikeConfig::default(),
220            distance: 0.0,
221            top_speed: 0.0,
222            time: 0.0,
223            font: None,
224        }
225    }
226}

228/// Obrot modelu względem osi obrotu silnika.
229///
230/// Niezależnie od konwencji osi w pliku (patrz `AxisUp` w importerze)
231/// przód motocykla w leży w `-X`, a silnik jedzie po `+Z` przy
232/// `yaw = 0`. Obrót o +90° wokół Y przenosi `-X` na `+Z`.
233///
234/// Znak jest tu nieoczywisty, bo zależy od tego, czy `Quat::from_rotation_y`
235/// obraca zgodnie z prawoskrętną regułą (czyli tak, jak wygląda obrót
236/// patrząc z góry na dół osi). Gdyby ta stała była zła, motocykl jechałby
237/// bokiem do drogi — stąd osobna nazwa i test `bike_matrix_points_the_nose_forward`.
238const BIKE_YAW_OFFSET: f32 = std::f32::consts::FRAC_PI_2;
239
240/// Macierz modelu motocykla w świecie.
241///
242/// Przechył to `roll` wokół osi Z lokalnej (osi kierunku jazdy), więc
243/// motocykl kładzie się w bok, a nie pochyla się do przodu.
244fn bike_matrix(b: &Bike) -> Mat4 {
245    model_matrix(
246        b.pos + Vec3::new(0.0, BIKE_LIFT, 0.0),
247        b.yaw + BIKE_YAW_OFFSET,
248        0.0,
249        b.lean,
250        Vec3::ONE,
251    )
252}

254/// Kamera podąża za motocyklem.
255fn update_camera(game: &mut Game) {
256    let b = &game.bike;
257    let f = b.forward();
258    // Kamerę trzymamy ZA motocykłem, ale nie na jego osi: przy pełnym
259    // skręcie kamera zjechałaby z drogi. Dlatego liczymy ją względem
260    // kierunku jazdy, a nie względem pozycji bezwzględnej.
261    let side = b.right() * (b.steer / game.cfg.max_steer) * CAM_SIDE;
262    let eye = b.pos - f * CAM_BACK + side + Vec3::new(0.0, CAM_UP, 0.0);
263    // Celujemy przed motocykl, nie w niego — inaczej kamera patrzyłaby
264    // w ogon i drogi by nie było widać.
265    let look = b.pos + f * 14.0 + Vec3::new(0.0, 1.2, 0.0);
266
267    if let Ok(mut scene) = game.scene.lock() {
268        let cam = scene.camera_mut();
269        cam.position = eye;
270        cam.target = look;
271    }
272}

274/// Lista rysowania na bieżącą klatkę.
275fn build_draw_list(game: &mut Game) {
276    let mut cmds = Vec::with_capacity(game.meshes.bike.len() + 3);
277    let m = bike_matrix(&game.bike);
278    for (id, mat) in &game.meshes.bike {
279        cmds.push(DrawCmd::new(*id, m).with_material(*mat));
280    }
281    // Droga leży w miejscu — jej transformacja to tylko przesunięcie
282    // świata, a ono wynika z cofnięcia motocykla (patrz `update`).
283    let z = game.bike.pos.z;
284    let road = model_matrix(Vec3::new(0.0, 0.0, z), 0.0, 0.0, 0.0, Vec3::ONE);
285    cmds.push(DrawCmd::new(game.meshes.verge, road));
286    cmds.push(DrawCmd::new(game.meshes.asphalt, road));
287    cmds.push(DrawCmd::new(game.meshes.paint, road));
288
289    if let Ok(mut scene) = game.scene.lock() {
290        scene.set_commands(cmds);
291    }
292}

294/// Krok symulacji + wejście.
295fn update(ctx: &mut Ctx, game: &mut Game) {
296    let dt = ctx.dt();
297    game.time += dt;
298    if let Ok(mut scene) = game.scene.lock() {
299        scene.advance(dt);
300    }
301
302    if ctx.input.just_pressed(Key::Escape) {
303        std::process::exit(0);
304    }
305    if ctx.input.just_pressed(Key::KeyR) {
306        // Pozycję czytamy PRZED pożyczką — `reset` bierze `&mut bike`,
307        // a `game.bike.pos.z` w tej samej wyrażeniu byłoby drugim
308        // pożyczeniem tego samego pola.
309        let z = game.bike.pos.z;
310        bike::reset(&mut game.bike, z);
311        game.distance = 0.0;
312        game.top_speed = 0.0;
313    }
314
315    // `axis(a, b)` daje -1/+1; przy W (gaz) ma być dodatnia.
316    let throttle = ctx.input.axis(Key::KeyS, Key::KeyW);
317    let steer = ctx.input.axis(Key::KeyA, Key::KeyD);
318    bike::step(&mut game.bike, dt, throttle, steer, &game.cfg);
319
320    // --- przebyta droga i rekord
321    //
322    // Liczymy ze *zmian* pozycji, a nie z samej prędkości: przy cofaniu
323    // prędkość jest ujemna i dystans malałby wtedy, gdy jedziemy do
324    // przodu. Używamy tu wartości bezwzględnej.
325    game.distance += game.bike.speed.abs() * dt;
326    game.top_speed = game.top_speed.max(game.bike.speed);
327
328    // --- nieskończona droga: przesuwamy świat
329    //
330    // Droga jest jednym pasmem zbudowanym w miejscu. Gdy motocykl
331    // zje daleko, cofamy go o stały krok — droga „przesuwa się" pod
332    // nim bez rysowania czegokolwiek nowego.
333    if game.bike.pos.z > road::RECENTER_AT {
334        game.bike.pos.z -= road::RECENTER_STEP;
335    } else if game.bike.pos.z < -road::RECENTER_AT {
336        game.bike.pos.z += road::RECENTER_STEP;
337    }
338
339    update_camera(game);
340    build_draw_list(game);
341}

343/// Predkosciomierz i wskaznik skretu.
344///
345/// HUD rysujemy w przestrzeni EKRANU (renderer 2D) — tak samo jak
346/// w `uran-tanks`, i z tej samego powodu: 2D nie przechodzi przez
347/// post-processing 3D i nie zostaje zniekształcony przez ACES.
348fn draw_hud(ctx: &mut Ctx, game: &Game) {
349    let size = ctx.window.logical_size();
350    let kmh = game.bike.speed.abs() * 3.6;
351
352    // --- wskaznik skretu: pasek u dołu ekranu
353    //
354    // Skręt w prawo daje ujemne `steer`, a na ekranie chcemy, żeby
355    // kierownica szła w prawo. Dlatego `steer` odwracamy znakiem.
356    let steer = -game.bike.steer / game.cfg.max_steer;
357    let half = size.x * 0.18;
358    let cx = size.x * 0.5;
359    let y = size.y - 46.0;
360    ctx.gfx
361        .screen_space()
362        .layer(100)
363        .color(Color::from_hex(0x141A24))
364        .draw_rect(Rect::new(
365            Vec2::new(cx - half, y),
366            Vec2::new(cx + half, y + 10.0),
367        ));
368    let knob = half * steer;
369    let tint = if steer.abs() > 0.85 {
370        Color::from_hex(0xFFB347)
371    } else {
372        Color::from_hex(0x66DDAA)
373    };
374    ctx.gfx.color(tint).draw_rect(Rect::new(
375        Vec2::new(cx + knob - 3.0, y - 2.0),
376        Vec2::new(cx + knob + 3.0, y + 12.0),
377    ));
378
379    // --- pasek gazu
380    let t = (ctx.input.axis(Key::KeyS, Key::KeyW) + 1.0) * 0.5;
381    ctx.gfx
382        .color(Color::from_hex(0x3A4A63))
383        .draw_rect(Rect::new(
384            Vec2::new(24.0, size.y - 74.0),
385            Vec2::new(24.0 + 220.0, size.y - 64.0),
386        ));
387    ctx.gfx
388        .color(Color::from_hex(0x66CCFF))
389        .draw_rect(Rect::new(
390            Vec2::new(26.0, size.y - 72.0),
391            Vec2::new(26.0 + 216.0 * t, size.y - 66.0),
392        ));
393
394    let Some(font) = game.font else { return };
395
396    // --- predkosciomierz
397    //
398    // UWAGA: tylko znaki ASCII. Polskie diakrytyczne wywoluja blad
399    // atlasu glifow w `uran-render` przy przepelnieniu (patrz `draw_hud`
400    // w uran-tanks), wiec HUD zostaje czytelny bez polskich znakow.
401    ctx.gfx.color(Color::from_hex(0xE8EEF8)).draw_text(
402        font,
403        &format!("{kmh:.0}"),
404        Vec2::new(cx, size.y - 118.0),
405        54.0,
406        TextAlign::Center,
407    );
408    ctx.gfx.color(Color::from_hex(0x7A8CA8)).draw_text(
409        font,
410        "km/h",
411        Vec2::new(cx, size.y - 60.0),
412        18.0,
413        TextAlign::Center,
414    );
415
416    // Wskazujemy promień skrętu tylko przy wyraźnym wychyle — przy
417    // kierownicy wprost jest nieskończony i `format!` wypisałoby `inf`.
418    let radius = game.bike.turn_radius(&game.cfg);
419    let radius_txt = if radius.is_finite() {
420        format!("R {:.0} m", radius)
421    } else {
422        "R -".to_string()
423    };
424
425    ctx.gfx.color(Color::from_hex(0x8FA6C4)).draw_text(
426        font,
427        &format!(
428            "distance {:.0} m    top {:.0} km/h    {}    parts {}",
429            game.distance,
430            game.top_speed * 3.6,
431            radius_txt,
432            game.meshes.bike.len()
433        ),
434        Vec2::new(24.0, 26.0),
435        16.0,
436        TextAlign::Left,
437    );
438
439    let fps = ctx.time.fps();
440    ctx.gfx.color(Color::from_hex(0x6C7A90)).draw_text(
441        font,
442        &format!("{fps:.0} FPS   W gas   S brake   A/D steer   R restart   ESC quit"),
443        Vec2::new(24.0, 48.0),
444        14.0,
445        TextAlign::Left,
446    );
447}

466/// Proxy 3D oddawane do renderera 2D.
467///
468/// Renderer 2D musi wywołać `draw` na scenie, a w tej samej chwili
469/// gra czyta i zmienia tę samą scenę (kamera, lista rysowania).
470/// Rozwiązanie: obaj trzymamy `Arc<Mutex<..>>` — proxy blokuje i rysuje,
471/// gra blokuje i aktualizuje.
472///
473/// Dlaczego `Mutex`, a nie `Rc<RefCell>`: `App` wymaga systemów `Send`,
474/// a `Rc` nie jest `Send`.
475struct SceneProxy(Arc<Mutex<Renderer3d>>);
476
477impl uran_render::Scene3d for SceneProxy {
478    fn draw(&mut self, target: &uran_render::Scene3dTarget<'_>) {
479        if let Ok(mut scene) = self.0.lock() {
480            uran_render::Scene3d::draw(&mut *scene, target);
481        }
482    }
483
484    fn resize(&mut self, w: u32, h: u32) {
485        if let Ok(mut scene) = self.0.lock() {
486            uran_render::Scene3d::resize(&mut *scene, w, h);
487        }
488    }
489}

491/// Wczytuje czcionkę — bez niej HUD działa, ale bez tekstu.
492fn setup(ctx: &mut Ctx, game: &mut Game) {
493    for path in FONT_FALLBACKS {
494        if let Ok(h) = ctx.assets.load_font(path) {
495            game.font = Some(h);
496            return;
497        }
498    }
499}

509    fn bike_model_points_along_the_direction_of_travel() {
510        // Nos modelu po imporcie leży na `-X` (oryginalna oś Blendera).
511        let nose_local = Vec3::new(-1.0, 0.0, 0.0);
512        let b = Bike::default();
513        let m = bike_matrix(&b);
514        let nose_world = m * uran_math::Vec4::new(nose_local.x, nose_local.y, nose_local.z, 1.0);
515        let nose = Vec3::new(nose_world.x, nose_world.y, nose_world.z);
516
517        // Postój = jazda po `+Z`, więc nos musi wskazywać `+Z`
518        assert!(
519            (nose.z - 1.0).abs() < 1e-4,
520            "nos wskazuje {nose:?} zamiast +Z — model jedzie bokiem"
521        );
522        assert!(nose.x.abs() < 1e-4, "nos ma składową X = {}", nose.x);
523    }
524
525    /// Skręt w prawo musi obrócić model zgodnie z tym, jak obróci się
526    /// kierunek jazdy — inaczej kierownica obraca bryłę w drugą stronę.
527    #[test]
528    fn model_follows_heading() {
529        let mut b = Bike::default();
530        b.yaw = 1.0; // ~57°
531        let nose_local = Vec3::new(-1.0, 0.0, 0.0);
532        let m = bike_matrix(&b);
533        let w = m * uran_math::Vec4::new(nose_local.x, nose_local.y, nose_local.z, 1.0);
534        let nose = Vec3::new(w.x, w.y, w.z).normalize_or_zero();
535        let want = b.forward();
536        assert!(
537            nose.dot(want) > 0.999,
538            "model {nose:?} nie zgadza się z kierunkiem jazdy {want:?}"
539        );
540    }
541
542    /// Model musi stać na ziemi, nie unosić się w powietrzu.
543    #[test]
544    fn bike_matrix_places_the_model_where_the_simulation_says() {
545        let mut b = Bike::default();
546        b.pos = Vec3::new(5.0, 0.0, -12.0);
547        let m = bike_matrix(&b);
548        let p = m * uran_math::Vec4::new(0.0, 0.0, 0.0, 1.0);
549        assert!(
550            (p.x - 5.0).abs() < 1e-4 && (p.z + 12.0).abs() < 1e-4,
551            "model jest w ({}, {}) zamiast (5, -12)",
552            p.x,
553            p.z
554        );
555    }
556
557    /// Kadr kamery: motocykl musi być WIDOCZNY, czyli mieścić się
558    /// między kamerą a punktem patrzenia.
559    #[test]
560    fn camera_sits_behind_and_above_the_bike() {
561        let mut b = Bike::default();
562        b.pos = Vec3::ZERO;
563        b.yaw = 0.0;
564        let f = b.forward();
565        let eye = b.pos - f * CAM_BACK + Vec3::new(0.0, CAM_UP, 0.0);
566        // okno jest 4,3 m za motocyklem i 1,5 m nad nim
567        assert!(
568            (eye.z + CAM_BACK).abs() < 1e-4,
569            "kamera nie jest za motocyklem"
570        );
571        assert!((eye.y - CAM_UP).abs() < 1e-4, "kamera nie jest nad ziemią");
572        // kamera NIE może być wewnątrz modelu (1,2 m wysokości)
573        assert!(eye.y > 1.2, "kamera jest w środku nadwozia");
574    }

577/// Punkt wejścia.
578fn main() {
579    // Stan gry dzielony między systemy — oba blokują ten sam mutex.
580    let game = Arc::new(Mutex::new(None::<Game>));
581
582    let setup_system = {
583        let game = Arc::clone(&game);
584        move |ctx: &mut Ctx| {
585            let mut guard = game.lock().expect("lock gry");
586            if guard.is_some() {
587                return;
588            }
589            // JEDEN `Device` obsługuje oba renderery (patrz
590            // `uran_render::scene3d`), a `samples` MUSI się zgadzać —
591            // inaczej hdr-owa tekstura post-processingu nie przejdzie
592            // walidacji wgpu.
593            let built = {
594                let r = ctx.renderer.as_ref().expect("renderer gotowy");
595                Game::new(r.gpu(), r.format(), r.samples())
596            };
597            *guard = Some(built);
598            let g = guard.as_mut().expect("właśnie utworzone");
599            setup(ctx, g);
600            let proxy = Box::new(SceneProxy(Arc::clone(&g.scene)));
601            if let Some(r) = ctx.renderer.as_mut() {
602                r.set_scene3d(proxy);
603            }
604        }
605    };
606
607    let update_system = {
608        let game = Arc::clone(&game);
609        move |ctx: &mut Ctx| {
610            let mut guard = game.lock().expect("lock gry");
611            if let Some(g) = guard.as_mut() {
612                update(ctx, g);
613            }
614        }
615    };
616
617    let draw_system = {
618        let game = Arc::clone(&game);
619        move |ctx: &mut Ctx| {
620            let guard = game.lock().expect("lock gry");
621            if let Some(g) = guard.as_ref() {
622                draw_hud(ctx, g);
623            }
624        }
625    };
626
627    App::new()
628        .window(
629            windowed(1280, 720)
630                .title("Junak Rider")
631                .background(0x7BA3CC)
632                .vsync(true)
633                .samples(1),
634        )
635        .screenshot(screenshot_path(), 30)
636        .add_startup_system(setup_system)
637        .add_system(update_system)
638        .add_system(draw_system)
639        .run();
640}
