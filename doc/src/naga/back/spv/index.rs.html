11/// The results of performing a bounds check.
12///
13/// On success, `write_bounds_check` returns a value of this type.
14pub(super) enum BoundsCheckResult {
15    /// The index is statically known and in bounds, with the given value.
16    KnownInBounds(u32),
17
18    /// The given instruction computes the index to be used.
19    Computed(Word),
20
21    /// The given instruction computes a boolean condition which is true
22    /// if the index is in bounds.
23    Conditional(Word),
24}
25
26/// A value that we either know at translation time, or need to compute at runtime.
27pub(super) enum MaybeKnown<T> {
28    /// The value is known at shader translation time.
29    Known(T),
30
31    /// The value is computed by the instruction with the given id.
32    Computed(Word),
33}

36    /// Emit code to compute the length of a run-time array.
37    ///
38    /// Given `array`, an expression referring a runtime-sized array, return the
39    /// instruction id for the array's length.
40    pub(super) fn write_runtime_array_length(
41        &mut self,
42        array: Handle<crate::Expression>,
43        block: &mut Block,
44    ) -> Result<Word, Error> {
45        // Naga IR permits runtime-sized arrays as global variables or as the
46        // final member of a struct that is a global variable. SPIR-V permits
47        // only the latter, so this back end wraps bare runtime-sized arrays
48        // in a made-up struct; see `helpers::global_needs_wrapper` and its uses.
49        // This code must handle both cases.
50        let (structure_id, last_member_index) = match self.ir_function.expressions[array] {
51            crate::Expression::AccessIndex { base, index } => {
52                match self.ir_function.expressions[base] {
53                    crate::Expression::GlobalVariable(handle) => (
54                        self.writer.global_variables[handle.index()].access_id,
55                        index,
56                    ),
57                    _ => return Err(Error::Validation("array length expression")),
58                }
59            }
60            crate::Expression::GlobalVariable(handle) => {
61                let global = &self.ir_module.global_variables[handle];
62                if !global_needs_wrapper(self.ir_module, global) {
63                    return Err(Error::Validation("array length expression"));
64                }
65
66                (self.writer.global_variables[handle.index()].var_id, 0)
67            }
68            _ => return Err(Error::Validation("array length expression")),
69        };
70
71        let length_id = self.gen_id();
72        block.body.push(Instruction::array_length(
73            self.writer.get_uint_type_id(),
74            length_id,
75            structure_id,
76            last_member_index,
77        ));
78
79        Ok(length_id)
80    }

82    /// Compute the length of a subscriptable value.
83    ///
84    /// Given `sequence`, an expression referring to some indexable type, return
85    /// its length. The result may either be computed by SPIR-V instructions, or
86    /// known at shader translation time.
87    ///
88    /// `sequence` may be a `Vector`, `Matrix`, or `Array`, a `Pointer` to any
89    /// of those, or a `ValuePointer`. An array may be fixed-size, dynamically
90    /// sized, or use a specializable constant as its length.
91    fn write_sequence_length(
92        &mut self,
93        sequence: Handle<crate::Expression>,
94        block: &mut Block,
95    ) -> Result<MaybeKnown<u32>, Error> {
96        let sequence_ty = self.fun_info[sequence].ty.inner_with(&self.ir_module.types);
97        match sequence_ty.indexable_length(self.ir_module) {
98            Ok(crate::proc::IndexableLength::Known(known_length)) => {
99                Ok(MaybeKnown::Known(known_length))
100            }
101            Ok(crate::proc::IndexableLength::Dynamic) => {
102                let length_id = self.write_runtime_array_length(sequence, block)?;
103                Ok(MaybeKnown::Computed(length_id))
104            }
105            Err(err) => {
106                log::error!("Sequence length for {:?} failed: {}", sequence, err);
107                Err(Error::Validation("indexable length"))
108            }
109        }
110    }

112    /// Compute the maximum valid index of a subscriptable value.
113    ///
114    /// Given `sequence`, an expression referring to some indexable type, return
115    /// its maximum valid index - one less than its length. The result may
116    /// either be computed, or known at shader translation time.
117    ///
118    /// `sequence` may be a `Vector`, `Matrix`, or `Array`, a `Pointer` to any
119    /// of those, or a `ValuePointer`. An array may be fixed-size, dynamically
120    /// sized, or use a specializable constant as its length.
121    fn write_sequence_max_index(
122        &mut self,
123        sequence: Handle<crate::Expression>,
124        block: &mut Block,
125    ) -> Result<MaybeKnown<u32>, Error> {
126        match self.write_sequence_length(sequence, block)? {
127            MaybeKnown::Known(known_length) => {
128                // We should have thrown out all attempts to subscript zero-length
129                // sequences during validation, so the following subtraction should never
130                // underflow.
131                assert!(known_length > 0);
132                // Compute the max index from the length now.
133                Ok(MaybeKnown::Known(known_length - 1))
134            }
135            MaybeKnown::Computed(length_id) => {
136                // Emit code to compute the max index from the length.
137                let const_one_id = self.get_index_constant(1);
138                let max_index_id = self.gen_id();
139                block.body.push(Instruction::binary(
140                    spirv::Op::ISub,
141                    self.writer.get_uint_type_id(),
142                    max_index_id,
143                    length_id,
144                    const_one_id,
145                ));
146                Ok(MaybeKnown::Computed(max_index_id))
147            }
148        }
149    }

151    /// Restrict an index to be in range for a vector, matrix, or array.
152    ///
153    /// This is used to implement `BoundsCheckPolicy::Restrict`. An in-bounds
154    /// index is left unchanged. An out-of-bounds index is replaced with some
155    /// arbitrary in-bounds index. Note,this is not necessarily clamping; for
156    /// example, negative indices might be changed to refer to the last element
157    /// of the sequence, not the first, as clamping would do.
158    ///
159    /// Either return the restricted index value, if known, or add instructions
160    /// to `block` to compute it, and return the id of the result. See the
161    /// documentation for `BoundsCheckResult` for details.
162    ///
163    /// The `sequence` expression may be a `Vector`, `Matrix`, or `Array`, a
164    /// `Pointer` to any of those, or a `ValuePointer`. An array may be
165    /// fixed-size, dynamically sized, or use a specializable constant as its
166    /// length.
167    pub(super) fn write_restricted_index(
168        &mut self,
169        sequence: Handle<crate::Expression>,
170        index: Handle<crate::Expression>,
171        block: &mut Block,
172    ) -> Result<BoundsCheckResult, Error> {
173        let index_id = self.cached[index];
174
175        // Get the sequence's maximum valid index. Return early if we've already
176        // done the bounds check.
177        let max_index_id = match self.write_sequence_max_index(sequence, block)? {
178            MaybeKnown::Known(known_max_index) => {
179                if let Ok(known_index) = self
180                    .ir_module
181                    .to_ctx()
182                    .eval_expr_to_u32_from(index, &self.ir_function.expressions)
183                {
184                    // Both the index and length are known at compile time.
185                    //
186                    // In strict WGSL compliance mode, out-of-bounds indices cannot be
187                    // reported at shader translation time, and must be replaced with
188                    // in-bounds indices at run time. So we cannot assume that
189                    // validation ensured the index was in bounds. Restrict now.
190                    let restricted = std::cmp::min(known_index, known_max_index);
191                    return Ok(BoundsCheckResult::KnownInBounds(restricted));
192                }
193
194                self.get_index_constant(known_max_index)
195            }
196            MaybeKnown::Computed(max_index_id) => max_index_id,
197        };
198
199        // One or the other of the index or length is dynamic, so emit code for
200        // BoundsCheckPolicy::Restrict.
201        let restricted_index_id = self.gen_id();
202        block.body.push(Instruction::ext_inst(
203            self.writer.gl450_ext_inst_id,
204            spirv::GLOp::UMin,
205            self.writer.get_uint_type_id(),
206            restricted_index_id,
207            &[index_id, max_index_id],
208        ));
209        Ok(BoundsCheckResult::Computed(restricted_index_id))
210    }

212    /// Write an index bounds comparison to `block`, if needed.
213    ///
214    /// If we're able to determine statically that `index` is in bounds for
215    /// `sequence`, return `KnownInBounds(value)`, where `value` is the actual
216    /// value of the index. (In principle, one could know that the index is in
217    /// bounds without knowing its specific value, but in our simple-minded
218    /// situation, we always know it.)
219    ///
220    /// If instead we must generate code to perform the comparison at run time,
221    /// return `Conditional(comparison_id)`, where `comparison_id` is an
222    /// instruction producing a boolean value that is true if `index` is in
223    /// bounds for `sequence`.
224    ///
225    /// The `sequence` expression may be a `Vector`, `Matrix`, or `Array`, a
226    /// `Pointer` to any of those, or a `ValuePointer`. An array may be
227    /// fixed-size, dynamically sized, or use a specializable constant as its
228    /// length.
229    fn write_index_comparison(
230        &mut self,
231        sequence: Handle<crate::Expression>,
232        index: Handle<crate::Expression>,
233        block: &mut Block,
234    ) -> Result<BoundsCheckResult, Error> {
235        let index_id = self.cached[index];
236
237        // Get the sequence's length. Return early if we've already done the
238        // bounds check.
239        let length_id = match self.write_sequence_length(sequence, block)? {
240            MaybeKnown::Known(known_length) => {
241                if let Ok(known_index) = self
242                    .ir_module
243                    .to_ctx()
244                    .eval_expr_to_u32_from(index, &self.ir_function.expressions)
245                {
246                    // Both the index and length are known at compile time.
247                    //
248                    // It would be nice to assume that, since we are using the
249                    // `ReadZeroSkipWrite` policy, we are not in strict WGSL
250                    // compliance mode, and thus we can count on the validator to have
251                    // rejected any programs with known out-of-bounds indices, and
252                    // thus just return `KnownInBounds` here without actually
253                    // checking.
254                    //
255                    // But it's also reasonable to expect that bounds check policies
256                    // and error reporting policies should be able to vary
257                    // independently without introducing security holes. So, we should
258                    // support the case where bad indices do not cause validation
259                    // errors, and are handled via `ReadZeroSkipWrite`.
260                    //
261                    // In theory, when `known_index` is bad, we could return a new
262                    // `KnownOutOfBounds` variant here. But it's simpler just to fall
263                    // through and let the bounds check take place. The shader is
264                    // broken anyway, so it doesn't make sense to invest in emitting
265                    // the ideal code for it.
266                    if known_index < known_length {
267                        return Ok(BoundsCheckResult::KnownInBounds(known_index));
268                    }
269                }
270
271                self.get_index_constant(known_length)
272            }
273            MaybeKnown::Computed(length_id) => length_id,
274        };
275
276        // Compare the index against the length.
277        let condition_id = self.gen_id();
278        block.body.push(Instruction::binary(
279            spirv::Op::ULessThan,
280            self.writer.get_bool_type_id(),
281            condition_id,
282            index_id,
283            length_id,
284        ));
285
286        // Indicate that we did generate the check.
287        Ok(BoundsCheckResult::Conditional(condition_id))
288    }

290    /// Emit a conditional load for `BoundsCheckPolicy::ReadZeroSkipWrite`.
291    ///
292    /// Generate code to load a value of `result_type` if `condition` is true,
293    /// and generate a null value of that type if it is false. Call `emit_load`
294    /// to emit the instructions to perform the load. Return the id of the
295    /// merged value of the two branches.
296    pub(super) fn write_conditional_indexed_load<F>(
297        &mut self,
298        result_type: Word,
299        condition: Word,
300        block: &mut Block,
301        emit_load: F,
302    ) -> Word
303    where
304        F: FnOnce(&mut IdGenerator, &mut Block) -> Word,
305    {
306        // For the out-of-bounds case, we produce a zero value.
307        let null_id = self.writer.get_constant_null(result_type);
308
309        let mut selection = Selection::start(block, result_type);
310
311        // As it turns out, we don't actually need a full 'if-then-else'
312        // structure for this: SPIR-V constants are declared up front, so the
313        // 'else' block would have no instructions. Instead we emit something
314        // like this:
315        //
316        //     result = zero;
317        //     if in_bounds {
318        //         result = do the load;
319        //     }
320        //     use result;
321
322        // Continue only if the index was in bounds. Otherwise, branch to the
323        // merge block.
324        selection.if_true(self, condition, null_id);
325
326        // The in-bounds path. Perform the access and the load.
327        let loaded_value = emit_load(&mut self.writer.id_gen, selection.block());
328
329        selection.finish(self, loaded_value)
330    }

332    /// Emit code for bounds checks for an array, vector, or matrix access.
333    ///
334    /// This implements either `index_bounds_check_policy` or
335    /// `buffer_bounds_check_policy`, depending on the address space of the
336    /// pointer being accessed.
337    ///
338    /// Return a `BoundsCheckResult` indicating how the index should be
339    /// consumed. See that type's documentation for details.
340    pub(super) fn write_bounds_check(
341        &mut self,
342        base: Handle<crate::Expression>,
343        index: Handle<crate::Expression>,
344        block: &mut Block,
345    ) -> Result<BoundsCheckResult, Error> {
346        let policy = self.writer.bounds_check_policies.choose_policy(
347            base,
348            &self.ir_module.types,
349            self.fun_info,
350        );
351
352        Ok(match policy {
353            BoundsCheckPolicy::Restrict => self.write_restricted_index(base, index, block)?,
354            BoundsCheckPolicy::ReadZeroSkipWrite => {
355                self.write_index_comparison(base, index, block)?
356            }
357            BoundsCheckPolicy::Unchecked => BoundsCheckResult::Computed(self.cached[index]),
358        })
359    }

361    /// Emit code to subscript a vector by value with a computed index.
362    ///
363    /// Return the id of the element value.
364    pub(super) fn write_vector_access(
365        &mut self,
366        expr_handle: Handle<crate::Expression>,
367        base: Handle<crate::Expression>,
368        index: Handle<crate::Expression>,
369        block: &mut Block,
370    ) -> Result<Word, Error> {
371        let result_type_id = self.get_expression_type_id(&self.fun_info[expr_handle].ty);
372
373        let base_id = self.cached[base];
374        let index_id = self.cached[index];
375
376        let result_id = match self.write_bounds_check(base, index, block)? {
377            BoundsCheckResult::KnownInBounds(known_index) => {
378                let result_id = self.gen_id();
379                block.body.push(Instruction::composite_extract(
380                    result_type_id,
381                    result_id,
382                    base_id,
383                    &[known_index],
384                ));
385                result_id
386            }
387            BoundsCheckResult::Computed(computed_index_id) => {
388                let result_id = self.gen_id();
389                block.body.push(Instruction::vector_extract_dynamic(
390                    result_type_id,
391                    result_id,
392                    base_id,
393                    computed_index_id,
394                ));
395                result_id
396            }
397            BoundsCheckResult::Conditional(comparison_id) => {
398                // Run-time bounds checks were required. Emit
399                // conditional load.
400                self.write_conditional_indexed_load(
401                    result_type_id,
402                    comparison_id,
403                    block,
404                    |id_gen, block| {
405                        // The in-bounds path. Generate the access.
406                        let element_id = id_gen.next();
407                        block.body.push(Instruction::vector_extract_dynamic(
408                            result_type_id,
409                            element_id,
410                            base_id,
411                            index_id,
412                        ));
413                        element_id
414                    },
415                )
416            }
417        };
418
419        Ok(result_id)
420    }
