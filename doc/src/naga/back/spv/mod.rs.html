91/// A SPIR-V block to which we are still adding instructions.
92///
93/// A `Block` represents a SPIR-V block that does not yet have a termination
94/// instruction like `OpBranch` or `OpReturn`.
95///
96/// The `OpLabel` that starts the block is implicit. It will be emitted based on
97/// `label_id` when we write the block to a `LogicalLayout`.
98///
99/// To terminate a `Block`, pass the block and the termination instruction to
100/// `Function::consume`. This takes ownership of the `Block` and transforms it
101/// into a `TerminatedBlock`.
102struct Block {
103    label_id: Word,
104    body: Vec<Instruction>,
105}
106
107/// A SPIR-V block that ends with a termination instruction.
108struct TerminatedBlock {
109    label_id: Word,
110    body: Vec<Instruction>,
111}

147impl Function {
148    fn consume(&mut self, mut block: Block, termination: Instruction) {
149        block.body.push(termination);
150        self.blocks.push(TerminatedBlock {
151            label_id: block.label_id,
152            body: block.body,
153        })
154    }
155
156    fn parameter_id(&self, index: u32) -> Word {
157        match self.entry_point_context {
158            Some(ref context) => context.argument_ids[index as usize],
159            None => self.parameters[index as usize]
160                .instruction
161                .result_id
162                .unwrap(),
163        }
164    }

167/// Characteristics of a SPIR-V `OpTypeImage` type.
168///
169/// SPIR-V requires non-composite types to be unique, including images. Since we
170/// use `LocalType` for this deduplication, it's essential that `LocalImageType`
171/// be equal whenever the corresponding `OpTypeImage`s would be. To reduce the
172/// likelihood of mistakes, we use fields that correspond exactly to the
173/// operands of an `OpTypeImage` instruction, using the actual SPIR-V types
174/// where practical.
175#[derive(Debug, PartialEq, Hash, Eq, Copy, Clone)]
176struct LocalImageType {
177    sampled_type: crate::ScalarKind,
178    dim: spirv::Dim,
179    flags: ImageTypeFlags,
180    image_format: spirv::ImageFormat,
181}

185    #[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
186    pub struct ImageTypeFlags: u8 {
187        const DEPTH = 0x1;
188        const ARRAYED = 0x2;
189        const MULTISAMPLED = 0x4;
190        const SAMPLED = 0x8;
191    }

194impl LocalImageType {
195    /// Construct a `LocalImageType` from the fields of a `TypeInner::Image`.
196    fn from_inner(dim: crate::ImageDimension, arrayed: bool, class: crate::ImageClass) -> Self {
197        let make_flags = |multi: bool, other: ImageTypeFlags| -> ImageTypeFlags {
198            let mut flags = other;
199            flags.set(ImageTypeFlags::ARRAYED, arrayed);
200            flags.set(ImageTypeFlags::MULTISAMPLED, multi);
201            flags
202        };
203
204        let dim = spirv::Dim::from(dim);
205
206        match class {
207            crate::ImageClass::Sampled { kind, multi } => LocalImageType {
208                sampled_type: kind,
209                dim,
210                flags: make_flags(multi, ImageTypeFlags::SAMPLED),
211                image_format: spirv::ImageFormat::Unknown,
212            },
213            crate::ImageClass::Depth { multi } => LocalImageType {
214                sampled_type: crate::ScalarKind::Float,
215                dim,
216                flags: make_flags(multi, ImageTypeFlags::DEPTH | ImageTypeFlags::SAMPLED),
217                image_format: spirv::ImageFormat::Unknown,
218            },
219            crate::ImageClass::Storage { format, access: _ } => LocalImageType {
220                sampled_type: crate::ScalarKind::from(format),
221                dim,
222                flags: make_flags(false, ImageTypeFlags::empty()),
223                image_format: format.into(),
224            },
225        }
226    }
227}

229/// A SPIR-V type constructed during code generation.
230///
231/// This is the variant of [`LookupType`] used to represent types that might not
232/// be available in the arena. Variants are present here for one of two reasons:
233///
234/// -   They represent types synthesized during code generation, as explained
235///     in the documentation for [`LookupType`].
236///
237/// -   They represent types for which SPIR-V forbids duplicate `OpType...`
238///     instructions, requiring deduplication.
239///
240/// This is not a complete copy of [`TypeInner`]: for example, SPIR-V generation
241/// never synthesizes new struct types, so `LocalType` has nothing for that.
242///
243/// Each `LocalType` variant should be handled identically to its analogous
244/// `TypeInner` variant. You can use the [`make_local`] function to help with
245/// this, by converting everything possible to a `LocalType` before inspecting
246/// it.
247///
248/// ## `Localtype` equality and SPIR-V `OpType` uniqueness
249///
250/// The definition of `Eq` on `LocalType` is carefully chosen to help us follow
251/// certain SPIR-V rules. SPIR-V §2.8 requires some classes of `OpType...`
252/// instructions to be unique; for example, you can't have two `OpTypeInt 32 1`
253/// instructions in the same module. All 32-bit signed integers must use the
254/// same type id.
255///
256/// All SPIR-V types that must be unique can be represented as a `LocalType`,
257/// and two `LocalType`s are always `Eq` if SPIR-V would require them to use the
258/// same `OpType...` instruction. This lets us avoid duplicates by recording the
259/// ids of the type instructions we've already generated in a hash table,
260/// [`Writer::lookup_type`], keyed by `LocalType`.
261///
262/// As another example, [`LocalImageType`], stored in the `LocalType::Image`
263/// variant, is designed to help us deduplicate `OpTypeImage` instructions. See
264/// its documentation for details.
265///
266/// `LocalType` also includes variants like `Pointer` that do not need to be
267/// unique - but it is harmless to avoid the duplication.
268///
269/// As it always must, the `Hash` implementation respects the `Eq` relation.
270///
271/// [`TypeInner`]: crate::TypeInner
272#[derive(Debug, PartialEq, Hash, Eq, Copy, Clone)]
273enum LocalType {
274    /// A scalar, vector, or pointer to one of those.
275    Value {
276        /// If `None`, this represents a scalar type. If `Some`, this represents
277        /// a vector type of the given size.
278        vector_size: Option<crate::VectorSize>,
279        scalar: crate::Scalar,
280        pointer_space: Option<spirv::StorageClass>,
281    },
282    /// A matrix of floating-point values.
283    Matrix {
284        columns: crate::VectorSize,
285        rows: crate::VectorSize,
286        width: crate::Bytes,
287    },
288    Pointer {
289        base: Handle<crate::Type>,
290        class: spirv::StorageClass,
291    },
292    Image(LocalImageType),
293    SampledImage {
294        image_type_id: Word,
295    },
296    Sampler,
297    /// Equivalent to a [`LocalType::Pointer`] whose `base` is a Naga IR [`BindingArray`]. SPIR-V
298    /// permits duplicated `OpTypePointer` ids, so it's fine to have two different [`LocalType`]
299    /// representations for pointer types.
300    ///
301    /// [`BindingArray`]: crate::TypeInner::BindingArray
302    PointerToBindingArray {
303        base: Handle<crate::Type>,
304        size: u32,
305        space: crate::AddressSpace,
306    },
307    BindingArray {
308        base: Handle<crate::Type>,
309        size: u32,
310    },
311    AccelerationStructure,
312    RayQuery,
313}

315/// A type encountered during SPIR-V generation.
316///
317/// In the process of writing SPIR-V, we need to synthesize various types for
318/// intermediate results and such: pointer types, vector/matrix component types,
319/// or even booleans, which usually appear in SPIR-V code even when they're not
320/// used by the module source.
321///
322/// However, we can't use `crate::Type` or `crate::TypeInner` for these, as the
323/// type arena may not contain what we need (it only contains types used
324/// directly by other parts of the IR), and the IR module is immutable, so we
325/// can't add anything to it.
326///
327/// So for local use in the SPIR-V writer, we use this type, which holds either
328/// a handle into the arena, or a [`LocalType`] containing something synthesized
329/// locally.
330///
331/// This is very similar to the [`proc::TypeResolution`] enum, with `LocalType`
332/// playing the role of `TypeInner`. However, `LocalType` also has other
333/// properties needed for SPIR-V generation; see the description of
334/// [`LocalType`] for details.
335///
336/// [`proc::TypeResolution`]: crate::proc::TypeResolution
337#[derive(Debug, PartialEq, Hash, Eq, Copy, Clone)]
338enum LookupType {
339    Handle(Handle<crate::Type>),
340    Local(LocalType),
341}
342
343impl From<LocalType> for LookupType {
344    fn from(local: LocalType) -> Self {
345        Self::Local(local)
346    }
347}

355fn make_local(inner: &crate::TypeInner) -> Option<LocalType> {
356    Some(match *inner {
357        crate::TypeInner::Scalar(scalar) | crate::TypeInner::Atomic(scalar) => LocalType::Value {
358            vector_size: None,
359            scalar,
360            pointer_space: None,
361        },
362        crate::TypeInner::Vector { size, scalar } => LocalType::Value {
363            vector_size: Some(size),
364            scalar,
365            pointer_space: None,
366        },
367        crate::TypeInner::Matrix {
368            columns,
369            rows,
370            scalar,
371        } => LocalType::Matrix {
372            columns,
373            rows,
374            width: scalar.width,
375        },
376        crate::TypeInner::Pointer { base, space } => LocalType::Pointer {
377            base,
378            class: helpers::map_storage_class(space),
379        },
380        crate::TypeInner::ValuePointer {
381            size,
382            scalar,
383            space,
384        } => LocalType::Value {
385            vector_size: size,
386            scalar,
387            pointer_space: Some(helpers::map_storage_class(space)),
388        },
389        crate::TypeInner::Image {
390            dim,
391            arrayed,
392            class,
393        } => LocalType::Image(LocalImageType::from_inner(dim, arrayed, class)),
394        crate::TypeInner::Sampler { comparison: _ } => LocalType::Sampler,
395        crate::TypeInner::AccelerationStructure => LocalType::AccelerationStructure,
396        crate::TypeInner::RayQuery => LocalType::RayQuery,
397        crate::TypeInner::Array { .. }
398        | crate::TypeInner::Struct { .. }
399        | crate::TypeInner::BindingArray { .. } => return None,
400    })
401}

410/// A map from evaluated [`Expression`](crate::Expression)s to their SPIR-V ids.
411///
412/// When we emit code to evaluate a given `Expression`, we record the
413/// SPIR-V id of its value here, under its `Handle<Expression>` index.
414///
415/// A `CachedExpressions` value can be indexed by a `Handle<Expression>` value.
416///
417/// [emit]: index.html#expression-evaluation-time-and-scope
418#[derive(Default)]
419struct CachedExpressions {
420    ids: Vec<Word>,
421}
422impl CachedExpressions {
423    fn reset(&mut self, length: usize) {
424        self.ids.clear();
425        self.ids.resize(length, 0);
426    }
427}
428impl ops::Index<Handle<crate::Expression>> for CachedExpressions {
429    type Output = Word;
430    fn index(&self, h: Handle<crate::Expression>) -> &Word {
431        let id = &self.ids[h.index()];
432        if *id == 0 {
433            unreachable!("Expression {:?} is not cached!", h);
434        }
435        id
436    }
437}
438impl ops::IndexMut<Handle<crate::Expression>> for CachedExpressions {
439    fn index_mut(&mut self, h: Handle<crate::Expression>) -> &mut Word {
440        let id = &mut self.ids[h.index()];
441        if *id != 0 {
442            unreachable!("Expression {:?} is already cached!", h);
443        }
444        id
445    }
446}
447impl recyclable::Recyclable for CachedExpressions {
448    fn recycle(self) -> Self {
449        CachedExpressions {
450            ids: self.ids.recycle(),
451        }
452    }
453}

465#[derive(Clone)]
466struct GlobalVariable {
467    /// ID of the OpVariable that declares the global.
468    ///
469    /// If you need the variable's value, use [`access_id`] instead of this
470    /// field. If we wrapped the Naga IR `GlobalVariable`'s type in a struct to
471    /// comply with Vulkan's requirements, then this points to the `OpVariable`
472    /// with the synthesized struct type, whereas `access_id` points to the
473    /// field of said struct that holds the variable's actual value.
474    ///
475    /// This is used to compute the `access_id` pointer in function prologues,
476    /// and used for `ArrayLength` expressions, which do need the struct.
477    ///
478    /// [`access_id`]: GlobalVariable::access_id
479    var_id: Word,
480
481    /// For `AddressSpace::Handle` variables, this ID is recorded in the function
482    /// prelude block (and reset before every function) as `OpLoad` of the variable.
483    /// It is then used for all the global ops, such as `OpImageSample`.
484    handle_id: Word,
485
486    /// Actual ID used to access this variable.
487    /// For wrapped buffer variables, this ID is `OpAccessChain` into the
488    /// wrapper. Otherwise, the same as `var_id`.
489    ///
490    /// Vulkan requires that globals in the `StorageBuffer` and `Uniform` storage
491    /// classes must be structs with the `Block` decoration, but WGSL and Naga IR
492    /// make no such requirement. So for such variables, we generate a wrapper struct
493    /// type with a single element of the type given by Naga, generate an
494    /// `OpAccessChain` for that member in the function prelude, and use that pointer
495    /// to refer to the global in the function body. This is the id of that access,
496    /// updated for each function in `write_function`.
497    access_id: Word,
498}
499
500impl GlobalVariable {
501    const fn dummy() -> Self {
502        Self {
503            var_id: 0,
504            handle_id: 0,
505            access_id: 0,
506        }
507    }
508
509    const fn new(id: Word) -> Self {
510        Self {
511            var_id: id,
512            handle_id: 0,
513            access_id: 0,
514        }
515    }
516
517    /// Prepare `self` for use within a single function.
518    fn reset_for_function(&mut self) {
519        self.handle_id = 0;
520        self.access_id = 0;
521    }
522}

698impl<'a> Default for Options<'a> {
699    fn default() -> Self {
700        let mut flags = WriterFlags::ADJUST_COORDINATE_SPACE
701            | WriterFlags::LABEL_VARYINGS
702            | WriterFlags::CLAMP_FRAG_DEPTH;
703        if cfg!(debug_assertions) {
704            flags |= WriterFlags::DEBUG;
705        }
706        Options {
707            lang_version: (1, 0),
708            flags,
709            binding_map: BindingMap::default(),
710            capabilities: None,
711            bounds_check_policies: crate::proc::BoundsCheckPolicies::default(),
712            zero_initialize_workgroup_memory: ZeroInitializeWorkgroupMemoryMode::Polyfill,
713            debug_info: None,
714        }
715    }
716}

718// A subset of options meant to be changed per pipeline.
719#[derive(Debug, Clone, PartialEq, Eq, Hash)]
720#[cfg_attr(feature = "serialize", derive(serde::Serialize))]
721#[cfg_attr(feature = "deserialize", derive(serde::Deserialize))]
722pub struct PipelineOptions {
723    /// The stage of the entry point.
724    pub shader_stage: crate::ShaderStage,
725    /// The name of the entry point.
726    ///
727    /// If no entry point that matches is found while creating a [`Writer`], a error will be thrown.
728    pub entry_point: String,
729}
730
731pub fn write_vec(
732    module: &crate::Module,
733    info: &crate::valid::ModuleInfo,
734    options: &Options,
735    pipeline_options: Option<&PipelineOptions>,
736) -> Result<Vec<u32>, Error> {
737    let mut words: Vec<u32> = Vec::new();
738    let mut w = Writer::new(options)?;
739
740    w.write(
741        module,
742        info,
743        pipeline_options,
744        &options.debug_info,
745        &mut words,
746    )?;
747    Ok(words)
748}
