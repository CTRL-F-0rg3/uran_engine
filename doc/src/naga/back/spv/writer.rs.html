24    fn to_words(&self, sink: &mut impl Extend<Word>) {
25        self.signature.as_ref().unwrap().to_words(sink);
26        for argument in self.parameters.iter() {
27            argument.instruction.to_words(sink);
28        }
29        for (index, block) in self.blocks.iter().enumerate() {
30            Instruction::label(block.label_id).to_words(sink);
31            if index == 0 {
32                for local_var in self.variables.values() {
33                    local_var.instruction.to_words(sink);
34                }
35            }
36            for instruction in block.body.iter() {
37                instruction.to_words(sink);
38            }
39        }
40    }
41}

84    /// Reset `Writer` to its initial state, retaining any allocations.
85    ///
86    /// Why not just implement `Recyclable` for `Writer`? By design,
87    /// `Recyclable::recycle` requires ownership of the value, not just
88    /// `&mut`; see the trait documentation. But we need to use this method
89    /// from functions like `Writer::write`, which only have `&mut Writer`.
90    /// Workarounds include unsafe code (`std::ptr::read`, then `write`, ugh)
91    /// or something like a `Default` impl that returns an oddly-initialized
92    /// `Writer`, which is worse.
93    fn reset(&mut self) {
94        use super::recyclable::Recyclable;
95        use std::mem::take;
96
97        let mut id_gen = IdGenerator::default();
98        let gl450_ext_inst_id = id_gen.next();
99        let void_type = id_gen.next();
100
101        // Every field of the old writer that is not determined by the `Options`
102        // passed to `Writer::new` should be reset somehow.
103        let fresh = Writer {
104            // Copied from the old Writer:
105            flags: self.flags,
106            bounds_check_policies: self.bounds_check_policies,
107            zero_initialize_workgroup_memory: self.zero_initialize_workgroup_memory,
108            capabilities_available: take(&mut self.capabilities_available),
109            binding_map: take(&mut self.binding_map),
110
111            // Initialized afresh:
112            id_gen,
113            void_type,
114            gl450_ext_inst_id,
115
116            // Recycled:
117            capabilities_used: take(&mut self.capabilities_used).recycle(),
118            extensions_used: take(&mut self.extensions_used).recycle(),
119            physical_layout: self.physical_layout.clone().recycle(),
120            logical_layout: take(&mut self.logical_layout).recycle(),
121            debugs: take(&mut self.debugs).recycle(),
122            annotations: take(&mut self.annotations).recycle(),
123            lookup_type: take(&mut self.lookup_type).recycle(),
124            lookup_function: take(&mut self.lookup_function).recycle(),
125            lookup_function_type: take(&mut self.lookup_function_type).recycle(),
126            constant_ids: take(&mut self.constant_ids).recycle(),
127            cached_constants: take(&mut self.cached_constants).recycle(),
128            global_variables: take(&mut self.global_variables).recycle(),
129            saved_cached: take(&mut self.saved_cached).recycle(),
130            temp_list: take(&mut self.temp_list).recycle(),
131        };
132
133        *self = fresh;
134
135        self.capabilities_used.insert(spirv::Capability::Shader);
136    }

138    /// Indicate that the code requires any one of the listed capabilities.
139    ///
140    /// If nothing in `capabilities` appears in the available capabilities
141    /// specified in the [`Options`] from which this `Writer` was created,
142    /// return an error. The `what` string is used in the error message to
143    /// explain what provoked the requirement. (If no available capabilities were
144    /// given, assume everything is available.)
145    ///
146    /// The first acceptable capability will be added to this `Writer`'s
147    /// [`capabilities_used`] table, and an `OpCapability` emitted for it in the
148    /// result. For this reason, more specific capabilities should be listed
149    /// before more general.
150    ///
151    /// [`capabilities_used`]: Writer::capabilities_used
152    pub(super) fn require_any(
153        &mut self,
154        what: &'static str,
155        capabilities: &[spirv::Capability],
156    ) -> Result<(), Error> {
157        match *capabilities {
158            [] => Ok(()),
159            [first, ..] => {
160                // Find the first acceptable capability, or return an error if
161                // there is none.
162                let selected = match self.capabilities_available {
163                    None => first,
164                    Some(ref available) => {
165                        match capabilities.iter().find(|cap| available.contains(cap)) {
166                            Some(&cap) => cap,
167                            None => {
168                                return Err(Error::MissingCapabilities(what, capabilities.to_vec()))
169                            }
170                        }
171                    }
172                };
173                self.capabilities_used.insert(selected);
174                Ok(())
175            }
176        }
177    }

215        arena: &UniqueArena<crate::Type>,
216        handle: Handle<crate::Type>,
217        class: spirv::StorageClass,

233            self.lookup_type.insert(lookup_type, id);

324    fn write_function(
325        &mut self,
326        ir_function: &crate::Function,
327        info: &FunctionInfo,
328        ir_module: &crate::Module,
329        mut interface: Option<FunctionInterface>,
330        debug_info: &Option<DebugInfoInner>,
331    ) -> Result<Word, Error> {
332        let mut function = Function::default();
333
334        let prelude_id = self.id_gen.next();
335        let mut prelude = Block::new(prelude_id);
336        let mut ep_context = EntryPointContext {
337            argument_ids: Vec::new(),
338            results: Vec::new(),
339        };
340
341        let mut local_invocation_id = None;
342
343        let mut parameter_type_ids = Vec::with_capacity(ir_function.arguments.len());
344        for argument in ir_function.arguments.iter() {
345            let class = spirv::StorageClass::Input;
346            let handle_ty = ir_module.types[argument.ty].inner.is_handle();
347            let argument_type_id = match handle_ty {
348                true => self.get_pointer_id(
349                    &ir_module.types,
350                    argument.ty,
351                    spirv::StorageClass::UniformConstant,
352                )?,
353                false => self.get_type_id(LookupType::Handle(argument.ty)),
354            };
355
356            if let Some(ref mut iface) = interface {
357                let id = if let Some(ref binding) = argument.binding {
358                    let name = argument.name.as_deref();
359
360                    let varying_id = self.write_varying(
361                        ir_module,
362                        iface.stage,
363                        class,
364                        name,
365                        argument.ty,
366                        binding,
367                    )?;
368                    iface.varying_ids.push(varying_id);
369                    let id = self.id_gen.next();
370                    prelude
371                        .body
372                        .push(Instruction::load(argument_type_id, id, varying_id, None));
373
374                    if binding == &crate::Binding::BuiltIn(crate::BuiltIn::LocalInvocationId) {
375                        local_invocation_id = Some(id);
376                    }
377
378                    id
379                } else if let crate::TypeInner::Struct { ref members, .. } =
380                    ir_module.types[argument.ty].inner
381                {
382                    let struct_id = self.id_gen.next();
383                    let mut constituent_ids = Vec::with_capacity(members.len());
384                    for member in members {
385                        let type_id = self.get_type_id(LookupType::Handle(member.ty));
386                        let name = member.name.as_deref();
387                        let binding = member.binding.as_ref().unwrap();
388                        let varying_id = self.write_varying(
389                            ir_module,
390                            iface.stage,
391                            class,
392                            name,
393                            member.ty,
394                            binding,
395                        )?;
396                        iface.varying_ids.push(varying_id);
397                        let id = self.id_gen.next();
398                        prelude
399                            .body
400                            .push(Instruction::load(type_id, id, varying_id, None));
401                        constituent_ids.push(id);
402
403                        if binding == &crate::Binding::BuiltIn(crate::BuiltIn::GlobalInvocationId) {
404                            local_invocation_id = Some(id);
405                        }
406                    }
407                    prelude.body.push(Instruction::composite_construct(
408                        argument_type_id,
409                        struct_id,
410                        &constituent_ids,
411                    ));
412                    struct_id
413                } else {
414                    unreachable!("Missing argument binding on an entry point");
415                };
416                ep_context.argument_ids.push(id);
417            } else {
418                let argument_id = self.id_gen.next();
419                let instruction = Instruction::function_parameter(argument_type_id, argument_id);
420                if self.flags.contains(WriterFlags::DEBUG) {
421                    if let Some(ref name) = argument.name {
422                        self.debugs.push(Instruction::name(argument_id, name));
423                    }
424                }
425                function.parameters.push(FunctionArgument {
426                    instruction,
427                    handle_id: if handle_ty {
428                        let id = self.id_gen.next();
429                        prelude.body.push(Instruction::load(
430                            self.get_type_id(LookupType::Handle(argument.ty)),
431                            id,
432                            argument_id,
433                            None,
434                        ));
435                        id
436                    } else {
437                        0
438                    },
439                });
440                parameter_type_ids.push(argument_type_id);
441            };
442        }
443
444        let return_type_id = match ir_function.result {
445            Some(ref result) => {
446                if let Some(ref mut iface) = interface {
447                    let mut has_point_size = false;
448                    let class = spirv::StorageClass::Output;
449                    if let Some(ref binding) = result.binding {
450                        has_point_size |=
451                            *binding == crate::Binding::BuiltIn(crate::BuiltIn::PointSize);
452                        let type_id = self.get_type_id(LookupType::Handle(result.ty));
453                        let varying_id = self.write_varying(
454                            ir_module,
455                            iface.stage,
456                            class,
457                            None,
458                            result.ty,
459                            binding,
460                        )?;
461                        iface.varying_ids.push(varying_id);
462                        ep_context.results.push(ResultMember {
463                            id: varying_id,
464                            type_id,
465                            built_in: binding.to_built_in(),
466                        });
467                    } else if let crate::TypeInner::Struct { ref members, .. } =
468                        ir_module.types[result.ty].inner
469                    {
470                        for member in members {
471                            let type_id = self.get_type_id(LookupType::Handle(member.ty));
472                            let name = member.name.as_deref();
473                            let binding = member.binding.as_ref().unwrap();
474                            has_point_size |=
475                                *binding == crate::Binding::BuiltIn(crate::BuiltIn::PointSize);
476                            let varying_id = self.write_varying(
477                                ir_module,
478                                iface.stage,
479                                class,
480                                name,
481                                member.ty,
482                                binding,
483                            )?;
484                            iface.varying_ids.push(varying_id);
485                            ep_context.results.push(ResultMember {
486                                id: varying_id,
487                                type_id,
488                                built_in: binding.to_built_in(),
489                            });
490                        }
491                    } else {
492                        unreachable!("Missing result binding on an entry point");
493                    }
494
495                    if self.flags.contains(WriterFlags::FORCE_POINT_SIZE)
496                        && iface.stage == crate::ShaderStage::Vertex
497                        && !has_point_size
498                    {
499                        // add point size artificially
500                        let varying_id = self.id_gen.next();
501                        let pointer_type_id = self.get_float_pointer_type_id(class);
502                        Instruction::variable(pointer_type_id, varying_id, class, None)
503                            .to_words(&mut self.logical_layout.declarations);
504                        self.decorate(
505                            varying_id,
506                            spirv::Decoration::BuiltIn,
507                            &[spirv::BuiltIn::PointSize as u32],
508                        );
509                        iface.varying_ids.push(varying_id);
510
511                        let default_value_id = self.get_constant_scalar(crate::Literal::F32(1.0));
512                        prelude
513                            .body
514                            .push(Instruction::store(varying_id, default_value_id, None));
515                    }
516                    self.void_type
517                } else {
518                    self.get_type_id(LookupType::Handle(result.ty))
519                }
520            }
521            None => self.void_type,
522        };
523
524        let lookup_function_type = LookupFunctionType {
525            parameter_type_ids,
526            return_type_id,
527        };
528
529        let function_id = self.id_gen.next();
530        if self.flags.contains(WriterFlags::DEBUG) {
531            if let Some(ref name) = ir_function.name {
532                self.debugs.push(Instruction::name(function_id, name));
533            }
534        }
535
536        let function_type = self.get_function_type(lookup_function_type);
537        function.signature = Some(Instruction::function(
538            return_type_id,
539            function_id,
540            spirv::FunctionControl::empty(),
541            function_type,
542        ));
543
544        if interface.is_some() {
545            function.entry_point_context = Some(ep_context);
546        }
547
548        // fill up the `GlobalVariable::access_id`
549        for gv in self.global_variables.iter_mut() {
550            gv.reset_for_function();
551        }
552        for (handle, var) in ir_module.global_variables.iter() {
553            if info[handle].is_empty() {
554                continue;
555            }
556
557            let mut gv = self.global_variables[handle.index()].clone();
558            if let Some(ref mut iface) = interface {
559                // Have to include global variables in the interface
560                if self.physical_layout.version >= 0x10400 {
561                    iface.varying_ids.push(gv.var_id);
562                }
563            }
564
565            // Handle globals are pre-emitted and should be loaded automatically.
566            //
567            // Any that are binding arrays we skip as we cannot load the array, we must load the result after indexing.
568            let is_binding_array = match ir_module.types[var.ty].inner {
569                crate::TypeInner::BindingArray { .. } => true,
570                _ => false,
571            };
572
573            if var.space == crate::AddressSpace::Handle && !is_binding_array {
574                let var_type_id = self.get_type_id(LookupType::Handle(var.ty));
575                let id = self.id_gen.next();
576                prelude
577                    .body
578                    .push(Instruction::load(var_type_id, id, gv.var_id, None));
579                gv.access_id = gv.var_id;
580                gv.handle_id = id;
581            } else if global_needs_wrapper(ir_module, var) {
582                let class = map_storage_class(var.space);
583                let pointer_type_id = self.get_pointer_id(&ir_module.types, var.ty, class)?;
584                let index_id = self.get_index_constant(0);
585
586                let id = self.id_gen.next();
587                prelude.body.push(Instruction::access_chain(
588                    pointer_type_id,
589                    id,
590                    gv.var_id,
591                    &[index_id],
592                ));
593                gv.access_id = id;
594            } else {
595                // by default, the variable ID is accessed as is
596                gv.access_id = gv.var_id;
597            };
598
599            // work around borrow checking in the presence of `self.xxx()` calls
600            self.global_variables[handle.index()] = gv;
601        }
602
603        // Create a `BlockContext` for generating SPIR-V for the function's
604        // body.
605        let mut context = BlockContext {
606            ir_module,
607            ir_function,
608            fun_info: info,
609            function: &mut function,
610            // Re-use the cached expression table from prior functions.
611            cached: std::mem::take(&mut self.saved_cached),
612
613            // Steal the Writer's temp list for a bit.
614            temp_list: std::mem::take(&mut self.temp_list),
615            writer: self,
616            expression_constness: crate::proc::ExpressionConstnessTracker::from_arena(
617                &ir_function.expressions,
618            ),
619        };
620
621        // fill up the pre-emitted and const expressions
622        context.cached.reset(ir_function.expressions.len());
623        for (handle, expr) in ir_function.expressions.iter() {
624            if (expr.needs_pre_emit() && !matches!(*expr, crate::Expression::LocalVariable(_)))
625                || context.expression_constness.is_const(handle)
626            {
627                context.cache_expression_value(handle, &mut prelude)?;
628            }
629        }
630
631        for (handle, variable) in ir_function.local_variables.iter() {
632            let id = context.gen_id();
633
634            if context.writer.flags.contains(WriterFlags::DEBUG) {
635                if let Some(ref name) = variable.name {
636                    context.writer.debugs.push(Instruction::name(id, name));
637                }
638            }
639
640            let init_word = variable.init.map(|constant| context.cached[constant]);
641            let pointer_type_id = context.writer.get_pointer_id(
642                &ir_module.types,
643                variable.ty,
644                spirv::StorageClass::Function,
645            )?;
646            let instruction = Instruction::variable(
647                pointer_type_id,
648                id,
649                spirv::StorageClass::Function,
650                init_word.or_else(|| match ir_module.types[variable.ty].inner {
651                    crate::TypeInner::RayQuery => None,
652                    _ => {
653                        let type_id = context.get_type_id(LookupType::Handle(variable.ty));
654                        Some(context.writer.write_constant_null(type_id))
655                    }
656                }),
657            );
658            context
659                .function
660                .variables
661                .insert(handle, LocalVariable { id, instruction });
662        }
663
664        // cache local variable expressions
665        for (handle, expr) in ir_function.expressions.iter() {
666            if matches!(*expr, crate::Expression::LocalVariable(_)) {
667                context.cache_expression_value(handle, &mut prelude)?;
668            }
669        }
670
671        let next_id = context.gen_id();
672
673        context
674            .function
675            .consume(prelude, Instruction::branch(next_id));
676
677        let workgroup_vars_init_exit_block_id =
678            match (context.writer.zero_initialize_workgroup_memory, interface) {
679                (
680                    super::ZeroInitializeWorkgroupMemoryMode::Polyfill,
681                    Some(
682                        ref mut interface @ FunctionInterface {
683                            stage: crate::ShaderStage::Compute,
684                            ..
685                        },
686                    ),
687                ) => context.writer.generate_workgroup_vars_init_block(
688                    next_id,
689                    ir_module,
690                    info,
691                    local_invocation_id,
692                    interface,
693                    context.function,
694                ),
695                _ => None,
696            };
697
698        let main_id = if let Some(exit_id) = workgroup_vars_init_exit_block_id {
699            exit_id
700        } else {
701            next_id
702        };
703
704        context.write_block(
705            main_id,
706            &ir_function.body,
707            super::block::BlockExit::Return,
708            LoopContext::default(),
709            debug_info.as_ref(),
710        )?;
711
712        // Consume the `BlockContext`, ending its borrows and letting the
713        // `Writer` steal back its cached expression table and temp_list.
714        let BlockContext {
715            cached, temp_list, ..
716        } = context;
717        self.saved_cached = cached;
718        self.temp_list = temp_list;
719
720        function.to_words(&mut self.logical_layout.function_definitions);
721        Instruction::function_end().to_words(&mut self.logical_layout.function_definitions);
722
723        Ok(function_id)
724    }

798    fn make_scalar(&mut self, id: Word, scalar: crate::Scalar) -> Instruction {
799        use crate::ScalarKind as Sk;
800
801        let bits = (scalar.width * BITS_PER_BYTE) as u32;
802        match scalar.kind {
803            Sk::Sint | Sk::Uint => {
804                let signedness = if scalar.kind == Sk::Sint {
805                    super::instructions::Signedness::Signed
806                } else {
807                    super::instructions::Signedness::Unsigned
808                };
809                let cap = match bits {
810                    8 => Some(spirv::Capability::Int8),
811                    16 => Some(spirv::Capability::Int16),
812                    64 => Some(spirv::Capability::Int64),
813                    _ => None,
814                };
815                if let Some(cap) = cap {
816                    self.capabilities_used.insert(cap);
817                }
818                Instruction::type_int(id, bits, signedness)
819            }
820            Sk::Float => {
821                if bits == 64 {
822                    self.capabilities_used.insert(spirv::Capability::Float64);
823                }
824                Instruction::type_float(id, bits)
825            }
826            Sk::Bool => Instruction::type_bool(id),
827            Sk::AbstractInt | Sk::AbstractFloat => {
828                unreachable!("abstract types should never reach the backend");
829            }
830        }
831    }

833    fn request_type_capabilities(&mut self, inner: &crate::TypeInner) -> Result<(), Error> {
834        match *inner {
835            crate::TypeInner::Image {
836                dim,
837                arrayed,
838                class,
839            } => {
840                let sampled = match class {
841                    crate::ImageClass::Sampled { .. } => true,
842                    crate::ImageClass::Depth { .. } => true,
843                    crate::ImageClass::Storage { format, .. } => {
844                        self.request_image_format_capabilities(format.into())?;
845                        false
846                    }
847                };
848
849                match dim {
850                    crate::ImageDimension::D1 => {
851                        if sampled {
852                            self.require_any("sampled 1D images", &[spirv::Capability::Sampled1D])?;
853                        } else {
854                            self.require_any("1D storage images", &[spirv::Capability::Image1D])?;
855                        }
856                    }
857                    crate::ImageDimension::Cube if arrayed => {
858                        if sampled {
859                            self.require_any(
860                                "sampled cube array images",
861                                &[spirv::Capability::SampledCubeArray],
862                            )?;
863                        } else {
864                            self.require_any(
865                                "cube array storage images",
866                                &[spirv::Capability::ImageCubeArray],
867                            )?;
868                        }
869                    }
870                    _ => {}
871                }
872            }
873            crate::TypeInner::AccelerationStructure => {
874                self.require_any("Acceleration Structure", &[spirv::Capability::RayQueryKHR])?;
875            }
876            crate::TypeInner::RayQuery => {
877                self.require_any("Ray Query", &[spirv::Capability::RayQueryKHR])?;
878            }
879            _ => {}
880        }
881        Ok(())
882    }

884    fn write_type_declaration_local(&mut self, id: Word, local_ty: LocalType) {
885        let instruction = match local_ty {
886            LocalType::Value {
887                vector_size: None,
888                scalar,
889                pointer_space: None,
890            } => self.make_scalar(id, scalar),
891            LocalType::Value {
892                vector_size: Some(size),
893                scalar,
894                pointer_space: None,
895            } => {
896                let scalar_id = self.get_type_id(LookupType::Local(LocalType::Value {
897                    vector_size: None,
898                    scalar,
899                    pointer_space: None,
900                }));
901                Instruction::type_vector(id, scalar_id, size)
902            }
903            LocalType::Matrix {
904                columns,
905                rows,
906                width,
907            } => {
908                let vector_id = self.get_type_id(LookupType::Local(LocalType::Value {
909                    vector_size: Some(rows),
910                    scalar: crate::Scalar::float(width),
911                    pointer_space: None,
912                }));

914            }
915            LocalType::Pointer { base, class } => {
916                let type_id = self.get_type_id(LookupType::Handle(base));
917                Instruction::type_pointer(id, class, type_id)
918            }
919            LocalType::Value {
920                vector_size,
921                scalar,
922                pointer_space: Some(class),
923            } => {
924                let type_id = self.get_type_id(LookupType::Local(LocalType::Value {
925                    vector_size,
926                    scalar,
927                    pointer_space: None,
928                }));
929                Instruction::type_pointer(id, class, type_id)
930            }
931            LocalType::Image(image) => {
932                let local_type = LocalType::Value {
933                    vector_size: None,
934                    scalar: crate::Scalar {
935                        kind: image.sampled_type,
936                        width: 4,
937                    },
938                    pointer_space: None,
939                };
940                let type_id = self.get_type_id(LookupType::Local(local_type));
941                Instruction::type_image(id, type_id, image.dim, image.flags, image.image_format)
942            }
943            LocalType::Sampler => Instruction::type_sampler(id),
944            LocalType::SampledImage { image_type_id } => {
945                Instruction::type_sampled_image(id, image_type_id)
946            }
947            LocalType::BindingArray { base, size } => {
948                let inner_ty = self.get_type_id(LookupType::Handle(base));
949                let scalar_id = self.get_constant_scalar(crate::Literal::U32(size));
950                Instruction::type_array(id, inner_ty, scalar_id)
951            }
952            LocalType::PointerToBindingArray { base, size, space } => {
953                let inner_ty =
954                    self.get_type_id(LookupType::Local(LocalType::BindingArray { base, size }));
955                let class = map_storage_class(space);
956                Instruction::type_pointer(id, class, inner_ty)
957            }
958            LocalType::AccelerationStructure => Instruction::type_acceleration_structure(id),
959            LocalType::RayQuery => Instruction::type_ray_query(id),
960        };
961
962        instruction.to_words(&mut self.logical_layout.declarations);
963    }

965    fn write_type_declaration_arena(
966        &mut self,
967        arena: &UniqueArena<crate::Type>,
968        handle: Handle<crate::Type>,
969    ) -> Result<Word, Error> {
970        let ty = &arena[handle];
971        let id = if let Some(local) = make_local(&ty.inner) {
972            // This type can be represented as a `LocalType`, so check if we've
973            // already written an instruction for it. If not, do so now, with
974            // `write_type_declaration_local`.
975            match self.lookup_type.entry(LookupType::Local(local)) {
976                // We already have an id for this `LocalType`.
977                Entry::Occupied(e) => *e.get(),
978
979                // It's a type we haven't seen before.
980                Entry::Vacant(e) => {
981                    let id = self.id_gen.next();
982                    e.insert(id);
983
984                    self.write_type_declaration_local(id, local);
985
986                    // If it's a type that needs SPIR-V capabilities, request them now,
987                    // so write_type_declaration_local can stay infallible.
988                    self.request_type_capabilities(&ty.inner)?;
989
990                    id
991                }
992            }
993        } else {
994            use spirv::Decoration;
995
996            let id = self.id_gen.next();
997            let instruction = match ty.inner {
998                crate::TypeInner::Array { base, size, stride } => {
999                    self.decorate(id, Decoration::ArrayStride, &[stride]);
1000
1001                    let type_id = self.get_type_id(LookupType::Handle(base));
1002                    match size {
1003                        crate::ArraySize::Constant(length) => {
1004                            let length_id = self.get_index_constant(length.get());
1005                            Instruction::type_array(id, type_id, length_id)
1006                        }
1007                        crate::ArraySize::Dynamic => Instruction::type_runtime_array(id, type_id),
1008                    }
1009                }
1010                crate::TypeInner::BindingArray { base, size } => {
1011                    let type_id = self.get_type_id(LookupType::Handle(base));
1012                    match size {
1013                        crate::ArraySize::Constant(length) => {
1014                            let length_id = self.get_index_constant(length.get());
1015                            Instruction::type_array(id, type_id, length_id)
1016                        }
1017                        crate::ArraySize::Dynamic => Instruction::type_runtime_array(id, type_id),
1018                    }
1019                }
1020                crate::TypeInner::Struct {
1021                    ref members,
1022                    span: _,
1023                } => {
1024                    let mut has_runtime_array = false;
1025                    let mut member_ids = Vec::with_capacity(members.len());
1026                    for (index, member) in members.iter().enumerate() {
1027                        let member_ty = &arena[member.ty];
1028                        match member_ty.inner {
1029                            crate::TypeInner::Array {
1030                                base: _,
1031                                size: crate::ArraySize::Dynamic,
1032                                stride: _,
1033                            } => {
1034                                has_runtime_array = true;
1035                            }
1036                            _ => (),
1037                        }
1038                        self.decorate_struct_member(id, index, member, arena)?;
1039                        let member_id = self.get_type_id(LookupType::Handle(member.ty));
1040                        member_ids.push(member_id);
1041                    }
1042                    if has_runtime_array {
1043                        self.decorate(id, Decoration::Block, &[]);
1044                    }
1045                    Instruction::type_struct(id, member_ids.as_slice())
1046                }
1047
1048                // These all have TypeLocal representations, so they should have been
1049                // handled by `write_type_declaration_local` above.
1050                crate::TypeInner::Scalar(_)
1051                | crate::TypeInner::Atomic(_)
1052                | crate::TypeInner::Vector { .. }
1053                | crate::TypeInner::Matrix { .. }
1054                | crate::TypeInner::Pointer { .. }
1055                | crate::TypeInner::ValuePointer { .. }
1056                | crate::TypeInner::Image { .. }
1057                | crate::TypeInner::Sampler { .. }
1058                | crate::TypeInner::AccelerationStructure
1059                | crate::TypeInner::RayQuery => unreachable!(),
1060            };
1061
1062            instruction.to_words(&mut self.logical_layout.declarations);
1063            id
1064        };
1065
1066        // Add this handle as a new alias for that type.
1067        self.lookup_type.insert(LookupType::Handle(handle), id);
1068
1069        if self.flags.contains(WriterFlags::DEBUG) {
1070            if let Some(ref name) = ty.name {
1071                self.debugs.push(Instruction::name(id, name));
1072            }
1073        }
1074
1075        Ok(id)
1076    }

1251        &mut self,
1252        handle: Handle<crate::Expression>,
1253        ir_module: &crate::Module,
1254        mod_info: &ModuleInfo,
1255    ) -> Result<Word, Error> {
1256        let id = match ir_module.const_expressions[handle] {
1257            crate::Expression::Literal(literal) => self.get_constant_scalar(literal),
1258            crate::Expression::Constant(constant) => {
1259                let constant = &ir_module.constants[constant];
1260                self.constant_ids[constant.init.index()]
1261            }
1262            crate::Expression::ZeroValue(ty) => {
1263                let type_id = self.get_type_id(LookupType::Handle(ty));
1264                self.get_constant_null(type_id)
1265            }
1266            crate::Expression::Compose { ty, ref components } => {
1267                let component_ids: Vec<_> = crate::proc::flatten_compose(
1268                    ty,
1269                    components,
1270                    &ir_module.const_expressions,
1271                    &ir_module.types,
1272                )
1273                .map(|component| self.constant_ids[component.index()])
1274                .collect();
1275                self.get_constant_composite(LookupType::Handle(ty), component_ids.as_slice())
1276            }
1277            crate::Expression::Splat { size, value } => {
1278                let value_id = self.constant_ids[value.index()];
1279                let component_ids = &[value_id; 4][..size as usize];
1280
1281                let ty = self.get_expression_lookup_type(&mod_info[handle]);
1282
1283                self.get_constant_composite(ty, component_ids)
1284            }
1285            _ => unreachable!(),
1286        };
1287
1288        self.constant_ids[handle.index()] = id;
1289
1290        Ok(id)
1291    }

1293    pub(super) fn write_barrier(&mut self, flags: crate::Barrier, block: &mut Block) {
1294        let memory_scope = if flags.contains(crate::Barrier::STORAGE) {
1295            spirv::Scope::Device
1296        } else {
1297            spirv::Scope::Workgroup
1298        };
1299        let mut semantics = spirv::MemorySemantics::ACQUIRE_RELEASE;
1300        semantics.set(
1301            spirv::MemorySemantics::UNIFORM_MEMORY,
1302            flags.contains(crate::Barrier::STORAGE),
1303        );
1304        semantics.set(
1305            spirv::MemorySemantics::WORKGROUP_MEMORY,
1306            flags.contains(crate::Barrier::WORK_GROUP),
1307        );
1308        let exec_scope_id = self.get_index_constant(spirv::Scope::Workgroup as u32);
1309        let mem_scope_id = self.get_index_constant(memory_scope as u32);
1310        let semantics_id = self.get_index_constant(semantics.bits());
1311        block.body.push(Instruction::control_barrier(
1312            exec_scope_id,
1313            mem_scope_id,
1314            semantics_id,
1315        ));

1318    fn generate_workgroup_vars_init_block(
1319        &mut self,
1320        entry_id: Word,
1321        ir_module: &crate::Module,
1322        info: &FunctionInfo,
1323        local_invocation_id: Option<Word>,
1324        interface: &mut FunctionInterface,
1325        function: &mut Function,
1326    ) -> Option<Word> {
1327        let body = ir_module
1328            .global_variables
1329            .iter()
1330            .filter(|&(handle, var)| {
1331                !info[handle].is_empty() && var.space == crate::AddressSpace::WorkGroup
1332            })
1333            .map(|(handle, var)| {
1334                // It's safe to use `var_id` here, not `access_id`, because only
1335                // variables in the `Uniform` and `StorageBuffer` address spaces
1336                // get wrapped, and we're initializing `WorkGroup` variables.
1337                let var_id = self.global_variables[handle.index()].var_id;
1338                let var_type_id = self.get_type_id(LookupType::Handle(var.ty));
1339                let init_word = self.get_constant_null(var_type_id);
1340                Instruction::store(var_id, init_word, None)
1341            })
1342            .collect::<Vec<_>>();
1343
1344        if body.is_empty() {
1345            return None;
1346        }
1347
1348        let uint3_type_id = self.get_uint3_type_id();
1349
1350        let mut pre_if_block = Block::new(entry_id);
1351
1352        let local_invocation_id = if let Some(local_invocation_id) = local_invocation_id {
1353            local_invocation_id
1354        } else {
1355            let varying_id = self.id_gen.next();
1356            let class = spirv::StorageClass::Input;
1357            let pointer_type_id = self.get_uint3_pointer_type_id(class);
1358
1359            Instruction::variable(pointer_type_id, varying_id, class, None)
1360                .to_words(&mut self.logical_layout.declarations);
1361
1362            self.decorate(
1363                varying_id,
1364                spirv::Decoration::BuiltIn,
1365                &[spirv::BuiltIn::LocalInvocationId as u32],
1366            );
1367
1368            interface.varying_ids.push(varying_id);
1369            let id = self.id_gen.next();
1370            pre_if_block
1371                .body
1372                .push(Instruction::load(uint3_type_id, id, varying_id, None));
1373
1374            id
1375        };
1376
1377        let zero_id = self.get_constant_null(uint3_type_id);
1378        let bool3_type_id = self.get_bool3_type_id();
1379
1380        let eq_id = self.id_gen.next();
1381        pre_if_block.body.push(Instruction::binary(
1382            spirv::Op::IEqual,
1383            bool3_type_id,
1384            eq_id,
1385            local_invocation_id,
1386            zero_id,
1387        ));
1388
1389        let condition_id = self.id_gen.next();
1390        let bool_type_id = self.get_bool_type_id();
1391        pre_if_block.body.push(Instruction::relational(
1392            spirv::Op::All,
1393            bool_type_id,
1394            condition_id,
1395            eq_id,
1396        ));
1397
1398        let merge_id = self.id_gen.next();
1399        pre_if_block.body.push(Instruction::selection_merge(
1400            merge_id,
1401            spirv::SelectionControl::NONE,
1402        ));
1403
1404        let accept_id = self.id_gen.next();
1405        function.consume(
1406            pre_if_block,
1407            Instruction::branch_conditional(condition_id, accept_id, merge_id),
1408        );
1409
1410        let accept_block = Block {
1411            label_id: accept_id,
1412            body,
1413        };
1414        function.consume(accept_block, Instruction::branch(merge_id));
1415
1416        let mut post_if_block = Block::new(merge_id);
1417
1418        self.write_barrier(crate::Barrier::WORK_GROUP, &mut post_if_block);
1419
1420        let next_id = self.id_gen.next();
1421        function.consume(post_if_block, Instruction::branch(next_id));
1422        Some(next_id)
1423    }

1425    /// Generate an `OpVariable` for one value in an [`EntryPoint`]'s IO interface.
1426    ///
1427    /// The [`Binding`]s of the arguments and result of an [`EntryPoint`]'s
1428    /// [`Function`] describe a SPIR-V shader interface. In SPIR-V, the
1429    /// interface is represented by global variables in the `Input` and `Output`
1430    /// storage classes, with decorations indicating which builtin or location
1431    /// each variable corresponds to.
1432    ///
1433    /// This function emits a single global `OpVariable` for a single value from
1434    /// the interface, and adds appropriate decorations to indicate which
1435    /// builtin or location it represents, how it should be interpolated, and so
1436    /// on. The `class` argument gives the variable's SPIR-V storage class,
1437    /// which should be either [`Input`] or [`Output`].
1438    ///
1439    /// [`Binding`]: crate::Binding
1440    /// [`Function`]: crate::Function
1441    /// [`EntryPoint`]: crate::EntryPoint
1442    /// [`Input`]: spirv::StorageClass::Input
1443    /// [`Output`]: spirv::StorageClass::Output
1444    fn write_varying(
1445        &mut self,
1446        ir_module: &crate::Module,
1447        stage: crate::ShaderStage,
1448        class: spirv::StorageClass,
1449        debug_name: Option<&str>,
1450        ty: Handle<crate::Type>,
1451        binding: &crate::Binding,
1452    ) -> Result<Word, Error> {
1453        let id = self.id_gen.next();
1454        let pointer_type_id = self.get_pointer_id(&ir_module.types, ty, class)?;
1455        Instruction::variable(pointer_type_id, id, class, None)
1456            .to_words(&mut self.logical_layout.declarations);
1457
1458        if self
1459            .flags
1460            .contains(WriterFlags::DEBUG | WriterFlags::LABEL_VARYINGS)
1461        {
1462            if let Some(name) = debug_name {
1463                self.debugs.push(Instruction::name(id, name));
1464            }
1465        }
1466
1467        use spirv::{BuiltIn, Decoration};
1468
1469        match *binding {
1470            crate::Binding::Location {
1471                location,
1472                interpolation,
1473                sampling,
1474                second_blend_source,
1475            } => {
1476                self.decorate(id, Decoration::Location, &[location]);
1477
1478                let no_decorations =
1479                    // VUID-StandaloneSpirv-Flat-06202
1480                    // > The Flat, NoPerspective, Sample, and Centroid decorations
1481                    // > must not be used on variables with the Input storage class in a vertex shader
1482                    (class == spirv::StorageClass::Input && stage == crate::ShaderStage::Vertex) ||
1483                    // VUID-StandaloneSpirv-Flat-06201
1484                    // > The Flat, NoPerspective, Sample, and Centroid decorations
1485                    // > must not be used on variables with the Output storage class in a fragment shader
1486                    (class == spirv::StorageClass::Output && stage == crate::ShaderStage::Fragment);
1487
1488                if !no_decorations {
1489                    match interpolation {
1490                        // Perspective-correct interpolation is the default in SPIR-V.
1491                        None | Some(crate::Interpolation::Perspective) => (),
1492                        Some(crate::Interpolation::Flat) => {
1493                            self.decorate(id, Decoration::Flat, &[]);
1494                        }
1495                        Some(crate::Interpolation::Linear) => {
1496                            self.decorate(id, Decoration::NoPerspective, &[]);
1497                        }
1498                    }
1499                    match sampling {
1500                        // Center sampling is the default in SPIR-V.
1501                        None | Some(crate::Sampling::Center) => (),
1502                        Some(crate::Sampling::Centroid) => {
1503                            self.decorate(id, Decoration::Centroid, &[]);
1504                        }
1505                        Some(crate::Sampling::Sample) => {
1506                            self.require_any(
1507                                "per-sample interpolation",
1508                                &[spirv::Capability::SampleRateShading],
1509                            )?;
1510                            self.decorate(id, Decoration::Sample, &[]);
1511                        }
1512                    }
1513                }
1514                if second_blend_source {
1515                    self.decorate(id, Decoration::Index, &[1]);
1516                }
1517            }
1518            crate::Binding::BuiltIn(built_in) => {
1519                use crate::BuiltIn as Bi;
1520                let built_in = match built_in {
1521                    Bi::Position { invariant } => {
1522                        if invariant {
1523                            self.decorate(id, Decoration::Invariant, &[]);
1524                        }
1525
1526                        if class == spirv::StorageClass::Output {
1527                            BuiltIn::Position
1528                        } else {
1529                            BuiltIn::FragCoord
1530                        }
1531                    }
1532                    Bi::ViewIndex => {
1533                        self.require_any("`view_index` built-in", &[spirv::Capability::MultiView])?;
1534                        BuiltIn::ViewIndex
1535                    }
1536                    // vertex
1537                    Bi::BaseInstance => BuiltIn::BaseInstance,
1538                    Bi::BaseVertex => BuiltIn::BaseVertex,
1539                    Bi::ClipDistance => {
1540                        self.require_any(
1541                            "`clip_distance` built-in",
1542                            &[spirv::Capability::ClipDistance],
1543                        )?;
1544                        BuiltIn::ClipDistance
1545                    }
1546                    Bi::CullDistance => {
1547                        self.require_any(
1548                            "`cull_distance` built-in",
1549                            &[spirv::Capability::CullDistance],
1550                        )?;
1551                        BuiltIn::CullDistance
1552                    }
1553                    Bi::InstanceIndex => BuiltIn::InstanceIndex,
1554                    Bi::PointSize => BuiltIn::PointSize,
1555                    Bi::VertexIndex => BuiltIn::VertexIndex,
1556                    // fragment
1557                    Bi::FragDepth => BuiltIn::FragDepth,
1558                    Bi::PointCoord => BuiltIn::PointCoord,
1559                    Bi::FrontFacing => BuiltIn::FrontFacing,
1560                    Bi::PrimitiveIndex => {
1561                        self.require_any(
1562                            "`primitive_index` built-in",
1563                            &[spirv::Capability::Geometry],
1564                        )?;
1565                        BuiltIn::PrimitiveId
1566                    }
1567                    Bi::SampleIndex => {
1568                        self.require_any(
1569                            "`sample_index` built-in",
1570                            &[spirv::Capability::SampleRateShading],
1571                        )?;
1572
1573                        BuiltIn::SampleId
1574                    }
1575                    Bi::SampleMask => BuiltIn::SampleMask,
1576                    // compute
1577                    Bi::GlobalInvocationId => BuiltIn::GlobalInvocationId,
1578                    Bi::LocalInvocationId => BuiltIn::LocalInvocationId,
1579                    Bi::LocalInvocationIndex => BuiltIn::LocalInvocationIndex,
1580                    Bi::WorkGroupId => BuiltIn::WorkgroupId,
1581                    Bi::WorkGroupSize => BuiltIn::WorkgroupSize,
1582                    Bi::NumWorkGroups => BuiltIn::NumWorkgroups,
1583                };
1584
1585                self.decorate(id, Decoration::BuiltIn, &[built_in as u32]);
1586
1587                use crate::ScalarKind as Sk;
1588
1589                // Per the Vulkan spec, `VUID-StandaloneSpirv-Flat-04744`:
1590                //
1591                // > Any variable with integer or double-precision floating-
1592                // > point type and with Input storage class in a fragment
1593                // > shader, must be decorated Flat
1594                if class == spirv::StorageClass::Input && stage == crate::ShaderStage::Fragment {
1595                    let is_flat = match ir_module.types[ty].inner {
1596                        crate::TypeInner::Scalar(scalar)
1597                        | crate::TypeInner::Vector { scalar, .. } => match scalar.kind {
1598                            Sk::Uint | Sk::Sint | Sk::Bool => true,
1599                            Sk::Float => false,
1600                            Sk::AbstractInt | Sk::AbstractFloat => {
1601                                return Err(Error::Validation(
1602                                    "Abstract types should not appear in IR presented to backends",
1603                                ))
1604                            }
1605                        },
1606                        _ => false,
1607                    };
1608
1609                    if is_flat {
1610                        self.decorate(id, Decoration::Flat, &[]);
1611                    }
1612                }
1613            }
1614        }
1615
1616        Ok(id)
1617    }

1619    fn write_global_variable(
1620        &mut self,
1621        ir_module: &crate::Module,
1622        global_variable: &crate::GlobalVariable,
1623    ) -> Result<Word, Error> {
1624        use spirv::Decoration;
1625
1626        let id = self.id_gen.next();
1627        let class = map_storage_class(global_variable.space);
1628
1629        //self.check(class.required_capabilities())?;
1630
1631        if self.flags.contains(WriterFlags::DEBUG) {
1632            if let Some(ref name) = global_variable.name {
1633                self.debugs.push(Instruction::name(id, name));
1634            }
1635        }
1636
1637        let storage_access = match global_variable.space {
1638            crate::AddressSpace::Storage { access } => Some(access),
1639            _ => match ir_module.types[global_variable.ty].inner {
1640                crate::TypeInner::Image {
1641                    class: crate::ImageClass::Storage { access, .. },
1642                    ..
1643                } => Some(access),
1644                _ => None,
1645            },
1646        };
1647        if let Some(storage_access) = storage_access {
1648            if !storage_access.contains(crate::StorageAccess::LOAD) {
1649                self.decorate(id, Decoration::NonReadable, &[]);
1650            }
1651            if !storage_access.contains(crate::StorageAccess::STORE) {
1652                self.decorate(id, Decoration::NonWritable, &[]);
1653            }
1654        }
1655
1656        // Note: we should be able to substitute `binding_array<Foo, 0>`,
1657        // but there is still code that tries to register the pre-substituted type,
1658        // and it is failing on 0.
1659        let mut substitute_inner_type_lookup = None;
1660        if let Some(ref res_binding) = global_variable.binding {
1661            self.decorate(id, Decoration::DescriptorSet, &[res_binding.group]);
1662            self.decorate(id, Decoration::Binding, &[res_binding.binding]);
1663
1664            if let Some(&BindingInfo {
1665                binding_array_size: Some(remapped_binding_array_size),
1666            }) = self.binding_map.get(res_binding)
1667            {
1668                if let crate::TypeInner::BindingArray { base, .. } =
1669                    ir_module.types[global_variable.ty].inner
1670                {
1671                    substitute_inner_type_lookup =
1672                        Some(LookupType::Local(LocalType::PointerToBindingArray {
1673                            base,
1674                            size: remapped_binding_array_size,
1675                            space: global_variable.space,
1676                        }))
1677                }
1678            }
1679        };
1680
1681        let init_word = global_variable
1682            .init
1683            .map(|constant| self.constant_ids[constant.index()]);
1684        let inner_type_id = self.get_type_id(
1685            substitute_inner_type_lookup.unwrap_or(LookupType::Handle(global_variable.ty)),
1686        );
1687
1688        // generate the wrapping structure if needed
1689        let pointer_type_id = if global_needs_wrapper(ir_module, global_variable) {
1690            let wrapper_type_id = self.id_gen.next();
1691
1692            self.decorate(wrapper_type_id, Decoration::Block, &[]);
1693            let member = crate::StructMember {
1694                name: None,
1695                ty: global_variable.ty,
1696                binding: None,
1697                offset: 0,
1698            };
1699            self.decorate_struct_member(wrapper_type_id, 0, &member, &ir_module.types)?;
1700
1701            Instruction::type_struct(wrapper_type_id, &[inner_type_id])
1702                .to_words(&mut self.logical_layout.declarations);
1703
1704            let pointer_type_id = self.id_gen.next();
1705            Instruction::type_pointer(pointer_type_id, class, wrapper_type_id)
1706                .to_words(&mut self.logical_layout.declarations);
1707
1708            pointer_type_id
1709        } else {
1710            // This is a global variable in the Storage address space. The only
1711            // way it could have `global_needs_wrapper() == false` is if it has
1712            // a runtime-sized or binding array.
1713            // Runtime-sized arrays were decorated when iterating through struct content.
1714            // Now binding arrays require Block decorating.
1715            if let crate::AddressSpace::Storage { .. } = global_variable.space {
1716                match ir_module.types[global_variable.ty].inner {
1717                    crate::TypeInner::BindingArray { base, .. } => {
1718                        let decorated_id = self.get_type_id(LookupType::Handle(base));
1719                        self.decorate(decorated_id, Decoration::Block, &[]);
1720                    }
1721                    _ => (),
1722                };
1723            }
1724            if substitute_inner_type_lookup.is_some() {
1725                inner_type_id
1726            } else {
1727                self.get_pointer_id(&ir_module.types, global_variable.ty, class)?
1728            }
1729        };
1730
1731        let init_word = match (global_variable.space, self.zero_initialize_workgroup_memory) {
1732            (crate::AddressSpace::Private, _)
1733            | (crate::AddressSpace::WorkGroup, super::ZeroInitializeWorkgroupMemoryMode::Native) => {
1734                init_word.or_else(|| Some(self.get_constant_null(inner_type_id)))
1735            }
1736            _ => init_word,
1737        };
1738
1739        Instruction::variable(pointer_type_id, id, class, init_word)
1740            .to_words(&mut self.logical_layout.declarations);
1741        Ok(id)
1742    }

1825    fn write_logical_layout(
1826        &mut self,
1827        ir_module: &crate::Module,

1833            ir_module: &crate::Module,
1834            binding: Option<&crate::Binding>,
1835            ty: Handle<crate::Type>,
1836        ) -> bool {
1837            match ir_module.types[ty].inner {
1838                crate::TypeInner::Struct { ref members, .. } => members.iter().any(|member| {
1839                    has_view_index_check(ir_module, member.binding.as_ref(), member.ty)
1840                }),
1841                _ => binding == Some(&crate::Binding::BuiltIn(crate::BuiltIn::ViewIndex)),
1842            }

1846            ir_module
1847                .global_variables
1848                .iter()
1849                .any(|(_, var)| match var.space {
1850                    crate::AddressSpace::Storage { .. } => true,
1851                    _ => false,
1852                });
1853        let has_view_index = ir_module
1854            .entry_points
1855            .iter()
1856            .flat_map(|entry| entry.function.arguments.iter())
1857            .any(|arg| has_view_index_check(ir_module, arg.binding.as_ref(), arg.ty));
1858        let has_ray_query = ir_module.special_types.ray_desc.is_some()
1859            | ir_module.special_types.ray_intersection.is_some();
1860

1862            // enable the storage buffer class on < SPV-1.3
1863            Instruction::extension("SPV_KHR_storage_buffer_storage_class")
1864                .to_words(&mut self.logical_layout.extensions);
1865        }
1866        if has_view_index {
1867            Instruction::extension("SPV_KHR_multiview")
1868                .to_words(&mut self.logical_layout.extensions)
1869        }
1870        if has_ray_query {
1871            Instruction::extension("SPV_KHR_ray_query")
1872                .to_words(&mut self.logical_layout.extensions)
1873        }
1874        Instruction::type_void(self.void_type).to_words(&mut self.logical_layout.declarations);
1875        Instruction::ext_inst_import(self.gl450_ext_inst_id, "GLSL.std.450")
1876            .to_words(&mut self.logical_layout.ext_inst_imports);
1877
1878        let mut debug_info_inner = None;
1879        if self.flags.contains(WriterFlags::DEBUG) {
1880            if let Some(debug_info) = debug_info.as_ref() {
1881                let source_file_id = self.id_gen.next();
1882                self.debugs.push(Instruction::string(
1883                    &debug_info.file_name.display().to_string(),
1884                    source_file_id,
1885                ));
1886
1887                debug_info_inner = Some(DebugInfoInner {
1888                    source_code: debug_info.source_code,
1889                    source_file_id,
1890                });
1891                self.debugs.push(Instruction::source(
1892                    spirv::SourceLanguage::Unknown,
1893                    0,
1894                    &debug_info_inner,
1895                ));
1896            }
1897        }
1898
1899        // write all types
1900        for (handle, _) in ir_module.types.iter() {
1901            self.write_type_declaration_arena(&ir_module.types, handle)?;
1902        }
1903
1904        // write all const-expressions as constants
1905        self.constant_ids
1906            .resize(ir_module.const_expressions.len(), 0);
1907        for (handle, _) in ir_module.const_expressions.iter() {
1908            self.write_constant_expr(handle, ir_module, mod_info)?;
1909        }
1910        debug_assert!(self.constant_ids.iter().all(|&id| id != 0));
1911
1912        // write the name of constants on their respective const-expression initializer
1913        if self.flags.contains(WriterFlags::DEBUG) {
1914            for (_, constant) in ir_module.constants.iter() {
1915                if let Some(ref name) = constant.name {
1916                    let id = self.constant_ids[constant.init.index()];
1917                    self.debugs.push(Instruction::name(id, name));
1918                }
1919            }
1920        }
1921
1922        // write all global variables
1923        for (handle, var) in ir_module.global_variables.iter() {
1924            // If a single entry point was specified, only write `OpVariable` instructions
1925            // for the globals it actually uses. Emit dummies for the others,
1926            // to preserve the indices in `global_variables`.
1927            let gvar = match ep_index {
1928                Some(index) if mod_info.get_entry_point(index)[handle].is_empty() => {
1929                    GlobalVariable::dummy()
1930                }
1931                _ => {
1932                    let id = self.write_global_variable(ir_module, var)?;
1933                    GlobalVariable::new(id)
1934                }
1935            };
1936            self.global_variables.push(gvar);
1937        }
1938
1939        // write all functions
1940        for (handle, ir_function) in ir_module.functions.iter() {
1941            let info = &mod_info[handle];
1942            if let Some(index) = ep_index {
1943                let ep_info = mod_info.get_entry_point(index);
1944                // If this function uses globals that we omitted from the SPIR-V
1945                // because the entry point and its callees didn't use them,
1946                // then we must skip it.
1947                if !ep_info.dominates_global_use(info) {
1948                    log::info!("Skip function {:?}", ir_function.name);
1949                    continue;
1950                }
1951
1952                // Skip functions that that are not compatible with this entry point's stage.
1953                //
1954                // When validation is enabled, it rejects modules whose entry points try to call
1955                // incompatible functions, so if we got this far, then any functions incompatible
1956                // with our selected entry point must not be used.
1957                //
1958                // When validation is disabled, `fun_info.available_stages` is always just
1959                // `ShaderStages::all()`, so this will write all functions in the module, and
1960                // the downstream GLSL compiler will catch any problems.
1961                if !info.available_stages.contains(ep_info.available_stages) {
1962                    continue;
1963                }
1964            }
1965            let id = self.write_function(ir_function, info, ir_module, None, &debug_info_inner)?;
1966            self.lookup_function.insert(handle, id);
1967        }
1968
1969        // write all or one entry points
1970        for (index, ir_ep) in ir_module.entry_points.iter().enumerate() {
1971            if ep_index.is_some() && ep_index != Some(index) {
1972                continue;
1973            }
1974            let info = mod_info.get_entry_point(index);
1975            let ep_instruction =
1976                self.write_entry_point(ir_ep, info, ir_module, &debug_info_inner)?;
1977            ep_instruction.to_words(&mut self.logical_layout.entry_points);
1978        }
1979
1980        for capability in self.capabilities_used.iter() {
1981            Instruction::capability(*capability).to_words(&mut self.logical_layout.capabilities);
1982        }
1983        for extension in self.extensions_used.iter() {
1984            Instruction::extension(extension).to_words(&mut self.logical_layout.extensions);
1985        }
1986        if ir_module.entry_points.is_empty() {
1987            // SPIR-V doesn't like modules without entry points
1988            Instruction::capability(spirv::Capability::Linkage)
1989                .to_words(&mut self.logical_layout.capabilities);
1990        }
1991
1992        let addressing_model = spirv::AddressingModel::Logical;
1993        let memory_model = spirv::MemoryModel::GLSL450;
1994        //self.check(addressing_model.required_capabilities())?;
1995        //self.check(memory_model.required_capabilities())?;
1996
1997        Instruction::memory_model(addressing_model, memory_model)
1998            .to_words(&mut self.logical_layout.memory_model);
1999
2000        if self.flags.contains(WriterFlags::DEBUG) {
2001            for debug in self.debugs.iter() {
2002                debug.to_words(&mut self.logical_layout.debugs);
2003            }
2004        }
2005
2006        for annotation in self.annotations.iter() {
2007            annotation.to_words(&mut self.logical_layout.annotations);
2008        }
2009
2010        Ok(())
2011    }

2013    pub fn write(
2014        &mut self,
2015        ir_module: &crate::Module,
2016        info: &ModuleInfo,
2017        pipeline_options: Option<&PipelineOptions>,
2018        debug_info: &Option<DebugInfo>,
2019        words: &mut Vec<Word>,
2020    ) -> Result<(), Error> {
2021        self.reset();
2022
2023        // Try to find the entry point and corresponding index
2024        let ep_index = match pipeline_options {
2025            Some(po) => {
2026                let index = ir_module
2027                    .entry_points
2028                    .iter()
2029                    .position(|ep| po.shader_stage == ep.stage && po.entry_point == ep.name)
2030                    .ok_or(Error::EntryPointNotFound)?;
2031                Some(index)
2032            }
2033            None => None,
2034        };
2035
2036        self.write_logical_layout(ir_module, info, ep_index, debug_info)?;
2037        self.write_physical_layout();
2038
2039        self.physical_layout.in_words(words);
2040        self.logical_layout.in_words(words);
2041        Ok(())
2042    }

2050        self.require_any("NonUniformEXT", &[spirv::Capability::ShaderNonUniform])?;
