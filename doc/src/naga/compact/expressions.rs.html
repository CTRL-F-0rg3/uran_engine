4pub struct ExpressionTracer<'tracer> {
5    pub constants: &'tracer Arena<crate::Constant>,
6
7    /// The arena in which we are currently tracing expressions.
8    pub expressions: &'tracer Arena<crate::Expression>,
9
10    /// The used map for `types`.
11    pub types_used: &'tracer mut HandleSet<crate::Type>,
12
13    /// The used map for `constants`.
14    pub constants_used: &'tracer mut HandleSet<crate::Constant>,
15
16    /// The used set for `arena`.
17    ///
18    /// This points to whatever arena holds the expressions we are
19    /// currently tracing: either a function's expression arena, or
20    /// the module's constant expression arena.
21    pub expressions_used: &'tracer mut HandleSet<crate::Expression>,
22
23    /// The used set for the module's `const_expressions` arena.
24    ///
25    /// If `None`, we are already tracing the constant expressions,
26    /// and `expressions_used` already refers to their handle set.
27    pub const_expressions_used: Option<&'tracer mut HandleSet<crate::Expression>>,
28}

30impl<'tracer> ExpressionTracer<'tracer> {
31    /// Propagate usage through `self.expressions`, starting with `self.expressions_used`.
32    ///
33    /// Treat `self.expressions_used` as the initial set of "known
34    /// live" expressions, and follow through to identify all
35    /// transitively used expressions.
36    ///
37    /// Mark types, constants, and constant expressions used directly
38    /// by `self.expressions` as used. Items used indirectly are not
39    /// marked.
40    ///
41    /// [fe]: crate::Function::expressions
42    /// [ce]: crate::Module::const_expressions
43    pub fn trace_expressions(&mut self) {
44        log::trace!(
45            "entering trace_expression of {}",
46            if self.const_expressions_used.is_some() {
47                "function expressions"
48            } else {
49                "const expressions"
50            }
51        );
52
53        // We don't need recursion or a work list. Because an
54        // expression may only refer to other expressions that precede
55        // it in the arena, it suffices to make a single pass over the
56        // arena from back to front, marking the referents of used
57        // expressions as used themselves.
58        for (handle, expr) in self.expressions.iter().rev() {
59            // If this expression isn't used, it doesn't matter what it uses.
60            if !self.expressions_used.contains(handle) {
61                continue;
62            }
63
64            log::trace!("tracing new expression {:?}", expr);
65
66            use crate::Expression as Ex;
67            match *expr {
68                // Expressions that do not contain handles that need to be traced.
69                Ex::Literal(_)
70                | Ex::FunctionArgument(_)
71                | Ex::GlobalVariable(_)
72                | Ex::LocalVariable(_)
73                | Ex::CallResult(_)
74                | Ex::RayQueryProceedResult => {}
75
76                Ex::Constant(handle) => {
77                    self.constants_used.insert(handle);
78                    // Constants and expressions are mutually recursive, which
79                    // complicates our nice one-pass algorithm. However, since
80                    // constants don't refer to each other, we can get around
81                    // this by looking *through* each constant and marking its
82                    // initializer as used. Since `expr` refers to the constant,
83                    // and the constant refers to the initializer, it must
84                    // precede `expr` in the arena.
85                    let init = self.constants[handle].init;
86                    match self.const_expressions_used {
87                        Some(ref mut used) => used.insert(init),
88                        None => self.expressions_used.insert(init),
89                    }
90                }
91                Ex::ZeroValue(ty) => self.types_used.insert(ty),
92                Ex::Compose { ty, ref components } => {
93                    self.types_used.insert(ty);
94                    self.expressions_used
95                        .insert_iter(components.iter().cloned());
96                }
97                Ex::Access { base, index } => self.expressions_used.insert_iter([base, index]),
98                Ex::AccessIndex { base, index: _ } => self.expressions_used.insert(base),
99                Ex::Splat { size: _, value } => self.expressions_used.insert(value),
100                Ex::Swizzle {
101                    size: _,
102                    vector,
103                    pattern: _,
104                } => self.expressions_used.insert(vector),
105                Ex::Load { pointer } => self.expressions_used.insert(pointer),
106                Ex::ImageSample {
107                    image,
108                    sampler,
109                    gather: _,
110                    coordinate,
111                    array_index,
112                    offset,
113                    ref level,
114                    depth_ref,
115                } => {
116                    self.expressions_used
117                        .insert_iter([image, sampler, coordinate]);
118                    self.expressions_used.insert_iter(array_index);
119                    match self.const_expressions_used {
120                        Some(ref mut used) => used.insert_iter(offset),
121                        None => self.expressions_used.insert_iter(offset),
122                    }
123                    use crate::SampleLevel as Sl;
124                    match *level {
125                        Sl::Auto | Sl::Zero => {}
126                        Sl::Exact(expr) | Sl::Bias(expr) => self.expressions_used.insert(expr),
127                        Sl::Gradient { x, y } => self.expressions_used.insert_iter([x, y]),
128                    }
129                    self.expressions_used.insert_iter(depth_ref);
130                }
131                Ex::ImageLoad {
132                    image,
133                    coordinate,
134                    array_index,
135                    sample,
136                    level,
137                } => {
138                    self.expressions_used.insert(image);
139                    self.expressions_used.insert(coordinate);
140                    self.expressions_used.insert_iter(array_index);
141                    self.expressions_used.insert_iter(sample);
142                    self.expressions_used.insert_iter(level);
143                }
144                Ex::ImageQuery { image, ref query } => {
145                    self.expressions_used.insert(image);
146                    use crate::ImageQuery as Iq;
147                    match *query {
148                        Iq::Size { level } => self.expressions_used.insert_iter(level),
149                        Iq::NumLevels | Iq::NumLayers | Iq::NumSamples => {}
150                    }
151                }
152                Ex::Unary { op: _, expr } => self.expressions_used.insert(expr),
153                Ex::Binary { op: _, left, right } => {
154                    self.expressions_used.insert_iter([left, right]);
155                }
156                Ex::Select {
157                    condition,
158                    accept,
159                    reject,
160                } => self
161                    .expressions_used
162                    .insert_iter([condition, accept, reject]),
163                Ex::Derivative {
164                    axis: _,
165                    ctrl: _,
166                    expr,
167                } => self.expressions_used.insert(expr),
168                Ex::Relational { fun: _, argument } => self.expressions_used.insert(argument),
169                Ex::Math {
170                    fun: _,
171                    arg,
172                    arg1,
173                    arg2,
174                    arg3,
175                } => {
176                    self.expressions_used.insert(arg);
177                    self.expressions_used.insert_iter(arg1);
178                    self.expressions_used.insert_iter(arg2);
179                    self.expressions_used.insert_iter(arg3);
180                }
181                Ex::As {
182                    expr,
183                    kind: _,
184                    convert: _,
185                } => self.expressions_used.insert(expr),
186                Ex::AtomicResult { ty, comparison: _ } => self.types_used.insert(ty),
187                Ex::WorkGroupUniformLoadResult { ty } => self.types_used.insert(ty),
188                Ex::ArrayLength(expr) => self.expressions_used.insert(expr),
189                Ex::RayQueryGetIntersection {
190                    query,
191                    committed: _,
192                } => self.expressions_used.insert(query),
193            }
194        }
195    }
196}

198impl ModuleMap {
199    /// Fix up all handles in `expr`.
200    ///
201    /// Use the expression handle remappings in `operand_map`, and all
202    /// other mappings from `self`.
203    pub fn adjust_expression(
204        &self,
205        expr: &mut crate::Expression,
206        operand_map: &HandleMap<crate::Expression>,
207    ) {
208        let adjust = |expr: &mut Handle<crate::Expression>| {
209            operand_map.adjust(expr);
210        };
211
212        use crate::Expression as Ex;
213        match *expr {
214            // Expressions that do not contain handles that need to be adjusted.
215            Ex::Literal(_)
216            | Ex::FunctionArgument(_)
217            | Ex::GlobalVariable(_)
218            | Ex::LocalVariable(_)
219            | Ex::CallResult(_)
220            | Ex::RayQueryProceedResult => {}
221
222            // Expressions that contain handles that need to be adjusted.
223            Ex::Constant(ref mut constant) => self.constants.adjust(constant),
224            Ex::ZeroValue(ref mut ty) => self.types.adjust(ty),
225            Ex::Compose {
226                ref mut ty,
227                ref mut components,
228            } => {
229                self.types.adjust(ty);
230                for component in components {
231                    adjust(component);
232                }
233            }
234            Ex::Access {
235                ref mut base,
236                ref mut index,
237            } => {
238                adjust(base);
239                adjust(index);
240            }
241            Ex::AccessIndex {
242                ref mut base,
243                index: _,
244            } => adjust(base),
245            Ex::Splat {
246                size: _,
247                ref mut value,
248            } => adjust(value),
249            Ex::Swizzle {
250                size: _,
251                ref mut vector,
252                pattern: _,
253            } => adjust(vector),
254            Ex::Load { ref mut pointer } => adjust(pointer),
255            Ex::ImageSample {
256                ref mut image,
257                ref mut sampler,
258                gather: _,
259                ref mut coordinate,
260                ref mut array_index,
261                ref mut offset,
262                ref mut level,
263                ref mut depth_ref,
264            } => {
265                adjust(image);
266                adjust(sampler);
267                adjust(coordinate);
268                operand_map.adjust_option(array_index);
269                if let Some(ref mut offset) = *offset {
270                    self.const_expressions.adjust(offset);
271                }
272                self.adjust_sample_level(level, operand_map);
273                operand_map.adjust_option(depth_ref);
274            }
275            Ex::ImageLoad {
276                ref mut image,
277                ref mut coordinate,
278                ref mut array_index,
279                ref mut sample,
280                ref mut level,
281            } => {
282                adjust(image);
283                adjust(coordinate);
284                operand_map.adjust_option(array_index);
285                operand_map.adjust_option(sample);
286                operand_map.adjust_option(level);
287            }
288            Ex::ImageQuery {
289                ref mut image,
290                ref mut query,
291            } => {
292                adjust(image);
293                self.adjust_image_query(query, operand_map);
294            }
295            Ex::Unary {
296                op: _,
297                ref mut expr,
298            } => adjust(expr),
299            Ex::Binary {
300                op: _,
301                ref mut left,
302                ref mut right,
303            } => {
304                adjust(left);
305                adjust(right);
306            }
307            Ex::Select {
308                ref mut condition,
309                ref mut accept,
310                ref mut reject,
311            } => {
312                adjust(condition);
313                adjust(accept);
314                adjust(reject);
315            }
316            Ex::Derivative {
317                axis: _,
318                ctrl: _,
319                ref mut expr,
320            } => adjust(expr),
321            Ex::Relational {
322                fun: _,
323                ref mut argument,
324            } => adjust(argument),
325            Ex::Math {
326                fun: _,
327                ref mut arg,
328                ref mut arg1,
329                ref mut arg2,
330                ref mut arg3,
331            } => {
332                adjust(arg);
333                operand_map.adjust_option(arg1);
334                operand_map.adjust_option(arg2);
335                operand_map.adjust_option(arg3);
336            }
337            Ex::As {
338                ref mut expr,
339                kind: _,
340                convert: _,
341            } => adjust(expr),
342            Ex::AtomicResult {
343                ref mut ty,
344                comparison: _,
345            } => self.types.adjust(ty),
346            Ex::WorkGroupUniformLoadResult { ref mut ty } => self.types.adjust(ty),
347            Ex::ArrayLength(ref mut expr) => adjust(expr),
348            Ex::RayQueryGetIntersection {
349                ref mut query,
350                committed: _,
351            } => adjust(query),
352        }
353    }
354
355    fn adjust_sample_level(
356        &self,
357        level: &mut crate::SampleLevel,
358        operand_map: &HandleMap<crate::Expression>,
359    ) {
360        let adjust = |expr: &mut Handle<crate::Expression>| operand_map.adjust(expr);
361
362        use crate::SampleLevel as Sl;
363        match *level {
364            Sl::Auto | Sl::Zero => {}
365            Sl::Exact(ref mut expr) => adjust(expr),
366            Sl::Bias(ref mut expr) => adjust(expr),
367            Sl::Gradient {
368                ref mut x,
369                ref mut y,
370            } => {
371                adjust(x);
372                adjust(y);
373            }
374        }
375    }
376
377    fn adjust_image_query(
378        &self,
379        query: &mut crate::ImageQuery,
380        operand_map: &HandleMap<crate::Expression>,
381    ) {
382        use crate::ImageQuery as Iq;
383
384        match *query {
385            Iq::Size { ref mut level } => operand_map.adjust_option(level),
386            Iq::NumLevels | Iq::NumLayers | Iq::NumSamples => {}
387        }
388    }
