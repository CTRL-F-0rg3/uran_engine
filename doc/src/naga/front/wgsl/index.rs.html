5/// A `GlobalDecl` list in which each definition occurs before all its uses.
6pub struct Index<'a> {
7    dependency_order: Vec<Handle<ast::GlobalDecl<'a>>>,
8}
9
10impl<'a> Index<'a> {
11    /// Generate an `Index` for the given translation unit.
12    ///
13    /// Perform a topological sort on `tu`'s global declarations, placing
14    /// referents before the definitions that refer to them.
15    ///
16    /// Return an error if the graph of references between declarations contains
17    /// any cycles.
18    pub fn generate(tu: &ast::TranslationUnit<'a>) -> Result<Self, Error<'a>> {
19        // Produce a map from global definitions' names to their `Handle<GlobalDecl>`s.
20        // While doing so, reject conflicting definitions.
21        let mut globals = FastHashMap::with_capacity_and_hasher(tu.decls.len(), Default::default());
22        for (handle, decl) in tu.decls.iter() {
23            let ident = decl_ident(decl);
24            let name = ident.name;
25            if let Some(old) = globals.insert(name, handle) {
26                return Err(Error::Redefinition {
27                    previous: decl_ident(&tu.decls[old]).span,
28                    current: ident.span,
29                });
30            }
31        }
32
33        let len = tu.decls.len();
34        let solver = DependencySolver {
35            globals: &globals,
36            module: tu,
37            visited: vec![false; len],
38            temp_visited: vec![false; len],
39            path: Vec::new(),
40            out: Vec::with_capacity(len),
41        };
42        let dependency_order = solver.solve()?;
43
44        Ok(Self { dependency_order })
45    }
46
47    /// Iterate over `GlobalDecl`s, visiting each definition before all its uses.
48    ///
49    /// Produce handles for all of the `GlobalDecl`s of the `TranslationUnit`
50    /// passed to `Index::generate`, ordered so that a given declaration is
51    /// produced before any other declaration that uses it.
52    pub fn visit_ordered(&self) -> impl Iterator<Item = Handle<ast::GlobalDecl<'a>>> + '_ {
53        self.dependency_order.iter().copied()
54    }
55}

57/// An edge from a reference to its referent in the current depth-first
58/// traversal.
59///
60/// This is like `ast::Dependency`, except that we've determined which
61/// `GlobalDecl` it refers to.
62struct ResolvedDependency<'a> {
63    /// The referent of some identifier used in the current declaration.
64    decl: Handle<ast::GlobalDecl<'a>>,
65
66    /// Where that use occurs within the current declaration.
67    usage: Span,
68}

70/// Local state for ordering a `TranslationUnit`'s module-scope declarations.
71///
72/// Values of this type are used temporarily by `Index::generate`
73/// to perform a depth-first sort on the declarations.
74/// Technically, what we want is a topological sort, but a depth-first sort
75/// has one key benefit - it's much more efficient in storing
76/// the path of each node for error generation.
77struct DependencySolver<'source, 'temp> {
78    /// A map from module-scope definitions' names to their handles.
79    globals: &'temp FastHashMap<&'source str, Handle<ast::GlobalDecl<'source>>>,
80
81    /// The translation unit whose declarations we're ordering.
82    module: &'temp ast::TranslationUnit<'source>,
83
84    /// For each handle, whether we have pushed it onto `out` yet.
85    visited: Vec<bool>,
86
87    /// For each handle, whether it is an predecessor in the current depth-first
88    /// traversal. This is used to detect cycles in the reference graph.
89    temp_visited: Vec<bool>,
90
91    /// The current path in our depth-first traversal. Used for generating
92    /// error messages for non-trivial reference cycles.
93    path: Vec<ResolvedDependency<'source>>,
94
95    /// The list of declaration handles, with declarations before uses.
96    out: Vec<Handle<ast::GlobalDecl<'source>>>,
97}
98
99impl<'a> DependencySolver<'a, '_> {
100    /// Produce the sorted list of declaration handles, and check for cycles.
101    fn solve(mut self) -> Result<Vec<Handle<ast::GlobalDecl<'a>>>, Error<'a>> {
102        for (id, _) in self.module.decls.iter() {
103            if self.visited[id.index()] {
104                continue;
105            }
106
107            self.dfs(id)?;
108        }
109
110        Ok(self.out)
111    }
112
113    /// Ensure that all declarations used by `id` have been added to the
114    /// ordering, and then append `id` itself.
115    fn dfs(&mut self, id: Handle<ast::GlobalDecl<'a>>) -> Result<(), Error<'a>> {
116        let decl = &self.module.decls[id];
117        let id_usize = id.index();
118
119        self.temp_visited[id_usize] = true;
120        for dep in decl.dependencies.iter() {
121            if let Some(&dep_id) = self.globals.get(dep.ident) {
122                self.path.push(ResolvedDependency {
123                    decl: dep_id,
124                    usage: dep.usage,
125                });
126                let dep_id_usize = dep_id.index();
127
128                if self.temp_visited[dep_id_usize] {
129                    // Found a cycle.
130                    return if dep_id == id {
131                        // A declaration refers to itself directly.
132                        Err(Error::RecursiveDeclaration {
133                            ident: decl_ident(decl).span,
134                            usage: dep.usage,
135                        })
136                    } else {
137                        // A declaration refers to itself indirectly, through
138                        // one or more other definitions. Report the entire path
139                        // of references.
140                        let start_at = self
141                            .path
142                            .iter()
143                            .rev()
144                            .enumerate()
145                            .find_map(|(i, dep)| (dep.decl == dep_id).then_some(i))
146                            .unwrap_or(0);
147
148                        Err(Error::CyclicDeclaration {
149                            ident: decl_ident(&self.module.decls[dep_id]).span,
150                            path: self.path[start_at..]
151                                .iter()
152                                .map(|curr_dep| {
153                                    let curr_id = curr_dep.decl;
154                                    let curr_decl = &self.module.decls[curr_id];
155
156                                    (decl_ident(curr_decl).span, curr_dep.usage)
157                                })
158                                .collect(),
159                        })
160                    };
161                } else if !self.visited[dep_id_usize] {
162                    self.dfs(dep_id)?;
163                }
164
165                // Remove this edge from the current path.
166                self.path.pop();
167            }
168
169            // Ignore unresolved identifiers; they may be predeclared objects.
170        }
171
172        // Remove this node from the current path.
173        self.temp_visited[id_usize] = false;
174
175        // Now everything this declaration uses has been visited, and is already
176        // present in `out`. That means we we can append this one to the
177        // ordering, and mark it as visited.
178        self.out.push(id);
179        self.visited[id_usize] = true;
180
181        Ok(())
182    }
183}

185const fn decl_ident<'a>(decl: &ast::GlobalDecl<'a>) -> ast::Ident<'a> {
186    match decl.kind {
187        ast::GlobalDeclKind::Fn(ref f) => f.name,
188        ast::GlobalDeclKind::Var(ref v) => v.name,
189        ast::GlobalDeclKind::Const(ref c) => c.name,
190        ast::GlobalDeclKind::Struct(ref s) => s.name,
191        ast::GlobalDeclKind::Type(ref t) => t.name,
192    }
193}
