152impl PartialEq for crate::Literal {
153    fn eq(&self, other: &Self) -> bool {
154        match (*self, *other) {
155            (Self::F64(a), Self::F64(b)) => a.to_bits() == b.to_bits(),
156            (Self::F32(a), Self::F32(b)) => a.to_bits() == b.to_bits(),
157            (Self::U32(a), Self::U32(b)) => a == b,
158            (Self::I32(a), Self::I32(b)) => a == b,
159            (Self::I64(a), Self::I64(b)) => a == b,
160            (Self::Bool(a), Self::Bool(b)) => a == b,
161            _ => false,
162        }
163    }
164}

166impl std::hash::Hash for crate::Literal {
167    fn hash<H: std::hash::Hasher>(&self, hasher: &mut H) {
168        match *self {
169            Self::F64(v) | Self::AbstractFloat(v) => {
170                hasher.write_u8(0);
171                v.to_bits().hash(hasher);
172            }
173            Self::F32(v) => {
174                hasher.write_u8(1);
175                v.to_bits().hash(hasher);
176            }
177            Self::U32(v) => {
178                hasher.write_u8(2);
179                v.hash(hasher);
180            }
181            Self::I32(v) => {
182                hasher.write_u8(3);
183                v.hash(hasher);
184            }
185            Self::Bool(v) => {
186                hasher.write_u8(4);
187                v.hash(hasher);
188            }
189            Self::I64(v) | Self::AbstractInt(v) => {
190                hasher.write_u8(5);
191                v.hash(hasher);
192            }
193        }
194    }

289    /// Get the size of this type.
290    pub fn size(&self, _gctx: GlobalCtx) -> u32 {
291        match *self {
292            Self::Scalar(scalar) | Self::Atomic(scalar) => scalar.width as u32,
293            Self::Vector { size, scalar } => size as u32 * scalar.width as u32,
294            // matrices are treated as arrays of aligned columns
295            Self::Matrix {
296                columns,
297                rows,
298                scalar,
299            } => Alignment::from(rows) * scalar.width as u32 * columns as u32,
300            Self::Pointer { .. } | Self::ValuePointer { .. } => POINTER_SPAN,
301            Self::Array {
302                base: _,
303                size,
304                stride,
305            } => {
306                let count = match size {
307                    super::ArraySize::Constant(count) => count.get(),
308                    // A dynamically-sized array has to have at least one element
309                    super::ArraySize::Dynamic => 1,
310                };
311                count * stride
312            }
313            Self::Struct { span, .. } => span,
314            Self::Image { .. }
315            | Self::Sampler { .. }
316            | Self::AccelerationStructure
317            | Self::RayQuery
318            | Self::BindingArray { .. } => 0,
319        }
320    }

322    /// Return the canonical form of `self`, or `None` if it's already in
323    /// canonical form.
324    ///
325    /// Certain types have multiple representations in `TypeInner`. This
326    /// function converts all forms of equivalent types to a single
327    /// representative of their class, so that simply applying `Eq` to the
328    /// result indicates whether the types are equivalent, as far as Naga IR is
329    /// concerned.
330    pub fn canonical_form(
331        &self,
332        types: &crate::UniqueArena<crate::Type>,
333    ) -> Option<crate::TypeInner> {
334        use crate::TypeInner as Ti;
335        match *self {
336            Ti::Pointer { base, space } => match types[base].inner {
337                Ti::Scalar(scalar) => Some(Ti::ValuePointer {
338                    size: None,
339                    scalar,
340                    space,
341                }),
342                Ti::Vector { size, scalar } => Some(Ti::ValuePointer {
343                    size: Some(size),
344                    scalar,
345                    space,
346                }),
347                _ => None,
348            },
349            _ => None,
350        }
351    }

353    /// Compare `self` and `rhs` as types.
354    ///
355    /// This is mostly the same as `<TypeInner as Eq>::eq`, but it treats
356    /// `ValuePointer` and `Pointer` types as equivalent.
357    ///
358    /// When you know that one side of the comparison is never a pointer, it's
359    /// fine to not bother with canonicalization, and just compare `TypeInner`
360    /// values with `==`.
361    pub fn equivalent(
362        &self,
363        rhs: &crate::TypeInner,
364        types: &crate::UniqueArena<crate::Type>,
365    ) -> bool {
366        let left = self.canonical_form(types);
367        let right = rhs.canonical_form(types);
368        left.as_ref().unwrap_or(self) == right.as_ref().unwrap_or(rhs)
369    }
370
371    pub fn is_dynamically_sized(&self, types: &crate::UniqueArena<crate::Type>) -> bool {
372        use crate::TypeInner as Ti;
373        match *self {
374            Ti::Array { size, .. } => size == crate::ArraySize::Dynamic,
375            Ti::Struct { ref members, .. } => members
376                .last()
377                .map(|last| types[last.ty].inner.is_dynamically_sized(types))
378                .unwrap_or(false),
379            _ => false,
380        }
381    }

514impl crate::Expression {
515    /// Returns true if the expression is considered emitted at the start of a function.
516    pub const fn needs_pre_emit(&self) -> bool {
517        match *self {
518            Self::Literal(_)
519            | Self::Constant(_)
520            | Self::ZeroValue(_)
521            | Self::FunctionArgument(_)
522            | Self::GlobalVariable(_)
523            | Self::LocalVariable(_) => true,
524            _ => false,
525        }
526    }
527
528    /// Return true if this expression is a dynamic array index, for [`Access`].
529    ///
530    /// This method returns true if this expression is a dynamically computed
531    /// index, and as such can only be used to index matrices and arrays when
532    /// they appear behind a pointer. See the documentation for [`Access`] for
533    /// details.
534    ///
535    /// Note, this does not check the _type_ of the given expression. It's up to
536    /// the caller to establish that the `Access` expression is well-typed
537    /// through other means, like [`ResolveContext`].
538    ///
539    /// [`Access`]: crate::Expression::Access
540    /// [`ResolveContext`]: crate::proc::ResolveContext
541    pub fn is_dynamic_index(&self, module: &crate::Module) -> bool {
542        match *self {
543            Self::Literal(_) | Self::ZeroValue(_) => false,
544            Self::Constant(handle) => {
545                let constant = &module.constants[handle];
546                !matches!(constant.r#override, crate::Override::None)
547            }
548            _ => true,
549        }
550    }

553impl crate::Function {
554    /// Return the global variable being accessed by the expression `pointer`.
555    ///
556    /// Assuming that `pointer` is a series of `Access` and `AccessIndex`
557    /// expressions that ultimately access some part of a `GlobalVariable`,
558    /// return a handle for that global.
559    ///
560    /// If the expression does not ultimately access a global variable, return
561    /// `None`.
562    pub fn originating_global(
563        &self,
564        mut pointer: crate::Handle<crate::Expression>,
565    ) -> Option<crate::Handle<crate::GlobalVariable>> {
566        loop {
567            pointer = match self.expressions[pointer] {
568                crate::Expression::Access { base, .. } => base,
569                crate::Expression::AccessIndex { base, .. } => base,
570                crate::Expression::GlobalVariable(handle) => return Some(handle),
571                crate::Expression::LocalVariable(_) => return None,
572                crate::Expression::FunctionArgument(_) => return None,
573                // There are no other expressions that produce pointer values.
574                _ => unreachable!(),
575            }
576        }
577    }

658impl GlobalCtx<'_> {
659    /// Try to evaluate the expression in `self.const_expressions` using its `handle` and return it as a `u32`.
660    #[allow(dead_code)]
661    pub(super) fn eval_expr_to_u32(
662        &self,
663        handle: crate::Handle<crate::Expression>,
664    ) -> Result<u32, U32EvalError> {
665        self.eval_expr_to_u32_from(handle, self.const_expressions)
666    }
667
668    /// Try to evaluate the expression in the `arena` using its `handle` and return it as a `u32`.
669    pub(super) fn eval_expr_to_u32_from(
670        &self,
671        handle: crate::Handle<crate::Expression>,
672        arena: &crate::Arena<crate::Expression>,
673    ) -> Result<u32, U32EvalError> {
674        match self.eval_expr_to_literal_from(handle, arena) {
675            Some(crate::Literal::U32(value)) => Ok(value),
676            Some(crate::Literal::I32(value)) => {
677                value.try_into().map_err(|_| U32EvalError::Negative)
678            }
679            _ => Err(U32EvalError::NonConst),
680        }
681    }

684    pub(crate) fn eval_expr_to_literal(
685        &self,
686        handle: crate::Handle<crate::Expression>,
687    ) -> Option<crate::Literal> {
688        self.eval_expr_to_literal_from(handle, self.const_expressions)
689    }
690
691    fn eval_expr_to_literal_from(
692        &self,
693        handle: crate::Handle<crate::Expression>,
694        arena: &crate::Arena<crate::Expression>,
695    ) -> Option<crate::Literal> {
696        fn get(
697            gctx: GlobalCtx,
698            handle: crate::Handle<crate::Expression>,
699            arena: &crate::Arena<crate::Expression>,
700        ) -> Option<crate::Literal> {
701            match arena[handle] {
702                crate::Expression::Literal(literal) => Some(literal),
703                crate::Expression::ZeroValue(ty) => match gctx.types[ty].inner {
704                    crate::TypeInner::Scalar(scalar) => crate::Literal::zero(scalar),
705                    _ => None,
706                },
707                _ => None,
708            }
709        }
710        match arena[handle] {
711            crate::Expression::Constant(c) => {
712                get(*self, self.constants[c].init, self.const_expressions)
713            }
714            _ => get(*self, handle, arena),
715        }
716    }
717}

719/// Return an iterator over the individual components assembled by a
720/// `Compose` expression.
721///
722/// Given `ty` and `components` from an `Expression::Compose`, return an
723/// iterator over the components of the resulting value.
724///
725/// Normally, this would just be an iterator over `components`. However,
726/// `Compose` expressions can concatenate vectors, in which case the i'th
727/// value being composed is not generally the i'th element of `components`.
728/// This function consults `ty` to decide if this concatenation is occuring,
729/// and returns an iterator that produces the components of the result of
730/// the `Compose` expression in either case.
731pub fn flatten_compose<'arenas>(
732    ty: crate::Handle<crate::Type>,
733    components: &'arenas [crate::Handle<crate::Expression>],
734    expressions: &'arenas crate::Arena<crate::Expression>,
735    types: &'arenas crate::UniqueArena<crate::Type>,
736) -> impl Iterator<Item = crate::Handle<crate::Expression>> + 'arenas {
737    // Returning `impl Iterator` is a bit tricky. We may or may not
738    // want to flatten the components, but we have to settle on a
739    // single concrete type to return. This function returns a single
740    // iterator chain that handles both the flattening and
741    // non-flattening cases.
742    let (size, is_vector) = if let crate::TypeInner::Vector { size, .. } = types[ty].inner {
743        (size as usize, true)
744    } else {
745        (components.len(), false)
746    };
747
748    /// Flatten `Compose` expressions if `is_vector` is true.
749    fn flatten_compose<'c>(
750        component: &'c crate::Handle<crate::Expression>,
751        is_vector: bool,
752        expressions: &'c crate::Arena<crate::Expression>,
753    ) -> &'c [crate::Handle<crate::Expression>] {
754        if is_vector {
755            if let crate::Expression::Compose {
756                ty: _,
757                components: ref subcomponents,
758            } = expressions[*component]
759            {
760                return subcomponents;
761            }
762        }
763        std::slice::from_ref(component)
764    }
765
766    /// Flatten `Splat` expressions if `is_vector` is true.
767    fn flatten_splat<'c>(
768        component: &'c crate::Handle<crate::Expression>,
769        is_vector: bool,
770        expressions: &'c crate::Arena<crate::Expression>,
771    ) -> impl Iterator<Item = crate::Handle<crate::Expression>> {
772        let mut expr = *component;
773        let mut count = 1;
774        if is_vector {
775            if let crate::Expression::Splat { size, value } = expressions[expr] {
776                expr = value;
777                count = size as usize;
778            }
779        }
780        std::iter::repeat(expr).take(count)
781    }
782
783    // Expressions like `vec4(vec3(vec2(6, 7), 8), 9)` require us to
784    // flatten up to two levels of `Compose` expressions.
785    //
786    // Expressions like `vec4(vec3(1.0), 1.0)` require us to flatten
787    // `Splat` expressions. Fortunately, the operand of a `Splat` must
788    // be a scalar, so we can stop there.
789    components
790        .iter()
791        .flat_map(move |component| flatten_compose(component, is_vector, expressions))
792        .flat_map(move |component| flatten_compose(component, is_vector, expressions))
793        .flat_map(move |component| flatten_splat(component, is_vector, expressions))
794        .take(size)
795}
