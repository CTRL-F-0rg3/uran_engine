35/// Uniform control flow characteristics.
36#[derive(Clone, Debug)]
37#[cfg_attr(feature = "serialize", derive(serde::Serialize))]
38#[cfg_attr(feature = "deserialize", derive(serde::Deserialize))]
39#[cfg_attr(test, derive(PartialEq))]
40pub struct Uniformity {
41    /// A child expression with non-uniform result.
42    ///
43    /// This means, when the relevant invocations are scheduled on a compute unit,
44    /// they have to use vector registers to store an individual value
45    /// per invocation.
46    ///
47    /// Whenever the control flow is conditioned on such value,
48    /// the hardware needs to keep track of the mask of invocations,
49    /// and process all branches of the control flow.
50    ///
51    /// Any operations that depend on non-uniform results also produce non-uniform.
52    pub non_uniform_result: NonUniformResult,
53    /// If this expression requires uniform control flow, store the reason here.
54    pub requirements: UniformityRequirements,
55}
56
57impl Uniformity {
58    const fn new() -> Self {
59        Uniformity {
60            non_uniform_result: None,
61            requirements: UniformityRequirements::empty(),
62        }
63    }
64}

66bitflags::bitflags! {
67    #[derive(Clone, Copy, Debug, PartialEq)]
68    struct ExitFlags: u8 {
69        /// Control flow may return from the function, which makes all the
70        /// subsequent statements within the current function (only!)
71        /// to be executed in a non-uniform control flow.
72        const MAY_RETURN = 0x1;
73        /// Control flow may be killed. Anything after `Statement::Kill` is
74        /// considered inside non-uniform context.
75        const MAY_KILL = 0x2;
76    }
77}

79/// Uniformity characteristics of a function.
80#[cfg_attr(test, derive(Debug, PartialEq))]
81struct FunctionUniformity {
82    result: Uniformity,
83    exit: ExitFlags,
84}

86impl ops::BitOr for FunctionUniformity {
87    type Output = Self;
88    fn bitor(self, other: Self) -> Self {
89        FunctionUniformity {
90            result: Uniformity {
91                non_uniform_result: self
92                    .result
93                    .non_uniform_result
94                    .or(other.result.non_uniform_result),
95                requirements: self.result.requirements | other.result.requirements,
96            },
97            exit: self.exit | other.exit,
98        }
99    }
100}

102impl FunctionUniformity {
103    const fn new() -> Self {
104        FunctionUniformity {
105            result: Uniformity::new(),
106            exit: ExitFlags::empty(),
107        }
108    }
109
110    /// Returns a disruptor based on the stored exit flags, if any.
111    const fn exit_disruptor(&self) -> Option<UniformityDisruptor> {
112        if self.exit.contains(ExitFlags::MAY_RETURN) {
113            Some(UniformityDisruptor::Return)
114        } else if self.exit.contains(ExitFlags::MAY_KILL) {
115            Some(UniformityDisruptor::Discard)
116        } else {
117            None
118        }
119    }

178impl GlobalOrArgument {
179    fn from_expression(
180        expression_arena: &Arena<crate::Expression>,
181        expression: Handle<crate::Expression>,
182    ) -> Result<GlobalOrArgument, ExpressionError> {
183        Ok(match expression_arena[expression] {
184            crate::Expression::GlobalVariable(var) => GlobalOrArgument::Global(var),
185            crate::Expression::FunctionArgument(i) => GlobalOrArgument::Argument(i),
186            crate::Expression::Access { base, .. }
187            | crate::Expression::AccessIndex { base, .. } => match expression_arena[base] {
188                crate::Expression::GlobalVariable(var) => GlobalOrArgument::Global(var),
189                _ => return Err(ExpressionError::ExpectedGlobalOrArgument),
190            },
191            _ => return Err(ExpressionError::ExpectedGlobalOrArgument),
192        })
193    }

207pub struct FunctionInfo {
208    /// Validation flags.
209    #[allow(dead_code)]
210    flags: ValidationFlags,
211    /// Set of shader stages where calling this function is valid.
212    pub available_stages: ShaderStages,
213    /// Uniformity characteristics.
214    pub uniformity: Uniformity,
215    /// Function may kill the invocation.
216    pub may_kill: bool,
217
218    /// All pairs of (texture, sampler) globals that may be used together in
219    /// sampling operations by this function and its callees. This includes
220    /// pairings that arise when this function passes textures and samplers as
221    /// arguments to its callees.
222    ///
223    /// This table does not include uses of textures and samplers passed as
224    /// arguments to this function itself, since we do not know which globals
225    /// those will be. However, this table *is* exhaustive when computed for an
226    /// entry point function: entry points never receive textures or samplers as
227    /// arguments, so all an entry point's sampling can be reported in terms of
228    /// globals.
229    ///
230    /// The GLSL back end uses this table to construct reflection info that
231    /// clients need to construct texture-combined sampler values.
232    pub sampling_set: crate::FastHashSet<SamplingKey>,
233
234    /// How this function and its callees use this module's globals.
235    ///
236    /// This is indexed by `Handle<GlobalVariable>` indices. However,
237    /// `FunctionInfo` implements `std::ops::Index<Handle<GlobalVariable>>`,
238    /// so you can simply index this struct with a global handle to retrieve
239    /// its usage information.
240    global_uses: Box<[GlobalUse]>,
241
242    /// Information about each expression in this function's body.
243    ///
244    /// This is indexed by `Handle<Expression>` indices. However, `FunctionInfo`
245    /// implements `std::ops::Index<Handle<Expression>>`, so you can simply
246    /// index this struct with an expression handle to retrieve its
247    /// `ExpressionInfo`.
248    expressions: Box<[ExpressionInfo]>,
249
250    /// All (texture, sampler) pairs that may be used together in sampling
251    /// operations by this function and its callees, whether they are accessed
252    /// as globals or passed as arguments.
253    ///
254    /// Participants are represented by [`GlobalVariable`] handles whenever
255    /// possible, and otherwise by indices of this function's arguments.
256    ///
257    /// When analyzing a function call, we combine this data about the callee
258    /// with the actual arguments being passed to produce the callers' own
259    /// `sampling_set` and `sampling` tables.
260    ///
261    /// [`GlobalVariable`]: crate::GlobalVariable
262    sampling: crate::FastHashSet<Sampling>,
263
264    /// Indicates that the function is using dual source blending.
265    pub dual_source_blending: bool,
266}

268impl FunctionInfo {
269    pub const fn global_variable_count(&self) -> usize {
270        self.global_uses.len()
271    }
272    pub const fn expression_count(&self) -> usize {
273        self.expressions.len()
274    }
275    pub fn dominates_global_use(&self, other: &Self) -> bool {
276        for (self_global_uses, other_global_uses) in
277            self.global_uses.iter().zip(other.global_uses.iter())
278        {
279            if !self_global_uses.contains(*other_global_uses) {
280                return false;
281            }
282        }
283        true
284    }
285}

301/// Disruptor of the uniform control flow.
302#[derive(Clone, Copy, Debug, thiserror::Error)]
303#[cfg_attr(test, derive(PartialEq))]
304pub enum UniformityDisruptor {
305    #[error("Expression {0:?} produced non-uniform result, and control flow depends on it")]
306    Expression(Handle<crate::Expression>),
307    #[error("There is a Return earlier in the control flow of the function")]
308    Return,
309    #[error("There is a Discard earlier in the entry point across all called functions")]
310    Discard,
311}

313impl FunctionInfo {
314    /// Adds a value-type reference to an expression.
315    #[must_use]
316    fn add_ref_impl(
317        &mut self,
318        handle: Handle<crate::Expression>,
319        global_use: GlobalUse,
320    ) -> NonUniformResult {
321        let info = &mut self.expressions[handle.index()];
322        info.ref_count += 1;
323        // mark the used global as read
324        if let Some(global) = info.assignable_global {
325            self.global_uses[global.index()] |= global_use;
326        }
327        info.uniformity.non_uniform_result
328    }
329
330    /// Adds a value-type reference to an expression.
331    #[must_use]
332    fn add_ref(&mut self, handle: Handle<crate::Expression>) -> NonUniformResult {
333        self.add_ref_impl(handle, GlobalUse::READ)
334    }
335
336    /// Adds a potentially assignable reference to an expression.
337    /// These are destinations for `Store` and `ImageStore` statements,
338    /// which can transit through `Access` and `AccessIndex`.
339    #[must_use]
340    fn add_assignable_ref(
341        &mut self,
342        handle: Handle<crate::Expression>,
343        assignable_global: &mut Option<Handle<crate::GlobalVariable>>,
344    ) -> NonUniformResult {
345        let info = &mut self.expressions[handle.index()];
346        info.ref_count += 1;
347        // propagate the assignable global up the chain, till it either hits
348        // a value-type expression, or the assignment statement.
349        if let Some(global) = info.assignable_global {
350            if let Some(_old) = assignable_global.replace(global) {
351                unreachable!()
352            }
353        }
354        info.uniformity.non_uniform_result
355    }
356
357    /// Inherit information from a called function.
358    fn process_call(
359        &mut self,
360        callee: &Self,
361        arguments: &[Handle<crate::Expression>],
362        expression_arena: &Arena<crate::Expression>,
363    ) -> Result<FunctionUniformity, WithSpan<FunctionError>> {
364        self.sampling_set
365            .extend(callee.sampling_set.iter().cloned());
366        for sampling in callee.sampling.iter() {
367            // If the callee was passed the texture or sampler as an argument,
368            // we may now be able to determine which globals those referred to.
369            let image_storage = match sampling.image {
370                GlobalOrArgument::Global(var) => GlobalOrArgument::Global(var),
371                GlobalOrArgument::Argument(i) => {
372                    let handle = arguments[i as usize];
373                    GlobalOrArgument::from_expression(expression_arena, handle).map_err(
374                        |source| {
375                            FunctionError::Expression { handle, source }
376                                .with_span_handle(handle, expression_arena)
377                        },
378                    )?
379                }
380            };
381
382            let sampler_storage = match sampling.sampler {
383                GlobalOrArgument::Global(var) => GlobalOrArgument::Global(var),
384                GlobalOrArgument::Argument(i) => {
385                    let handle = arguments[i as usize];
386                    GlobalOrArgument::from_expression(expression_arena, handle).map_err(
387                        |source| {
388                            FunctionError::Expression { handle, source }
389                                .with_span_handle(handle, expression_arena)
390                        },
391                    )?
392                }
393            };
394
395            // If we've managed to pin both the image and sampler down to
396            // specific globals, record that in our `sampling_set`. Otherwise,
397            // record as much as we do know in our own `sampling` table, for our
398            // callers to sort out.
399            match (image_storage, sampler_storage) {
400                (GlobalOrArgument::Global(image), GlobalOrArgument::Global(sampler)) => {
401                    self.sampling_set.insert(SamplingKey { image, sampler });
402                }
403                (image, sampler) => {
404                    self.sampling.insert(Sampling { image, sampler });
405                }
406            }
407        }
408
409        // Inherit global use from our callees.
410        for (mine, other) in self.global_uses.iter_mut().zip(callee.global_uses.iter()) {
411            *mine |= *other;
412        }
413
414        Ok(FunctionUniformity {
415            result: callee.uniformity.clone(),
416            exit: if callee.may_kill {
417                ExitFlags::MAY_KILL
418            } else {
419                ExitFlags::empty()
420            },
421        })
422    }
423
424    /// Compute the [`ExpressionInfo`] for `handle`.
425    ///
426    /// Replace the dummy entry in [`self.expressions`] for `handle`
427    /// with a real `ExpressionInfo` value describing that expression.
428    ///
429    /// This function is called as part of a forward sweep through the
430    /// arena, so we can assume that all earlier expressions in the
431    /// arena already have valid info. Since expressions only depend
432    /// on earlier expressions, this includes all our subexpressions.
433    ///
434    /// Adjust the reference counts on all expressions we use.
435    ///
436    /// Also populate the [`sampling_set`], [`sampling`] and
437    /// [`global_uses`] fields of `self`.
438    ///
439    /// [`self.expressions`]: FunctionInfo::expressions
440    /// [`sampling_set`]: FunctionInfo::sampling_set
441    /// [`sampling`]: FunctionInfo::sampling
442    /// [`global_uses`]: FunctionInfo::global_uses
443    #[allow(clippy::or_fun_call)]
444    fn process_expression(
445        &mut self,
446        handle: Handle<crate::Expression>,
447        expression_arena: &Arena<crate::Expression>,
448        other_functions: &[FunctionInfo],
449        resolve_context: &ResolveContext,
450        capabilities: super::Capabilities,
451    ) -> Result<(), ExpressionError> {
452        use crate::{Expression as E, SampleLevel as Sl};
453
454        let expression = &expression_arena[handle];
455        let mut assignable_global = None;
456        let uniformity = match *expression {
457            E::Access { base, index } => {
458                let base_ty = self[base].ty.inner_with(resolve_context.types);
459
460                // build up the caps needed if this is indexed non-uniformly
461                let mut needed_caps = super::Capabilities::empty();
462                let is_binding_array = match *base_ty {
463                    crate::TypeInner::BindingArray {
464                        base: array_element_ty_handle,
465                        ..
466                    } => {
467                        // these are nasty aliases, but these idents are too long and break rustfmt
468                        let ub_st = super::Capabilities::UNIFORM_BUFFER_AND_STORAGE_TEXTURE_ARRAY_NON_UNIFORM_INDEXING;
469                        let st_sb = super::Capabilities::SAMPLED_TEXTURE_AND_STORAGE_BUFFER_ARRAY_NON_UNIFORM_INDEXING;
470                        let sampler = super::Capabilities::SAMPLER_NON_UNIFORM_INDEXING;
471
472                        // We're a binding array, so lets use the type of _what_ we are array of to determine if we can non-uniformly index it.
473                        let array_element_ty =
474                            &resolve_context.types[array_element_ty_handle].inner;
475
476                        needed_caps |= match *array_element_ty {
477                            // If we're an image, use the appropriate limit.
478                            crate::TypeInner::Image { class, .. } => match class {
479                                crate::ImageClass::Storage { .. } => ub_st,
480                                _ => st_sb,
481                            },
482                            crate::TypeInner::Sampler { .. } => sampler,
483                            // If we're anything but an image, assume we're a buffer and use the address space.
484                            _ => {
485                                if let E::GlobalVariable(global_handle) = expression_arena[base] {
486                                    let global = &resolve_context.global_vars[global_handle];
487                                    match global.space {
488                                        crate::AddressSpace::Uniform => ub_st,
489                                        crate::AddressSpace::Storage { .. } => st_sb,
490                                        _ => unreachable!(),
491                                    }
492                                } else {
493                                    unreachable!()
494                                }
495                            }
496                        };
497
498                        true
499                    }
500                    _ => false,
501                };
502
503                if self[index].uniformity.non_uniform_result.is_some()
504                    && !capabilities.contains(needed_caps)
505                    && is_binding_array
506                {
507                    return Err(ExpressionError::MissingCapabilities(needed_caps));
508                }
509
510                Uniformity {
511                    non_uniform_result: self
512                        .add_assignable_ref(base, &mut assignable_global)
513                        .or(self.add_ref(index)),
514                    requirements: UniformityRequirements::empty(),
515                }
516            }
517            E::AccessIndex { base, .. } => Uniformity {
518                non_uniform_result: self.add_assignable_ref(base, &mut assignable_global),
519                requirements: UniformityRequirements::empty(),
520            },
521            // always uniform
522            E::Splat { size: _, value } => Uniformity {
523                non_uniform_result: self.add_ref(value),
524                requirements: UniformityRequirements::empty(),
525            },
526            E::Swizzle { vector, .. } => Uniformity {
527                non_uniform_result: self.add_ref(vector),
528                requirements: UniformityRequirements::empty(),
529            },
530            E::Literal(_) | E::Constant(_) | E::ZeroValue(_) => Uniformity::new(),
531            E::Compose { ref components, .. } => {
532                let non_uniform_result = components
533                    .iter()
534                    .fold(None, |nur, &comp| nur.or(self.add_ref(comp)));
535                Uniformity {
536                    non_uniform_result,
537                    requirements: UniformityRequirements::empty(),
538                }
539            }
540            // depends on the builtin or interpolation
541            E::FunctionArgument(index) => {
542                let arg = &resolve_context.arguments[index as usize];
543                let uniform = match arg.binding {
544                    Some(crate::Binding::BuiltIn(built_in)) => match built_in {
545                        // per-polygon built-ins are uniform
546                        crate::BuiltIn::FrontFacing
547                        // per-work-group built-ins are uniform
548                        | crate::BuiltIn::WorkGroupId
549                        | crate::BuiltIn::WorkGroupSize
550                        | crate::BuiltIn::NumWorkGroups => true,
551                        _ => false,
552                    },
553                    // only flat inputs are uniform
554                    Some(crate::Binding::Location {
555                        interpolation: Some(crate::Interpolation::Flat),
556                        ..
557                    }) => true,
558                    _ => false,
559                };
560                Uniformity {
561                    non_uniform_result: if uniform { None } else { Some(handle) },
562                    requirements: UniformityRequirements::empty(),
563                }
564            }
565            // depends on the address space
566            E::GlobalVariable(gh) => {
567                use crate::AddressSpace as As;
568                assignable_global = Some(gh);
569                let var = &resolve_context.global_vars[gh];
570                let uniform = match var.space {
571                    // local data is non-uniform
572                    As::Function | As::Private => false,
573                    // workgroup memory is exclusively accessed by the group
574                    As::WorkGroup => true,
575                    // uniform data
576                    As::Uniform | As::PushConstant => true,
577                    // storage data is only uniform when read-only
578                    As::Storage { access } => !access.contains(crate::StorageAccess::STORE),
579                    As::Handle => false,
580                };
581                Uniformity {
582                    non_uniform_result: if uniform { None } else { Some(handle) },
583                    requirements: UniformityRequirements::empty(),
584                }
585            }
586            E::LocalVariable(_) => Uniformity {
587                non_uniform_result: Some(handle),
588                requirements: UniformityRequirements::empty(),
589            },
590            E::Load { pointer } => Uniformity {
591                non_uniform_result: self.add_ref(pointer),
592                requirements: UniformityRequirements::empty(),
593            },
594            E::ImageSample {
595                image,
596                sampler,
597                gather: _,
598                coordinate,
599                array_index,
600                offset: _,
601                level,
602                depth_ref,
603            } => {
604                let image_storage = GlobalOrArgument::from_expression(expression_arena, image)?;
605                let sampler_storage = GlobalOrArgument::from_expression(expression_arena, sampler)?;
606
607                match (image_storage, sampler_storage) {
608                    (GlobalOrArgument::Global(image), GlobalOrArgument::Global(sampler)) => {
609                        self.sampling_set.insert(SamplingKey { image, sampler });
610                    }
611                    _ => {
612                        self.sampling.insert(Sampling {
613                            image: image_storage,
614                            sampler: sampler_storage,
615                        });
616                    }
617                }
618
619                // "nur" == "Non-Uniform Result"
620                let array_nur = array_index.and_then(|h| self.add_ref(h));
621                let level_nur = match level {
622                    Sl::Auto | Sl::Zero => None,
623                    Sl::Exact(h) | Sl::Bias(h) => self.add_ref(h),
624                    Sl::Gradient { x, y } => self.add_ref(x).or(self.add_ref(y)),
625                };
626                let dref_nur = depth_ref.and_then(|h| self.add_ref(h));
627                Uniformity {
628                    non_uniform_result: self
629                        .add_ref(image)
630                        .or(self.add_ref(sampler))
631                        .or(self.add_ref(coordinate))
632                        .or(array_nur)
633                        .or(level_nur)
634                        .or(dref_nur),
635                    requirements: if level.implicit_derivatives() {
636                        UniformityRequirements::IMPLICIT_LEVEL
637                    } else {
638                        UniformityRequirements::empty()
639                    },
640                }
641            }
642            E::ImageLoad {
643                image,
644                coordinate,
645                array_index,
646                sample,
647                level,
648            } => {
649                let array_nur = array_index.and_then(|h| self.add_ref(h));
650                let sample_nur = sample.and_then(|h| self.add_ref(h));
651                let level_nur = level.and_then(|h| self.add_ref(h));
652                Uniformity {
653                    non_uniform_result: self
654                        .add_ref(image)
655                        .or(self.add_ref(coordinate))
656                        .or(array_nur)
657                        .or(sample_nur)
658                        .or(level_nur),
659                    requirements: UniformityRequirements::empty(),
660                }
661            }
662            E::ImageQuery { image, query } => {
663                let query_nur = match query {
664                    crate::ImageQuery::Size { level: Some(h) } => self.add_ref(h),
665                    _ => None,
666                };
667                Uniformity {
668                    non_uniform_result: self.add_ref_impl(image, GlobalUse::QUERY).or(query_nur),
669                    requirements: UniformityRequirements::empty(),
670                }
671            }
672            E::Unary { expr, .. } => Uniformity {
673                non_uniform_result: self.add_ref(expr),
674                requirements: UniformityRequirements::empty(),
675            },
676            E::Binary { left, right, .. } => Uniformity {
677                non_uniform_result: self.add_ref(left).or(self.add_ref(right)),
678                requirements: UniformityRequirements::empty(),
679            },
680            E::Select {
681                condition,
682                accept,
683                reject,
684            } => Uniformity {
685                non_uniform_result: self
686                    .add_ref(condition)
687                    .or(self.add_ref(accept))
688                    .or(self.add_ref(reject)),
689                requirements: UniformityRequirements::empty(),
690            },
691            // explicit derivatives require uniform
692            E::Derivative { expr, .. } => Uniformity {
693                //Note: taking a derivative of a uniform doesn't make it non-uniform
694                non_uniform_result: self.add_ref(expr),
695                requirements: UniformityRequirements::DERIVATIVE,
696            },
697            E::Relational { argument, .. } => Uniformity {
698                non_uniform_result: self.add_ref(argument),
699                requirements: UniformityRequirements::empty(),
700            },
701            E::Math {
702                fun: _,
703                arg,
704                arg1,
705                arg2,
706                arg3,
707            } => {
708                let arg1_nur = arg1.and_then(|h| self.add_ref(h));
709                let arg2_nur = arg2.and_then(|h| self.add_ref(h));
710                let arg3_nur = arg3.and_then(|h| self.add_ref(h));
711                Uniformity {
712                    non_uniform_result: self.add_ref(arg).or(arg1_nur).or(arg2_nur).or(arg3_nur),
713                    requirements: UniformityRequirements::empty(),
714                }
715            }
716            E::As { expr, .. } => Uniformity {
717                non_uniform_result: self.add_ref(expr),
718                requirements: UniformityRequirements::empty(),
719            },
720            E::CallResult(function) => other_functions[function.index()].uniformity.clone(),
721            E::AtomicResult { .. } | E::RayQueryProceedResult => Uniformity {
722                non_uniform_result: Some(handle),
723                requirements: UniformityRequirements::empty(),
724            },
725            E::WorkGroupUniformLoadResult { .. } => Uniformity {
726                // The result of WorkGroupUniformLoad is always uniform by definition
727                non_uniform_result: None,
728                // The call is what cares about uniformity, not the expression
729                // This expression is never emitted, so this requirement should never be used anyway?
730                requirements: UniformityRequirements::empty(),
731            },
732            E::ArrayLength(expr) => Uniformity {
733                non_uniform_result: self.add_ref_impl(expr, GlobalUse::QUERY),
734                requirements: UniformityRequirements::empty(),
735            },
736            E::RayQueryGetIntersection {
737                query,
738                committed: _,
739            } => Uniformity {
740                non_uniform_result: self.add_ref(query),
741                requirements: UniformityRequirements::empty(),
742            },
743        };
744
745        let ty = resolve_context.resolve(expression, |h| Ok(&self[h].ty))?;
746        self.expressions[handle.index()] = ExpressionInfo {
747            uniformity,
748            ref_count: 0,
749            assignable_global,
750            ty,
751        };
752        Ok(())
753    }
754
755    /// Analyzes the uniformity requirements of a block (as a sequence of statements).
756    /// Returns the uniformity characteristics at the *function* level, i.e.
757    /// whether or not the function requires to be called in uniform control flow,
758    /// and whether the produced result is not disrupting the control flow.
759    ///
760    /// The parent control flow is uniform if `disruptor.is_none()`.
761    ///
762    /// Returns a `NonUniformControlFlow` error if any of the expressions in the block
763    /// require uniformity, but the current flow is non-uniform.
764    #[allow(clippy::or_fun_call)]
765    fn process_block(
766        &mut self,
767        statements: &crate::Block,
768        other_functions: &[FunctionInfo],
769        mut disruptor: Option<UniformityDisruptor>,
770        expression_arena: &Arena<crate::Expression>,
771    ) -> Result<FunctionUniformity, WithSpan<FunctionError>> {
772        use crate::Statement as S;
773
774        let mut combined_uniformity = FunctionUniformity::new();
775        for statement in statements {
776            let uniformity = match *statement {
777                S::Emit(ref range) => {
778                    let mut requirements = UniformityRequirements::empty();
779                    for expr in range.clone() {
780                        let req = self.expressions[expr.index()].uniformity.requirements;
781                        if self
782                            .flags
783                            .contains(super::ValidationFlags::CONTROL_FLOW_UNIFORMITY)
784                            && !req.is_empty()
785                        {
786                            if let Some(cause) = disruptor {
787                                return Err(FunctionError::NonUniformControlFlow(req, expr, cause)
788                                    .with_span_handle(expr, expression_arena));
789                            }
790                        }
791                        requirements |= req;
792                    }
793                    FunctionUniformity {
794                        result: Uniformity {
795                            non_uniform_result: None,
796                            requirements,
797                        },
798                        exit: ExitFlags::empty(),
799                    }
800                }
801                S::Break | S::Continue => FunctionUniformity::new(),
802                S::Kill => FunctionUniformity {
803                    result: Uniformity::new(),
804                    exit: if disruptor.is_some() {
805                        ExitFlags::MAY_KILL
806                    } else {
807                        ExitFlags::empty()
808                    },
809                },
810                S::Barrier(_) => FunctionUniformity {
811                    result: Uniformity {
812                        non_uniform_result: None,
813                        requirements: UniformityRequirements::WORK_GROUP_BARRIER,
814                    },
815                    exit: ExitFlags::empty(),
816                },
817                S::WorkGroupUniformLoad { pointer, .. } => {
818                    let _condition_nur = self.add_ref(pointer);
819
820                    // Don't check that this call occurs in uniform control flow until Naga implements WGSL's standard
821                    // uniformity analysis (https://github.com/gfx-rs/naga/issues/1744).
822                    // The uniformity analysis Naga uses now is less accurate than the one in the WGSL standard,
823                    // causing Naga to reject correct uses of `workgroupUniformLoad` in some interesting programs.
824
825                    /*
826                    if self
827                        .flags
828                        .contains(super::ValidationFlags::CONTROL_FLOW_UNIFORMITY)
829                    {
830                        let condition_nur = self.add_ref(pointer);
831                        let this_disruptor =
832                            disruptor.or(condition_nur.map(UniformityDisruptor::Expression));
833                        if let Some(cause) = this_disruptor {
834                            return Err(FunctionError::NonUniformWorkgroupUniformLoad(cause)
835                                .with_span_static(*span, "WorkGroupUniformLoad"));
836                        }
837                    } */
838                    FunctionUniformity {
839                        result: Uniformity {
840                            non_uniform_result: None,
841                            requirements: UniformityRequirements::WORK_GROUP_BARRIER,
842                        },
843                        exit: ExitFlags::empty(),
844                    }
845                }
846                S::Block(ref b) => {
847                    self.process_block(b, other_functions, disruptor, expression_arena)?
848                }
849                S::If {
850                    condition,
851                    ref accept,
852                    ref reject,
853                } => {
854                    let condition_nur = self.add_ref(condition);
855                    let branch_disruptor =
856                        disruptor.or(condition_nur.map(UniformityDisruptor::Expression));
857                    let accept_uniformity = self.process_block(
858                        accept,
859                        other_functions,
860                        branch_disruptor,
861                        expression_arena,
862                    )?;
863                    let reject_uniformity = self.process_block(
864                        reject,
865                        other_functions,
866                        branch_disruptor,
867                        expression_arena,
868                    )?;
869                    accept_uniformity | reject_uniformity
870                }
871                S::Switch {
872                    selector,
873                    ref cases,
874                } => {
875                    let selector_nur = self.add_ref(selector);
876                    let branch_disruptor =
877                        disruptor.or(selector_nur.map(UniformityDisruptor::Expression));
878                    let mut uniformity = FunctionUniformity::new();
879                    let mut case_disruptor = branch_disruptor;
880                    for case in cases.iter() {
881                        let case_uniformity = self.process_block(
882                            &case.body,
883                            other_functions,
884                            case_disruptor,
885                            expression_arena,
886                        )?;
887                        case_disruptor = if case.fall_through {
888                            case_disruptor.or(case_uniformity.exit_disruptor())
889                        } else {
890                            branch_disruptor
891                        };
892                        uniformity = uniformity | case_uniformity;
893                    }
894                    uniformity
895                }
896                S::Loop {
897                    ref body,
898                    ref continuing,
899                    break_if,
900                } => {
901                    let body_uniformity =
902                        self.process_block(body, other_functions, disruptor, expression_arena)?;
903                    let continuing_disruptor = disruptor.or(body_uniformity.exit_disruptor());
904                    let continuing_uniformity = self.process_block(
905                        continuing,
906                        other_functions,
907                        continuing_disruptor,
908                        expression_arena,
909                    )?;
910                    if let Some(expr) = break_if {
911                        let _ = self.add_ref(expr);
912                    }
913                    body_uniformity | continuing_uniformity
914                }
915                S::Return { value } => FunctionUniformity {
916                    result: Uniformity {
917                        non_uniform_result: value.and_then(|expr| self.add_ref(expr)),
918                        requirements: UniformityRequirements::empty(),
919                    },
920                    exit: if disruptor.is_some() {
921                        ExitFlags::MAY_RETURN
922                    } else {
923                        ExitFlags::empty()
924                    },
925                },
926                // Here and below, the used expressions are already emitted,
927                // and their results do not affect the function return value,
928                // so we can ignore their non-uniformity.
929                S::Store { pointer, value } => {
930                    let _ = self.add_ref_impl(pointer, GlobalUse::WRITE);
931                    let _ = self.add_ref(value);
932                    FunctionUniformity::new()
933                }
934                S::ImageStore {
935                    image,
936                    coordinate,
937                    array_index,
938                    value,
939                } => {
940                    let _ = self.add_ref_impl(image, GlobalUse::WRITE);
941                    if let Some(expr) = array_index {
942                        let _ = self.add_ref(expr);
943                    }
944                    let _ = self.add_ref(coordinate);
945                    let _ = self.add_ref(value);
946                    FunctionUniformity::new()
947                }
948                S::Call {
949                    function,
950                    ref arguments,
951                    result: _,
952                } => {
953                    for &argument in arguments {
954                        let _ = self.add_ref(argument);
955                    }
956                    let info = &other_functions[function.index()];
957                    //Note: the result is validated by the Validator, not here
958                    self.process_call(info, arguments, expression_arena)?
959                }
960                S::Atomic {
961                    pointer,
962                    ref fun,
963                    value,
964                    result: _,
965                } => {
966                    let _ = self.add_ref_impl(pointer, GlobalUse::WRITE);
967                    let _ = self.add_ref(value);
968                    if let crate::AtomicFunction::Exchange { compare: Some(cmp) } = *fun {
969                        let _ = self.add_ref(cmp);
970                    }
971                    FunctionUniformity::new()
972                }
973                S::RayQuery { query, ref fun } => {
974                    let _ = self.add_ref(query);
975                    if let crate::RayQueryFunction::Initialize {
976                        acceleration_structure,
977                        descriptor,
978                    } = *fun
979                    {
980                        let _ = self.add_ref(acceleration_structure);
981                        let _ = self.add_ref(descriptor);
982                    }
983                    FunctionUniformity::new()
984                }
985            };
986
987            disruptor = disruptor.or(uniformity.exit_disruptor());
988            combined_uniformity = combined_uniformity | uniformity;
989        }
990        Ok(combined_uniformity)
991    }
992}

1007    /// Builds the `FunctionInfo` based on the function, and validates the
1008    /// uniform control flow if required by the expressions of this function.
1009    pub(super) fn process_function(
1010        &self,
1011        fun: &crate::Function,
1012        module: &crate::Module,
1013        flags: ValidationFlags,
1014        capabilities: super::Capabilities,
1015    ) -> Result<FunctionInfo, WithSpan<FunctionError>> {
1016        let mut info = FunctionInfo {
1017            flags,
1018            available_stages: ShaderStages::all(),
1019            uniformity: Uniformity::new(),
1020            may_kill: false,
1021            sampling_set: crate::FastHashSet::default(),
1022            global_uses: vec![GlobalUse::empty(); module.global_variables.len()].into_boxed_slice(),
1023            expressions: vec![ExpressionInfo::new(); fun.expressions.len()].into_boxed_slice(),
1024            sampling: crate::FastHashSet::default(),
1025            dual_source_blending: false,
1026        };
1027        let resolve_context =
1028            ResolveContext::with_locals(module, &fun.local_variables, &fun.arguments);
1029
1030        for (handle, _) in fun.expressions.iter() {
1031            if let Err(source) = info.process_expression(
1032                handle,
1033                &fun.expressions,
1034                &self.functions,
1035                &resolve_context,
1036                capabilities,
1037            ) {
1038                return Err(FunctionError::Expression { handle, source }
1039                    .with_span_handle(handle, &fun.expressions));
1040            }
1041        }
1042
1043        for (_, expr) in fun.local_variables.iter() {
1044            if let Some(init) = expr.init {
1045                let _ = info.add_ref(init);
1046            }
1047        }
1048
1049        let uniformity = info.process_block(&fun.body, &self.functions, None, &fun.expressions)?;
1050        info.uniformity = uniformity.result;
1051        info.may_kill = uniformity.exit.contains(ExitFlags::MAY_KILL);
1052
1053        Ok(info)
1054    }
1055
1056    pub fn get_entry_point(&self, index: usize) -> &FunctionInfo {
1057        &self.entry_points[index]
1058    }
1059}
