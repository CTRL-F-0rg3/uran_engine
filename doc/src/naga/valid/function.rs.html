163    struct ControlFlowAbility: u8 {
164        /// The control can return out of this block.
165        const RETURN = 0x1;
166        /// The control can break.
167        const BREAK = 0x2;
168        /// The control can continue.

178struct BlockContext<'a> {
179    abilities: ControlFlowAbility,
180    info: &'a FunctionInfo,
181    expressions: &'a Arena<crate::Expression>,
182    types: &'a UniqueArena<crate::Type>,
183    local_vars: &'a Arena<crate::LocalVariable>,
184    global_vars: &'a Arena<crate::GlobalVariable>,
185    functions: &'a Arena<crate::Function>,
186    special_types: &'a crate::SpecialTypes,
187    prev_infos: &'a [FunctionInfo],
188    return_type: Option<Handle<crate::Type>>,
189}
190
191impl<'a> BlockContext<'a> {
192    fn new(
193        fun: &'a crate::Function,
194        module: &'a crate::Module,
195        info: &'a FunctionInfo,
196        prev_infos: &'a [FunctionInfo],
197    ) -> Self {
198        Self {
199            abilities: ControlFlowAbility::RETURN,
200            info,
201            expressions: &fun.expressions,
202            types: &module.types,
203            local_vars: &fun.local_variables,
204            global_vars: &module.global_variables,
205            functions: &module.functions,
206            special_types: &module.special_types,
207            prev_infos,
208            return_type: fun.result.as_ref().map(|fr| fr.ty),
209        }
210    }
211
212    const fn with_abilities(&self, abilities: ControlFlowAbility) -> Self {
213        BlockContext { abilities, ..*self }
214    }
215
216    fn get_expression(&self, handle: Handle<crate::Expression>) -> &'a crate::Expression {
217        &self.expressions[handle]
218    }
219
220    fn resolve_type_impl(
221        &self,
222        handle: Handle<crate::Expression>,
223        valid_expressions: &BitSet,
224    ) -> Result<&crate::TypeInner, WithSpan<ExpressionError>> {
225        if handle.index() >= self.expressions.len() {
226            Err(ExpressionError::DoesntExist.with_span())
227        } else if !valid_expressions.contains(handle.index()) {
228            Err(ExpressionError::NotInScope.with_span_handle(handle, self.expressions))
229        } else {
230            Ok(self.info[handle].ty.inner_with(self.types))
231        }
232    }
233
234    fn resolve_type(
235        &self,
236        handle: Handle<crate::Expression>,
237        valid_expressions: &BitSet,
238    ) -> Result<&crate::TypeInner, WithSpan<FunctionError>> {
239        self.resolve_type_impl(handle, valid_expressions)
240            .map_err_inner(|source| FunctionError::Expression { handle, source }.with_span())
241    }
242
243    fn resolve_pointer_type(
244        &self,
245        handle: Handle<crate::Expression>,
246    ) -> Result<&crate::TypeInner, FunctionError> {
247        if handle.index() >= self.expressions.len() {
248            Err(FunctionError::Expression {
249                handle,
250                source: ExpressionError::DoesntExist,
251            })
252        } else {
253            Ok(self.info[handle].ty.inner_with(self.types))
254        }
255    }
256}

258impl super::Validator {
259    fn validate_call(
260        &mut self,
261        function: Handle<crate::Function>,
262        arguments: &[Handle<crate::Expression>],
263        result: Option<Handle<crate::Expression>>,
264        context: &BlockContext,
265    ) -> Result<super::ShaderStages, WithSpan<CallError>> {
266        let fun = &context.functions[function];
267        if fun.arguments.len() != arguments.len() {
268            return Err(CallError::ArgumentCount {
269                required: fun.arguments.len(),
270                seen: arguments.len(),
271            }
272            .with_span());
273        }
274        for (index, (arg, &expr)) in fun.arguments.iter().zip(arguments).enumerate() {
275            let ty = context
276                .resolve_type_impl(expr, &self.valid_expression_set)
277                .map_err_inner(|source| {
278                    CallError::Argument { index, source }
279                        .with_span_handle(expr, context.expressions)
280                })?;
281            let arg_inner = &context.types[arg.ty].inner;
282            if !ty.equivalent(arg_inner, context.types) {
283                return Err(CallError::ArgumentType {
284                    index,
285                    required: arg.ty,
286                    seen_expression: expr,
287                }
288                .with_span_handle(expr, context.expressions));
289            }
290        }
291
292        if let Some(expr) = result {
293            if self.valid_expression_set.insert(expr.index()) {
294                self.valid_expression_list.push(expr);
295            } else {
296                return Err(CallError::ResultAlreadyInScope(expr)
297                    .with_span_handle(expr, context.expressions));
298            }
299            match context.expressions[expr] {
300                crate::Expression::CallResult(callee)
301                    if fun.result.is_some() && callee == function => {}
302                _ => {
303                    return Err(CallError::ExpressionMismatch(result)
304                        .with_span_handle(expr, context.expressions))
305                }
306            }
307        } else if fun.result.is_some() {
308            return Err(CallError::ExpressionMismatch(result).with_span());
309        }
310
311        let callee_info = &context.prev_infos[function.index()];
312        Ok(callee_info.available_stages)
313    }

315    fn emit_expression(
316        &mut self,
317        handle: Handle<crate::Expression>,
318        context: &BlockContext,
319    ) -> Result<(), WithSpan<FunctionError>> {
320        if self.valid_expression_set.insert(handle.index()) {
321            self.valid_expression_list.push(handle);
322            Ok(())
323        } else {
324            Err(FunctionError::ExpressionAlreadyInScope(handle)
325                .with_span_handle(handle, context.expressions))
326        }
327    }

329    fn validate_atomic(
330        &mut self,
331        pointer: Handle<crate::Expression>,
332        fun: &crate::AtomicFunction,
333        value: Handle<crate::Expression>,
334        result: Handle<crate::Expression>,
335        context: &BlockContext,
336    ) -> Result<(), WithSpan<FunctionError>> {
337        let pointer_inner = context.resolve_type(pointer, &self.valid_expression_set)?;
338        let ptr_scalar = match *pointer_inner {
339            crate::TypeInner::Pointer { base, .. } => match context.types[base].inner {
340                crate::TypeInner::Atomic(scalar) => scalar,
341                ref other => {
342                    log::error!("Atomic pointer to type {:?}", other);
343                    return Err(AtomicError::InvalidPointer(pointer)
344                        .with_span_handle(pointer, context.expressions)
345                        .into_other());
346                }
347            },
348            ref other => {
349                log::error!("Atomic on type {:?}", other);
350                return Err(AtomicError::InvalidPointer(pointer)
351                    .with_span_handle(pointer, context.expressions)
352                    .into_other());
353            }
354        };
355
356        let value_inner = context.resolve_type(value, &self.valid_expression_set)?;
357        match *value_inner {
358            crate::TypeInner::Scalar(scalar) if scalar == ptr_scalar => {}
359            ref other => {
360                log::error!("Atomic operand type {:?}", other);
361                return Err(AtomicError::InvalidOperand(value)
362                    .with_span_handle(value, context.expressions)
363                    .into_other());
364            }
365        }
366
367        if let crate::AtomicFunction::Exchange { compare: Some(cmp) } = *fun {
368            if context.resolve_type(cmp, &self.valid_expression_set)? != value_inner {
369                log::error!("Atomic exchange comparison has a different type from the value");
370                return Err(AtomicError::InvalidOperand(cmp)
371                    .with_span_handle(cmp, context.expressions)
372                    .into_other());
373            }
374        }
375
376        self.emit_expression(result, context)?;
377        match context.expressions[result] {
378            crate::Expression::AtomicResult { ty, comparison }
379                if {
380                    let scalar_predicate =
381                        |ty: &crate::TypeInner| *ty == crate::TypeInner::Scalar(ptr_scalar);
382                    match &context.types[ty].inner {
383                        ty if !comparison => scalar_predicate(ty),
384                        &crate::TypeInner::Struct { ref members, .. } if comparison => {
385                            validate_atomic_compare_exchange_struct(
386                                context.types,
387                                members,
388                                scalar_predicate,
389                            )
390                        }
391                        _ => false,
392                    }
393                } => {}
394            _ => {
395                return Err(AtomicError::ResultTypeMismatch(result)
396                    .with_span_handle(result, context.expressions)
397                    .into_other())
398            }
399        }
400        Ok(())
401    }

403    fn validate_block_impl(
404        &mut self,
405        statements: &crate::Block,
406        context: &BlockContext,
407    ) -> Result<BlockInfo, WithSpan<FunctionError>> {
408        use crate::{AddressSpace, Statement as S, TypeInner as Ti};
409        let mut finished = false;
410        let mut stages = super::ShaderStages::all();
411        for (statement, &span) in statements.span_iter() {
412            if finished {
413                return Err(FunctionError::InstructionsAfterReturn
414                    .with_span_static(span, "instructions after return"));
415            }
416            match *statement {
417                S::Emit(ref range) => {
418                    for handle in range.clone() {
419                        self.emit_expression(handle, context)?;
420                    }
421                }
422                S::Block(ref block) => {
423                    let info = self.validate_block(block, context)?;
424                    stages &= info.stages;
425                    finished = info.finished;
426                }
427                S::If {
428                    condition,
429                    ref accept,
430                    ref reject,
431                } => {
432                    match *context.resolve_type(condition, &self.valid_expression_set)? {
433                        Ti::Scalar(crate::Scalar {
434                            kind: crate::ScalarKind::Bool,
435                            width: _,
436                        }) => {}
437                        _ => {
438                            return Err(FunctionError::InvalidIfType(condition)
439                                .with_span_handle(condition, context.expressions))
440                        }
441                    }
442                    stages &= self.validate_block(accept, context)?.stages;
443                    stages &= self.validate_block(reject, context)?.stages;
444                }
445                S::Switch {
446                    selector,
447                    ref cases,
448                } => {
449                    let uint = match context
450                        .resolve_type(selector, &self.valid_expression_set)?
451                        .scalar_kind()
452                    {
453                        Some(crate::ScalarKind::Uint) => true,
454                        Some(crate::ScalarKind::Sint) => false,
455                        _ => {
456                            return Err(FunctionError::InvalidSwitchType(selector)
457                                .with_span_handle(selector, context.expressions))
458                        }
459                    };
460                    self.switch_values.clear();
461                    for case in cases {
462                        match case.value {
463                            crate::SwitchValue::I32(_) if !uint => {}
464                            crate::SwitchValue::U32(_) if uint => {}
465                            crate::SwitchValue::Default => {}
466                            _ => {
467                                return Err(FunctionError::ConflictingCaseType.with_span_static(
468                                    case.body
469                                        .span_iter()
470                                        .next()
471                                        .map_or(Default::default(), |(_, s)| *s),
472                                    "conflicting switch arm here",
473                                ));
474                            }
475                        };
476                        if !self.switch_values.insert(case.value) {
477                            return Err(match case.value {
478                                crate::SwitchValue::Default => FunctionError::MultipleDefaultCases
479                                    .with_span_static(
480                                        case.body
481                                            .span_iter()
482                                            .next()
483                                            .map_or(Default::default(), |(_, s)| *s),
484                                        "duplicated switch arm here",
485                                    ),
486                                _ => FunctionError::ConflictingSwitchCase(case.value)
487                                    .with_span_static(
488                                        case.body
489                                            .span_iter()
490                                            .next()
491                                            .map_or(Default::default(), |(_, s)| *s),
492                                        "conflicting switch arm here",
493                                    ),
494                            });
495                        }
496                    }
497                    if !self.switch_values.contains(&crate::SwitchValue::Default) {
498                        return Err(FunctionError::MissingDefaultCase
499                            .with_span_static(span, "missing default case"));
500                    }
501                    if let Some(case) = cases.last() {
502                        if case.fall_through {
503                            return Err(FunctionError::LastCaseFallTrough.with_span_static(
504                                case.body
505                                    .span_iter()
506                                    .next()
507                                    .map_or(Default::default(), |(_, s)| *s),
508                                "bad switch arm here",
509                            ));
510                        }
511                    }
512                    let pass_through_abilities = context.abilities
513                        & (ControlFlowAbility::RETURN | ControlFlowAbility::CONTINUE);
514                    let sub_context =
515                        context.with_abilities(pass_through_abilities | ControlFlowAbility::BREAK);
516                    for case in cases {
517                        stages &= self.validate_block(&case.body, &sub_context)?.stages;
518                    }
519                }
520                S::Loop {
521                    ref body,
522                    ref continuing,
523                    break_if,
524                } => {
525                    // special handling for block scoping is needed here,
526                    // because the continuing{} block inherits the scope
527                    let base_expression_count = self.valid_expression_list.len();
528                    let pass_through_abilities = context.abilities & ControlFlowAbility::RETURN;
529                    stages &= self
530                        .validate_block_impl(
531                            body,
532                            &context.with_abilities(
533                                pass_through_abilities
534                                    | ControlFlowAbility::BREAK
535                                    | ControlFlowAbility::CONTINUE,
536                            ),
537                        )?
538                        .stages;
539                    stages &= self
540                        .validate_block_impl(
541                            continuing,
542                            &context.with_abilities(ControlFlowAbility::empty()),
543                        )?
544                        .stages;
545
546                    if let Some(condition) = break_if {
547                        match *context.resolve_type(condition, &self.valid_expression_set)? {
548                            Ti::Scalar(crate::Scalar {
549                                kind: crate::ScalarKind::Bool,
550                                width: _,
551                            }) => {}
552                            _ => {
553                                return Err(FunctionError::InvalidIfType(condition)
554                                    .with_span_handle(condition, context.expressions))
555                            }
556                        }
557                    }
558
559                    for handle in self.valid_expression_list.drain(base_expression_count..) {
560                        self.valid_expression_set.remove(handle.index());
561                    }
562                }
563                S::Break => {
564                    if !context.abilities.contains(ControlFlowAbility::BREAK) {
565                        return Err(FunctionError::BreakOutsideOfLoopOrSwitch
566                            .with_span_static(span, "invalid break"));
567                    }
568                    finished = true;
569                }
570                S::Continue => {
571                    if !context.abilities.contains(ControlFlowAbility::CONTINUE) {
572                        return Err(FunctionError::ContinueOutsideOfLoop
573                            .with_span_static(span, "invalid continue"));
574                    }
575                    finished = true;
576                }
577                S::Return { value } => {
578                    if !context.abilities.contains(ControlFlowAbility::RETURN) {
579                        return Err(FunctionError::InvalidReturnSpot
580                            .with_span_static(span, "invalid return"));
581                    }
582                    let value_ty = value
583                        .map(|expr| context.resolve_type(expr, &self.valid_expression_set))
584                        .transpose()?;
585                    let expected_ty = context.return_type.map(|ty| &context.types[ty].inner);
586                    // We can't return pointers, but it seems best not to embed that
587                    // assumption here, so use `TypeInner::equivalent` for comparison.
588                    let okay = match (value_ty, expected_ty) {
589                        (None, None) => true,
590                        (Some(value_inner), Some(expected_inner)) => {
591                            value_inner.equivalent(expected_inner, context.types)
592                        }
593                        (_, _) => false,
594                    };
595
596                    if !okay {
597                        log::error!(
598                            "Returning {:?} where {:?} is expected",
599                            value_ty,
600                            expected_ty
601                        );
602                        if let Some(handle) = value {
603                            return Err(FunctionError::InvalidReturnType(value)
604                                .with_span_handle(handle, context.expressions));
605                        } else {
606                            return Err(FunctionError::InvalidReturnType(value)
607                                .with_span_static(span, "invalid return"));
608                        }
609                    }
610                    finished = true;
611                }
612                S::Kill => {
613                    stages &= super::ShaderStages::FRAGMENT;
614                    finished = true;
615                }
616                S::Barrier(_) => {
617                    stages &= super::ShaderStages::COMPUTE;
618                }
619                S::Store { pointer, value } => {
620                    let mut current = pointer;
621                    loop {
622                        let _ = context
623                            .resolve_pointer_type(current)
624                            .map_err(|e| e.with_span())?;
625                        match context.expressions[current] {
626                            crate::Expression::Access { base, .. }
627                            | crate::Expression::AccessIndex { base, .. } => current = base,
628                            crate::Expression::LocalVariable(_)
629                            | crate::Expression::GlobalVariable(_)
630                            | crate::Expression::FunctionArgument(_) => break,
631                            _ => {
632                                return Err(FunctionError::InvalidStorePointer(current)
633                                    .with_span_handle(pointer, context.expressions))
634                            }
635                        }
636                    }
637
638                    let value_ty = context.resolve_type(value, &self.valid_expression_set)?;
639                    match *value_ty {
640                        Ti::Image { .. } | Ti::Sampler { .. } => {
641                            return Err(FunctionError::InvalidStoreValue(value)
642                                .with_span_handle(value, context.expressions));
643                        }
644                        _ => {}
645                    }
646
647                    let pointer_ty = context
648                        .resolve_pointer_type(pointer)
649                        .map_err(|e| e.with_span())?;
650
651                    let good = match *pointer_ty {
652                        Ti::Pointer { base, space: _ } => match context.types[base].inner {
653                            Ti::Atomic(scalar) => *value_ty == Ti::Scalar(scalar),
654                            ref other => value_ty == other,
655                        },
656                        Ti::ValuePointer {
657                            size: Some(size),
658                            scalar,
659                            space: _,
660                        } => *value_ty == Ti::Vector { size, scalar },
661                        Ti::ValuePointer {
662                            size: None,
663                            scalar,
664                            space: _,
665                        } => *value_ty == Ti::Scalar(scalar),
666                        _ => false,
667                    };
668                    if !good {
669                        return Err(FunctionError::InvalidStoreTypes { pointer, value }
670                            .with_span()
671                            .with_handle(pointer, context.expressions)
672                            .with_handle(value, context.expressions));
673                    }
674
675                    if let Some(space) = pointer_ty.pointer_space() {
676                        if !space.access().contains(crate::StorageAccess::STORE) {
677                            return Err(FunctionError::InvalidStorePointer(pointer)
678                                .with_span_static(
679                                    context.expressions.get_span(pointer),
680                                    "writing to this location is not permitted",
681                                ));
682                        }
683                    }
684                }
685                S::ImageStore {
686                    image,
687                    coordinate,
688                    array_index,
689                    value,
690                } => {
691                    //Note: this code uses a lot of `FunctionError::InvalidImageStore`,
692                    // and could probably be refactored.
693                    let var = match *context.get_expression(image) {
694                        crate::Expression::GlobalVariable(var_handle) => {
695                            &context.global_vars[var_handle]
696                        }
697                        // We're looking at a binding index situation, so punch through the index and look at the global behind it.
698                        crate::Expression::Access { base, .. }
699                        | crate::Expression::AccessIndex { base, .. } => {
700                            match *context.get_expression(base) {
701                                crate::Expression::GlobalVariable(var_handle) => {
702                                    &context.global_vars[var_handle]
703                                }
704                                _ => {
705                                    return Err(FunctionError::InvalidImageStore(
706                                        ExpressionError::ExpectedGlobalVariable,
707                                    )
708                                    .with_span_handle(image, context.expressions))
709                                }
710                            }
711                        }
712                        _ => {
713                            return Err(FunctionError::InvalidImageStore(
714                                ExpressionError::ExpectedGlobalVariable,
715                            )
716                            .with_span_handle(image, context.expressions))
717                        }
718                    };
719
720                    // Punch through a binding array to get the underlying type
721                    let global_ty = match context.types[var.ty].inner {
722                        Ti::BindingArray { base, .. } => &context.types[base].inner,
723                        ref inner => inner,
724                    };
725
726                    let value_ty = match *global_ty {
727                        Ti::Image {
728                            class,
729                            arrayed,
730                            dim,
731                        } => {
732                            match context
733                                .resolve_type(coordinate, &self.valid_expression_set)?
734                                .image_storage_coordinates()
735                            {
736                                Some(coord_dim) if coord_dim == dim => {}
737                                _ => {
738                                    return Err(FunctionError::InvalidImageStore(
739                                        ExpressionError::InvalidImageCoordinateType(
740                                            dim, coordinate,
741                                        ),
742                                    )
743                                    .with_span_handle(coordinate, context.expressions));
744                                }
745                            };
746                            if arrayed != array_index.is_some() {
747                                return Err(FunctionError::InvalidImageStore(
748                                    ExpressionError::InvalidImageArrayIndex,
749                                )
750                                .with_span_handle(coordinate, context.expressions));
751                            }
752                            if let Some(expr) = array_index {
753                                match *context.resolve_type(expr, &self.valid_expression_set)? {
754                                    Ti::Scalar(crate::Scalar {
755                                        kind: crate::ScalarKind::Sint | crate::ScalarKind::Uint,
756                                        width: _,
757                                    }) => {}
758                                    _ => {
759                                        return Err(FunctionError::InvalidImageStore(
760                                            ExpressionError::InvalidImageArrayIndexType(expr),
761                                        )
762                                        .with_span_handle(expr, context.expressions));
763                                    }
764                                }
765                            }
766                            match class {
767                                crate::ImageClass::Storage { format, .. } => {
768                                    crate::TypeInner::Vector {
769                                        size: crate::VectorSize::Quad,
770                                        scalar: crate::Scalar {
771                                            kind: format.into(),
772                                            width: 4,
773                                        },
774                                    }
775                                }
776                                _ => {
777                                    return Err(FunctionError::InvalidImageStore(
778                                        ExpressionError::InvalidImageClass(class),
779                                    )
780                                    .with_span_handle(image, context.expressions));
781                                }
782                            }
783                        }
784                        _ => {
785                            return Err(FunctionError::InvalidImageStore(
786                                ExpressionError::ExpectedImageType(var.ty),
787                            )
788                            .with_span()
789                            .with_handle(var.ty, context.types)
790                            .with_handle(image, context.expressions))
791                        }
792                    };
793
794                    if *context.resolve_type(value, &self.valid_expression_set)? != value_ty {
795                        return Err(FunctionError::InvalidStoreValue(value)
796                            .with_span_handle(value, context.expressions));
797                    }
798                }
799                S::Call {
800                    function,
801                    ref arguments,
802                    result,
803                } => match self.validate_call(function, arguments, result, context) {
804                    Ok(callee_stages) => stages &= callee_stages,
805                    Err(error) => {
806                        return Err(error.and_then(|error| {
807                            FunctionError::InvalidCall { function, error }
808                                .with_span_static(span, "invalid function call")
809                        }))
810                    }
811                },
812                S::Atomic {
813                    pointer,
814                    ref fun,
815                    value,
816                    result,
817                } => {
818                    self.validate_atomic(pointer, fun, value, result, context)?;
819                }
820                S::WorkGroupUniformLoad { pointer, result } => {
821                    stages &= super::ShaderStages::COMPUTE;
822                    let pointer_inner =
823                        context.resolve_type(pointer, &self.valid_expression_set)?;
824                    match *pointer_inner {
825                        Ti::Pointer {
826                            space: AddressSpace::WorkGroup,
827                            ..
828                        } => {}
829                        Ti::ValuePointer {
830                            space: AddressSpace::WorkGroup,
831                            ..
832                        } => {}
833                        _ => {
834                            return Err(FunctionError::WorkgroupUniformLoadInvalidPointer(pointer)
835                                .with_span_static(span, "WorkGroupUniformLoad"))
836                        }
837                    }
838                    self.emit_expression(result, context)?;
839                    let ty = match &context.expressions[result] {
840                        &crate::Expression::WorkGroupUniformLoadResult { ty } => ty,
841                        _ => {
842                            return Err(FunctionError::WorkgroupUniformLoadExpressionMismatch(
843                                result,
844                            )
845                            .with_span_static(span, "WorkGroupUniformLoad"));
846                        }
847                    };
848                    let expected_pointer_inner = Ti::Pointer {
849                        base: ty,
850                        space: AddressSpace::WorkGroup,
851                    };
852                    if !expected_pointer_inner.equivalent(pointer_inner, context.types) {
853                        return Err(FunctionError::WorkgroupUniformLoadInvalidPointer(pointer)
854                            .with_span_static(span, "WorkGroupUniformLoad"));
855                    }
856                }
857                S::RayQuery { query, ref fun } => {
858                    let query_var = match *context.get_expression(query) {
859                        crate::Expression::LocalVariable(var) => &context.local_vars[var],
860                        ref other => {
861                            log::error!("Unexpected ray query expression {other:?}");
862                            return Err(FunctionError::InvalidRayQueryExpression(query)
863                                .with_span_static(span, "invalid query expression"));
864                        }
865                    };
866                    match context.types[query_var.ty].inner {
867                        Ti::RayQuery => {}
868                        ref other => {
869                            log::error!("Unexpected ray query type {other:?}");
870                            return Err(FunctionError::InvalidRayQueryType(query_var.ty)
871                                .with_span_static(span, "invalid query type"));
872                        }
873                    }
874                    match *fun {
875                        crate::RayQueryFunction::Initialize {
876                            acceleration_structure,
877                            descriptor,
878                        } => {
879                            match *context
880                                .resolve_type(acceleration_structure, &self.valid_expression_set)?
881                            {
882                                Ti::AccelerationStructure => {}
883                                _ => {
884                                    return Err(FunctionError::InvalidAccelerationStructure(
885                                        acceleration_structure,
886                                    )
887                                    .with_span_static(span, "invalid acceleration structure"))
888                                }
889                            }
890                            let desc_ty_given =
891                                context.resolve_type(descriptor, &self.valid_expression_set)?;
892                            let desc_ty_expected = context
893                                .special_types
894                                .ray_desc
895                                .map(|handle| &context.types[handle].inner);
896                            if Some(desc_ty_given) != desc_ty_expected {
897                                return Err(FunctionError::InvalidRayDescriptor(descriptor)
898                                    .with_span_static(span, "invalid ray descriptor"));
899                            }
900                        }
901                        crate::RayQueryFunction::Proceed { result } => {
902                            self.emit_expression(result, context)?;
903                        }
904                        crate::RayQueryFunction::Terminate => {}
905                    }
906                }
907            }
908        }
909        Ok(BlockInfo { stages, finished })
910    }

912    fn validate_block(
913        &mut self,
914        statements: &crate::Block,
915        context: &BlockContext,
916    ) -> Result<BlockInfo, WithSpan<FunctionError>> {
917        let base_expression_count = self.valid_expression_list.len();
918        let info = self.validate_block_impl(statements, context)?;
919        for handle in self.valid_expression_list.drain(base_expression_count..) {
920            self.valid_expression_set.remove(handle.index());
921        }
922        Ok(info)
923    }

925    fn validate_local_var(
926        &self,
927        var: &crate::LocalVariable,
928        gctx: crate::proc::GlobalCtx,
929        fun_info: &FunctionInfo,
930        expression_constness: &crate::proc::ExpressionConstnessTracker,
931    ) -> Result<(), LocalVariableError> {
932        log::debug!("var {:?}", var);
933        let type_info = self
934            .types
935            .get(var.ty.index())
936            .ok_or(LocalVariableError::InvalidType(var.ty))?;
937        if !type_info.flags.contains(super::TypeFlags::CONSTRUCTIBLE) {
938            return Err(LocalVariableError::InvalidType(var.ty));
939        }
940
941        if let Some(init) = var.init {
942            let decl_ty = &gctx.types[var.ty].inner;
943            let init_ty = fun_info[init].ty.inner_with(gctx.types);
944            if !decl_ty.equivalent(init_ty, gctx.types) {
945                return Err(LocalVariableError::InitializerType);
946            }
947
948            if !expression_constness.is_const(init) {
949                return Err(LocalVariableError::NonConstInitializer);
950            }
951        }
952
953        Ok(())
954    }

956    pub(super) fn validate_function(
957        &mut self,
958        fun: &crate::Function,
959        module: &crate::Module,
960        mod_info: &ModuleInfo,
961        entry_point: bool,
962    ) -> Result<FunctionInfo, WithSpan<FunctionError>> {
963        let mut info = mod_info.process_function(fun, module, self.flags, self.capabilities)?;
964
965        let expression_constness =
966            crate::proc::ExpressionConstnessTracker::from_arena(&fun.expressions);
967
968        for (var_handle, var) in fun.local_variables.iter() {
969            self.validate_local_var(var, module.to_ctx(), &info, &expression_constness)
970                .map_err(|source| {
971                    FunctionError::LocalVariable {
972                        handle: var_handle,
973                        name: var.name.clone().unwrap_or_default(),
974                        source,
975                    }
976                    .with_span_handle(var.ty, &module.types)
977                    .with_handle(var_handle, &fun.local_variables)
978                })?;
979        }
980
981        for (index, argument) in fun.arguments.iter().enumerate() {
982            match module.types[argument.ty].inner.pointer_space() {
983                Some(crate::AddressSpace::Private | crate::AddressSpace::Function) | None => {}
984                Some(other) => {
985                    return Err(FunctionError::InvalidArgumentPointerSpace {
986                        index,
987                        name: argument.name.clone().unwrap_or_default(),
988                        space: other,
989                    }
990                    .with_span_handle(argument.ty, &module.types))
991                }
992            }
993            // Check for the least informative error last.
994            if !self.types[argument.ty.index()]
995                .flags
996                .contains(super::TypeFlags::ARGUMENT)
997            {
998                return Err(FunctionError::InvalidArgumentType {
999                    index,
1000                    name: argument.name.clone().unwrap_or_default(),
1001                }
1002                .with_span_handle(argument.ty, &module.types));
1003            }
1004
1005            if !entry_point && argument.binding.is_some() {
1006                return Err(FunctionError::PipelineInputRegularFunction {
1007                    name: argument.name.clone().unwrap_or_default(),
1008                }
1009                .with_span_handle(argument.ty, &module.types));
1010            }
1011        }
1012
1013        if let Some(ref result) = fun.result {
1014            if !self.types[result.ty.index()]
1015                .flags
1016                .contains(super::TypeFlags::CONSTRUCTIBLE)
1017            {
1018                return Err(FunctionError::NonConstructibleReturnType
1019                    .with_span_handle(result.ty, &module.types));
1020            }
1021
1022            if !entry_point && result.binding.is_some() {
1023                return Err(FunctionError::PipelineOutputRegularFunction
1024                    .with_span_handle(result.ty, &module.types));
1025            }
1026        }
1027
1028        self.valid_expression_set.clear();
1029        self.valid_expression_list.clear();
1030        for (handle, expr) in fun.expressions.iter() {
1031            if expr.needs_pre_emit() {
1032                self.valid_expression_set.insert(handle.index());
1033            }
1034            if self.flags.contains(super::ValidationFlags::EXPRESSIONS) {
1035                match self.validate_expression(handle, expr, fun, module, &info, mod_info) {
1036                    Ok(stages) => info.available_stages &= stages,
1037                    Err(source) => {
1038                        return Err(FunctionError::Expression { handle, source }
1039                            .with_span_handle(handle, &fun.expressions))
1040                    }
1041                }
1042            }
1043        }
1044
1045        if self.flags.contains(super::ValidationFlags::BLOCKS) {
1046            let stages = self
1047                .validate_block(
1048                    &fun.body,
1049                    &BlockContext::new(fun, module, &info, &mod_info.functions),
1050                )?
1051                .stages;
1052            info.available_stages &= stages;
1053        }
1054        Ok(info)
1055    }
