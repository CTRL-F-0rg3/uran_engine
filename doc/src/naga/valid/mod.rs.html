35bitflags::bitflags! {
36    /// Validation flags.
37    ///
38    /// If you are working with trusted shaders, then you may be able
39    /// to save some time by skipping validation.
40    ///
41    /// If you do not perform full validation, invalid shaders may
42    /// cause Naga to panic. If you do perform full validation and
43    /// [`Validator::validate`] returns `Ok`, then Naga promises that
44    /// code generation will either succeed or return an error; it
45    /// should never panic.
46    ///
47    /// The default value for `ValidationFlags` is
48    /// `ValidationFlags::all()`.
49    #[cfg_attr(feature = "serialize", derive(serde::Serialize))]
50    #[cfg_attr(feature = "deserialize", derive(serde::Deserialize))]
51    #[derive(Clone, Copy, Debug, Eq, PartialEq)]
52    pub struct ValidationFlags: u8 {
53        /// Expressions.
54        const EXPRESSIONS = 0x1;
55        /// Statements and blocks of them.
56        const BLOCKS = 0x2;
57        /// Uniformity of control flow for operations that require it.
58        const CONTROL_FLOW_UNIFORMITY = 0x4;
59        /// Host-shareable structure layouts.
60        const STRUCT_LAYOUTS = 0x8;
61        /// Constants.
62        const CONSTANTS = 0x10;
63        /// Group, binding, and location attributes.
64        const BINDINGS = 0x20;
65    }
66}
67
68impl Default for ValidationFlags {
69    fn default() -> Self {
70        Self::all()
71    }
72}

185#[derive(Clone, Debug, thiserror::Error)]
186pub enum ValidationError {
187    #[error(transparent)]
188    InvalidHandle(#[from] InvalidHandleError),
189    #[error(transparent)]
190    Layouter(#[from] LayoutError),
191    #[error("Type {handle:?} '{name}' is invalid")]
192    Type {
193        handle: Handle<crate::Type>,
194        name: String,
195        source: TypeError,
196    },
197    #[error("Constant expression {handle:?} is invalid")]
198    ConstExpression {
199        handle: Handle<crate::Expression>,
200        source: ConstExpressionError,
201    },
202    #[error("Constant {handle:?} '{name}' is invalid")]
203    Constant {
204        handle: Handle<crate::Constant>,
205        name: String,
206        source: ConstantError,
207    },
208    #[error("Global variable {handle:?} '{name}' is invalid")]
209    GlobalVariable {
210        handle: Handle<crate::GlobalVariable>,
211        name: String,
212        source: GlobalVariableError,
213    },
214    #[error("Function {handle:?} '{name}' is invalid")]
215    Function {
216        handle: Handle<crate::Function>,
217        name: String,
218        source: FunctionError,
219    },
220    #[error("Entry point {name} at {stage:?} is invalid")]
221    EntryPoint {
222        stage: crate::ShaderStage,
223        name: String,
224        source: EntryPointError,
225    },
226    #[error("Module is corrupted")]
227    Corrupted,

230impl crate::TypeInner {
231    const fn is_sized(&self) -> bool {
232        match *self {
233            Self::Scalar { .. }
234            | Self::Vector { .. }
235            | Self::Matrix { .. }
236            | Self::Array {
237                size: crate::ArraySize::Constant(_),
238                ..
239            }
240            | Self::Atomic { .. }
241            | Self::Pointer { .. }
242            | Self::ValuePointer { .. }
243            | Self::Struct { .. } => true,
244            Self::Array { .. }
245            | Self::Image { .. }
246            | Self::Sampler { .. }
247            | Self::AccelerationStructure
248            | Self::RayQuery
249            | Self::BindingArray { .. } => false,
250        }
251    }
252
253    /// Return the `ImageDimension` for which `self` is an appropriate coordinate.
254    const fn image_storage_coordinates(&self) -> Option<crate::ImageDimension> {
255        match *self {
256            Self::Scalar(crate::Scalar {
257                kind: crate::ScalarKind::Sint | crate::ScalarKind::Uint,
258                ..
259            }) => Some(crate::ImageDimension::D1),
260            Self::Vector {
261                size: crate::VectorSize::Bi,
262                scalar:
263                    crate::Scalar {
264                        kind: crate::ScalarKind::Sint | crate::ScalarKind::Uint,
265                        ..
266                    },
267            } => Some(crate::ImageDimension::D2),
268            Self::Vector {
269                size: crate::VectorSize::Tri,
270                scalar:
271                    crate::Scalar {
272                        kind: crate::ScalarKind::Sint | crate::ScalarKind::Uint,
273                        ..
274                    },
275            } => Some(crate::ImageDimension::D3),
276            _ => None,
277        }
278    }

308    fn validate_constant(
309        &self,
310        handle: Handle<crate::Constant>,
311        gctx: crate::proc::GlobalCtx,
312        mod_info: &ModuleInfo,
313    ) -> Result<(), ConstantError> {
314        let con = &gctx.constants[handle];
315
316        let type_info = &self.types[con.ty.index()];
317        if !type_info.flags.contains(TypeFlags::CONSTRUCTIBLE) {
318            return Err(ConstantError::NonConstructibleType);
319        }
320
321        let decl_ty = &gctx.types[con.ty].inner;
322        let init_ty = mod_info[con.init].inner_with(gctx.types);
323        if !decl_ty.equivalent(init_ty, gctx.types) {
324            return Err(ConstantError::InvalidType);
325        }
326
327        Ok(())
328    }

330    /// Check the given module to be valid.
331    pub fn validate(
332        &mut self,
333        module: &crate::Module,
334    ) -> Result<ModuleInfo, WithSpan<ValidationError>> {
335        self.reset();
336        self.reset_types(module.types.len());
337
338        Self::validate_module_handles(module).map_err(|e| e.with_span())?;
339
340        self.layouter.update(module.to_ctx()).map_err(|e| {
341            let handle = e.ty;
342            ValidationError::from(e).with_span_handle(handle, &module.types)
343        })?;
344
345        // These should all get overwritten.
346        let placeholder = TypeResolution::Value(crate::TypeInner::Scalar(crate::Scalar {
347            kind: crate::ScalarKind::Bool,
348            width: 0,
349        }));
350
351        let mut mod_info = ModuleInfo {
352            type_flags: Vec::with_capacity(module.types.len()),
353            functions: Vec::with_capacity(module.functions.len()),
354            entry_points: Vec::with_capacity(module.entry_points.len()),
355            const_expression_types: vec![placeholder; module.const_expressions.len()]
356                .into_boxed_slice(),
357        };
358
359        for (handle, ty) in module.types.iter() {
360            let ty_info = self
361                .validate_type(handle, module.to_ctx())
362                .map_err(|source| {
363                    ValidationError::Type {
364                        handle,
365                        name: ty.name.clone().unwrap_or_default(),
366                        source,
367                    }
368                    .with_span_handle(handle, &module.types)
369                })?;
370            mod_info.type_flags.push(ty_info.flags);
371            self.types[handle.index()] = ty_info;
372        }
373
374        {
375            let t = crate::Arena::new();
376            let resolve_context = crate::proc::ResolveContext::with_locals(module, &t, &[]);
377            for (handle, _) in module.const_expressions.iter() {
378                mod_info
379                    .process_const_expression(handle, &resolve_context, module.to_ctx())
380                    .map_err(|source| {
381                        ValidationError::ConstExpression { handle, source }
382                            .with_span_handle(handle, &module.const_expressions)
383                    })?
384            }
385        }
386
387        if self.flags.contains(ValidationFlags::CONSTANTS) {
388            for (handle, _) in module.const_expressions.iter() {
389                self.validate_const_expression(handle, module.to_ctx(), &mod_info)
390                    .map_err(|source| {
391                        ValidationError::ConstExpression { handle, source }
392                            .with_span_handle(handle, &module.const_expressions)
393                    })?
394            }
395
396            for (handle, constant) in module.constants.iter() {
397                self.validate_constant(handle, module.to_ctx(), &mod_info)
398                    .map_err(|source| {
399                        ValidationError::Constant {
400                            handle,
401                            name: constant.name.clone().unwrap_or_default(),
402                            source,
403                        }
404                        .with_span_handle(handle, &module.constants)
405                    })?
406            }
407        }
408
409        for (var_handle, var) in module.global_variables.iter() {
410            self.validate_global_var(var, module.to_ctx(), &mod_info)
411                .map_err(|source| {
412                    ValidationError::GlobalVariable {
413                        handle: var_handle,
414                        name: var.name.clone().unwrap_or_default(),
415                        source,
416                    }
417                    .with_span_handle(var_handle, &module.global_variables)
418                })?;
419        }
420
421        for (handle, fun) in module.functions.iter() {
422            match self.validate_function(fun, module, &mod_info, false) {
423                Ok(info) => mod_info.functions.push(info),
424                Err(error) => {
425                    return Err(error.and_then(|source| {
426                        ValidationError::Function {
427                            handle,
428                            name: fun.name.clone().unwrap_or_default(),
429                            source,
430                        }
431                        .with_span_handle(handle, &module.functions)
432                    }))
433                }
434            }
435        }
436
437        let mut ep_map = FastHashSet::default();
438        for ep in module.entry_points.iter() {
439            if !ep_map.insert((ep.stage, &ep.name)) {
440                return Err(ValidationError::EntryPoint {
441                    stage: ep.stage,
442                    name: ep.name.clone(),
443                    source: EntryPointError::Conflict,
444                }
445                .with_span()); // TODO: keep some EP span information?
446            }
447
448            match self.validate_entry_point(ep, module, &mod_info) {
449                Ok(info) => mod_info.entry_points.push(info),
450                Err(error) => {
451                    return Err(error.and_then(|source| {
452                        ValidationError::EntryPoint {
453                            stage: ep.stage,
454                            name: ep.name.clone(),
455                            source,
456                        }
457                        .with_span()
458                    }));
459                }
460            }
461        }
462
463        Ok(mod_info)
464    }

467fn validate_atomic_compare_exchange_struct(
468    types: &crate::UniqueArena<crate::Type>,
469    members: &[crate::StructMember],
470    scalar_predicate: impl FnOnce(&crate::TypeInner) -> bool,
471) -> bool {
472    members.len() == 2
473        && members[0].name.as_deref() == Some("old_value")
474        && scalar_predicate(&types[members[0].ty].inner)
475        && members[1].name.as_deref() == Some("exchanged")
476        && types[members[1].ty].inner == crate::TypeInner::Scalar(crate::Scalar::BOOL)
