1//! Piechota — tysiące żołnierzy symulowanych wyłącznie na GPU.
2//!
3//! Cała armia żyje w jednym `storage` bufferze na karcie. CPU wykonuje
4//! w klatce kilka operacji: aktualizuje punkt zgrupowania, pozycję
5//! gracza i wysyła 96 B parametrów. Ani jedna pozycja żołnierza nie
6//! wraca na CPU — HUD czyta wyłącznie cztery liczniki statystyk, które
7//! GPU wystawia do asynchronicznego odczytu.
8//!
9//! Podział na drużyny to zwykły podział zakresu indeksów w buforze:
10//! `0..friendly` to poddani gracza, dalej przeciwnicy, na końcu rezerwa.
11
12use uran_engine::prelude::*;
13// `GpuSim`, `SimParams`, `SimStats`, `Team`, `UnitFlags` są w prelude silnika.
14use uran_render::{GpuUnit, SimParams, Team, UnitFlags};
15
16/// Łączna liczba żołnierzy obu drużyn.
17///
18/// To jest właśnie sprawdzian: czy płynność zależy od liczby jednostek,
19/// czy od liczby *systemów* na CPU.
20pub const ARMY_CAPACITY: usize = 100_000;
21
22/// Zapas „rezerwy" (procent pojemności), którą wskrzesza klawisz F.
23const RESERVE_PERCENT: usize = 20;
24
25/// Ziarno losowości jednostki o indeksie `index` w globalnym buforze.
26///
27/// Używamy indeksu GŁÓWALNEGO (a nie numeru wewnątrz drużyny), bo obie
28/// drużyny numerują się od zera — inaczej żołnierz 7 gracza i żołnierz 7
29/// przeciwnika dostaliby identyczne ziarno, a armia wyglądałaby sztucznie
30/// równo. Mieszamy z „złotą" stałą, żeby zbiory nie pokrywały się.
31fn seed_of(index: usize) -> u32 {
32    (index as u32) ^ 0x9E37_79B9
33}
34
35/// Ile żołnierzy startuje w jednej partii przy uzupełnieniu (1/5 rezerwy).
36const REINFORCE_BATCHES: u32 = 5;
37
38/// Konfiguracja armii.
39#[derive(Debug, Clone, Copy)]
40pub struct InfantryConfig {
41    /// Pojemność bufora GPU (alokowana raz, przy starcie).
42    pub capacity: usize,
43    /// Ile żołnierzy startuje po stronie gracza.
44    pub friendly: usize,
45    /// Ile po stronie przeciwnika.
46    pub enemy: usize,
47    /// Promień pojedynczego żołnierza (jednostki świata).
48    pub unit_size: f32,
49    /// Prędkość marszu.
50    pub speed: f32,
51    /// Zasięg ostrzału.
52    pub attack_range: f32,
53    /// Obrażenia za strzał.
54    pub attack_damage: f32,
55    /// Czas między strzałami.
56    pub attack_cooldown: f32,
57}
58
59impl Default for InfantryConfig {
60    fn default() -> Self {
61        let capacity = ARMY_CAPACITY;
62        // po połowie na drużynę, reszta to rezerwa
63        let reserve = capacity * RESERVE_PERCENT / 100;
64        let per_team = (capacity - reserve) / 2;
65        Self {
66            capacity,
67            friendly: per_team,
68            enemy: per_team,
69            unit_size: 2.6,
70            speed: 78.0,
71            // zasięg musi obejmować kilka rzędów szyku, inaczej jednostki
72            // w środku kolumny nigdy nie widzą wroga i armia nie walczy
73            attack_range: 14.0,
74            attack_damage: 2.0,
75            attack_cooldown: 0.45,
76        }
77    }
78}
79
80impl InfantryConfig {
81    /// Pierwszy wolny indeks poza drużynami (początek rezerwy).
82    pub fn reserve_start(&self) -> usize {
83        self.friendly + self.enemy
84    }
85
86    /// Ta sama konfiguracja, ale o innej pojemności.
87    ///
88    /// Podział na drużyny i rezerwę liczymy OD NOWA — inaczej zmiana samego
89    /// `capacity` zostawiłaby `friendly + enemy > capacity` i armia
90    /// wyszłaby poza zaalokowany bufor.
91    pub fn with_capacity(&self, capacity: usize) -> Self {
92        let reserve = capacity * RESERVE_PERCENT / 100;
93        let per_team = (capacity - reserve) / 2;
94        Self {
95            capacity,
96            friendly: per_team,
97            enemy: per_team,
98            ..*self
99        }
100    }
101}

103/// Stan dowodzenia armią po stronie gry.
104pub struct Infantry {

115impl Infantry {
116    pub fn new(config: InfantryConfig) -> Self {
117        Self {
118            config,
119            rally: Vec2::ZERO,
120            time: 0.0,
121            reinforcements: 0,
122            rng_state: 0x5eed_1234,
123        }
124    }
125
126    pub fn config(&self) -> &InfantryConfig {
127        &self.config
128    }

130    /// Punkt zgrupowania (potrzebny do rysowania markera).
131    pub fn rally(&self) -> Vec2 {
132        self.rally
133    }

135    /// Generator xorshift32 — powtarzalny, więc demo wygląda tak samo
136    /// po każdym uruchomieniu (ułatwia porównywanie zrzutów).
137    fn rand(&mut self) -> f32 {
138        let mut x = self.rng_state;
139        x ^= x << 13;
140        x ^= x >> 17;
141        x ^= x << 5;
142        self.rng_state = x;
143        (x & 0x00FF_FFFF) as f32 / 0x00FF_FFFF as f32
144    }
145
146    fn rand_range(&mut self, min: f32, max: f32) -> f32 {
147        min + self.rand() * (max - min)
148    }
149
150    /// Formacja: szyk zwarty (kolumny) albo rozproszony.
151    ///
152    /// Zwarty szyk wygląda lepiej w boju i jest realistyczny; rozproszony
153    /// jest szybszy przy ogromnej armii (mniej kolizji w siatce). Dajemy
154    /// szykowi 3/4 armii, reszta rozproszona.
155    ///
156    /// CAŁY szyk musi zmieścić się w prostokącie `bounds` — inaczej
157    /// skrajne kolumny wystawałyby poza mapę. Dlatego liczbę kolumn
158    /// ograniczamy tak, aby ostatnia kolumna mieściła się w szerokości,
159    /// a rozstaw dobieramy z obu osi naraz (węższa wygrywa).
160    fn formation_point(
161        &mut self,
162        index: usize,
163        count: usize,
164        center: Vec2,
165        bounds: Rect,
166        min_spread: f32,
167    ) -> Vec2 {
168        if index * 4 < count * 3 {
169            // ile kolumn zmieści się w poziomie przy minimalnym rozstawie
170            let cols_by_width = ((bounds.width() / min_spread).floor() as usize).max(1);
171            // ile kolumn zmieści się w pionie przy minimalnym rozstawie
172            let rows_by_height = ((bounds.height() / min_spread).floor() as usize).max(1);
173            // ile kolumn potrzebujemy, żeby zmieścić tyle a tyle rzędów
174            let cols_needed = count.div_ceil(rows_by_height).max(1);
175            let cols = cols_needed.min(cols_by_width).max(1);
176            let rows = count.div_ceil(cols).max(1);
177
178            // rozstaw liczony osobno dla każdej osi; bierzemy mniejszy,
179            // żeby blok na pewno zmieścił się w OBU wymiarach
180            let col_spread = if cols > 1 {
181                bounds.width() / (cols as f32 - 1.0)
182            } else {
183                0.0
184            };
185            let row_spread = if rows > 1 {
186                bounds.height() / (rows as f32 - 1.0)
187            } else {
188                0.0
189            };
190            let spread = col_spread.min(row_spread);
191
192            // pozycja w szyku wyśrodkowana względem środka (o -0.5 kolumny/rzędu)
193            let col = (index % cols) as f32 - (cols as f32 - 1.0) * 0.5;
194            let row = (index / cols) as f32 - (rows as f32 - 1.0) * 0.5;
195            let p = center + Vec2::new(col * spread, row * spread);
196            // ostatnia linia: szyk mógł mieć mniej rzędów niż wyliczyliśmy,
197            // więc przesuwamy z powrotem do środka, żeby nie zostawić
198            // pustego przesunięcia na dole bloku
199            let rows_actual = count.div_ceil(cols).max(1) as f32;
200            let shift = (rows_actual - rows as f32) * 0.5 * spread;
201            Vec2::new(p.x, p.y + shift)
202        } else {
203            // rozproszona bryła: elipsa dopasowana do prostokąta szyku
204            let a = self.rand_range(0.0, std::f32::consts::TAU);
205            let r = self.rand().sqrt();
206            center
207                + Vec2::new(
208                    a.cos() * r * bounds.width() * 0.5,
209                    a.sin() * r * bounds.height() * 0.5,
210                )
211        }
212    }

214    /// Buduje początkową armię do jednorazowego wgrania na GPU.
215    ///
216    /// Zwraca dokładnie `capacity` wpisów, w kolejności:
217    /// `0..friendly` gracze, `friendly..friendly+enemy` przeciwnicy,
218    /// dalej martwa rezerwka. Dzięki stałemu podziałowi indeksów
219    /// `reinforce_range` wystarczy do wskrzeszania kolejnych partii.
220    pub fn build_army(&mut self, arena: Rect) -> Vec<GpuUnit> {
221        let cfg = self.config;
222        let mut units = Vec::with_capacity(cfg.capacity);
223        let min_spread = cfg.unit_size * 1.6;
224
225        // poddani gracza: szyk przy punkcie zgrupowania (środek areny).
226        // Margines 0.04 zostawiamy na promień żołnierza, żeby skrajne
227        // jednostki nie wychodziły poza krawędź areny.
228        let friendly_bounds = Rect::from_center(
229            Vec2::ZERO,
230            Vec2::new(arena.width() * 0.30, arena.height() * 0.56),
231        );
232        for i in 0..cfg.friendly {
233            let p = self.formation_point(i, cfg.friendly, Vec2::ZERO, friendly_bounds, min_spread);
234            units.push(GpuUnit::spawn(
235                p,
236                Vec2::ZERO,
237                Team::Friendly,
238                100.0,
239                seed_of(i),
240            ));
241        }
242
243        // przeciwnicy: dwa skrzydełka po bokach, żeby ruch formacji był widoczny.
244        // Środek skrzydła trzymamy w połowie odstępu między skrzydłami, a sam
245        // prostokąt ograniczamy do połowy tego odstępu — inaczej skrzydełka
246        // nachodziłyby na siebie i wychodziłyby poza arenę.
247        let wing_span = arena.width() * 0.30;
248        let wing_size = Vec2::new(wing_span, arena.height() * 0.46);
249        let wings = [
250            Vec2::new(arena.center().x - wing_span * 0.5, arena.center().y),
251            Vec2::new(arena.center().x + wing_span * 0.5, arena.center().y),
252        ];
253        for i in 0..cfg.enemy {
254            let center = wings[i % wings.len()];
255            let bounds = Rect::from_center(center, wing_size);
256            let p = self.formation_point(i, cfg.enemy, center, bounds, min_spread);
257            // ziarno z GLOBALNEGO indeksu slotu — dzięki temu przeciwnik
258            // nie powtarza wzoru jasności gracza
259            units.push(GpuUnit::spawn(
260                p,
261                Vec2::ZERO,
262                Team::Enemy,
263                100.0,
264                seed_of(cfg.friendly + i),
265            ));
266        }
267
268        // reszta bufora: martwa rezerwa do wskrzeszenia klawiszem F
269        for i in cfg.reserve_start()..cfg.capacity {
270            units.push(GpuUnit {
271                position: [0.0; 2],
272                velocity: [0.0; 2],
273                target: [0.0; 2],
274                health: 0.0,
275                cooldown: 0.0,
276                flags: UnitFlags::dead().with_team(Team::Friendly).0,
277                seed: i as u32,
278                _pad: [0.0; 2],
279            });
280        }
281        units
282    }

284    /// Wgrywa armię i ustawia parametry symulacji (jednorazowo, przy starcie).
285    ///
286    /// Zwraca `false`, gdy gra nie włączyła symulacji GPU — wtedy demo
287    /// po prostu nie ma piechoty, ale reszta działa.
288    pub fn install(&mut self, ctx: &mut Ctx, arena: Rect) -> bool {
289        // Najpierw budujemy armię na CPU — `ctx.sim_mut()` pożycza cały
290        // renderer, więc w trakcie tego pożyczania nie wolno sięgać po `ctx`.
291        let army = self.build_army(arena);
292        {
293            let Some(sim) = ctx.sim_mut() else {
294                return false;
295            };
296            *sim.params_mut() = SimParams::for_arena(arena, self.config.capacity, 8.0);
297            let style = sim.style_mut();
298            style.radius = self.config.unit_size;
299            style.alpha = 0.95;
300        }
301        ctx.upload_army(&army);
302        true
303    }

305    /// Dowodzenie: rozkazy gracza + wysyłka parametrów do GPU.
306    pub fn update(&mut self, ctx: &mut Ctx, dt: f32) {
307        self.time += dt;
308
309        // rozkaz: prawy przycisk = nowy punkt zgrupowania
310        if ctx.input.mouse_just_pressed(MouseButton::Right) {
311            self.rally = ctx.mouse_world();
312        }
313        // F = uzupełnienie z rezerwy
314        if ctx.input.just_pressed(Key::KeyF) {
315            self.reinforce(ctx);
316        }
317
318        // Wartości z kontekstu czytamy PRZED pożyczeniem symulatora —
319        // `sim_mut()` pożycza cały renderer na czas wywołania.
320        let player_pos = ctx.camera.position;
321        let Some(sim) = ctx.sim_mut() else { return };
322        let cfg = self.config;
323        let rally = self.rally;
324        let p = sim.params_mut();
325        p.dt = dt;
326        p.time = self.time;
327        p.player_pos = player_pos.to_array();
328        p.rally = rally.to_array();
329        p.max_speed = cfg.speed;
330        // Waga odpychania po normalizacji kierunku (patrz `think` w sim.wgsl).
331        // Za mało = jednostki wchodzą na siebie i armia zlewa się w jeden
332        // punkt; za dużo = szyk się rozrywa na całą mapę. 0.5 = odpychanie
333        // waży tyle co marsz, co utrzymuje formację w ruchu.
334        p.separation = 0.5;
335        p.separation_radius = cfg.unit_size * 2.2;
336        p.attack_range = cfg.attack_range;
337        p.attack_damage = cfg.attack_damage;
338        p.attack_cooldown = cfg.attack_cooldown;
339        p.unit_size = cfg.unit_size;
340    }

342    /// Wskrzesza kolejną partię z rezerwy wokół punktu zgrupowania.
343    fn reinforce(&mut self, ctx: &mut Ctx) {
344        let cfg = self.config;
345        let start_of_reserve = cfg.reserve_start();
346        if start_of_reserve >= cfg.capacity {
347            return;
348        }
349        let reserve = cfg.capacity - start_of_reserve;
350        let batch = reserve / REINFORCE_BATCHES as usize;
351        let which = (self.reinforcements % REINFORCE_BATCHES) as usize;
352        let from = start_of_reserve + which * batch;
353        let to = (from + batch).min(cfg.capacity);
354        ctx.reinforce_range(from, to, self.rally, Team::Friendly);
355        self.reinforcements += 1;
356    }

358    /// Rysuje marker punktu zgrupowania (w świecie, pod armią).
359    pub fn draw_rally_marker(&self, ctx: &mut Ctx) {
360        let pulse = 1.0 + 0.12 * (self.time * 3.0).sin();
361        ctx.gfx
362            .world_space()
363            .layer(-5)
364            .blend(BlendMode::Additive)
365            .color(Color::from_hex(0x4FC3F7).with_alpha(0.55))
366            .draw_rect(Rect::from_center(self.rally, Vec2::splat(22.0 * pulse)));
367        ctx.gfx
368            .color(Color::from_hex(0x4FC3F7).with_alpha(0.22))
369            .draw_rect(Rect::from_center(self.rally, Vec2::splat(70.0 * pulse)));
370    }

372    /// HUD armii. Statystyki pochodzą z GPU, więc mogą być kilka klatek
373    /// stare — to cena asynchronicznego odczytu.
374    ///
375    /// `font` podajemy jawnie (a nie przez `Ctx::font`), bo to ta sama
376    /// czcionka, którą gra już trzyma w swoim stanie.
377    pub fn draw_hud(&self, ctx: &mut Ctx, font: Font, _size: Vec2) {
378        let stats: SimStats = ctx.sim_stats();
379        let alive = stats.total_alive();
380        let samples = stats.samples;
381        // przed pierwszym odczytem liczników nie pokazujemy zer
382        let line = if samples == 0 {
383            "Piechota — czekam na pierwsze statystyki z GPU…".to_string()
384        } else {
385            format!(
386                "Piechota {} żołnierzy na GPU  ·  {} vs {} ({} żywych)  ·  {} strzałów  ·  {} poległo",
387                self.config.capacity,
388                stats.alive_friendly,
389                stats.alive_enemy,
390                alive,
391                stats.shots,
392                stats.kills,
393            )
394        };
395
396        ctx.gfx
397            .screen_space()
398            .layer(101)
399            .color(Color::from_hex(0x8FA6C4))
400            .draw_text(font, &line, Vec2::new(24.0, 58.0), 16.0, TextAlign::Left);
401        ctx.gfx.color(Color::from_hex(0x64789A)).draw_text(
402            font,
403            "RMB — nowy punkt zgrupowania    F — uzupełnienie z rezerwy",
404            Vec2::new(24.0, 80.0),
405            14.0,
406            TextAlign::Left,
407        );
408    }

507    #[test]
508    fn units_are_not_stacked_on_one_spot() {
509        // szyk ma rozłożyć jednostki, nie kłaść je w jednym punkcie
510        let mut inf = Infantry::new(small_config());
511        let army = inf.build_army(arena());
512        let friendly: Vec<Vec2> = army[..inf.config().friendly]
513            .iter()
514            .map(|u| u.pos())

516        let mut min_x = f32::MAX;
517        let mut max_x = f32::MIN;
518        for p in &friendly {
519            min_x = min_x.min(p.x);
520            max_x = max_x.max(p.x);
521        }
522        assert!(
523            max_x - min_x > 100.0,
524            "formacja nie rozłożyła się w poziomie (szerokość {})",

530    fn reinforce_batch_stays_within_reserve() {
531        // partie uzupełnienia nigdy nie wychodzą poza rezerwę
532        let c = small_config();
533        let reserve = c.capacity - c.reserve_start();
534        let batch = reserve / REINFORCE_BATCHES as usize;
535        for which in 0..REINFORCE_BATCHES as usize {
536            let from = c.reserve_start() + which * batch;
537            let to = (from + batch).min(c.capacity);
538            assert!(from >= c.reserve_start());
539            assert!(to <= c.capacity);
540            assert!(to > from, "partia {which} jest pusta");
541        }
