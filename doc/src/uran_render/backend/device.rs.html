9/// Błąd inicjalizacji renderera.
10#[derive(Debug)]
11pub enum RenderError {
12    Surface(wgpu::CreateSurfaceError),
13    Adapter(String),
14    Device(wgpu::RequestDeviceError),
15    NoSurfaceFormat,
16}
17
18impl fmt::Display for RenderError {
19    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
20        match self {
21            Self::Surface(e) => write!(f, "nie udało się utworzyć powierzchni: {e}"),
22            Self::Adapter(name) => write!(f, "brak adaptera GPU: {name}"),
23            Self::Device(e) => write!(f, "nie udało się utworzyć urządzenia: {e}"),
24            Self::NoSurfaceFormat => {
25                write!(f, "adapter nie udostępnia żadnego formatu powierzchni")
26            }
27        }
28    }
29}
30
31impl std::error::Error for RenderError {}
32
33impl From<wgpu::CreateSurfaceError> for RenderError {
34    fn from(e: wgpu::CreateSurfaceError) -> Self {
35        Self::Surface(e)
36    }
37}
38
39impl From<wgpu::RequestDeviceError> for RenderError {
40    fn from(e: wgpu::RequestDeviceError) -> Self {
41        Self::Device(e)
42    }
43}

45/// Rozmiar push constants potrzebny przez potok siatek
46/// (mat3x3<f32> = 48 bajtów + vec4 = 64).
47pub const PUSH_CONSTANT_SIZE: u32 = 64;
48
49/// Wszystkie zasoby GPU wymagane do renderowania 2D.
50pub struct GpuContext {
51    pub instance: wgpu::Instance,
52    pub surface: wgpu::Surface<'static>,
53    pub adapter: wgpu::Adapter,
54    pub device: wgpu::Device,
55    pub queue: wgpu::Queue,
56    pub config: wgpu::SurfaceConfiguration,
57    pub size: PhysicalSize<u32>,
58    /// Informacja o adapterze (do logów startowych).
59    pub info: wgpu::AdapterInfo,
60}
61
62impl GpuContext {
63    pub async fn new(window: Arc<Window>, vsync: bool, _samples: u32) -> Result<Self, RenderError> {
64        let size = window.inner_size();
65        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor {
66            backends: wgpu::Backends::all(),
67            ..Default::default()
68        });
69        let surface = instance.create_surface(window)?;
70
71        let adapter = instance
72            .request_adapter(&wgpu::RequestAdapterOptions {
73                power_preference: wgpu::PowerPreference::HighPerformance,
74                compatible_surface: Some(&surface),
75                force_fallback_adapter: false,
76            })
77            .await
78            .ok_or_else(|| {
79                RenderError::Adapter("żaden adapter nie pasuje do tej powierzchni".into())
80            })?;
81
82        let (device, queue) = adapter
83            .request_device(
84                &wgpu::DeviceDescriptor {
85                    label: Some("Uran Device"),
86                    required_features: wgpu::Features::PUSH_CONSTANTS,
87                    required_limits: wgpu::Limits {
88                        max_push_constant_size: PUSH_CONSTANT_SIZE,
89                        ..wgpu::Limits::downlevel_defaults()
90                    },
91                },
92                None,
93            )
94            .await?;
95
96        let caps = surface.get_capabilities(&adapter);
97        // sRGB daje poprawne mieszanie alfa; format bez sufiksu to plan B
98        let format = caps
99            .formats
100            .iter()
101            .copied()
102            .find(wgpu::TextureFormat::is_srgb)
103            .unwrap_or(caps.formats[0]);
104
105        let present_mode = if vsync {
106            wgpu::PresentMode::AutoVsync
107        } else {
108            match caps.present_modes.contains(&wgpu::PresentMode::AutoNoVsync) {
109                true => wgpu::PresentMode::AutoNoVsync,
110                false => wgpu::PresentMode::Immediate,
111            }
112        };
113
114        let config = wgpu::SurfaceConfiguration {
115            // COPY_SRC pozwala robić zrzuty ekranu (czytanie klatki z GPU)
116            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::COPY_SRC,
117            format,
118            width: size.width.max(1),
119            height: size.height.max(1),
120            present_mode,
121            alpha_mode: caps.alpha_modes[0],
122            view_formats: vec![],
123            desired_maximum_frame_latency: 2,
124        };
125        surface.configure(&device, &config);
126
127        Ok(Self {
128            info: adapter.get_info(),
129            instance,
130            surface,
131            adapter,
132            device,
133            queue,
134            config,
135            size,
136        })
137    }
138
139    /// Rozmiar okna w fizycznych pikselach.
140    pub fn size(&self) -> (u32, u32) {
141        (self.config.width, self.config.height)
142    }
143
144    /// Czy powierzchnia jest formatu sRGB (wpływa na format tekstur).
145    pub fn is_srgb_surface(&self) -> bool {
146        self.config.format.is_srgb()
147    }
148
149    /// Ponowna konfiguracja po zmianie rozmiaru okna.
150    pub fn resize(&mut self, size: PhysicalSize<u32>) {
151        if size.width == 0 || size.height == 0 {
152            return;
153        }
154        self.size = size;
155        self.config.width = size.width;
156        self.config.height = size.height;
157        self.surface.configure(&self.device, &self.config);
158    }
159
160    /// Limit rozmiaru tekstury (2D).
161    pub fn max_texture_size(&self) -> u32 {
162        self.device.limits().max_texture_dimension_2d
163    }
164}
