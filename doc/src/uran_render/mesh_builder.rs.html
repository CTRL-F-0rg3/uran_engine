9use uran_ecs::Mesh;
10use uran_math::{Color, Rect, Vec2};
11
12use crate::shape;
13
14/// Buduje [`Mesh`] krok po kroku.
15#[derive(Debug, Clone)]
16pub struct MeshBuilder {
17    mesh: Mesh,
18    color: Color,
19}
20
21impl MeshBuilder {
22    fn new(mesh: Mesh) -> Self {
23        Self {
24            mesh,
25            color: Color::WHITE,
26        }
27    }
28
29    /// Prostokąt (anchor: lewy dolny róg).
30    pub fn quad(rect: Rect) -> Self {
31        Self::new(Mesh::quad(rect.corners(), Color::WHITE))
32    }
33
34    /// Kwadrat o boku `size` wyśrodkowany w podanym punkcie.
35    pub fn square(center: Vec2, size: f32) -> Self {
36        Self::quad(Rect::from_center(center, Vec2::splat(size)))
37    }
38
39    /// Trójkąt o podanych wierzchołkach.
40    pub fn triangle(a: Vec2, b: Vec2, c: Vec2) -> Self {
41        Self::new(Mesh::triangle(a, b, c, Color::WHITE))
42    }
43
44    /// Okrąg wypełniony.
45    pub fn circle(center: Vec2, radius: f32, segments: u32) -> Self {
46        Self::new(shape::circle(center, radius, segments, Color::WHITE))
47    }
48
49    /// Prostokąt z zaokrąglonymi rogami.
50    pub fn rounded_rect(rect: Rect, radius: f32) -> Self {
51        Self::new(shape::rounded_rect(rect, radius, 4, Color::WHITE))
52    }
53
54    /// Wielokąt (triangulowany automatycznie).
55    pub fn polygon(points: &[Vec2]) -> Self {
56        Self::new(shape::polygon(points, Color::WHITE))
57    }
58
59    /// Odcinek o grubości.
60    pub fn line(from: Vec2, to: Vec2, width: f32) -> Self {
61        Self::new(shape::thick_line(from, to, width, Color::WHITE))
62    }
63
64    /// Z gotowej siatki (np. z ECS).
65    pub fn from_mesh(mesh: Mesh) -> Self {
66        Self::new(mesh)
67    }
68
69    /// Ustawia kolor wszystkich wierzchołków.
70    pub fn color(mut self, color: Color) -> Self {
71        self.color = color;
72        for vertex in &mut self.mesh.vertices {
73            vertex.color = color.to_array();
74        }
75        self
76    }
77
78    /// Kolor wierzchołków narastająco (gradient wzdłuż indeksów).
79    pub fn gradient(mut self, from: Color, to: Color) -> Self {
80        let count = self.mesh.vertices.len();
81        for (i, vertex) in self.mesh.vertices.iter_mut().enumerate() {
82            let t = if count <= 1 {
83                0.0
84            } else {
85                i as f32 / (count - 1) as f32
86            };
87            vertex.color = from.lerp(to, t).to_array();
88        }
89        self
90    }
91
92    /// Przesuwa całą siatkę.
93    pub fn translated(mut self, offset: Vec2) -> Self {
94        for vertex in &mut self.mesh.vertices {
95            vertex.position[0] += offset.x;
96            vertex.position[1] += offset.y;
97        }
98        self
99    }
100
101    /// Obraca siatkę wokół środka jej prostokąta ograniczającego.
102    pub fn rotated(mut self, radians: f32) -> Self {
103        let rect = self.bounds();
104        let center = rect.center();
105        let (sin, cos) = radians.sin_cos();
106        for vertex in &mut self.mesh.vertices {
107            let p = Vec2::new(vertex.position[0], vertex.position[1]) - center;
108            vertex.position[0] = (p.x * cos - p.y * sin) + center.x;
109            vertex.position[1] = (p.x * sin + p.y * cos) + center.y;
110        }
111        self
112    }
113
114    /// Prostokąt ograniczający siatkę.
115    pub fn bounds(&self) -> Rect {
116        let mut iter = self
117            .mesh
118            .vertices
119            .iter()
120            .map(|v| Vec2::new(v.position[0], v.position[1]));
121        let Some(first) = iter.next() else {
122            return Rect::ZERO;
123        };
124        let mut min = first;
125        let mut max = first;
126        for p in iter {
127            min = min.min(p);
128            max = max.max(p);
129        }
130        Rect::new(min, max)
131    }
132
133    /// Zwraca gotową siatkę.
134    pub fn build(self) -> Mesh {
135        self.mesh
136    }
137
138    /// Budowa z poprawieniem koloru (przydatne po `from_mesh`).
139    pub fn build_colored(mut self, color: Color) -> Mesh {
140        self.color = color;
141        self.build()
142    }
143}

145#[cfg(test)]
146mod tests {
147    use super::*;
148
149    #[test]
150    fn quad_has_two_triangles() {
151        let mesh = MeshBuilder::quad(Rect::from_xywh(0.0, 0.0, 10.0, 10.0)).build();
152        assert_eq!(mesh.triangle_count(), 2);
153        assert!(mesh.validate());
154        assert_eq!(mesh.vertices.len(), 4);
155    }
156
157    #[test]
158    fn color_applies_to_all_vertices() {
159        let mesh = MeshBuilder::quad(Rect::from_xywh(0.0, 0.0, 4.0, 4.0))
160            .color(Color::RED)
161            .build();
162        assert!(mesh
163            .vertices
164            .iter()
165            .all(|v| v.color == Color::RED.to_array()));
166    }
167
168    #[test]
169    fn gradient_varies_colors() {
170        let mesh = MeshBuilder::quad(Rect::from_xywh(0.0, 0.0, 4.0, 4.0))
171            .gradient(Color::BLACK, Color::WHITE)
172            .build();
173        let first = mesh.vertices[0].color;
174        let last = mesh.vertices[3].color;
175        assert!(first[0] < last[0]);
176    }
177
178    #[test]
179    fn translation_moves_bounds() {
180        let mesh = MeshBuilder::quad(Rect::from_xywh(0.0, 0.0, 10.0, 10.0))
181            .translated(Vec2::new(5.0, 5.0))
182            .build();
183        assert_eq!(mesh.vertices[0].position, [5.0, 5.0]);
184    }
185
186    #[test]
187    fn rotation_preserves_size() {
188        let rect = Rect::from_xywh(0.0, 0.0, 10.0, 20.0);
189        let mesh = MeshBuilder::quad(rect)
190            .rotated(std::f32::consts::FRAC_PI_4)
191            .build();
192        let bounds = MeshBuilder::from_mesh(mesh.clone()).bounds();
193        // środek zostaje na miejscu...
194        assert!((bounds.center() - rect.center()).length() < 1e-4);
195        // ...a realne pole siatki (nie AABB!) jest niezmienne
196        let total: f32 = mesh
197            .indices
198            .chunks(3)
199            .map(|t| {
200                let p = |i: u32| uran_math::Vec2::from_array(mesh.vertices[i as usize].position);
201                (p(t[1]) - p(t[0])).perp_dot(p(t[2]) - p(t[0])).abs() * 0.5
202            })
203            .sum();
204        assert!((total - 200.0).abs() < 0.1, "pole {total} != 200");
205    }
206
207    #[test]
208    fn bounds_of_empty_mesh_is_zero() {
209        assert_eq!(MeshBuilder::from_mesh(Mesh::default()).bounds(), Rect::ZERO);
210    }
