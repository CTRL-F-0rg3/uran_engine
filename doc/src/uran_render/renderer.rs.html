25/// Statystyki pojedynczej klatki (do HUD-a i debugu).
26#[derive(Debug, Clone, Copy, Default, PartialEq)]
27pub struct FrameStats {
28    /// Liczba sprite'ów (w tym glifów tekstu).
29    pub sprites: usize,
30    /// Liczba siatek.
31    pub meshes: usize,
32    /// Faktyczne draw calle — to jest liczba, która boli, gdy rośnie.
33    pub draw_calls: usize,
34    /// Liczba instancji przesłanych do GPU.
35    pub instances: usize,
36    /// Tekstury wgrane w tej klatce (0 = wszystko już w pamięci).
37    pub textures_uploaded: usize,
38}
39
40impl std::fmt::Display for FrameStats {
41    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
42        write!(
43            f,
44            "{} sprite'ów, {} siatek, {} draw calli, {} instancji",
45            self.sprites, self.meshes, self.draw_calls, self.instances
46        )
47    }
48}

112    pub async fn new(window: Arc<Window>, vsync: bool, samples: u32) -> Result<Self, RenderError> {
113        let gpu = GpuContext::new(window, vsync, samples).await?;
114        let device = &gpu.device;
115
116        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
117            label: Some("Uran 2D Shader"),
118            source: wgpu::ShaderSource::Wgsl(include_str!("shader.wgsl").into()),
119        });
120
121        // sampler i layout bind grupy tworzymy raz i współdzielimy między
122        // potokami a rejestrem tekstur (w wgpu 0.19 nie implementują `Clone`)
123        let pipelines_layout = create_texture_bind_group_layout(device);
124        let sampler = create_sampler(device);
125
126        let pipelines = PipelineCache::new(
127            device,
128            gpu.config.format,
129            samples,
130            &shader,
131            &pipelines_layout,
132            &sampler,
133        );
134        let textures = TextureRegistry::new(
135            device,
136            &gpu.queue,
137            pipelines_layout,
138            sampler,
139            gpu.is_srgb_surface(),
140        );
141
142        let quad_vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
143            label: Some("Uran Quad Vertices"),
144            contents: bytemuck::cast_slice(&unit_quad_vertices()),
145            usage: wgpu::BufferUsages::VERTEX,
146        });
147        let quad_index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
148            label: Some("Uran Quad Indices"),
149            contents: bytemuck::cast_slice(&quad_indices()),
150            usage: wgpu::BufferUsages::INDEX,
151        });
152
153        let globals_buffer = device.create_buffer(&wgpu::BufferDescriptor {
154            label: Some("Uran Globals"),
155            size: std::mem::size_of::<Globals>() as u64,
156            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
157            mapped_at_creation: false,
158        });
159        let globals_bind_group = pipelines.globals_bind_group(device, &globals_buffer);
160
161        let screen_globals_buffer = device.create_buffer(&wgpu::BufferDescriptor {
162            label: Some("Uran Screen Globals"),
163            size: std::mem::size_of::<Globals>() as u64,
164            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
165            mapped_at_creation: false,
166        });
167        let screen_globals_bind_group =
168            pipelines.globals_bind_group(device, &screen_globals_buffer);
169
170        let instance_capacity = 1024;
171        // bufor instancji tworzymy przed przeniesieniem `gpu` — inaczej
172        // pożyczka `&gpu.device` żyłaby dłużej niż samo `gpu`
173        let instance_buffer = create_instance_buffer(device, instance_capacity);
174
175        Ok(Self {
176            gpu,
177            pipelines,
178            textures,
179            fonts: FontRegistry::new(),
180            quad_vertex_buffer,
181            quad_index_buffer,
182            quad_index_count: 6,
183            instance_buffer,
184            instance_capacity,
185            globals_buffer,
186            globals_bind_group,
187            screen_globals_buffer,
188            screen_globals_bind_group,
189            meshes: HashMap::new(),
190            msaa_texture: None,
191            frame_index: 0,
192            mesh_ttl: 600,
193            sim: None,
194            scene3d: None,
195            screenshot: None,
196            atlas_dumped: false,
197            stats: FrameStats::default(),
198        })
199    }

339    /// Rozszerza listę o tekst: zamienia `TextDraw` na sprite'y glifów.
340    ///
341    /// Robimy to **przed** sortowaniem, żeby glify brały udział w batchowaniu
342    /// razem z resztą sceny.
343    pub fn prepare_text(&mut self, list: &mut DrawList, assets: &AssetServer) {
344        if list.texts().is_empty() {
345            return;
346        }
347        let texts = list.texts().to_vec();
348        let mut sprites = list.sprites().to_vec();
349        for text in &texts {
350            if let Some(data) = assets.font(text.font) {
351                let _ = self.fonts.load(text.font, &data.data);
352            }
353            self.fonts
354                .layout(text, Mat3::from_translation(text.position), &mut sprites);
355        }
356        list.replace_sprites(sprites);
357        list.clear_texts();
358    }

360    /// Wgrywa atlasy czcionek, które zmieniły się od ostatniej klatki.
361    pub fn sync_font_atlases(&mut self, fonts_used: &[uran_asset::HandleId]) {
362        if std::env::var("URAN_DUMP_ATLAS").is_ok() && !self.atlas_dumped {
363            for id in fonts_used {
364                if let Some(data) = self.fonts.atlas_data(*id) {
365                    let size = crate::text::ATLAS_SIZE;
366                    let mut rgba = Vec::with_capacity(data.len() * 4);
367                    for &c in data {
368                        // pokrycie -> czarno-biały obraz dla oka
369                        rgba.extend_from_slice(&[c, c, c, 255]);
370                    }
371                    if let Ok(img) = uran_asset::Image::from_rgba(size, size, rgba) {
372                        let path = format!("/tmp/uran_atlas_{}.png", id.index());
373                        let _ = img.save_png(std::path::Path::new(&path));
374                        eprintln!("[uran] zapisano atlas czcionki: {path}");
375                    }
376                }
377            }
378            self.atlas_dumped = true;
379        }
380        for key in fonts_used {
381            if !self.fonts.is_dirty(*key) {
382                continue;
383            }
384            if let Some(data) = self.fonts.atlas_data(*key) {
385                // `data` to wycinek rejestru, a `upload_atlas` potrzebuje
386                // dostępu do `self` — kopiujemy
387                let data = data.to_vec();
388                self.textures.upload_atlas(
389                    &self.gpu.device,
390                    TextureKey::FontAtlas(*key),
391                    crate::text::ATLAS_SIZE,
392                    &data,
393                    &self.gpu.queue,
394                );
395                self.fonts.mark_clean(*key);
396            }
397        }
398    }

400    /// Rozwijanie 9-slice: 9 kafelków w zależności od rozmiaru obrazu.
401    ///
402    /// Robimy to przed sortowaniem, bo renderer zna rozmiar tekstury, którego
403    /// nie ma w `DrawList`.
404    pub fn expand_nine_slices(&mut self, list: &mut DrawList, assets: &AssetServer) {
405        if list.nine_slices().is_empty() {
406            return;
407        }
408        let slices = list.nine_slices().to_vec();
409        let mut sprites = list.sprites().to_vec();
410
411        for slice in slices {
412            let image_size = assets
413                .image(slice.texture)
414                .map(|i| Vec2::new(i.width as f32, i.height as f32))
415                .unwrap_or(Vec2::ONE);
416
417            let dst = slice.dst;
418            let size = dst.size();
419            // margines w pikselach przeliczamy na UV i na piksele docelowe
420            let bx = (slice.border / image_size.x).clamp(0.0, 0.5);
421            let by = (slice.border / image_size.y).clamp(0.0, 0.5);
422            let px = (slice.border).min(size.x * 0.5);
423            let py = (slice.border).min(size.y * 0.5);
424
425            // (u0, v0, u1, v1) dla 3x3 kafelków w kolejności: L-D, Ś-D, P-D,
426            // L-Ś, Ś-Ś, P-Ś, L-G, Ś-G, P-G
427            let src = [
428                (0.0, 0.0, bx, by),
429                (bx, 0.0, 1.0 - bx, by),
430                (1.0 - bx, 0.0, 1.0, by),
431                (0.0, by, bx, 1.0 - by),
432                (bx, by, 1.0 - bx, 1.0 - by),
433                (1.0 - bx, by, 1.0, 1.0 - by),
434                (0.0, 1.0 - by, bx, 1.0),
435                (bx, 1.0 - by, 1.0 - bx, 1.0),
436                (1.0 - bx, 1.0 - by, 1.0, 1.0),
437            ];
438            let targets = [
439                Rect::new(dst.min, dst.min + Vec2::new(px, py)),
440                Rect::new(
441                    Vec2::new(dst.min.x + px, dst.min.y),
442                    Vec2::new(dst.max.x - px, dst.min.y + py),
443                ),
444                Rect::new(
445                    Vec2::new(dst.max.x - px, dst.min.y),
446                    Vec2::new(dst.max.x, dst.min.y + py),
447                ),
448                Rect::new(
449                    Vec2::new(dst.min.x, dst.min.y + py),
450                    Vec2::new(dst.min.x + px, dst.max.y - py),
451                ),
452                Rect::new(
453                    Vec2::new(dst.min.x + px, dst.min.y + py),
454                    Vec2::new(dst.max.x - px, dst.max.y - py),
455                ),
456                Rect::new(
457                    Vec2::new(dst.max.x - px, dst.min.y + py),
458                    Vec2::new(dst.max.x, dst.max.y - py),
459                ),
460                Rect::new(
461                    Vec2::new(dst.min.x, dst.max.y - py),
462                    Vec2::new(dst.min.x + px, dst.max.y),
463                ),
464                Rect::new(
465                    Vec2::new(dst.min.x + px, dst.max.y - py),
466                    Vec2::new(dst.max.x - px, dst.max.y),
467                ),
468                Rect::new(
469                    Vec2::new(dst.max.x - px, dst.max.y - py),
470                    Vec2::new(dst.max.x, dst.max.y),
471                ),
472            ];
473
474            for ((u0, v0, u1, v1), target) in src.iter().zip(targets) {
475                let mut sprite = SpriteDraw::rect(
476                    target,
477                    slice.transform,
478                    uran_ecs::UvRect::new(Vec2::new(*u0, *v0), Vec2::new(*u1, *v1)),
479                    slice.color,
480                );
481                sprite.texture = TextureKey::Image(slice.texture.into());
482                sprite.blend = slice.blend;
483                sprite.layer = slice.layer;
484                sprite.screen_space = slice.screen_space;
485                sprites.push(sprite);
486            }
487        }
488        list.replace_sprites(sprites);
489        list.clear_nine_slices();
490    }

492    /// Renderuje jedną klatkę. Lista zostanie posortowana (zmienia kolejność
493    /// elementów wewnątrz `list`).
494    pub fn render(
495        &mut self,
496        list: &mut DrawList,
497        camera: &Camera2d,
498        clear_color: uran_math::Color,
499        assets: &AssetServer,
500    ) -> FrameStats {
501        self.frame_index += 1;
502        self.textures.begin_frame();
503
504        // 1) tekst -> sprite'y, 9-slice -> 9 prostokątów, potem sortowanie
505        self.expand_nine_slices(list, assets);
506        self.prepare_text(list, assets);
507        list.sort();
508
509        // 2) atlasy czcionek, których używamy w tej klatce
510        let mut fonts_used: Vec<uran_asset::HandleId> = Vec::new();
511        for sprite in list.sprites() {
512            if let TextureKey::FontAtlas(id) = sprite.texture {
513                if !fonts_used.contains(&id) {
514                    fonts_used.push(id);
515                }
516            }
517        }
518        self.sync_font_atlases(&fonts_used);
519
520        // 3) instancje sprite'ów w kolejności rysowania
521        let sprites = list.sprites();
522        let capacity = (sprites.len().max(1) * 2).next_power_of_two();
523        if capacity > self.instance_capacity {
524            self.instance_buffer = create_instance_buffer(&self.gpu.device, capacity);
525            self.instance_capacity = capacity;
526        }
527        if !sprites.is_empty() {
528            let instances: Vec<SpriteInstance> = sprites
529                .iter()
530                .map(|s| {
531                    // Projekcja ekranowa ma oś Y skierowaną w dół, więc górna
532                    // krawędź kwadratu ląduje na dole ekranu. Bez zamiany V
533                    // tekst i obrazy w HUD-ie byłyby odwrócone pionowo.
534                    let uv = if s.screen_space { s.uv.flip_y() } else { s.uv };
535                    SpriteInstance::new(s.transform, uv, s.color)
536                })
537                .collect();
538            self.gpu
539                .queue
540                .write_buffer(&self.instance_buffer, 0, bytemuck::cast_slice(&instances));
541        }
542
543        // 3) faza przygotowania (potrzebuje &mut): wgrywamy tekstury i tworzymy
544        // bufory siatek, żeby w pętli rysującej móc korzystać z &self
545        for sprite in list.sprites() {
546            self.textures
547                .ensure_bound(&self.gpu.device, sprite.texture, assets, &self.gpu.queue);
548        }
549        for mesh in list.meshes() {
550            if let Some(geo) = list.geometry().get(mesh.geometry as usize) {
551                self.ensure_mesh(geo);
552            }
553        }
554
555        // 4) macierz świata -> NDC oraz macierz ekranu (dla HUD-u)
556        let (width, height) = self.gpu.size();
557        let window_size = Vec2::new(width as f32, height as f32);
558        let globals = Globals::new(camera.view_projection(window_size));
559        let screen_globals = Globals::new(crate::camera::screen_matrix(window_size));
560        self.gpu
561            .queue
562            .write_buffer(&self.globals_buffer, 0, bytemuck::bytes_of(&globals));
563        self.gpu.queue.write_buffer(
564            &self.screen_globals_buffer,
565            0,
566            bytemuck::bytes_of(&screen_globals),
567        );
568
569        if std::env::var("URAN_DEBUG").is_ok() && self.frame_index <= 1 {
570            eprintln!("[uran-render] okno={window_size:?}");
571            eprintln!("[uran-render]   świat view_proj={:?}", globals.view_proj);
572            eprintln!(
573                "[uran-render]   ekran view_proj={:?}",
574                screen_globals.view_proj
575            );
576        }
577
578        // 5) render pass
579        let surface_texture = match self.gpu.surface.get_current_texture() {
580            Ok(texture) => texture,
581            Err(wgpu::SurfaceError::Lost | wgpu::SurfaceError::Outdated) => {
582                self.gpu
583                    .surface
584                    .configure(&self.gpu.device, &self.gpu.config);
585                return FrameStats::default();
586            }
587            Err(wgpu::SurfaceError::Timeout) => return FrameStats::default(),
588            Err(e) => {
589                eprintln!("⚠️  błąd powierzchni: {e}");
590                return FrameStats::default();
591            }
592        };
593
594        let view = surface_texture
595            .texture
596            .create_view(&wgpu::TextureViewDescriptor::default());
597        let clear = clear_color.to_linear();
598        // `wgpu::TextureView` nie implementuje `Clone`, a potrzebujemy
599        // pożyczenia widoku MSAA bez trzymania pożyczki `&mut self` na czas
600        // wywołań rysujących. Dlatego „wypożyczamy" go z `Option`
601        // i wkładamy z powrotem na końcu klatki.
602        let mut msaa = self.msaa_texture.take();
603        if msaa.is_none() && self.pipelines.samples > 1 {
604            msaa = Some(self.create_msaa_view());
605        }
606        let (attachment, resolve) = match msaa.as_ref() {
607            Some(view) => (view, Some(view)),
608            None => (&view, None),
609        };
610
611        let mut encoder = self
612            .gpu
613            .device
614            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
615                label: Some("Uran Encoder"),
616            });
617
618        // 5a) symulacja jednostek na GPU — compute w tym samym encoderze,
619        // więc w tej samej klatce render pass widzi już nowe pozycje
620        if let Some(sim) = &mut self.sim {
621            sim.poll_stats();
622            sim.step(&mut encoder, &self.gpu.queue);
623        }
624
625        let mut stats = FrameStats {
626            sprites: sprites.len(),
627            meshes: list.meshes().len(),
628            ..Default::default()
629        };
630
631        // --- scena 3D: OSOBNY render pass, własny depth buffer ----------
632        // Idzie PRZED passem 2D i czyści kolor + głębokość. 2D dostaje potem
633        // `LoadOp::Load`, więc farma trafia na wierzech świata 3D (HUD,
634        // celownik, panele), a nie pod niego. Gdy 3D nie ma, 2D czyści sam.
635        let scene3d_active = self.scene3d.as_ref().is_some_and(|s| s.is_active());
636        if scene3d_active {
637            let target = crate::scene3d::Scene3dTarget {
638                color: attachment,
639                resolve,
640                format: self.pipelines.format,
641                samples: self.pipelines.samples,
642                gpu: &self.gpu,
643            };
644            if let Some(scene) = self.scene3d.as_mut() {
645                scene.draw(&target);
646            }
647        }
648
649        {
650            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
651                label: Some("Uran Render Pass"),
652                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
653                    view: attachment,
654                    resolve_target: resolve,
655                    ops: wgpu::Operations {
656                        // 3D zdążyło już wyrysować kadr — nie wolno go
657                        // wyczyścić, bo zniknęłaby cała scena.
658                        load: if scene3d_active {
659                            wgpu::LoadOp::Load
660                        } else {
661                            wgpu::LoadOp::Clear(wgpu::Color {
662                                r: clear.r as f64,
663                                g: clear.g as f64,
664                                b: clear.b as f64,
665                                a: clear.a as f64,
666                            })
667                        },
668                        store: wgpu::StoreOp::Store,
669                    },
670                })],
671                // 2D nie potrzebuje głębi: kolejność zależy od listy rysowania
672                depth_stencil_attachment: None,
673                timestamp_writes: None,
674                occlusion_query_set: None,
675            });
676
677            // Kolejność ma znaczenie dla czytelności:
678            //   1. armia z GPU (jest „pod" resztą sceny),
679            //   2. świat: sprite'y i siatki z listy rysowania,
680            //   3. HUD w przestrzeni ekranu.
681            //
682            // Armia musi być PRZED sprite'ami świata, inaczej 100k kółek
683            // zasłoniłoby teren; HUD zostaje na wierzchu, bo narysowany
684            // sprite'ami 2D i nie ma głębi.
685            if let Some(sim) = &self.sim {
686                let world_globals = &self.globals_bind_group;
687                if let Some(n) = sim.draw(&mut pass, world_globals) {
688                    stats.draw_calls += 1;
689                    stats.instances += n;
690                }
691            }
692            // faza rysowania — obie metody tylko odczytują stan renderera
693            stats.draw_calls += self.draw_sprites(&mut pass, list.sprites(), &mut stats);
694            stats.draw_calls += self.draw_meshes(&mut pass, list);
695        }
696
697        self.gpu.queue.submit(Some(encoder.finish()));
698
699        // Zmapowanie bufora statystyk DOPIERO po submicie — wcześniej
700        // walidacja wgpu odrzuca submit, bo bufor byłby zmapowany, a dopiero
701        // potem miał zostać zapisany przez `copy_buffer_to_buffer`.
702        if let Some(sim) = &mut self.sim {
703            sim.after_submit();
704        }
705
706        // Callback z `map_async` wywołuje wgpu dopiero przy `poll`. Bez tego
707        // statystyki nigdy nie dotarłyby z GPU i HUD pokazywałby zera.
708        // `Poll` nie czeka na GPU — tylko obsługuje callbacki, więc
709        // nie kosztuje przyszłego klatki.
710        self.gpu.device.poll(wgpu::Maintain::Poll);
711
712        if std::env::var("URAN_DEBUG").is_ok() && self.frame_index <= 1 {
713            eprintln!("[uran-render] okno={window_size:?}");
714            eprintln!("[uran-render]   świat  view_proj={:?}", globals.view_proj);
715            eprintln!(
716                "[uran-render]   ekran  view_proj={:?}",
717                screen_globals.view_proj
718            );
719            for (i, s) in list.sprites().iter().take(4).enumerate() {
720                let c = s.transform.to_cols_array_2d();
721                eprintln!(
722                    "[uran-render]   sprite[{i}] tekstura={:?} ekran={} macierz kolumny={:?}",
723                    s.texture, s.screen_space, c
724                );
725            }
726        }
727
728        // zrzut ekranu — kopiujemy zanim klatka zniknie z swapchainu
729        if let Some(request) = self.screenshot.as_mut() {
730            if request.frames == 0 {
731                let path = request.path.clone();
732                self.save_screenshot(&surface_texture, &path);
733                self.screenshot = None;
734            } else {
735                request.frames -= 1;
736            }
737        }
738
739        surface_texture.present();
740
741        // oddajemy widok MSAA do renderera
742        self.msaa_texture = msaa;
743        self.cleanup_unused_meshes();
744        self.stats = stats;
745        stats
746    }

748    /// Tworzy teksturę multisamplingową w aktualnym rozmiarze okna.
749    fn create_msaa_view(&self) -> wgpu::TextureView {
750        let (width, height) = self.gpu.size();
751        let texture = self.gpu.device.create_texture(&wgpu::TextureDescriptor {
752            label: Some("Uran MSAA"),
753            size: wgpu::Extent3d {
754                width,
755                height,
756                depth_or_array_layers: 1,
757            },
758            mip_level_count: 1,
759            sample_count: self.pipelines.samples,
760            dimension: wgpu::TextureDimension::D2,
761            format: self.pipelines.format,
762            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
763            view_formats: &[],
764        });
765        texture.create_view(&wgpu::TextureViewDescriptor::default())
766    }

768    /// Rysuje sprite'y: instancje leżą już w kolejności rysowania, więc
769    /// wystarczy rozciąć listę na przebiegi o jednakowej (tekstura, tryb).
770    ///
771    /// Wymaga wcześniejszego `ensure_bound` dla użytych tekstur (faza
772    /// przygotowania) — wtedy tu wystarczy niemutowalny dostęp.
773    fn draw_sprites<'a>(
774        &'a self,
775        pass: &mut wgpu::RenderPass<'a>,
776        sprites: &[SpriteDraw],
777        stats: &mut FrameStats,
778    ) -> usize {
779        if sprites.is_empty() {
780            return 0;
781        }
782        stats.instances = sprites.len();
783        pass.set_vertex_buffer(0, self.quad_vertex_buffer.slice(..));
784        pass.set_vertex_buffer(1, self.instance_buffer.slice(..));
785        pass.set_index_buffer(self.quad_index_buffer.slice(..), wgpu::IndexFormat::Uint32);
786
787        let mut draw_calls = 0;
788        let mut run_start = 0usize;
789        while run_start < sprites.len() {
790            let first = &sprites[run_start];
791            let mut run_end = run_start + 1;
792            while run_end < sprites.len() {
793                let next = &sprites[run_end];
794                // `screen_space` musi rozdzielać przebiegi: HUD i scena mają
795                // różne macierze, więc bez tego warunek cały HUD dostałby
796                // macierz świata (albo odwrotnie) i pojechałby w złym miejscu.
797                if next.texture != first.texture
798                    || next.blend != first.blend
799                    || next.screen_space != first.screen_space
800                {
801                    break;
802                }
803                run_end += 1;
804            }
805
806            pass.set_pipeline(self.pipelines.sprite(first.blend));
807            if first.screen_space {
808                pass.set_bind_group(1, &self.screen_globals_bind_group, &[]);
809            } else {
810                pass.set_bind_group(1, &self.globals_bind_group, &[]);
811            }
812            pass.set_bind_group(0, self.textures.get_bound(first.texture), &[]);
813            pass.draw_indexed(
814                0..self.quad_index_count,
815                0,
816                run_start as u32..run_end as u32,
817            );
818            draw_calls += 1;
819            run_start = run_end;
820        }
821        draw_calls
822    }

824    /// Rysuje siatki — każda to osobny draw call (mają własną geometrię).
825    ///
826    /// Wymaga wcześniejszego `ensure_mesh` (faza przygotowania).
827    fn draw_meshes<'a>(&'a self, pass: &mut wgpu::RenderPass<'a>, list: &DrawList) -> usize {
828        let geometry = list.geometry();
829        let mut draw_calls = 0;
830
831        for mesh in list.meshes() {
832            let Some(geo) = geometry.get(mesh.geometry as usize) else {
833                continue;
834            };
835            if geo.indices.is_empty() {
836                continue;
837            }
838            let Some(gpu_mesh) = self.meshes.get(&geo.hash) else {
839                continue;
840            };
841
842            let push = MeshPushConstants::new(mesh.transform, mesh.color);
843            pass.set_pipeline(self.pipelines.mesh(mesh.blend));
844            // HUD rysowany z siatek (np. ikony UI) musi dostać macierz ekranu
845            if mesh.screen_space {
846                pass.set_bind_group(1, &self.screen_globals_bind_group, &[]);
847            } else {
848                pass.set_bind_group(1, &self.globals_bind_group, &[]);
849            }
850            pass.set_bind_group(0, self.textures.get_bound(mesh.texture), &[]);
851            pass.set_push_constants(wgpu::ShaderStages::VERTEX, 0, bytemuck::bytes_of(&push));
852            pass.set_vertex_buffer(0, gpu_mesh.vertex_buffer.slice(..));
853            pass.set_index_buffer(gpu_mesh.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
854            pass.draw_indexed(0..gpu_mesh.index_count, 0, 0..1);
855            draw_calls += 1;
856        }
857        draw_calls
858    }

860    /// Tworzy bufory GPU dla geometrii, jeśli jeszcze ich nie ma (faza
861    /// przygotowania; `last_used_frame` odświeżamy tu, bo w pętli rysującej
862    /// mamy tylko `&self`).
863    fn ensure_mesh(&mut self, geometry: &MeshGeometry) {
864        let frame = self.frame_index;
865        if let Some(mesh) = self.meshes.get_mut(&geometry.hash) {
866            mesh.last_used_frame = frame;
867            return;
868        }
869        // Kolory wierzchołków konwertujemy sRGB -> liniowe TUTAJ, przy
870        // uploadzie. Dzięki temu shader nie musi tego robić, a tint
871        // (który `MeshPushConstants::new` też konwertuje) mnoży już
872        // liniowe wartości — inaczej siatki wychodzą niemal czarne.
873        let mut vertices = geometry.vertices.clone();
874        for v in &mut vertices {
875            let c = uran_math::Color::rgba(v.color[0], v.color[1], v.color[2], v.color[3]);
876            v.color = c.to_linear().to_array();
877        }
878        let vertex_buffer = self
879            .gpu
880            .device
881            .create_buffer_init(&wgpu::util::BufferInitDescriptor {
882                label: Some("Uran Mesh Vertices"),
883                contents: bytemuck::cast_slice(&vertices),
884                usage: wgpu::BufferUsages::VERTEX,
885            });
886        let index_buffer = self
887            .gpu
888            .device
889            .create_buffer_init(&wgpu::util::BufferInitDescriptor {
890                label: Some("Uran Mesh Indices"),
891                contents: bytemuck::cast_slice(&geometry.indices),
892                usage: wgpu::BufferUsages::INDEX,
893            });
894        self.meshes.insert(
895            geometry.hash,
896            GpuMesh {
897                vertex_buffer,
898                index_buffer,
899                index_count: geometry.indices.len() as u32,
900                last_used_frame: frame,
901            },
902        );
903    }

905    /// Zwalnia siatki nieużywane od dawna (rosnący ślad w cache'u).
906    fn cleanup_unused_meshes(&mut self) {
907        if self.meshes.len() < 256 {
908            return;
909        }
910        let frame = self.frame_index;
911        let ttl = self.mesh_ttl;
912        self.meshes
913            .retain(|_, mesh| frame.saturating_sub(mesh.last_used_frame) < ttl);
914    }

916    /// Zapisuje bieżącą klatkę swapchainu do pliku PNG.
917    ///
918    /// Czyta piksele z GPU i zamienia format powierzchni (może być BGRA)
919    /// na RGBA, którego oczekuje `image`.
920    fn save_screenshot(&self, surface: &wgpu::SurfaceTexture, path: &std::path::Path) {
921        let (width, height) = self.gpu.size();
922        let format = self.gpu.config.format;
923        let bytes_per_row = width * 4;
924        // `write_buffer`/kopiowanie wymaga wiersza wyrównanego do 256 bajtów
925        let padded = bytes_per_row.div_ceil(256) * 256;
926        let buffer = self.gpu.device.create_buffer(&wgpu::BufferDescriptor {
927            label: Some("Uran Screenshot"),
928            size: (padded * height) as u64,
929            usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
930            mapped_at_creation: false,
931        });
932
933        let mut encoder = self
934            .gpu
935            .device
936            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
937                label: Some("Screenshot Copy"),
938            });
939        encoder.copy_texture_to_buffer(
940            wgpu::ImageCopyTexture {
941                texture: &surface.texture,
942                mip_level: 0,
943                origin: wgpu::Origin3d::ZERO,
944                aspect: wgpu::TextureAspect::All,
945            },
946            wgpu::ImageCopyBuffer {
947                buffer: &buffer,
948                layout: wgpu::ImageDataLayout {
949                    offset: 0,
950                    bytes_per_row: Some(padded),
951                    rows_per_image: Some(height),
952                },
953            },
954            wgpu::Extent3d {
955                width,
956                height,
957                depth_or_array_layers: 1,
958            },
959        );
960        self.gpu.queue.submit(Some(encoder.finish()));
961
962        // mapowanie bufora jest asynchroniczne — blokujemy wątek, bo
963        // zrzut ekranu nie jest operacją hot pathu
964        let slice = buffer.slice(..);
965        let (tx, rx) = std::sync::mpsc::channel();
966        slice.map_async(wgpu::MapMode::Read, move |r| {
967            let _ = tx.send(r);
968        });
969        self.gpu.device.poll(wgpu::Maintain::Wait);
970        if rx.recv().is_err() {
971            eprintln!("⚠️  nie udało się zmapować bufora zrzutu");
972            return;
973        }
974
975        let data = slice.get_mapped_range();
976        // zdekodowanie wierszy wyrównanych do 256 bajtów + ewentualna zamiana
977        // BGRA -> RGBA
978        let bgra = format == wgpu::TextureFormat::Bgra8Unorm
979            || format == wgpu::TextureFormat::Bgra8UnormSrgb;
980        let mut rgba = Vec::with_capacity((bytes_per_row * height) as usize);
981        for row in 0..height {
982            let start = (row * padded) as usize;
983            let line = &data[start..start + bytes_per_row as usize];
984            if bgra {
985                rgba.extend(line.chunks(4).flat_map(|p| [p[2], p[1], p[0], p[3]]));
986            } else {
987                rgba.extend_from_slice(line);
988            }
989        }
990        drop(data);
991        buffer.unmap();
992
993        match uran_asset::Image::from_rgba(width, height, rgba) {
994            Ok(image) => match image.save_png(path) {
995                Ok(()) => println!("📸 zapisano zrzut ekranu: {}", path.display()),
996                Err(e) => eprintln!("⚠️  nie udało się zapisać PNG: {e}"),
997            },
998            Err(e) => eprintln!("⚠️  nie udało się zbudować obrazu: {e}"),
999        }
1000    }
