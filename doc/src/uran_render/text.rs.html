1//! Tekst: atlas glifów rasteryzowanych na żądanie + składanie tekstu.
2//!
3//! Podejście jest takie samo jak w większości silników 2D: glif rasteryzujemy
4//! raz, wkładamy do atlasu R8 i od tej chwili każdy znak jest zwykłym
5//! sprite'em z prostokątnym UV. Dzięki temu cały HUD to nadal jeden batch,
6//! bez osobnego potoku do tekstu.

16/// Bok atlasu glifów w pikselach.
17pub const ATLAS_SIZE: u32 = 1024;
18
19/// Odstęp między glifami w atlasie (zapobiega „prześwitywaniu" przy filtrowaniu).
20const ATLAS_PADDING: u32 = 1;
21
22/// Limit znaków przetwarzanych w jednym wywołaniu.
23const MAX_TEXT_CHARS: usize = 8192;
24
25/// Pojedynczy glif w atlasie.
26#[derive(Debug, Clone, Copy, PartialEq)]
27pub struct GlyphInfo {
28    /// Prostokąt glifu w atlasie (piksele).
29    pub rect: Rect,
30    /// Przesunięcie od początku linii bazowej (piksele).
31    pub offset: Vec2,
32    /// Rozmiar glifu (piksele).
33    pub size: Vec2,
34    /// Szerokość posuwu (piksele).
35    pub advance: f32,
36}
37
38/// Stan jednej czcionki: atlas + cache glifów.
39struct FontEntry {
40    font: FontVec,
41    units_per_em: f32,
42    ascent: f32,
43    descent: f32,
44    line_gap: f32,
45    /// Zawartość atlasu (R8, `ATLAS_SIZE` x `ATLAS_SIZE`).
46    pixels: Vec<u8>,
47    /// Kolejna wolna kolumna na bieżącej półce.
48    pen_x: u32,
49    /// Wysokość zajętych półek.
50    shelf_y: u32,
51    /// Największa wysokość glifu na BIEŻĄCEJ półce.
52    ///
53    /// Bez tego nowa półka zaczynałaby się zbyt wysoko (lub zbyt nisko,
54    /// jeśli przesuwamy ją o wysokość pojedynczego glifu) i glify z
55    /// różnych półek nachodziłyby na siebie.
56    shelf_height: u32,
57    /// Rasteryzowany rozmiar w pikselach.
58    raster_size: u32,
59    /// Czy atlas zmienił się od ostatniego wgrania na GPU.
60    dirty: bool,
61    glyphs: HashMap<GlyphId, Option<GlyphInfo>>,
62    /// Ile razy atlas się zapełnił (diagnostyka).
63    resets: u32,
64}
65
66impl FontEntry {
67    fn new(data: &[u8], raster_size: u32) -> Result<Self, InvalidFont> {
68        let font = FontVec::try_from_vec(data.to_vec())?;
69        Ok(Self {
70            units_per_em: font.units_per_em().unwrap_or(1000.0),
71            ascent: font.ascent_unscaled(),
72            descent: font.descent_unscaled(),
73            line_gap: font.line_gap_unscaled(),
74            font,
75            pixels: vec![0; (ATLAS_SIZE * ATLAS_SIZE) as usize],
76            pen_x: 0,
77            shelf_y: 0,
78            shelf_height: 0,
79            raster_size,
80            dirty: true,
81            glyphs: HashMap::new(),
82            resets: 0,
83        })
84    }
85
86    /// Zwalnia miejsce w atlasie (pełne wypełnienie = zaczynamy od nowa).
87    fn reset_atlas(&mut self) {
88        self.pixels.fill(0);
89        self.pen_x = 0;
90        self.shelf_y = 0;
91        self.shelf_height = 0;
92        self.glyphs.clear();
93        self.resets += 1;
94        self.dirty = true;
95    }
96
97    /// Rezerwuje prostokąt w atlasie; `None` = brak miejsca.
98    ///
99    /// UWAGA na `shelf_height`: nowa półka zaczyna się PO
100    /// `shelf_y + shelf_height`, a NIE po `shelf_y + h`. Glify mają różne
101    /// wysokości (kropka 2 px, „W" 20 px) i przesuwanie półki o wysokość
102    /// pojedynczego glifu kładło niski znak NA bitmapach wyżej położonych
103    /// liter. Sąsiadujące glify nachodziły na siebie i w HUD-zie „o"
104    /// wyglądało jak „q".
105    fn allocate(&mut self, w: u32, h: u32) -> Option<(u32, u32)> {
106        if w + ATLAS_PADDING > ATLAS_SIZE || h + ATLAS_PADDING > ATLAS_SIZE {
107            return None;
108        }
109        if self.pen_x + w + ATLAS_PADDING > ATLAS_SIZE {
110            // zamykamy bieżącą półkę i zaczynamy nową na jej wysokości
111            self.shelf_y += self.shelf_height + ATLAS_PADDING;
112            self.pen_x = 0;
113            self.shelf_height = 0;
114        }
115        if self.shelf_y + h + ATLAS_PADDING > ATLAS_SIZE {
116            // brak miejsca — wywołujący zdecyduje, czy zrobić reset
117            return None;
118        }
119        let at = (self.pen_x, self.shelf_y);
120        self.pen_x += w + ATLAS_PADDING;
121        self.shelf_height = self.shelf_height.max(h);
122        self.dirty = true;
123        Some(at)
124    }
125
126    /// Rasteryzuje glif (raz) i zwraca jego pozycję w atlasie.
127    fn rasterize(&mut self, id: GlyphId) -> Option<GlyphInfo> {
128        if let Some(cached) = self.glyphs.get(&id) {
129            return *cached;
130        }
131        // UWAGA: `PxScale` w ab_glyph oznacza liczbę pikseli na `em`, a NIE
132        // na jednostkę fontu — ab_glyph sam dzieli przez `units_per_em`.
133        // Podanie `raster_size / upem` kurczyło cały em do ułamka piksela
134        // i każdy glif wychodził 1x1 (tekst był niewidoczny).
135        let scale = PxScale::from(self.raster_size as f32);
136        let outlined = self.font.outline_glyph(ab_glyph::Glyph {
137            id,
138            scale,
139            position: ab_glyph::Point { x: 0.0, y: 0.0 },
140        })?;
141        let bounds = outlined.px_bounds();
142        let w = bounds.width().ceil().max(0.0) as u32;
143        let h = bounds.height().ceil().max(0.0) as u32;
144        let advance =
145            self.font.h_advance_unscaled(id) / self.units_per_em * self.raster_size as f32;
146
147        if w == 0 || h == 0 {
148            // spacja albo znak bez pokrycia
149            let info = GlyphInfo {
150                rect: Rect::ZERO,
151                offset: Vec2::ZERO,
152                size: Vec2::ZERO,
153                advance,
154            };
155            self.glyphs.insert(id, Some(info));
156            return Some(info);
157        }
158
159        // nie zmieścimy się na żadnej półce — czyścimy atlas i próbujemy od nowa
160        let at = match self.allocate(w, h) {
161            Some(at) => at,
162            None => {
163                self.reset_atlas();
164                self.allocate(w, h)?
165            }
166        };
167        let (x, y) = at;
168
169        // Rysujemy z marginesem, żeby filtrowanie liniowe nie łapało sąsiadów.
170        //
171        // UWAGA: `Outlined::draw` podaje współrzędne JUŻ przesunięte o
172        // `-px_bounds.min` i z osią Y skierowaną w dół, czyli dokładnie
173        // jako indeksy w bitmapie glifu. Odejmowanie `bounds.min` po raz
174        // drugi wyrzucało poza bitmapę i zostawał ledwie widoczny pasek.
175        outlined.draw(|px, py, coverage| {
176            if px >= w || py >= h {
177                return;
178            }
179            let dst = ((y + py) * ATLAS_SIZE + x + px) as usize;
180            self.pixels[dst] = (coverage.clamp(0.0, 1.0) * 255.0).round() as u8;
181        });
182
183        // `size` i `offset` podajemy jako UŁAMEK wysokości `em` (bitmap
184        // dzielony przez `raster_size`), a nie w jednostkach fontu — dzięki
185        // temu `layout` mnoży je przez `cmd.size / units_per_em` i dostaje
186        // rozmiar w jednostkach świata, niezależnie od rozdzielczości atlasu.
187        let em = self.raster_size as f32;
188        let info = GlyphInfo {
189            rect: Rect::from_xywh(x as f32, y as f32, w as f32, h as f32),
190            offset: Vec2::new(bounds.min.x / em, bounds.min.y / em),
191            size: Vec2::new(w as f32 / em, h as f32 / em),
192            advance: advance / em,
193        };
194        self.glyphs.insert(id, Some(info));
195        Some(info)
196    }
197}

220/// Rejestr czcionek: trzyma atlasy i składa tekst w sprite'y.
221pub struct FontRegistry {
222    fonts: Vec<Option<FontEntry>>,
223    raster_size: u32,
224}

226impl Default for FontRegistry {
227    fn default() -> Self {
228        Self::new()
229    }

232impl FontRegistry {
233    /// Domyślny rozmiar rasteryzacji. 48 px to dobry kompromis jakość/miejsce
234    /// dla typowych rozmiarów czcionek 12–32 px.
235    pub const DEFAULT_RASTER_SIZE: u32 = 48;
236
237    pub fn new() -> Self {
238        Self {
239            fonts: Vec::new(),
240            raster_size: Self::DEFAULT_RASTER_SIZE,
241        }
242    }
243
244    /// Zmiana skali rasteryzacji unieważnia atlasy.
245    pub fn set_raster_size(&mut self, size: u32) {
246        let size = size.clamp(8, 96);
247        if size != self.raster_size {
248            self.raster_size = size;
249            self.fonts.clear();
250        }
251    }
252
253    pub fn raster_size(&self) -> u32 {
254        self.raster_size
255    }
256
257    /// Wczytuje czcionkę (idempotentne — powtórne wywołanie nic nie robi).
258    pub fn load(&mut self, handle: Handle<FontData>, data: &[u8]) -> Result<(), InvalidFont> {
259        let index = handle.index() as usize;
260        if self.fonts.len() <= index {
261            self.fonts.resize_with(index + 1, || None);
262        }
263        if self.fonts[index].is_some() {
264            return Ok(());
265        }
266        self.fonts[index] = Some(FontEntry::new(data, self.raster_size)?);
267        Ok(())
268    }
269
270    /// Czy atlas czcionki zmienił się od ostatniego wgrania na GPU.
271    pub fn is_dirty(&self, handle: HandleId) -> bool {
272        self.fonts
273            .get(handle.index() as usize)
274            .and_then(|f| f.as_ref())
275            .map(|f| f.dirty)
276            .unwrap_or(false)
277    }
278
279    pub fn mark_clean(&mut self, handle: HandleId) {
280        if let Some(Some(entry)) = self.fonts.get_mut(handle.index() as usize) {
281            entry.dirty = false;
282        }
283    }
284
285    /// Bajty atlasu do wgrania na GPU (R8).
286    pub fn atlas_data(&self, handle: HandleId) -> Option<&[u8]> {
287        self.fonts
288            .get(handle.index() as usize)
289            .and_then(|f| f.as_ref())
290            .map(|f| f.pixels.as_slice())
291    }

293    /// Metryki czcionki dla zadanego rozmiaru tekstu.
294    pub fn metrics(&self, handle: HandleId, size: f32) -> FontMetrics {
295        let Some(entry) = self
296            .fonts
297            .get(handle.index() as usize)
298            .and_then(|f| f.as_ref())
299        else {
300            return FontMetrics {
301                line_height: size,
302                ascent: size,
303                descent: 0.0,
304            };
305        };
306        let s = size / entry.units_per_em;
307        FontMetrics {
308            line_height: (entry.ascent - entry.descent + entry.line_gap) * s,
309            ascent: entry.ascent * s,
310            descent: entry.descent * s,
311        }
312    }
313
314    /// Szerokość tekstu w jednostkach świata (bez łamania wierszy).
315    pub fn measure(&mut self, handle: HandleId, text: &str, size: f32) -> Vec2 {
316        let Some(entry) = self
317            .fonts
318            .get_mut(handle.index() as usize)
319            .and_then(|f| f.as_mut())
320        else {
321            return Vec2::ZERO;
322        };
323        let scale = size / entry.units_per_em;
324        let mut width = 0.0_f32;
325        let mut previous = GlyphId::default();
326        for c in text.chars().take(MAX_TEXT_CHARS) {
327            let id = entry.font.glyph_id(c);
328            width += (entry.font.h_advance_unscaled(id) + entry.font.kern_unscaled(previous, id))
329                * scale;
330            previous = id;
331        }
332        Vec2::new(width, self.metrics(handle, size).line_height)
333    }

335    /// Składa tekst i wypisuje gotowe sprite'y do `out`.
336    ///
337    /// Zajmuje się kerningiem, łamaniem wierszy, wyrównaniem i wielkością
338    /// (`cmd.size` to wysokość `em` w jednostkach świata).
339    pub fn layout(&mut self, cmd: &TextDraw, transform: Mat3, out: &mut Vec<SpriteDraw>) {
340        let key = HandleId::from(cmd.font);
341        let Some(entry) = self
342            .fonts
343            .get_mut(key.index() as usize)
344            .and_then(|f| f.as_mut())
345        else {
346            return;
347        };
348
349        let world_per_unit = cmd.size / entry.units_per_em;
350        let line_height = (entry.ascent - entry.descent + entry.line_gap) * world_per_unit;
351        let ascent = entry.ascent * world_per_unit;
352
353        // --- 1) ułożenie: identyfikatory glifów, posuwy i podział na wiersze
354        let mut lines: Vec<(usize, usize, f32)> = Vec::new();
355        let mut line_start = 0usize;
356        let mut pen = 0.0f32;
357        let mut previous = GlyphId::default();
358        let mut word_start = 0usize;
359        let mut space_width = 0.0f32;
360        let mut index = 0usize;
361
362        for c in cmd.text.chars().take(MAX_TEXT_CHARS) {
363            let id = entry.font.glyph_id(c);
364            if c == '\n' {
365                lines.push((line_start, index, pen));
366                line_start = index + 1;
367                pen = 0.0;
368                previous = GlyphId::default();
369                index += 1;
370                continue;
371            }
372            let kern = entry.font.kern_unscaled(previous, id);
373            let advance = (entry.font.h_advance_unscaled(id) + kern) * world_per_unit;
374
375            if c == ' ' {
376                word_start = index;
377                space_width = advance;
378            } else if let Some(max_w) = cmd.max_width {
379                // łamanie wierszy po spacji, gdy przekroczymy `max_width`
380                if pen + advance > max_w && index > word_start {
381                    lines.push((line_start, word_start, pen - space_width));
382                    line_start = word_start;
383                    pen = 0.0;
384                    previous = GlyphId::default();
385                }
386            }
387
388            pen += advance;
389            previous = id;
390            index += 1;
391        }
392        lines.push((line_start, index, pen));
393
394        // --- 2) rasteryzacja i produkcja sprite'ów
395        let atlas = ATLAS_SIZE as f32;
396        for (line_no, (start, end, width)) in lines.into_iter().enumerate() {
397            let shift = match cmd.align {
398                TextAlign::Left => 0.0,
399                TextAlign::Center => -width * 0.5,
400                TextAlign::Right => -width,
401            };
402            let baseline = -ascent - line_no as f32 * line_height * cmd.line_height;
403            let mut pen = 0.0f32;
404            let mut previous = GlyphId::default();
405
406            for c in cmd.text.chars().take(end).skip(start) {
407                let id = entry.font.glyph_id(c);
408                let kern = entry.font.kern_unscaled(previous, id);
409                let advance = (entry.font.h_advance_unscaled(id) + kern) * world_per_unit;
410
411                if c != ' ' {
412                    if let Some(glyph) = entry.rasterize(id) {
413                        if glyph.size.x > 0.0 && glyph.size.y > 0.0 {
414                            let uv = UvRect::new(
415                                Vec2::new(glyph.rect.min.x / atlas, glyph.rect.min.y / atlas),
416                                Vec2::new(glyph.rect.max.x / atlas, glyph.rect.max.y / atlas),
417                            );
418                            // Środek glifu: posuw + przesunięcie od początku
419                            // linii bazowej, wszystko w jednostkach świata.
420                            //
421                            // `GlyphInfo::size/offset` to UŁAMKI `em`, więc
422                            // mnożymy je wprost przez `cmd.size`. Posuwy
423                            // (advance) pochodzą z jednostek fontu, stąd
424                            // osobny przelicznik `world_per_unit`.
425                            let size = glyph.size * cmd.size;
426                            let local = Vec2::new(
427                                shift + pen + glyph.offset.x * cmd.size,
428                                baseline - glyph.offset.y * cmd.size,
429                            );
430                            out.push(SpriteDraw {
431                                transform: transform
432                                    * Mat3::from_translation(
433                                        local + Vec2::new(size.x, -size.y) * 0.5,
434                                    )
435                                    * Mat3::from_scale(size),
436                                uv,
437                                color: cmd.color,
438                                texture: TextureKey::FontAtlas(key),
439                                blend: cmd.blend,
440                                layer: cmd.layer,
441                                screen_space: cmd.screen_space,
442                            });
443                        }
444                    }
445                }
446                // UWAGA: posuw PRZESUWAMY DOPIERO PO narysowaniu glifu.
447                //
448                // `pen` to pozycja, w której stoi BIEŻĄCY glif — tak jak
449                // `pen` w `measure` i w fazie układania. Robienie `pen +=`
450                // przed rysowaniem przesuwało każdą literę o jej własny
451                // advance, więc pierwsza litera leciała o `advance` za daleko
452                // i litery nachodziły na siebie ("Dzien" => "Dżen", nachodzące
453                // na siebie "z" i "D"), a tekst z biegiem czasu rozjeżdżał się
454                // coraz bardziej.
455                pen += advance;
456                previous = id;
457            }
458        }
459    }

500    #[test]
501    fn sprite_reaches_expected_pixels_end_to_end() {
502        // Regresja dla całego łańcucha: kamera -> Globals -> jednostkowy quad.
503        // Sprite o szerokości 1800 jednostek świata przy kamerze dopasowanej
504        // do okna 1072 px MUSI zająć całą szerokość okna.
505        let window = Vec2::new(1072.0, 756.0);
506        let arena = Rect::new(Vec2::new(-900.0, -560.0), Vec2::new(900.0, 560.0));
507        let mut camera = Camera2d::new();
508        camera.fit_world(arena, window);
509
510        let globals = Globals::new(camera.view_projection(window));
511        // prostokąt wypełniający arenę, przekształcony jak w SpriteDraw::rect
512        let transform = Mat3::from_translation(arena.center()) * Mat3::from_scale(arena.size());
513        // lewy dolny róg jednostkowego quada (-0.5, -0.5) i prawy górny (0.5, 0.5)
514        let bl =
515            globals.transform_point((transform * Vec2::new(-0.5, -0.5).extend(1.0)).truncate());
516        let tr = globals.transform_point((transform * Vec2::new(0.5, 0.5).extend(1.0)).truncate());
517
518        // szerokość w NDC pomnożona przez połowę okna daje piksele
519        let width_px = (tr.x - bl.x).abs() * window.x * 0.5;
520        assert!(
521            (width_px - window.x).abs() < 1.0,
522            "arena zajęła {width_px} px zamiast {} px (NDC {bl:?}..{tr:?})",
523            window.x
524        );
525    }

527    /// Ile pikseli bitmapy glifu jest wypełnionych.
528    fn coverage(entry: &mut FontEntry, c: char) -> (u32, u32, usize) {
529        let id = entry.font.glyph_id(c);
530        let g = entry.rasterize(id).expect("brak konturu");
531        let (w, h) = (g.rect.width() as u32, g.rect.height() as u32);
532        let atlas = ATLAS_SIZE as usize;
533        let mut filled = 0;
534        for row in 0..h {
535            let start = ((g.rect.min.y as usize) + row as usize) * atlas + g.rect.min.x as usize;

565    #[test]
566    fn glyphs_start_at_pen_and_follow_advances() {
567        // Regresja: posuw `pen` musi wskazywac pozycje BIEZACEGO glifu.
568        //
569        // Wczesniej `pen += advance` szlo PRZED rysowaniem, wiec kazda litera
570        // ladowala o swoj wlasny advance za daleko: pierwsza litera tekstu
571        // zaczynala sie zamiast w `shift`, a kolejne nakladaly sie na siebie
572        // ("Dzien" wygladalo jak "Dzen" z plamiastymi literami).
573        //
574        // Sprawdzamy to niezaleznie od czcionki, na SAMEJ metryce fontu:
575        //   * lewa krawedz pierwszego glifu == jego `offset.x`,
576        //   * odleglosc miedzy sasiednimi glifami == `advance` poprzedniego,
577        //   * ostatni glif konczy sie wylacznie przed `measure()`.
578        let Some(data) = system_font() else { return };
579        let handle: Handle<FontData> = Handle::new(0, 0);
580        let mut registry = FontRegistry::new();
581        registry.load(handle, &data).unwrap();
582
583        let text = "Dzien 1";
584        let size = 15.0f32;
585        let mut out = Vec::new();
586        let mut cmd = text_cmd(handle, text);
587        cmd.size = size;
588        registry.layout(&cmd, Mat3::IDENTITY, &mut out);
589        assert!(!out.is_empty(), "layout nie wyprodukowal glifow");
590
591        // Lewa krawedz i szerokosc kazdego quada w jednostkach swiata.
592        let edges: Vec<(f32, f32)> = out
593            .iter()
594            .map(|s| {
595                let c = s.transform.to_cols_array_2d();
596                let w = c[0][0].abs();
597                (c[2][0] - w * 0.5, c[2][0] + w * 0.5)
598            })
599            .collect();
600
601        // 1) pierwszy glif zaczyna sie w `shift` (= 0 dla Left), a NIE
602        //    o jeden advance dalej.
603        let first_offset = {
604            let e = registry.fonts[0].as_mut().unwrap();
605            let g = e.rasterize(e.font.glyph_id('D')).unwrap();
606            g.offset.x * size
607        };
608        assert!(
609            (edges[0].0 - first_offset).abs() < 0.01,
610            "pierwszy glif zaczyna sie w {:.3}, oczekiwano {:.3} (offset.x * size)",
611            edges[0].0,
612            first_offset
613        );
614
615        // 2) sasiednie glify NIE zachodza na siebie: kazda prawa krawedz
616        //    lezy w najwyzej na lewej krawedzi nastepnego (+ tolerancja na
617        //    kerning i na kolidujace bounding boxy w bitmape).
618        for w in edges.windows(2) {
619            assert!(
620                w[0].1 <= w[1].0 + 0.5,
621                "glify nachodza na siebie: prawa krawedz {:.3} > lewa nastepnego {:.3}",
622                w[0].1,
623                w[1].0
624            );
625        }
626
627        // 3) ostatni glif konczy sie w `measure()` — calkowita szerokosc
628        //    tekstu zgadza sie z suma posuwow (tu `shift` = 0).
629        let measured = registry.measure(HandleId::from(handle), text, size).x;
630        let last = *edges.last().unwrap();
631        assert!(
632            last.1 <= measured + 0.5,
633            "tekst wider niz `measure()`: koniec {:.3} > {:.3}",
634            last.1,
635            measured
636        );
637        // `measure` liczy posuwy w jednostkach fontu przeliczonych przez
638        // `size / upem` — sprawdzamy, ze ta szerokosc jest wogole sensowna
639        // (kiedys `pen` byl liczony w surowych jednostkach fontu i tekst
640        // rozjeżdżal sie o czynnik `upem`).
641        assert!(
642            (10.0..400.0).contains(&measured),
643            "`measure` zwrocilo bledna szerokosc {measured} dla rozmiaru {size}"
644        );
645    }

660        let g = entry.rasterize(id).expect("brak konturu glifu");
661        // regresja: przy em = `raster` px litera 'M' musi mieć kilkadziesiąt
662        // pikseli, a nie 1x1 (pomyłka z interpretacją PxScale)
663        let px = g.size * raster as f32;
664        assert!(
665            px.x > 4.0 && px.y > 4.0,
666            "glif 'M' ma tylko {px:?} px przy em = {raster} px"
667        );
668        assert!(px.x <= raster as f32 && px.y <= raster as f32);
669        eprintln!(
670            "[test] 'M': atlas rect={:?} ({}x{} px) offset={:?} em-size={:?} => {px:?} px",

673            g.rect.height(),
674            g.offset,
675            g.size
676        );
677
678        let size = 32.0f32;

692        // równą skali kolumny X; wielkość 'M' to ok. 0.7 em
693        assert!(
694            c[0][0] > size * 0.3 && c[0][0] < size * 1.2,
695            "szerokość glifu {:?}, oczekiwano ~{size}",
