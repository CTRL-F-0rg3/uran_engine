1//! Kamera perspektywiczna 3D.
2//!
3//! Trzyma pozycję i cel, a macierz `view_proj` liczy dopiero przy
4//! rysowaniu — dzięki temu poruszanie kamerą nie kosztuje nic poza
5//! jednym mnożeniem macierzy na klatkę.
6
7use uran_math::{Mat4, Vec3};
8
9/// Kamera 3D: pozycja, cel i pole widzenia.
10///
11/// W prawej ręce: patrzymy wzdłuż osi `-Z`, jak w OpenGL/wgpu.
12#[derive(Debug, Clone, Copy)]
13pub struct Camera3d {
14    /// Pozycja oka w świecie.
15    pub position: Vec3,
16    /// Punkt, na który patrzymy (nie musi być znormalizowany).
17    pub target: Vec3,
18    /// Kąt pionowy pola widzenia w radianach.
19    pub fov_y: f32,
20    /// Stosunek szerokości do wysokości (aspect).
21    pub aspect: f32,
22    /// Odległość bliższej i dalszej płaszczyzny obcinania.
23    pub near: f32,
24    pub far: f32,
25}
26
27impl Default for Camera3d {
28    fn default() -> Self {
29        Self {
30            position: Vec3::new(0.0, 3.0, 8.0),
31            target: Vec3::ZERO,
32            // 60° to wygodny kompromis: widać szeroko, ale nie ma
33            // charakterystycznego „rybiego oka" z 90°
34            fov_y: std::f32::consts::FRAC_PI_3,
35            aspect: 16.0 / 9.0,
36            near: 0.1,
37            far: 1000.0,
38        }
39    }
40}
41
42impl Camera3d {
43    /// Ustawia proporcje obrazu (po resize okna).
44    pub fn set_aspect(&mut self, aspect: f32) {
45        if aspect.is_finite() && aspect > 0.0 {
46            self.aspect = aspect;
47        }
48    }
49
50    /// Kierunek patrzenia (jednostkowy).
51    pub fn forward(&self) -> Vec3 {
52        (self.target - self.position).normalize_or_zero()
53    }
54
55    /// Wektor „w górę" świata — w kamerze wolnajazdowej to po prostu +Y.
56    pub fn up(&self) -> Vec3 {
57        Vec3::Y
58    }
59    /// Wektor „w prawo" (poprzeczny do kierunku patrzenia).
60    pub fn right(&self) -> Vec3 {
61        self.forward().cross(Vec3::Y).normalize_or_zero()
62    }
63
64    /// Macierz świata (kolumnowa) -> NDC.
65    ///
66    /// Kolejność `look_at * perspective` wynika z tego, że w
67    /// kolumnowym zapisie `A * B * v` najpierw działa `B` — czyli
68    /// najpierw przesuwamy punkt do przestrzeni oka, potem rzutujemy.
69    pub fn view_proj(&self) -> Mat4 {
70        let view = Mat4::look_at_rh(self.position, self.target, Vec3::Y);
71        let proj = Mat4::perspective_rh(self.fov_y, self.aspect, self.near, self.far);
72        proj * view
73    }
74
75    /// Sama macierz widoku (świat -> przestrzeń oka).
76    ///
77    /// Osobno od [`Self::view_proj`], bo G-Bufer trzyma normalne
78    /// w przestrzeni oka, a shader nieba potrzebuje odwrotności całej
79    /// macierzy rzutowania. Liczenie `look_at` drugi raz w shaderze
80    /// nie wchodzi w grę — koszt jest po stronie GPU na każdy piksel.
81    pub fn view(&self) -> Mat4 {
82        Mat4::look_at_rh(self.position, self.target, Vec3::Y)
83    }
84
85    /// Tangens połowy kąta widzenia w pionie.
86    ///
87    /// Wpisujemy go do uniformu sceny, bo passy pełnoekranowe muszą
88    /// odtworzyć promień przez piksel, a liczenie `tan` w każdym z nich
89    /// powielałoby tę samą definicję pola widzenia w czterech miejscach.
90    pub fn tan_half_fov(&self) -> f32 {
91        (self.fov_y * 0.5).tan()
92    }
93
94    /// Promień z oka w kierunku `dir` (kierunek NIE musi być jednostkowy —
95    /// normalizujemy tutaj, więc można podać różnicę `target - position`).
96    pub fn ray(&self, dir: Vec3) -> Ray {
97        Ray {
98            origin: self.position,
99            dir: dir.normalize_or_zero(),
100        }
101    }
102
103    /// Promień przez punkt ekranu w znormalizowanych współrzędnych.
104    ///
105    /// `ndc` to standardowy zakres `[-1, 1]` (oś Y w górę, przeciwnie do
106    /// konwencji ekranowej silnika). Środek to `(0,0)`, prawy górny róg
107    /// to `(1,1)`.
108    pub fn ray_through_ndc(&self, ndc_x: f32, ndc_y: f32) -> Ray {
109        let f = self.forward();
110        let r = self.right();
111        let u = f.cross(r);
112        // tangens połowy FOV-a: podstawa trójkąta na płaszczyźnie bliższej
113        let ty = (self.fov_y * 0.5).tan();
114        let tx = ty * self.aspect;
115        let dir = (f + r * (ndc_x * tx) + u * (ndc_y * ty)).normalize_or_zero();
116        self.ray(dir)
117    }
118}
119
120/// Promień w świecie (początek + kierunek jednostkowy).
121#[derive(Debug, Clone, Copy)]
122pub struct Ray {
123    pub origin: Vec3,
124    pub dir: Vec3,
125}
126
127impl Ray {
128    /// Punkt w odległości `t` od początku.
129    pub fn at(&self, t: f32) -> Vec3 {
130        self.origin + self.dir * t
131    }
132
133    /// Promień przesunięty o wektor `offset` (np. o pozycję lufy).
134    pub fn shifted(&self, offset: Vec3) -> Ray {
135        Ray {
136            origin: self.origin + offset,
137            dir: self.dir,
138        }
139    }
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn default_camera_looks_from_above_towards_origin() {
148        // kamera startuje NAD osią Z i patrzy w dół-ku środkowi, więc
149        // kierunek ma ujemne Y i Z
150        let c = Camera3d::default();
151        let f = c.forward();
152        assert!(f.z < -0.9, "kamera z +Z patrzy na -Z, a jest {f:?}");
153        assert!(f.y < 0.0, "kamera jest nad osią, więc patrzy w dół: {f:?}");
154    }
155
156    #[test]
157    fn right_is_perpendicular_to_forward() {
158        let c = Camera3d::default();
159        let dot = c.right().dot(c.forward());
160        assert!(dot.abs() < 1e-5, "wektory nie są prostopadłe: {dot}");
161    }
162
163    #[test]
164    fn aspect_guard_rejects_nonsense() {
165        let mut c = Camera3d::default();
166        let before = c.aspect;
167        c.set_aspect(0.0);
168        c.set_aspect(f32::NAN);
169        assert_eq!(c.aspect, before, "złe aspect nie mogą psuć macierzy");
170    }
171
172    #[test]
173    fn centre_ray_matches_forward() {
174        let c = Camera3d::default();
175        let r = c.ray_through_ndc(0.0, 0.0);
176        assert!(
177            r.dir.dot(c.forward()) > 0.999,
178            "środek ekranu to kierunek patrzenia"
179        );
180    }
181
182    #[test]
183    fn rays_fan_out_towards_screen_edges() {
184        let c = Camera3d::default();
185        let centre = c.ray_through_ndc(0.0, 0.0);
186        let right = c.ray_through_ndc(1.0, 0.0);
187        let up = c.ray_through_ndc(0.0, 1.0);
188        // krawędź ekranu musi zbaczać w bok, inaczej mamy równoległy rzut
189        assert!(right.dir.dot(c.right()) > 0.2);
190        // górna krawędź patrzy WYŻEJ niż środek, więc jej kierunek
191        // ma mniejsze Y (kamera patrzy w dół) — stąd `-`
192        assert!(up.dir.y < centre.dir.y, "górna krawędź nie idzie w górę");
193    }
194
195    #[test]
196    fn ray_direction_is_unit_length() {
197        // `Ray` NIE normalizuje w konstruktorze — to świadome: wołujący
198        // przekazuje już znormalizowany kierunek z kamery.
199        let r = Ray {
200            origin: Vec3::ZERO,
201            dir: Vec3::new(0.0, 0.0, 7.0).normalize(),
202        };
203        let n = r.at(2.0);
204        assert!(
205            (n.length() - 2.0).abs() < 1e-5,
206            "punkt {} nie leży 2 od początku",
207            n.length()
208        );
209    }
