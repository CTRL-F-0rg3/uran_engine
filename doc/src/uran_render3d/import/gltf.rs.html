1//! Import modeli glTF 2.0 (`.glb`) ze skinningiem i materiałami.
2//!
3//! ## Po co to, skoro jest `.obj`
4//!
5//! `.obj` obsługuje statyczne bryły. Postać to co innego: ma szkielet
6//! (macierze kości), wagi skinningu na wierzchołkach i klipy animacji.
7//! Wavefront tego nie wyraża, a większość postaci z Blendera ma to
8//! właśnie w glTF/FBX.
9//!
10//! ## Zakres — świadomie wąski
11//!
12//! | Obszar | Stan |
13//! |--------|------|
14//! | `.glb` (JSON + BIN w jednym pliku) | ✅ |
15//! | `.gltf` + osobne pliki | ❌ nieobsługiwane |
16//! | skinning (`JOINTS_0`/`WEIGHTS_0`) | ✅ do 4 wpływów |
17//! | animacje z pliku | ❌ (klipy czytamy z `.urananm`) |
18//! | `KHR_draco_mesh_compression` | ❌ |
19//! | morph targete | ❌ |
20//!
21//! Nieobsługiwane rzeczy warto znać, bo plik może je zawierać, a my
22//! po cichu je pomijamy: [`GltfScene::unsupported`] wymienia je po nazwie.
23//!
24//! ## Oś
25//!
26//! glTF jest Y-up, tak jak silnik — **nie przenosimy osi**. To różni ten
27//! loader od `import::obj`, który przeskakuje z Z-up Blendera na Y-up.
28//! Dzięki temu pozycje ze skinningiem, macierze kości i UV pasują do
29//! pliku bajt w bajt i nie trzeba ich nigdzie obracać.
30

38/// Błąd importu z kontekstem (który element, który plik).
39#[derive(Debug)]
40pub struct GltfError {
41    pub msg: String,
42}
43
44impl fmt::Display for GltfError {
45    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
46        write!(f, "glTF: {}", self.msg)
47    }
48}
49
50impl std::error::Error for GltfError {}
51
52fn err<T>(msg: impl Into<String>) -> Result<T, GltfError> {
53    Err(GltfError { msg: msg.into() })
54}

60    /// Buduje minimalny `.glb` z podanym JSON-em i (opcjonalnie) BIN-em.
61    ///
62    /// Testy na prawdziwym pliku 16 MB byłyby niepraktyczne, a błąd
63    /// wczytywania kontenera to dokładnie ten sam kod, co przy realnym
64    /// pliku — więc testujemy go na syntetycznych bajtach.
65    fn glb(json: &str, bin: &[u8]) -> Vec<u8> {
66        // Chunki GLB muszą mieć długość wielokrotnością 4, ale **offset
67        // w JSON-ie liczymy względem początku danych, nie ogonka** —
68        // więc `bin` zostawiamy nietknięty, a dopełnienie dopisujemy
69        // dopiero w nagłówku. Inaczej `bin[32..36]` w teście wskazywałoby
70        // nie na to, co loader zobaczy.
71        let js_len = json.len().div_ceil(4) * 4;
72        let bin_len = bin.len().div_ceil(4) * 4;
73        let total = 12 + 8 + js_len + 8 + bin_len;
74        let mut out = Vec::new();
75        out.extend_from_slice(&0x4654_6C67u32.to_le_bytes());
76        out.extend_from_slice(&2u32.to_le_bytes());
77        out.extend_from_slice(&(total as u32).to_le_bytes());
78        out.extend_from_slice(&(js_len as u32).to_le_bytes());
79        out.extend_from_slice(&0x4E4F_534Au32.to_le_bytes());
80        out.extend_from_slice(json.as_bytes());
81        out.resize(12 + 8 + js_len, b' ');
82        out.extend_from_slice(&(bin_len as u32).to_le_bytes());
83        out.extend_from_slice(&0x004E_4942u32.to_le_bytes());
84        out.extend_from_slice(bin);
85        out.resize(total, 0);
86        out
87    }

121    /// Bajty dla [`MINIMAL`], ułożone w kolejności, w której je czyta
122    /// `read_f32`.
123    ///
124    /// ## Dlaczego nie wektor `Vec<f32>`
125    ///
126    /// Indeksy są `UNSIGNED_INT` (4 B), a pozycje i wagi `FLOAT` (też
127    /// 4 B, ale inna interpretacja bajtów). Wypełnienie jednym wektorem
128    /// `f32` dajełoby bajty, które dla indeksów są niepoprawnym
129    /// `u32` — stąd osobne `copy_from_slice` na każdy blok.
130    fn minimal_bin() -> Vec<u8> {
131        let mut b = vec![0u8; 256];
132        // Jeden helper zamiast dwóch domknięć: oba pożyczałyby `&mut b`
133        // naraz, a Rust tego nie pozwala.
134        fn put(b: &mut [u8], off: usize, bytes: &[u8]) {
135            b[off..off + bytes.len()].copy_from_slice(bytes);
136        }
137        // POSITION: VEC3 f32 pod offsetem 0
138        put(&mut b, 0, &1.0f32.to_le_bytes());
139        // JOINTS_0: VEC4 f32 pod offsetem 12
140        put(&mut b, 12, &0.0f32.to_le_bytes());
141        // WEIGHTS_0: VEC4 f32 pod offsetem 28
142        put(&mut b, 28, &1.0f32.to_le_bytes());
143        // indices: SCALAR u32 pod offsetem 44
144        put(&mut b, 44, &0u32.to_le_bytes());
145        put(&mut b, 48, &0u32.to_le_bytes());
146        put(&mut b, 52, &0u32.to_le_bytes());
147        // inverseBindMatrices: MAT4 f32 pod offsetem 64 (przekątna = 1)
148        put(&mut b, 64, &1.0f32.to_le_bytes());
149        put(&mut b, 80, &1.0f32.to_le_bytes());
150        put(&mut b, 96, &1.0f32.to_le_bytes());
151        put(&mut b, 112, &1.0f32.to_le_bytes());
152        b
153    }

333/// Kontener GLB: nagłówek + chunki (JSON i BIN).
334struct Container<'a> {
335    json: Json,
336    bin: &'a [u8],
337}
338
339fn read_container(bytes: &[u8]) -> Result<Container<'_>, GltfError> {
340    if bytes.len() < 12 {
341        return err("plik za krótki na nagłówek GLB");
342    }
343    let magic = u32::from_le_bytes(bytes[0..4].try_into().unwrap());
344    if magic != 0x4654_6C67 {
345        return err("brak sygnatury `glTF` — to nie jest plik .glb");
346    }
347    let version = u32::from_le_bytes(bytes[4..8].try_into().unwrap());
348    if version != 2 {
349        return err(format!("wersja glTF {version} (obsługujemy 2)"));
350    }
351    let total = u32::from_le_bytes(bytes[8..12].try_into().unwrap()) as usize;
352    if total > bytes.len() {
353        return err(format!(
354            "nagłówek obiecuje {total} B, a plik ma {}",
355            bytes.len()
356        ));
357    }
358
359    // Chunki: (długość, typ) + dane. Kolejność bywa JSON, BIN, ale
360    // szukamy ich po typie, nie po pozycji.
361    let mut json_bytes: Option<&[u8]> = None;
362    let mut bin: &[u8] = &[];
363    let mut off = 12usize;
364    while off + 8 <= total {
365        let len = u32::from_le_bytes(bytes[off..off + 4].try_into().unwrap()) as usize;
366        let kind = u32::from_le_bytes(bytes[off + 4..off + 8].try_into().unwrap());
367        let start = off + 8;
368        if start + len > total {
369            return err("chunk sięga za koniec pliku");
370        }
371        match kind {
372            0x4E4F_534A => json_bytes = Some(&bytes[start..start + len]),
373            0x004E_4942 => bin = &bytes[start..start + len],
374            _ => {} // nieznany chunk pomijamy zgodnie ze specyfikacją
375        }
376        off = start + len + ((4 - len % 4) % 4);
377    }
378
379    let Some(jb) = json_bytes else {
380        return err("brak chunku JSON");
381    };
382    // Chunk JSON bywa dopełniony spacjami do wielokrotności 4.
383    let text = std::str::from_utf8(jb).map_err(|e| GltfError {
384        msg: format!("JSON nie jest UTF-8: {e}"),
385    })?;
386    let json = json::parse(text.trim_end()).map_err(|e| GltfError {
387        msg: format!("JSON: {e}"),
388    })?;
389    Ok(Container { json, bin })
390}

392/// Wierzchołek ze skinningiem: pozycja, normalna, UV, 4 kości, 4 wagi.
393///
394/// ## Dlaczego osobny typ, a nie rozszerzenie [`crate::mesh::Vertex`]
395///
396/// `Vertex` ma 44 B i jest częścią kontraktu z potokiem w `scene.rs` —
397/// test `vertex_is_44_bytes_and_uniform_is_aligned` pilnuje tego rozmiaru,
398/// a jego zmiana cicho psuje cały renderer 3D dla wszystkich demek.
399/// Dokładanie 24 B (4×`u16` + 4×`f32`) rozszerzyłoby każdą bryłę świata
400/// o 55% rozmiaru wierzchołka, żeby skinning miały tylko postacie.
401///
402/// Dlatego postacie mają własny format i własny potok, a reszta silnika
403/// nawet o nim nie wie.
404#[repr(C)]
405#[derive(Debug, Clone, Copy, bytemuck::Pod, bytemuck::Zeroable)]
406pub struct SkinVertex {
407    pub position: [f32; 3],
408    pub normal: [f32; 3],
409    pub uv: [f32; 2],
410    /// Indeksy kości — 4 na wierzchołek, bo tyle obsługuje sprzęt.
411    ///
412    /// W pliku są `u16`; trzymamy je jako `u32`, bo WGSL nie ma sensownego
413    /// `array<u16, 4>` w uniformie. Nieużywane sloty mają indeks 0 i wagę 0.
414    pub joints: [u32; 4],
415    /// Wagi — suma nie musi wynosić 1, bo plik może mieć mniej niż 4
416    /// wpływy na wierzchołek; shader normalizuje.
417    pub weights: [f32; 4],
418}

425    pub const STRIDE: u64 = 64;

465/// Tryb przezroczystości z glTF.
466#[derive(Debug, Clone, Copy, PartialEq, Eq)]
467pub enum AlphaMode {
468    Opaque,
469    Mask,
470    Blend,
471}
472
473impl Default for AlphaMode {
474    fn default() -> Self {
475        Self::Opaque
476    }
477}
478
479impl AlphaMode {
480    /// Czy materiał potrzebuje rysowania w kolejce alfa.
481    ///
482    /// `Mask` (wyrzynanie włosów) da się narysować w passie głównym,
483    /// bo jest binarny — potrzebuje odrzucania, nie mieszania.
484    pub fn needs_blend(&self) -> bool {
485        matches!(self, AlphaMode::Blend)
486    }
487}

520/// Model postaci: siatki, materiały, tekstury, szkielet.
521///
522/// Wszystko jest w jednej strukturze, bo zwykle ładujemy plik w całości
523/// i trzymamy w pamięci przez całą grę. Podział na osobne uchwyt
524/// (`MeshHandle`, `SkelHandle`) byłby pozorną oszczędnością — plik ma
525/// 19 tys. wierzchołków i mieści się w kilku MB.
526#[derive(Debug, Default)]
527pub struct GltfScene {
528    pub name: String,
529    pub meshes: Vec<SkinMesh>,
530    pub materials: Vec<GltfMaterial>,
531    pub textures: Vec<GltfTexture>,
532    pub joints: Vec<Joint>,
533    /// `inverseBindMatrices` w kolejności `skins[].joints`.
534    pub inverse_bind: Vec<Mat4>,
535    /// Czego w pliku nie umieliśmy przeczytać.
536    pub unsupported: Vec<&'static str>,
537    /// Skala dopasowania modelu: po zaimportowaniu człowiek ma 1.7 m.
538    ///
539    /// glTF traktuje jednostki jako metry, a ten asset jest tak
540    /// przygotowany, ale inne eksporty bywają w centymetrach. Pole
541    /// daje to wyraźnie do korekty bez zgadywania.
542    pub unit_scale: f32,
543}

545impl GltfScene {
546    /// Indeks kości o podanej nazwie — dla animacji po nazwie.
547    pub fn joint_by_name(&self, name: &str) -> Option<usize> {
548        self.joints.iter().position(|j| j.name == name)
549    }
550
551    /// Wysokość modelu w jednostkach pliku (najwyższy wierzchołek).
552    pub fn bounds_height(&self) -> f32 {
553        let mut lo = f32::MAX;
554        let mut hi = f32::MIN;
555        for m in &self.meshes {
556            for v in &m.vertices {
557                lo = lo.min(v.position[1]);
558                hi = hi.max(v.position[1]);
559            }
560        }
561        if lo > hi {
562            0.0
563        } else {
564            hi - lo
565        }
566    }
567}

569/// Kody typów komponentów z glTF.
570const F32: u32 = 5126;
571const U16: u32 = 5123;
572const U8: u32 = 5121;
573
574/// `UNSIGNED_INT` — typ komponentu, którego glTF używa dla niemal
575/// wszystkich indeksów wierzchołków. Bez wariantu poniżej `match`
576/// zwracałby `0.0`, czyli **każdy trójkąt degenerowałby się do punktu**
577/// i postać zniknęłaby z ekranu.
578const U32: u32 = 5125;

580/// `componentType` + `type` jako (rozmiar elementu, liczba składowych).
581fn comp_info(component: u32, ty: &str) -> Option<(usize, usize)> {
582    let size = match component {
583        5120 | U8 => 1,
584        U16 => 2,
585        5122 => 2,
586        5125 => 4,
587        F32 => 4,
588        _ => return None,
589    };
590    let n = match ty {
591        "SCALAR" => 1,
592        "VEC2" => 2,
593        "VEC3" => 3,
594        "VEC4" => 4,
595        "MAT4" => 16,
596        _ => return None,
597    };
598    Some((size, n))
599}

601/// Czyta `bufferView` i zwraca bajty z zakresu `offset..offset+len`.
602fn view_bytes<'a>(c: &Container<'a>, idx: usize) -> Result<&'a [u8], GltfError> {
603    let bv = c
604        .json
605        .get("bufferViews")
606        .and_then(|v| v.idx(idx))
607        .ok_or_else(|| GltfError {
608            msg: format!("brak bufferView {idx}"),
609        })?;
610    let off = bv.get("byteOffset").and_then(Json::as_u32).unwrap_or(0) as usize;
611    let len = bv
612        .get("byteLength")
613        .and_then(Json::as_u32)
614        .ok_or_else(|| GltfError {
615            msg: format!("bufferView {idx} bez byteLength"),
616        })? as usize;
617    let end = off.checked_add(len).ok_or_else(|| GltfError {
618        msg: "przepełnienie offsetu bufferView".into(),
619    })?;
620    c.bin.get(off..end).ok_or_else(|| GltfError {
621        msg: format!(
622            "bufferView {idx} wychodzi poza BIN chunk ({} B)",
623            c.bin.len()
624        ),
625    })
626}

628/// Czytuje accessor jako płaskie `Vec<f32>`, konwertując typy całkowite.
629///
630/// Wagi skinningu bywają `u8`/`u16` znormalizowane, a `f32` — więc
631/// jedna ścieżka odczytu zamiast osobnej dla każdego typu.
632fn read_f32(c: &Container<'_>, idx: usize) -> Result<Vec<f32>, GltfError> {
633    let acc = c
634        .json
635        .get("accessors")
636        .and_then(|a| a.idx(idx))
637        .ok_or_else(|| GltfError {
638            msg: format!("brak accessor {idx}"),
639        })?;
640    let component = acc
641        .get("componentType")
642        .and_then(Json::as_u32)
643        .ok_or_else(|| GltfError {
644            msg: format!("accessor {idx} bez componentType"),
645        })?;
646    let ty = acc.get("type").and_then(Json::as_str).unwrap_or("SCALAR");
647    let count = acc
648        .get("count")
649        .and_then(Json::as_u32)
650        .ok_or_else(|| GltfError {
651            msg: format!("accessor {idx} bez count"),
652        })? as usize;
653    let normalized = acc
654        .get("normalized")
655        .and_then(Json::as_bool)
656        .unwrap_or(false);
657
658    let Some(bvi) = acc.get("bufferView").and_then(Json::as_u32) else {
659        // Accessor bez `bufferView` to same zera — dozwolone przez specyfikację.
660        return Ok(vec![
661            0.0;
662            count * comp_info(component, ty).map_or(1, |x| x.1)
663        ]);
664    };
665    let bv = c
666        .json
667        .get("bufferViews")
668        .and_then(|v| v.idx(bvi as usize))
669        .ok_or_else(|| GltfError {
670            msg: format!("brak bufferView {bvi}"),
671        })?;
672    // `view_bytes` zwraca już dane **przesunięte** o `byteOffset`
673    // bufferView, więc tutaj wolno dodać wyłącznie przesunięcie samego
674    // accessora. Dodanie tu jeszcze raz `bv.byteOffset` to klasyczny
675    // podwójny offset — na małych plikach (offset 0) przechodzi, a na
676    // realnym assecte 16 MB wywala się, bo `u32` się przepełnia zakres.
677    let base = acc.get("byteOffset").and_then(Json::as_u32).unwrap_or(0) as usize;
678    let (size, ncomp) = comp_info(component, ty).ok_or_else(|| GltfError {
679        msg: format!("accessor {idx}: typ {ty} nieobsługiwany"),
680    })?;
681    let data = view_bytes(c, bvi as usize)?;
682
683    // `byteStride` bywa większy niż rozmiar elementu (interleaving).
684    let stride = bv.get("byteStride").and_then(Json::as_u32).unwrap_or(0) as usize;
685    let stride = if stride == 0 { size * ncomp } else { stride };
686
687    let mut out = Vec::with_capacity(count * ncomp);
688    for i in 0..count {
689        let start = base + i * stride;
690        for k in 0..ncomp {
691            let o = start + k * size;
692            let b = data.get(o..o + size).ok_or_else(|| GltfError {
693                msg: format!("accessor {idx}: brak bajtów dla elementu {i}"),
694            })?;
695            let v = match component {
696                F32 => f32::from_le_bytes(b.try_into().unwrap()),
697                U16 => {
698                    let v = u16::from_le_bytes(b.try_into().unwrap()) as f32;
699                    // `normalized` znaczy dzielenie przez 65535, nie surowa wartość
700                    if normalized {
701                        v / 65535.0
702                    } else {
703                        v
704                    }
705                }
706                5122 => i16::from_le_bytes(b.try_into().unwrap()) as f32,
707                U32 => u32::from_le_bytes(b.try_into().unwrap()) as f32,
708                U8 => {
709                    let v = b[0] as f32;
710                    if normalized {
711                        v / 255.0
712                    } else {
713                        v
714                    }
715                }
716                5120 => b[0] as i8 as f32,
717                _ => 0.0,
718            };
719            out.push(v);
720        }
721    }
722    Ok(out)
723}

725/// Wierzchołek w układzie pliku przed zastosowaniem macierzy węzła.
726///
727/// glTF pozwala, żeby pozycje były już w przestrzeni świata, a wtedy
728/// transformacja węzła to tylko orientacja. Dla skinningu to nie ma
729/// znaczenia — i tak liczymy macierz `bone * IBM` i mnożymy wierzchołek —
730/// ale normalne muszą dostać tę samą bazę, inaczej oświetlenie jest
731/// przekrzywione o pozycję modelu.
732struct RawPrimitives {
733    positions: Vec<f32>,
734    normals: Vec<f32>,
735    uvs: Vec<f32>,
736    joints: Vec<f32>,
737    weights: Vec<f32>,
738    indices: Vec<u32>,
739    material: usize,
740}
741
742/// Wczytuje jeden prymityw siatki ze skinningiem.
743fn read_primitive(
744    c: &Container<'_>,
745    prim: &Json,
746    fallback_mat: usize,
747) -> Result<RawPrimitives, GltfError> {
748    let mode = prim.get("mode").and_then(Json::as_u32).unwrap_or(4);
749    if mode != 4 {
750        return err(format!("prymityw w trybie {mode} (chcemy tylko trójkąty)"));
751    }
752    let attrs = prim.get("attributes").ok_or_else(|| GltfError {
753        msg: "prymityw bez `attributes`".into(),
754    })?;
755    let attr = |name: &str| -> Result<u32, GltfError> {
756        attrs
757            .get(name)
758            .and_then(Json::as_u32)
759            .ok_or_else(|| GltfError {
760                msg: format!("prymityw bez atrybutu {name}"),
761            })
762    };
763
764    let positions = read_f32(c, attr("POSITION")? as usize)?;
765    let nverts = positions.len() / 3;
766
767    // Normalna i UV są opcjonalne w specyfikacji — bez nich dostajemy
768    // wartości neutralne, a nie błąd, bo plik nadal da się wyświetlić.
769    let normals = match attrs.get("NORMAL").and_then(Json::as_u32) {
770        Some(i) => read_f32(c, i as usize)?,
771        None => vec![0.0; nverts * 3],
772    };
773    let uvs = match attrs.get("TEXCOORD_0").and_then(Json::as_u32) {
774        Some(i) => read_f32(c, i as usize)?,
775        None => vec![0.0; nverts * 2],
776    };
777
778    // Skinning: brak `JOINTS_0` znaczy statyczną siatkę, wtedy
779    // przypisujemy kość 0 z wagą 1 — geometria zostaje nieruchoma.
780    let (joints, weights) = match attrs.get("JOINTS_0").and_then(Json::as_u32) {
781        Some(ji) => {
782            let w = attrs
783                .get("WEIGHTS_0")
784                .and_then(Json::as_u32)
785                .ok_or_else(|| GltfError {
786                    msg: "JOINTS_0 bez WEIGHTS_0".into(),
787                })?;
788            (read_f32(c, ji as usize)?, read_f32(c, w as usize)?)
789        }
790        None => {
791            let mut j = vec![0.0; nverts * 4];
792            let mut w = vec![0.0; nverts * 4];
793            for v in 0..nverts {
794                w[v * 4] = 1.0;
795            }
796            (j, w)
797        }
798    };
799
800    let indices = match prim.get("indices").and_then(Json::as_u32) {
801        Some(i) => read_f32(c, i as usize)?
802            .into_iter()
803            .map(|v| v as u32)
804            .collect(),
805        // Bez indeksów to kolejność wierzchołków 0..n.
806        None => (0..nverts as u32).collect(),
807    };
808
809    let material = prim
810        .get("material")
811        .and_then(Json::as_u32)
812        .unwrap_or(fallback_mat as u32) as usize;
813
814    Ok(RawPrimitives {
815        positions,
816        normals,
817        uvs,
818        joints,
819        weights,
820        indices,
821        material,
822    })
823}

838fn read_materials(c: &Container<'_>, default_tex: Option<usize>) -> Vec<GltfMaterial> {
839    let Some(list) = c.json.get("materials").and_then(Json::as_array) else {
840        return Vec::new();
841    };
842    list.iter()
843        .map(|m| {
844            let pbr = m.get("pbrMetallicRoughness");
845            let base_color = pbr
846                .and_then(|p| p.get("baseColorFactor"))
847                .and_then(Json::as_f64_vec)
848                .map(|v| {
849                    let mut c = [1.0f32; 4];
850                    for (i, x) in v.iter().take(4).enumerate() {
851                        c[i] = *x as f32;
852                    }
853                    c
854                })
855                .unwrap_or([1.0; 4]);
856            GltfMaterial {
857                name: m
858                    .get("name")
859                    .and_then(Json::as_str)
860                    .unwrap_or("")
861                    .to_string(),
862                base_color,
863                base_texture: pbr
864                    .and_then(|p| p.get("baseColorTexture"))
865                    .and_then(tex_ref)
866                    .or(default_tex),
867                normal_texture: m.get("normalTexture").and_then(tex_ref),
868                orm_texture: pbr
869                    .and_then(|p| p.get("metallicRoughnessTexture"))
870                    .and_then(tex_ref),
871                // glTF trzyma roughness/metallic w jednym mnożniku
872                // (G rozkład, B metaliczność), a silnik ma je osobno —
873                // dlatego czytamy obie składowe niezależnie.
874                roughness: pbr
875                    .and_then(|p| p.get("roughnessFactor"))
876                    .and_then(Json::as_f64)
877                    .unwrap_or(1.0) as f32,
878                metallic: pbr
879                    .and_then(|p| p.get("metallicFactor"))
880                    .and_then(Json::as_f64)
881                    .unwrap_or(1.0) as f32,
882                alpha_mode: alpha_mode(m.get("alphaMode").and_then(Json::as_str)),
883                alpha_cutoff: m.get("alphaCutoff").and_then(Json::as_f64).unwrap_or(0.5) as f32,
884                double_sided: m
885                    .get("doubleSided")
886                    .and_then(Json::as_bool)
887                    .unwrap_or(false),
888                emissive: m
889                    .get("emissiveFactor")
890                    .and_then(Json::as_f64_vec)
891                    .map(|v| {
892                        let mut e = [0.0f32; 3];
893                        for (i, x) in v.iter().take(3).enumerate() {
894                            e[i] = *x as f32;
895                        }
896                        e
897                    })
898                    .unwrap_or([0.0; 3]),
899                emissive_texture: m.get("emissiveTexture").and_then(tex_ref),
900            }
901        })
902        .collect()
903}

905/// Które tekstury są kolorowe, a które danymi.
906///
907/// Robimy to po tym, co z nich faktycznie czyta dany materiał, a nie po
908/// nazwie pliku: eksporter nazywa je dowolnie.
909fn mark_color_textures(mats: &[GltfMaterial], texs: &mut [GltfTexture]) {
910    for m in mats {
911        for t in [m.base_texture, m.emissive_texture].into_iter().flatten() {
912            if let Some(x) = texs.get_mut(t) {
913                x.is_color = true;
914            }
915        }
916    }
917}

919fn read_textures(c: &Container<'_>, mats: &[GltfMaterial]) -> Vec<GltfTexture> {
920    let mut out = Vec::new();
921    let Some(images) = c.json.get("images").and_then(Json::as_array) else {
922        return out;
923    };
924    let Some(samplers) = c.json.get("samplers").and_then(Json::as_array) else {
925        return out;
926    };
927    for im in images {
928        let Some(bvi) = im.get("bufferView").and_then(Json::as_u32) else {
929            // Obraz z zewnętrznym `uri` wymagałby czytania z dysku —
930            // przy `.glb` nie występuje, więc to cicha pominięcie.
931            continue;
932        };
933        let Ok(bytes) = view_bytes(c, bvi as usize) else {
934            continue;
935        };
936        let mime = im
937            .get("mimeType")
938            .and_then(Json::as_str)
939            .unwrap_or("image/png");
940        // `samplers[i].source` wskazuje obraz; ścieżka idzie przez teksturę.
941        let _ = (mime, samplers);
942        out.push(GltfTexture {
943            name: im
944                .get("name")
945                .and_then(Json::as_str)
946                .unwrap_or("tex")
947                .to_string(),
948            bytes: bytes.to_vec(),
949            is_color: false,
950        });
951    }
952    mark_color_textures(mats, &mut out);
953    out
954}

956/// Buduje listę kości: rodzic, macierz lokalna, indeks w skinie.
957fn read_skeleton(
958    c: &Container<'_>,
959    out: &mut Vec<Joint>,
960    inverse: &mut Vec<Mat4>,
961) -> Result<(), GltfError> {
962    let skins = c
963        .json
964        .get("skins")
965        .and_then(Json::as_array)
966        .ok_or_else(|| GltfError {
967            msg: "plik nie ma `skins` — to nie postać".into(),
968        })?;
969    let Some(skin) = skins.first() else {
970        return err("pusta lista `skins`");
971    };
972    let joint_ids: Vec<usize> = skin
973        .get("joints")
974        .and_then(Json::as_array)
975        .map(|a| {
976            a.iter()
977                .filter_map(Json::as_u32)
978                .map(|i| i as usize)
979                .collect()
980        })
981        .unwrap_or_default();
982    if joint_ids.is_empty() {
983        return err("skin bez listy `joints`");
984    }
985
986    let nodes = c
987        .json
988        .get("nodes")
989        .and_then(Json::as_array)
990        .ok_or_else(|| GltfError {
991            msg: "brak `nodes`".into(),
992        })?;
993
994    // Rodzic po indeksie węzła — potrzebny do złożenia hierarchii.
995    let mut parent_of = vec![None; nodes.len()];
996    for (i, n) in nodes.iter().enumerate() {
997        if let Some(kids) = n.get("children").and_then(Json::as_array) {
998            for k in kids.iter().filter_map(Json::as_u32) {
999                parent_of[k as usize] = Some(i);
1000            }
1001        }
1002    }
1003
1004    let mut index_of = BTreeMap::new();
1005    for (si, id) in joint_ids.iter().enumerate() {
1006        index_of.insert(*id, si);
1007    }
1008
1009    for (si, id) in joint_ids.iter().enumerate() {
1010        let n = nodes.get(*id).ok_or_else(|| GltfError {
1011            msg: format!("węzeł {id} poza zakresem"),
1012        })?;
1013        out.push(Joint {
1014            name: n
1015                .get("name")
1016                .and_then(Json::as_str)
1017                .unwrap_or("")
1018                .to_string(),
1019            parent: parent_of[*id].and_then(|p| index_of.get(&p).copied()),
1020            local: node_matrix(n),
1021            skin_index: si as u32,
1022        });
1023    }
1024
1025    // `inverseBindMatrices` to macierze 4×4 kolumnowo, w kolejności
1026    // `skins[].joints`. Brak pola = same macierze tożsamości.
1027    if let Some(acc) = skin.get("inverseBindMatrices").and_then(Json::as_u32) {
1028        let flat = read_f32(c, acc as usize)?;
1029        for m in flat.chunks_exact(16) {
1030            inverse.push(mat_from_cols(m));
1031        }
1032    }
1033    while inverse.len() < out.len() {
1034        inverse.push(Mat4::IDENTITY);
1035    }
1036    Ok(())
1037}

1039/// Macierz lokalna węzła z TRS (domyślnie tożsamość).
1040fn node_matrix(n: &Json) -> Mat4 {
1041    let t = n
1042        .get("translation")
1043        .and_then(Json::as_f64_vec)
1044        .map(|v| Vec3::new(v[0] as f32, v[1] as f32, v[2] as f32))
1045        .unwrap_or(Vec3::ZERO);
1046    let r = n
1047        .get("rotation")
1048        .and_then(Json::as_f64_vec)
1049        .map(|v| Quat::from_xyzw(v[0] as f32, v[1] as f32, v[2] as f32, v[3] as f32))
1050        .unwrap_or(Quat::IDENTITY);
1051    let s = n
1052        .get("scale")
1053        .and_then(Json::as_f64_vec)
1054        .map(|v| Vec3::new(v[0] as f32, v[1] as f32, v[2] as f32))
1055        .unwrap_or(Vec3::ONE);
1056    // Złożenie T*R*S z osobnych konstruktorów — tak robi to `mesh.rs`
1057    // w `model_matrix`, więc zgodność z resztą silnika jest widoczna
1058    // na pierwszy rzut oka.
1059    Mat4::from_translation(t) * Mat4::from_mat3(Mat3::from_quat(r)) * Mat4::from_scale(s)
1060}

1062/// Buduje macierz z 16 liczb w kolejności kolumnowej (jak glTF).
1063fn mat_from_cols(m: &[f32]) -> Mat4 {
1064    // `Mat4::from_cols` składa kolumny, a glTF zapisuje macierz
1065    // kolumnowo — te same cztery kolumny, tylko w innym zapisie.
1066    Mat4::from_cols(
1067        Vec4::new(m[0], m[1], m[2], m[3]),
1068        Vec4::new(m[4], m[5], m[6], m[7]),
1069        Vec4::new(m[8], m[9], m[10], m[11]),
1070        Vec4::new(m[12], m[13], m[14], m[15]),
1071    )
1072}

1080/// Wczytuje model z bajtów pliku `.glb`.
1081///
1082/// Każdy prymityw staje się osobną [`SkinMesh`], bo każdy ma własny
1083/// materiał — a `DrawCmd` w silniku niesie jeden `MaterialId`.
1084pub fn load(bytes: &[u8]) -> Result<GltfScene, Box<dyn std::error::Error>> {
1085    let c = read_container(bytes)?;
1086
1087    let mut scene = GltfScene {
1088        unit_scale: 1.0,
1089        ..Default::default()
1090    };
1091
1092    read_skeleton(&c, &mut scene.joints, &mut scene.inverse_bind)?;
1093    scene.materials = read_materials(&c, None);
1094    scene.textures = read_textures(&c, &scene.materials);
1095
1096    let nodes = c.json.get("nodes").and_then(Json::as_array);
1097    let meshes = c.json.get("meshes").and_then(Json::as_array);
1098
1099    if let Some(nodes) = nodes {
1100        for n in nodes {
1101            let Some(mi) = n.get("mesh").and_then(Json::as_u32) else {
1102                continue;
1103            };
1104            let Some(m) = meshes.and_then(|m| m.get(mi as usize)) else {
1105                return Err(Box::new(GltfError {
1106                    msg: format!("węzeł wskazuje na nieistniejący mesh {mi}"),
1107                }) as Box<dyn std::error::Error>);
1108            };
1109            let prims = m.get("primitives").and_then(Json::as_array).unwrap_or(&[]);
1110            let mesh_name = m.get("name").and_then(Json::as_str).unwrap_or("mesh");
1111            let node_name = n.get("name").and_then(Json::as_str).unwrap_or("");
1112            let world = node_matrix(n);
1113
1114            for (pi, prim) in prims.iter().enumerate() {
1115                let raw = read_primitive(&c, prim, 0)?;
1116                // Nazwa musi rozróżniać prymitywy jednego mesha —
1117                // inaczej w logu zobaczysz trzy „Body".
1118                let name = if node_name.is_empty() {
1119                    format!("{mesh_name}#{pi}")
1120                } else {
1121                    format!("{node_name}/{mesh_name}#{pi}")
1122                };
1123                scene.meshes.push(build_skin_mesh(name, raw, world));
1124            }
1125        }
1126    }
1127
1128    if scene.meshes.is_empty() {
1129        return Err(Box::new(GltfError {
1130            msg: "plik nie zawiera żadnej geometrii".into(),
1131        }) as Box<dyn std::error::Error>);
1132    }
1133    if !c
1134        .json
1135        .get("animations")
1136        .and_then(Json::as_array)
1137        .map_or(false, |a| a.is_empty())
1138    {
1139        scene
1140            .unsupported
1141            .push("animacje osadzone w glTF (czytamy z .urananm)");
1142    }
1143    Ok(scene)
1144}

1146/// Składa [`SkinVertex`] z surowego prymitywu.
1147fn build_skin_mesh(name: String, raw: RawPrimitives, world: Mat4) -> SkinMesh {
1148    let n = raw.positions.len() / 3;
1149    let mut verts = Vec::with_capacity(n);
1150    // Normalna potrzebuje macierzy **odwrotnej transponowanej**, nie
1151    // zwykłej: przy skalowaniu modelu zwykła macierz przekrzywia
1152    // kierunek światła. W glam `inverse()` zwraca same wartości, więc
1153    // nie ma czego obsługiwać — osobliwe macierze dają NaN, a tych
1154    // w poprawnym pliku glTF nie ma.
1155    let inv = world.inverse().transpose();
1156    let normal_mat = Mat3::from_cols(
1157        inv.x_axis.truncate(),
1158        inv.y_axis.truncate(),
1159        inv.z_axis.truncate(),
1160    );
1161    for v in 0..n {
1162        let p = Vec3::new(
1163            raw.positions[v * 3],
1164            raw.positions[v * 3 + 1],
1165            raw.positions[v * 3 + 2],
1166        );
1167        // `Mat4 * Vec3` nie istnieje w glam — punkt mnożymy przez kolumnę
1168        // z jedynką na końcu, inaczej dostalibyśmy dzielenie przez `w`.
1169        let p = (world * p.extend(1.0)).truncate();
1170        let nv = if raw.normals.len() >= (v + 1) * 3 {
1171            let x = Vec3::new(
1172                raw.normals[v * 3],
1173                raw.normals[v * 3 + 1],
1174                raw.normals[v * 3 + 2],
1175            );
1176            (normal_mat * x).normalize_or_zero()
1177        } else {
1178            Vec3::Y
1179        };
1180        let uv = if raw.uvs.len() >= (v + 1) * 2 {
1181            Vec2::new(raw.uvs[v * 2], raw.uvs[v * 2 + 1])
1182        } else {
1183            Vec2::ZERO
1184        };
1185        let mut joints = [0u32; 4];
1186        let mut weights = [0.0f32; 4];
1187        for k in 0..4 {
1188            let ji = (v * 4 + k) as usize;
1189            let wi = (v * 4 + k) as usize;
1190            if raw.joints.get(ji).is_some() {
1191                joints[k] = raw.joints[ji] as u32;
1192            }
1193            if raw.weights.get(wi).is_some() {
1194                weights[k] = raw.weights[wi];
1195            }
1196        }
1197        verts.push(SkinVertex {
1198            position: p.to_array(),
1199            normal: nv.to_array(),
1200            uv: uv.to_array(),
1201            joints,
1202            weights,
1203        });
1204    }
1205    SkinMesh {
1206        name,
1207        vertices: verts,
1208        indices: raw.indices,
1209        material: raw.material,
1210    }
1211}
