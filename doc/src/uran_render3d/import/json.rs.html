1//! Minimalny parser JSON dla importu glTF.
2//!
3//! ## Dlaczego własny, a nie `serde_json`
4//!
5//! Workspace nie zależy od `serde` — w tym środowisku sieć jest
6//! zablokowana, więc dorzucanie zależności to ryzyko, że build przestanie
7//! się odtwarzać. JSON w glTF jest prosty: obiekty, tablice, liczby,
8//! napisy, `true`/`false`/`null`. Nie potrzebujemy tu niczego więcej, a
9//! parser ma ~200 linii i zero zależności.
10//!
11//! ## Czego świadomie nie robimy
12//!
13//! Nie implementujemy składania par `\uD83D\uDE00` w emoji. glTF używa
14//! escapów wyłącznie w nazwach, a tam nie występują. `Json::Str` zwraca
15//! wtedy znak zastępczy; gdyby kiedyś to przestało wystarczać, obsługa
16//! par surrogate to jedyne miejsce do poprawy.
17
18use std::collections::BTreeMap;
19use std::fmt;
20
21/// Wartość JSON — wystarczający podzbiór do odczytu pliku glTF.
22#[derive(Debug, Clone, PartialEq)]
23pub enum Json {
24    Null,
25    Bool(bool),
26    Num(f64),
27    Str(String),
28    Arr(Vec<Json>),
29    /// Kolejność kluczy nie jest tu potrzebna, więc `BTreeMap` wystarcza
30    /// i daje wygodne porównywanie w testach.
31    Obj(BTreeMap<String, Json>),
32}
33
34impl Json {
35    /// Pole obiektu po nazwie; `None` gdy go nie ma lub to nie obiekt.
36    pub fn get(&self, key: &str) -> Option<&Json> {
37        match self {
38            Json::Obj(m) => m.get(key),
39            _ => None,
40        }
41    }
42
43    /// Pole z wartością domyślną — najczęstszy przypadek w glTF, gdzie
44    /// brak pola znaczy „zachowaj resztę”.
45    pub fn field(&self, key: &str, default: Json) -> Json {
46        self.get(key).cloned().unwrap_or(default)
47    }
48
49    pub fn as_f64(&self) -> Option<f64> {
50        match self {
51            Json::Num(n) => Some(*n),
52            _ => None,
53        }
54    }
55
56    pub fn as_u32(&self) -> Option<u32> {
57        self.as_f64().map(|n| n as u32)
58    }
59
60    pub fn as_i32(&self) -> Option<i32> {
61        self.as_f64().map(|n| n as i32)
62    }
63
64    pub fn as_str(&self) -> Option<&str> {
65        match self {
66            Json::Str(s) => Some(s),
67            _ => None,
68        }
69    }
70
71    pub fn as_bool(&self) -> Option<bool> {
72        match self {
73            Json::Bool(b) => Some(*b),
74            _ => None,
75        }
76    }
77
78    pub fn as_array(&self) -> Option<&[Json]> {
79        match self {
80            Json::Arr(v) => Some(v),
81            _ => None,
82        }
83    }
84
85    /// Tablica liczb jako `Vec<f64>`; `None` gdy to nie tablica.
86    pub fn as_f64_vec(&self) -> Option<Vec<f64>> {
87        Some(self.as_array()?.iter().filter_map(Json::as_f64).collect())
88    }
89
90    /// Element tablicy — używane przy `nodes`, `meshes`, `skins`.
91    pub fn idx(&self, i: usize) -> Option<&Json> {
92        self.as_array()?.get(i)
93    }
94}
95
96struct Parser<'a> {
97    b: &'a [u8],
98    i: usize,
99}
100
101impl<'a> Parser<'a> {
102    fn err(&self, msg: &str) -> ParseError {
103        ParseError {
104            at: self.i,
105            msg: msg.to_string(),
106        }
107    }
108
109    /// Pomija biały znak. W glTF bywa go dużo — eksporter formatuje
110    /// dokument czytelnie, a my i tak czytamy bajty.
111    fn ws(&mut self) {
112        while let Some(c) = self.b.get(self.i) {
113            if matches!(c, b' ' | b'\t' | b'\n' | b'\r') {
114                self.i += 1;
115            } else {
116                break;
117            }
118        }
119    }
120
121    fn eat(&mut self, c: u8) -> Result<(), ParseError> {
122        if self.b.get(self.i) == Some(&c) {
123            self.i += 1;
124            Ok(())
125        } else {
126            Err(self.err(&format!("oczekiwano `{}`", c as char)))
127        }
128    }
129
130    fn lit(&mut self, s: &str, v: Json) -> Result<Json, ParseError> {
131        if self.b[self.i..].starts_with(s.as_bytes()) {
132            self.i += s.len();
133            Ok(v)
134        } else {
135            Err(self.err("nieznany literał"))
136        }
137    }
138
139    fn value(&mut self) -> Result<Json, ParseError> {
140        match self.b.get(self.i) {
141            None => Err(self.err("nieskończony koniec")),
142            Some(b'n') => self.lit("null", Json::Null),
143            Some(b't') => self.lit("true", Json::Bool(true)),
144            Some(b'f') => self.lit("false", Json::Bool(false)),
145            Some(b'"') => self.string().map(Json::Str),
146            Some(b'[') => self.array(),
147            Some(b'{') => self.object(),
148            Some(_) => self.number(),
149        }
150    }
151
152    fn string(&mut self) -> Result<String, ParseError> {
153        self.eat(b'"')?;
154        let mut out = String::new();
155        loop {
156            let c = *self
157                .b
158                .get(self.i)
159                .ok_or_else(|| self.err("niezamknięty napis"))?;
160            self.i += 1;
161            match c {
162                b'"' => return Ok(out),
163                b'\\' => {
164                    let e = *self
165                        .b
166                        .get(self.i)
167                        .ok_or_else(|| self.err("ucięty escape"))?;
168                    self.i += 1;
169                    match e {
170                        b'n' => out.push('\n'),
171                        b't' => out.push('\t'),
172                        b'r' => out.push('\r'),
173                        b'b' => out.push('\u{8}'),
174                        b'f' => out.push('\u{c}'),
175                        b'u' => {
176                            let hex = self
177                                .b
178                                .get(self.i..self.i + 4)
179                                .ok_or_else(|| self.err("ucięty \\u"))?;
180                            let s =
181                                std::str::from_utf8(hex).map_err(|_| self.err("zły hex w \\u"))?;
182                            let cp = u32::from_str_radix(s, 16)
183                                .map_err(|_| self.err("zły hex w \\u"))?;
184                            out.push(char::from_u32(cp).unwrap_or('\u{fffd}'));
185                            self.i += 4;
186                        }
187                        other => out.push(other as char),
188                    }
189                }
190                _ => out.push(c as char),
191            }
192        }
193    }
194
195    fn number(&mut self) -> Result<Json, ParseError> {
196        let start = self.i;
197        if self.b.get(self.i) == Some(&b'-') {
198            self.i += 1;
199        }
200        while let Some(c) = self.b.get(self.i) {
201            if c.is_ascii_digit() || matches!(c, b'.' | b'e' | b'E' | b'+' | b'-') {
202                self.i += 1;
203            } else {
204                break;
205            }
206        }
207        let s = std::str::from_utf8(&self.b[start..self.i])
208            .map_err(|_| self.err("zły UTF-8 w liczbie"))?;
209        s.parse::<f64>().map(Json::Num).map_err(|_| ParseError {
210            at: start,
211            msg: format!("zła liczba `{s}`"),
212        })
213    }
214
215    fn array(&mut self) -> Result<Json, ParseError> {
216        self.eat(b'[')?;
217        let mut v = Vec::new();
218        self.ws();
219        if self.b.get(self.i) == Some(&b']') {
220            self.i += 1;
221            return Ok(Json::Arr(v));
222        }
223        loop {
224            self.ws();
225            v.push(self.value()?);
226            self.ws();
227            match self.b.get(self.i) {
228                Some(b',') => self.i += 1,
229                Some(b']') => {
230                    self.i += 1;
231                    return Ok(Json::Arr(v));
232                }
233                _ => return Err(self.err("brak `,` lub `]` w tablicy")),
234            }
235        }
236    }
237
238    fn object(&mut self) -> Result<Json, ParseError> {
239        self.eat(b'{')?;
240        let mut m = BTreeMap::new();
241        self.ws();
242        if self.b.get(self.i) == Some(&b'}') {
243            self.i += 1;
244            return Ok(Json::Obj(m));
245        }
246        loop {
247            self.ws();
248            let k = self.string()?;
249            self.ws();
250            self.eat(b':')?;
251            self.ws();
252            let v = self.value()?;
253            m.insert(k, v);
254            self.ws();
255            match self.b.get(self.i) {
256                Some(b',') => self.i += 1,
257                Some(b'}') => {
258                    self.i += 1;
259                    return Ok(Json::Obj(m));
260                }
261                _ => return Err(self.err("brak `,` lub `}` w obiekcie")),
262            }
263        }
264    }
265}

267#[cfg(test)]
268mod tests {
269    use super::*;
270
271    fn j(s: &str) -> Json {
272        parse(s).unwrap_or_else(|e| panic!("nie sparsowano `{s}`: {e}"))
273    }
274
275    #[test]
276    fn czyta_scalars() {
277        assert_eq!(j("1"), Json::Num(1.0));
278        assert_eq!(j("-2.5"), Json::Num(-2.5));
279        assert_eq!(j("true"), Json::Bool(true));
280        assert_eq!(j("null"), Json::Null);
281        assert_eq!(j("\"tekst\""), Json::Str("tekst".into()));
282    }
283
284    #[test]
285    fn czyta_wykladniki() {
286        // Accessory glTF zapisują małe liczby jako `1e-2`.
287        assert_eq!(j("1e-2").as_f64(), Some(0.01));
288        assert_eq!(j("-1.5E3").as_f64(), Some(-1500.0));
289    }
290
291    #[test]
292    fn czyta_tablicy() {
293        assert_eq!(j("[1,2,3]").as_f64_vec(), Some(vec![1.0, 2.0, 3.0]));
294        assert_eq!(j("[]").as_array().map(|a| a.len()), Some(0));
295    }
296
297    #[test]
298    fn czyta_zagniezdzone_obiektow() {
299        let v = j(r#"{"a":{"b":[1,{"c":2}]},"d":true}"#);
300        assert_eq!(v.get("d").and_then(Json::as_bool), Some(true));
301        let inner = v.get("a").and_then(|a| a.get("b")).unwrap();
302        assert_eq!(inner.idx(0).and_then(Json::as_f64), Some(1.0));
303        assert_eq!(
304            inner.idx(1).and_then(|o| o.get("c")).and_then(Json::as_f64),
305            Some(2.0)
306        );
307    }
308
309    #[test]
310    fn pole_z_wartoscia_domyslna() {
311        let v = j(r#"{"a":1}"#);
312        assert_eq!(v.field("a", Json::Null).as_f64(), Some(1.0));
313        // brakujące pole dostaje wartość domyślną, nie `None`
314        assert_eq!(v.field("b", Json::Num(7.0)).as_f64(), Some(7.0));
315    }
316
317    #[test]
318    fn odporne_na_biale_znaki_i_escapy() {
319        let v = j("  { \"a\" : [ 1 , 2 ] ,\n\t\"s\" : \"x\\ty\" }  ");
320        assert_eq!(v.get("a").unwrap().as_f64_vec(), Some(vec![1.0, 2.0]));
321        assert_eq!(v.get("s").and_then(Json::as_str), Some("x\ty"));
322    }
323
324    #[test]
325    fn zglasza_blad_z_pozycja() {
326        let e = parse("{").unwrap_err();
327        // komunikat musi mówić GDZIE, bo glTF to jeden długi wiersz
328        assert!(format!("{e}").contains("bajcie"), "komunikat: {e}");
329        assert!(parse("[1,]").is_err());
330        assert!(parse("{} smiec").is_err());
331        assert!(parse("\"niezamkniete").is_err());
332    }

335/// Błąd z pozycją — bez niej komunikat jest bezużyteczny, bo JSON
336/// w glTF bywa jednym wierszem długości 90 kB.
337#[derive(Debug)]
338pub struct ParseError {
339    pub at: usize,
340    pub msg: String,
341}
342
343impl fmt::Display for ParseError {
344    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
345        write!(f, "błąd JSON przy bajcie {}: {}", self.at, self.msg)
346    }
347}
348
349impl std::error::Error for ParseError {}
350
351/// Parsuje cały dokument.
352pub fn parse(src: &str) -> Result<Json, ParseError> {
353    let b = src.as_bytes();
354    let mut p = Parser { b, i: 0 };
355    p.ws();
356    let v = p.value()?;
357    p.ws();
358    if p.i != b.len() {
359        return Err(p.err("śmieci za wartością"));
360    }
361    Ok(v)
362}
