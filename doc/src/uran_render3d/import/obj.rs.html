1//! Parser plików `.obj` (Wavefront) + `.mtl` do gotowych siatek silnika.
2//!
3//! ## Co tu robimy
4//!
5//! 1. Czytamy `v` / `vn` / `vt` / `f` / `usemtl` / `o` / `mtllib`.
6//! 2. Rozwijamy wielokąty na trójkąty (wianek) i rozdzielamy indeksy,
7//!    bo `f 1/5/9 2/6/10` to trzy różne pary (pozycja, UV, normalna).
8//! 3. Zamieniamy osie Z-up (Blender) na Y-up (silnik).
9//! 4. Normalizujemy skalę i stawiamy model na `y = 0`.

18/// Referencja na wierzchołek: `pozycja/uv/normalna` z 1-based indeksów.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub struct FaceRef {
21    pub pos: usize,
22    pub uv: Option<usize>,
23    pub nrm: Option<usize>,
24}
25
26impl FaceRef {
27    /// Parsuje token `1/2/3`, `1//3`, `1/2` lub `1`.
28    ///
29    /// Ujemne indeksy są względne do KOŃCA listy (`-1` = ostatni), więc
30    /// nie wolno ich liczyć od zera. Zwracamy indeks 0-based.
31    fn parse(tok: &str, npos: usize, nuv: usize, nnrm: usize) -> Result<Self, String> {
32        let mut it = tok.split('/');
33        let pos = it.next().unwrap_or("").trim();
34        let uv = it.next().map(str::trim).filter(|s| !s.is_empty());
35        let nrm = it.next().map(str::trim).filter(|s| !s.is_empty());
36
37        if pos.is_empty() {
38            return Err(format!("pusta referencja wierzchołka: {tok}"));
39        }
40        let pos = parse_index(pos, npos)?;
41        let uv = match uv {
42            Some(v) => Some(parse_index(v, nuv).map_err(|e| format!("uv: {e}"))?),
43            None => None,
44        };
45        let nrm = match nrm {
46            Some(v) => Some(parse_index(v, nnrm).map_err(|e| format!("normalna: {e}"))?),
47            None => None,
48        };
49        Ok(Self { pos, uv, nrm })
50    }
51}
52
53/// 1-based (lub ujemny od końca) indeks na 0-based.
54fn parse_index(tok: &str, len: usize) -> Result<usize, String> {
55    let raw: i64 = tok.parse().map_err(|_| format!("zły indeks: {tok}"))?;
56    let idx = if raw < 0 { len as i64 + raw } else { raw - 1 };
57    if idx < 0 || idx as usize >= len {
58        return Err(format!("indeks {tok} poza zakresem 1..={len}"));
59    }
60    Ok(idx as usize)
61}

63/// Wynik parsowania tekstu `.obj`, jeszcze bez siatek.
64#[derive(Debug, Clone, Default)]
65pub struct ObjScene {
66    pub positions: Vec<[f32; 3]>,
67    pub normals: Vec<[f32; 3]>,
68    pub uvs: Vec<[f32; 2]>,
69    /// Trójkąty po rozwinięciu wielokątów.
70    ///
71    /// Trzymamy pełne [`FaceRef`], a nie sam indeks pozycji: w `.obj`
72    /// te trzy listy są niezależne i ten sam wierzchołek często ma
73    /// inne UV niż sąsiad.
74    pub triangles: Vec<[FaceRef; 3]>,

77    /// Nazwy obiektów z `o`/`g`, równoległe do `tri_object`.

79    /// Indeks obiektu każdego trójkąta.
80    pub tri_object: Vec<usize>,
81    /// Nazwa pliku `.mtl` z `mtllib`.
82    pub mtllib: Option<String>,
83}

101/// Parsuje tekst `.obj` do [`ObjScene`].
102pub fn parse_obj(text: &str) -> Result<ObjScene, Box<dyn std::error::Error>> {
103    let mut scene = ObjScene::default();
104    let mut cur_material = String::new();
105    let mut cur_object = 0usize;
106    scene.objects.push(String::new());
107
108    for (i, raw) in text.lines().enumerate() {
109        let line_no = i + 1;
110        let line = raw.trim();
111        if line.is_empty() || line.starts_with('#') {
112            continue;
113        }
114        let mut it = line.split_whitespace();
115        let Some(key) = it.next() else { continue };
116        let rest: Vec<&str> = it.collect();
117        let err = |m: String| ParseError {
118            line: line_no,
119            message: m,
120        };
121
122        match key {
123            "v" => {
124                let p = parse_vec3(&rest).ok_or_else(|| err("zły wiersz v".into()))?;
125                scene.positions.push(p);
126            }
127            "vn" => {
128                let p = parse_vec3(&rest).ok_or_else(|| err("zły wiersz vn".into()))?;
129                scene.normals.push(p);
130            }
131            "vt" => {
132                // `vt u v` albo `vt u v w` (w ignorujemy — mamy tylko 2D)
133                let u = rest.first().and_then(|s| s.parse::<f32>().ok());
134                let v = rest.get(1).and_then(|s| s.parse::<f32>().ok());
135                match (u, v) {
136                    (Some(u), Some(v)) => scene.uvs.push([u, v]),
137                    (Some(u), None) => scene.uvs.push([u, 0.0]),
138                    _ => return Err(Box::new(err("zły wiersz vt".into()))),
139                }
140            }
141            "f" => {
142                if rest.len() < 3 {
143                    return Err(Box::new(err(format!(
144                        "wielokąt ma {} wierzchołków",
145                        rest.len()
146                    ))));
147                }
148                let mut face = Vec::with_capacity(rest.len());
149                for tok in &rest {
150                    // `?` na `String` zamieniłby błąd w gołą treść i
151                    // zgubił numer linii — dlatego pakujemy go tu w `ParseError`
152                    let r = FaceRef::parse(
153                        tok,
154                        scene.positions.len(),
155                        scene.uvs.len(),
156                        scene.normals.len(),
157                    );
158                    face.push(r.map_err(|e| err(format!("wiersz twarzy: {e}")))?);
159                }
160                push_face(&mut scene, &face, &cur_material, cur_object);
161            }
162            "usemtl" => cur_material = rest.join(" "),
163            "o" | "g" => {
164                // Przy pustej nazwie używamy materiału — dzięki temu
165                // lista części nigdy nie zawiera pustego obiektu.
166                let name = rest.join(" ");
167                let name = if name.is_empty() {
168                    cur_material.clone()
169                } else {
170                    name
171                };
172                cur_object = scene.objects.len();
173                scene.objects.push(name);
174            }
175            // W pliku z Blendera bywa „junak m10 .mtl" ze spacją w nazwie
176            "mtllib" => scene.mtllib = Some(rest.join(" ")),
177            // `s`, `vp`, `l` — pomijamy: normalne dostarcza sam plik
178            _ => {}
179        }
180    }
181    Ok(scene)
182}

184/// Rozwijamy wielokąt na trójkąty (wianek) i pilnujemy orientacji.
185///
186/// ## Dlaczego sprawdzamy CAŁY wielokąt, a nie każdy trójkąt osobno
187///
188/// Iloczyn wektorowy działa tylko dla trójkąta. Przy kwadracie na
189/// zakrzywionej powierzchni (opona, nadwozie) każdy trójkąt wianka leży
190/// w innej płaszczyźnie, a różnica bywa większa niż 90°. Porównanie
191/// jednego trójkąta z gładką normalną wierzchołka daje wtedy przypadkowe
192/// odwrócenia — a każde z nich to dziura, bo przy `cull_mode: Back`
193/// trójkąt znika i widać nieoświetlone wnętrze. Widać to dokładnie tak,
194/// jak opisuje zgłoszenie: **czarne kwadraty** w miejscach, gdzie sam
195/// plik jest poprawny.
196///
197/// Dlatego decyzja jest JEDNA na cały wielokąt, z normalną policzoną
198/// metodą Newella (suma wkładów wszystkich krawędzi — stabilna także dla
199/// wielokąta niepłaskiego) i z **marginesem**: odwracamy tylko przy
200/// jednoznacznym sprzeczności, a nie na granicy szumu.
201fn push_face(scene: &mut ObjScene, face: &[FaceRef], material: &str, object: usize) {
202    if face.len() < 3 {
203        return;
204    }
205
206    let reverse = match face_avg_normal(scene, face) {
207        Some(avg) => {
208            let geo = polygon_normal(scene, face);
209            let (gl, al) = (geo.length(), avg.length());
210            if gl < 1e-9 || al < 1e-9 {
211                false
212            } else {
213                // cos kąta między nimi; ujemny = skierowane przeciwnie.
214                // -0.2 to próg: powyżej 101° sprzeczności, więc przy
215                // legalnym łuku wielokąta nigdy nie wchodzimy w tę gałąź.
216                geo.dot(avg) / (gl * al) < -0.2
217            }
218        }
219        // Bez normalnych nie mamy czym zweryfikować — ufamy plikowi.
220        None => false,
221    };
222
223    for k in 1..face.len() - 1 {
224        // Odwrócenie całego wianka naraz: inaczej sąsiednie trójkąty
225        // tej samej ściany dostałyby przeciwne zwroty.
226        let (a, b, c) = if reverse {
227            (face[0], face[k + 1], face[k])
228        } else {
229            (face[0], face[k], face[k + 1])
230        };
231        // Trójkąt o zerowym polu nie pokryje żadnego piksela, a jego
232        // normalna geometryczna to (0,0,0) -> `normalize` daje NaN ->
233        // czarny piksel. Pomijamy go całkowicie.
234        let gn = face_normal(
235            scene.positions[a.pos],
236            scene.positions[b.pos],
237            scene.positions[c.pos],
238        );
239        if gn.length() < 1e-12 {
240            continue;
241        }
242        scene.triangles.push([a, b, c]);
243        scene.tri_material.push(material.to_string());
244        scene.tri_object.push(object);
245    }
246}

248/// Normalna wielokątu metodą Newella.
249///
250/// Suma `(p_i - p_{i+1}) x (p_i + p_{i+1})` po wszystkich krawędziach.
251/// W przeciwieństwie do iloczynu wektorowego działa dla dowolnej liczby
252/// wierzchołków i dla wielokąta **niepłaskiego** — a takie właśnie są
253/// kwadraty z Blendera na częściach cylindrycznych.
254fn polygon_normal(scene: &ObjScene, face: &[FaceRef]) -> Vec3 {
255    let mut n = Vec3::ZERO;
256    for i in 0..face.len() {
257        let a = scene.positions[face[i].pos];
258        let b = scene.positions[face[(i + 1) % face.len()].pos];
259        n += Vec3::new(
260            (a[1] - b[1]) * (a[2] + b[2]),
261            (a[2] - b[2]) * (a[0] + b[0]),
262            (a[0] - b[0]) * (a[1] + b[1]),
263        );
264    }
265    n
266}

268/// Średnia z normalnych wierzchołków wielokątu, albo `None`, gdy plik
269/// ich nie podał.
270fn face_avg_normal(scene: &ObjScene, face: &[FaceRef]) -> Option<Vec3> {
271    let mut sum = Vec3::ZERO;
272    let mut count = 0u32;
273    for f in face {
274        if let Some(i) = f.nrm {
275            let n = scene.normals[i];
276            sum += Vec3::new(n[0], n[1], n[2]);
277            count += 1;
278        }
279    }
280    if count == 0 {
281        None
282    } else {
283        Some(sum / count as f32)
284    }
285}

287fn parse_vec3(rest: &[&str]) -> Option<[f32; 3]> {
288    if rest.len() < 3 {
289        return None;
290    }
291    let x: f32 = rest[0].parse().ok()?;
292    let y: f32 = rest[1].parse().ok()?;
293    let z: f32 = rest[2].parse().ok()?;
294    if !x.is_finite() || !y.is_finite() || !z.is_finite() {
295        return None;
296    }
297    Some([x, y, z])
298}

300/// Iloczyn wektorowy × wektorowy (normalna ściany, przed normalizacją).
301fn face_normal(a: [f32; 3], b: [f32; 3], c: [f32; 3]) -> Vec3 {
302    let u = Vec3::new(b[0] - a[0], b[1] - a[1], b[2] - a[2]);
303    let v = Vec3::new(c[0] - a[0], c[1] - a[1], c[2] - a[2]);
304    u.cross(v)
305}

307/// Która oś pliku `.obj` wskazuje w górę.
308///
309/// **Nie zgadujemy tego po wymiarach.** Dla motocykla długość, wysokość
310/// i szerokość różnią się nieznacznie, a wczytanie modelu „na czuja"
311/// daje bryłę leżącą na boku — i wygląda to jak błąd silnika, a nie
312/// konfiguracji. Konwencja jest więc jawna i zapisywana przez tego, kto
313/// zna model.
314#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
315pub enum AxisUp {
316    /// `+Z` w górę — tak eksportuje Blender z ustawieniem domyślnym.
317    #[default]
318    Z,
319    /// `+Y` w górę — tak eksportuje Blender po włączeniu `Y up`
320    /// w opcjach eksportu Wavefront (oraz np. assimp, gdy nie
321    /// skaluje osi).
322    Y,
323}
324
325/// Zamienia oś pliku na układ silnika (Y-up, prawoskrętny).
326///
327/// ## Dlaczego to osobna funkcja, a nie `powiązanie` w kodzie
328///
329/// Obie konwencje muszą dać w macierzy **det = +1**. Mapowanie
330/// `(x, y, z) -> (x, z, y)` ma det = -1, czyli ODWRACA układ, a wraz
331/// nim kierunek zwrotu trójkątów: przy włączonym `cull_mode: Back`
332/// cały model znikałby jako „odwrócony do wewnątrz". Dlatego przy
333/// `Z up` schodzimy do przodu (`-y`), a przy `Y up` nie ruszamy
334/// niczego — to już układ docelowy.
335fn to_engine(p: [f32; 3], up: AxisUp) -> Vec3 {
336    match up {
337        // Blender Z-up: X = długość, Y = szerokość, Z = wysokość.
338        // Silnik: X = długość, Y = wysokość, Z = szerokość.
339        // Obrót o -90° wokół X (jedna składowa z minusem = det +1).
340        AxisUp::Z => Vec3::new(p[0], p[2], -p[1]),
341        // Blender Y-up: X = długość, Y = wysokość, Z = szerokość.
342        // Identyczność, więc wiatraka trójkątów nie ruszamy.
343        AxisUp::Y => Vec3::new(p[0], p[1], p[2]),
344    }
345}

366/// Buduje siatki z [`ObjScene`] i przypisuje materiały.
367///
368/// Nie optymalizujemy wspólnych wierzchołków między ścianami: pliki
369/// z Blendera i tak rzadko je dzielą, a scalanie po pozycji psułoby
370/// mapowanie UV, które akurat w tych modelach jest istotne.
371fn build(scene: ObjScene, lib: &MaterialLib, up: AxisUp) -> ImportedModel {
372    // Tylko obiekty, które faktycznie mają trójkąty.
373    let mut order: Vec<usize> = (0..scene.objects.len()).collect();
374    order.retain(|&o| scene.tri_object.contains(&o));
375
376    // Czesci grupujemy po PARZE (obiekcie, materiale), a nie tylko po
377    // obiekcie. Pliki z Blendera trzymaja w jednym obiekcie kilka
378    // `usemtl` — np. `Barn_Cube.005` ma DarkRed, White, LightRed i
379    // RoofBlack. Grupowanie tylko po `o` dawalo caly budynek w kolorze
380    // PIERWSZEGO materialu: czerwony dach, kremowe sciany i czarne
381    // okna splynaly w jeden kolor. Po jednej czesci na (obiek, material)
382    // kazda sciana dostaje swoj wlasny material.
383    let mut order: Vec<(usize, String)> = Vec::new();
384    for ti in 0..scene.triangles.len() {
385        if !scene.tri_object.contains(&scene.tri_object[ti]) {
386            continue;
387        }
388        let key = (scene.tri_object[ti], scene.tri_material[ti].clone());
389        if !order.contains(&key) {
390            order.push(key);
391        }
392    }
393
394    let mut parts = Vec::with_capacity(order.len());
395    for (obj, part_material) in order {
396        let mut mesh = Mesh::new();
397
398        for (ti, tri) in scene.triangles.iter().enumerate() {
399            if scene.tri_object[ti] != obj || scene.tri_material[ti] != part_material {
400                continue;
401            }
402            let material = lib.get(&scene.tri_material[ti]);
403            // Normalna z geometrii — awaryjnie, gdy plik nie ma `vn`.
404            // Przechodzi tę samą transformację co pozycja; inaczej przy
405            // pliku Y-up (gdzie `to_engine` jest identycznością) zgadza
406            // się przypadkiem, a przy Z-up wskazywałaby w złym układzie.
407            let gn = to_engine(
408                face_normal(
409                    scene.positions[tri[0].pos],
410                    scene.positions[tri[1].pos],
411                    scene.positions[tri[2].pos],
412                )
413                .to_array(),
414                up,
415            )
416            .normalize_or_zero();
417
418            for corner in tri {
419                let p = to_engine(scene.positions[corner.pos], up);
420                // Normalna przechodzi TĘ SAMĄ transformację co pozycja.
421                // Pominięcie jej tutaj daje model, który wygląda poprawnie
422                // z każdego kąta, ale jest zamalowany czarnym: światło
423                // pada na wnętrzność bryły, bo wektor wskazuje w stronę
424                // przeciwną. Pozycje i normalne były w układzie pliku,
425                // a silnik oczekuje obu w układzie silnika.
426                let n = corner
427                    .nrm
428                    .map(|i| to_engine(scene.normals[i], up).normalize_or_zero())
429                    // `unwrap_or` NIE wystarcza: `.map()` da tu `Some(ZERO)`,
430                    // gdy plik zapisał zerową normalną, a `normalize(vec3(0))`
431                    // w WGSL to NaN, który wychodzi czarnym pikselem.
432                    // Dopiero porównanie długości odsiewa taki przypadek
433                    // i sprowadza go do normalnej geometrycznej.
434                    .filter(|v| v.length() > 0.5)
435                    .unwrap_or(gn);
436                let uv = corner.uv.map(|i| scene.uvs[i]).unwrap_or([0.0, 0.0]);
437                // Kd jako kolor wierzchołka: gdy jest tekstura, shader
438                // zdominuje ją i tak; gdy jej nie ma, dostajemy kolor
439                // materiału zamiast bieli.
440                mesh.push(Vertex::new_uv(p, n, uv, material.albedo));
441            }
442            let i = mesh.vertices.len() as u32 - 3;
443            mesh.indices.extend_from_slice(&[i, i + 1, i + 2]);
444        }
445
446        if mesh.vertices.is_empty() {
447            continue;
448        }
449        let material = lib.get(&part_material);
450        let obj_name = scene.objects[obj].clone();
451        parts.push(ModelPart {
452            name: if obj_name.is_empty() {
453                material.name.clone()
454            } else {
455                obj_name
456            },
457            mesh,
458            material,
459        });
460    }
461
462    // Środek i rozmiar surowej geometrii
463    let mut lo = Vec3::splat(f32::INFINITY);
464    let mut hi = Vec3::splat(f32::NEG_INFINITY);
465    for p in &scene.positions {
466        let p = Vec3::new(p[0], p[1], p[2]);
467        lo = lo.min(p);
468        hi = hi.max(p);
469    }
470    if lo.x == f32::INFINITY {
471        lo = Vec3::ZERO;
472        hi = Vec3::ZERO;
473    }
474
475    ImportedModel {
476        parts,
477        raw_center: (lo + hi) * 0.5,
478        raw_size: hi - lo,
479    }
480}

482impl ImportedModel {
483    /// Aktualny rozmiar modelu (po normalizacji).
484    pub fn bounds_size(&self) -> Vec3 {
485        let (mut lo, mut hi) = (Vec3::splat(f32::INFINITY), Vec3::splat(f32::NEG_INFINITY));
486        for part in &self.parts {
487            for v in &part.mesh.vertices {
488                lo = lo.min(v.pos());
489                hi = hi.max(v.pos());
490            }
491        }
492        if lo.x == f32::INFINITY {
493            return Vec3::ZERO;
494        }
495        hi - lo
496    }
497
498    /// Przesuwa model tak, by stał na `y = 0` i był wyśrodkowany w XZ.
499    ///
500    /// Dzięki temu siatka wstawiona do sceny ma sensowne początki bez
501    /// wiedzy o tym, gdzie w pliku leżał oryginał.
502    pub fn normalize_to_ground(&mut self) {
503        let size = self.bounds_size();
504        if size.x <= 0.0 && size.y <= 0.0 {
505            return;
506        }
507        let (mut lo, mut hi) = (Vec3::splat(f32::INFINITY), Vec3::splat(f32::NEG_INFINITY));
508        for part in &self.parts {
509            for v in &part.mesh.vertices {
510                lo = lo.min(v.pos());
511                hi = hi.max(v.pos());
512            }
513        }
514        if lo.x == f32::INFINITY {
515            return;
516        }
517        let offset = Vec3::new(-(lo.x + hi.x) * 0.5, -lo.y, -(lo.z + hi.z) * 0.5);
518        for part in &mut self.parts {
519            for v in &mut part.mesh.vertices {
520                v.position = (v.pos() + offset).to_array();
521            }
522        }
523    }
524
525    /// Skaluje model do zadanej wysokości (w jednostkach świata).
526    ///
527    /// Modele z Blendera bywają w centymetrach albo w dowolnej skali
528    /// eksportu, a scena ma metry. Wysokość jest tu wybrana, bo motocykl
529    /// rozpoznajemy po wysokości — po długości różnią się motocykle
530    /// i samochody.
531    pub fn scale_to_height(&mut self, height: f32) {
532        let size = self.bounds_size();
533        if size.y <= 1e-6 || !size.y.is_finite() || height <= 0.0 {
534            return;
535        }
536        let k = height / size.y;
537        for part in &mut self.parts {
538            for v in &mut part.mesh.vertices {
539                // Skala jednorodna nie zmienia kierunku, więc normalnych
540                // nie ruszamy — wystarczy je znormalizować w shaderze.
541                v.position = (v.pos() * k).to_array();
542            }
543        }
544    }
545
546    /// Łączy części o tym samym materiale (mniej draw calli).
547    pub fn merge_by_material(&mut self) {
548        use std::collections::BTreeMap;
549        let mut groups: BTreeMap<String, Mesh> = BTreeMap::new();
550        let mut names: BTreeMap<String, String> = BTreeMap::new();
551        for part in &self.parts {
552            let e = groups.entry(part.material.name.clone()).or_default();
553            let base = e.vertices.len() as u32;
554            e.vertices.extend_from_slice(&part.mesh.vertices);
555            for i in &part.mesh.indices {
556                e.indices.push(base + i);
557            }
558            names
559                .entry(part.material.name.clone())
560                .or_insert_with(|| part.name.clone());
561        }
562        let mut out = Vec::new();
563        for (mat, mesh) in groups {
564            if mesh.vertices.is_empty() {
565                continue;
566            }
567            out.push(ModelPart {
568                name: names.get(&mat).cloned().unwrap_or_else(|| mat.clone()),
569                material: Material::fallback(&mat),
570                mesh,
571            });
572        }
573        self.parts = out;
574    }
575
576    /// Łączy WSZYSTKIE części w jedną siatkę (jeden materiał).
577    pub fn merge_all(&mut self) {
578        let mut merged = Mesh::new();
579        for part in &self.parts {
580            let base = merged.vertices.len() as u32;
581            merged.vertices.extend_from_slice(&part.mesh.vertices);
582            for i in &part.mesh.indices {
583                merged.indices.push(base + i);
584            }
585        }
586        let material = self
587            .parts
588            .first()
589            .map(|p| p.material.clone())
590            .unwrap_or_default();
591        self.parts = vec![ModelPart {
592            name: "merged".into(),
593            mesh: merged,
594            material,
595        }];
596    }

599/// Wczytuje `.obj` wraz z `.mtl`, zakładając plik **Z-up**.
600///
601/// To skrót dla [`load_from_file_with_axes`] z [`AxisUp::Z`], czyli
602/// domyślnym ustawieniem eksportu Blendera.
603pub fn load_from_file(path: impl AsRef<Path>) -> Result<ImportedModel, Box<dyn std::error::Error>> {
604    load_from_file_with_axes(path, AxisUp::Z)
605}

607/// Wczytuje `.obj` wraz z `.mtl` z jawną konwencją osi pionowej.
608///
609/// `path` wskazuje na plik `.obj`. Katalog `.mtl` bierzemy z dyrektywy
610/// `mtllib`, a jeśli go nie ma — z katalogu samego `.obj`. Brakujący
611/// `.mtl` to nie błąd: model dostanie materiały zastępcze i wciąż
612/// się załaduje.
613///
614/// ## Pojawia się parametr `up`
615///
616/// Zly wybór tej wartości nie daje błędu parsowania ani pustego kadru —
617/// model wczytuje się „dobrze", tylko leży na boku albo stoi na
618/// głowie. Dlatego `up` jest parametrem jawnym, a nie heurystyką
619/// wyliczaną z wymiarów: wymiary motocykla (długość ≈ 2,5 m, wysokość
620/// ≈ 1,2 m, szerokość ≈ 0,8 m) nie pozwalają odróżnić osi wiarygodnie.
621pub fn load_from_file_with_axes(
622    path: impl AsRef<Path>,
623    up: AxisUp,
624) -> Result<ImportedModel, Box<dyn std::error::Error>> {
625    let path = path.as_ref();
626    let text = std::fs::read_to_string(path)?;
627    let scene = parse_obj(&text)?;
628
629    let dir = path.parent().unwrap_or(Path::new("."));
630    let mtl_path = scene
631        .mtllib
632        .as_deref()
633        .map(str::trim)
634        .filter(|s| !s.is_empty())
635        .map(|n| dir.join(n))
636        .unwrap_or_else(|| path.with_extension("mtl"));
637
638    let lib = match std::fs::read_to_string(&mtl_path) {
639        Ok(t) => MaterialLib::parse(&t, mtl_path.parent().unwrap_or(dir)),
640        Err(_) => MaterialLib::default(),
641    };
642
643    let mut model = build(scene, &lib, up);
644    model.normalize_to_ground();
645    Ok(model)
646}

963    /// Test integracyjny na modelu junaka prosto z Blendera.
964    ///
965    /// ## Dlaczego to był „czarny kwadrat", a nie „dziura w siatce"
966    ///
967    /// Plik jest złożony z kwadratów na zakrzywionych powierzchniach
968    /// (opona, nadwozie). Stary `push_face` porównywał iloczyn wektorowy
969    /// **jednego trójkąta wianka** z **gładką normalną wierzchołka**; na
970    /// łuku te dwa wektory bywają od siebie ponad 90°, więc iloczyn
971    /// wychodził ujemny dla poprawnie zwiniętego trójkąta. Po odwróceniu
972    /// `cull_mode: Back` go odsiewał i widoczne było nieoświetlone wnętrze.
973    ///
974    /// ## Czego ten test NIE robi
975    ///
976    /// Nie porównuje normalnej **trójkąta wianka** z normalnymi pliku —
977    /// to byłby powtórzony ten sam błąd. Na zakrzywionym kwadacie poprawne
978    /// trójkąty mają tam `dot < 0` (sprawdzone na tym pliku: 27 z 64 450),
979    /// bo powierzchnia zakrzywiona jest krzywą, a nie płaszczyzną.
980    ///
981    /// Sprawdzamy niezmiennik **wielokąta**: normalna Newella vs średnia
982    /// z normalnych wierzchołków. Dla tego pliku daje to cos = +1 dla
983    /// wszystkich 24 999 ścian, czyli plik jest spójny i nie wolno go
984    /// „naprawiać" na siłę.
985    #[test]
986    fn real_model_polygons_are_wound_consistently() {
987        let Some(path) = test_model_path() else {
988            eprintln!("pomijam — brak pliku .obj w repo");
989            return;
990        };
991        let text = std::fs::read_to_string(&path).unwrap();
992        let s = parse_obj(&text).unwrap();
993
994        // Odtwarzamy wielokąty z linii `f` — `parse_obj` zwraca już
995        // rozwinięte trójkąty, a tu potrzebujemy całej ściany naraz.
996        let mut polys: Vec<Vec<FaceRef>> = Vec::new();
997        for line in text.lines() {
998            let Some(rest) = line.strip_prefix("f ") else {
999                continue;
1000            };
1001            let mut face = Vec::new();
1002            for (i, tok) in rest.split_whitespace().enumerate() {
1003                let npos = s.positions.len();
1004                let nuv = s.uvs.len();
1005                let nnrm = s.normals.len();
1006                match FaceRef::parse(tok, npos, nuv, nnrm) {
1007                    Ok(fr) => face.push(fr),
1008                    Err(e) => panic!("wielokąt {i}: {e}"),
1009                }
1010            }
1011            if face.len() >= 3 {
1012                polys.push(face);
1013            }
1014        }
1015
1016        let mut bad = 0usize;
1017        let mut min_cos = 1.0f32;
1018        for face in &polys {
1019            let Some(avg) = face_avg_normal(&s, face) else {
1020                continue;
1021            };
1022            let geo = polygon_normal(&s, face);
1023            let (gl, al) = (geo.length(), avg.length());
1024            if gl < 1e-9 || al < 1e-9 {
1025                continue;
1026            }
1027            let cos = geo.dot(avg) / (gl * al);
1028            min_cos = min_cos.min(cos);
1029            if cos < 0.0 {
1030                bad += 1;
1031            }
1032        }
1033        assert!(
1034            polys.len() > 1000,
1035            "przeanalizowano za mało: {}",
1036            polys.len()
1037        );
1038        assert_eq!(
1039            bad,
1040            0,
1041            "{bad} z {} ścian ma wiatrak przeciwny do normalnych pliku \
1042             (min cos = {min_cos})",
1043            polys.len()
1044        );
1045        assert!(
1046            min_cos > 0.9,
1047            "najgorsza ściana ma cos = {min_cos} — plik nie jest wiarygodnym źródłem"
1048        );

1081    /// Test integracyjny na modelu junaka prosto z Blendera.
1082    ///
1083    /// Ładujemy prawdziwy plik z repo, bo parser syntaktycznie poprawny
1084    /// wciąż potrafi zgubić UV albo odwrócić model — a to widać dopiero
1085    /// na geometrii, nie w testach na stringach.
1086    ///
1087    /// Test pomijamy, gdy pliku nie ma (np. kopia bez zasobów).
1088    #[test]
1089    fn loads_real_blender_motorcycle() {
1090        let Some(path) = test_model_path() else {
1091            eprintln!("pomijam: brak pliku .obj w repo");
1092            return;
1093        };
1094        // Ten plik jest Y-up (dowód w `junak-rider/src/main.rs`), więc
1095        // wczytujemy go właściwą konwencją. Przy Z-up test przechodził mimo
1096        // modelu leżącego na boku — mierzył tylko rzeczy niezależne od
1097        // obrotu (liczbę trójkątów i obecność UV).
1098        let Ok(mut model) = load_from_file_with_axes(&path, AxisUp::Y) else {
1099            eprintln!("pomijam: nie udało się wczytać {}", path.display());
1100            return;
1101        };
1102        model.scale_to_height(1.2);
1103
1104        let verts: usize = model.parts.iter().map(|p| p.mesh.vertices.len()).sum();
1105        let idx: usize = model.parts.iter().map(|p| p.mesh.indices.len()).sum();
1106        let tris = idx / 3;
1107        eprintln!(
1108            "junak: {} części, {verts} wierzchołków, {tris} trójkątów, rozmiar {:?}",
1109            model.parts.len(),
1110            model.bounds_size()
1111        );
1112
1113        assert!(model.parts.len() > 5, "motocykl ma wiele obiektów");
1114        assert!(tris > 10_000, "zachowaliśmy geometrię: {tris} trójkątów");
1115        // Import NIE scala wierzchołków (patrz `build`), więc na trójkąt
1116        // przypada dokładnie 3 wierzchołki. To świadomy koszt: 193 tys.
1117        // wierzchołków × 44 B ≈ 8,5 MB, ale za to UV są dokładne.
1118        assert_eq!(verts, tris * 3, "wierzchołki i trójkąty się nie zgadzają");
1119
1120        // UV muszą dojechać do wierzchołków — bez tego model wczytuje
1121        // się, ale jest nieoteksturowany mimo pliku z Blendera
1122        let with_uv = model
1123            .parts
1124            .iter()
1125            .flat_map(|p| p.mesh.vertices.iter())
1126            .filter(|v| v.has_uv())
1127            .count();
1128        assert!(
1129            with_uv > 1000,
1130            "UV nie dotarły: tylko {with_uv} wierzchołków"
1131        );
1132
1133        // Normalne MUSZĄ dostać tę samą transformację co pozycje.
1134        // Bez tego model jest czarny: światło pada na wnętrzność bryły.
1135        // Test patrzy na górne ściany motocykla (siodło, bak) — tam
1136        // normalna musi wskazywać w górę, czyli `n.y > 0`.
1137        let mut up = 0usize;
1138        let mut total = 0usize;
1139        for part in &model.parts {
1140            for v in &part.mesh.vertices {
1141                total += 1;
1142                if v.normal().y > 0.2 {
1143                    up += 1;
1144                }
1145            }
1146        }
1147        let up_ratio = up as f32 / total.max(1) as f32;
1148        eprintln!("normalne skierowane w górę: {up_ratio:.3} ({up}/{total})");
1149        assert!(
1150            up_ratio > 0.15,
1151            "tylko {up_ratio:.3} normalnych w górę — normalne nie dostały \
1152             transformacji osi i model będzie czarny"
1153        );
1154
1155        // po skalowaniu motocykl ma sensowne wymiary w metrach
1156        let size = model.bounds_size();
1157        assert!(
1158            size.y > 0.5 && size.y < 3.0,
1159            "wysokość {size:?} to nie motocykl"
1160        );
1161        assert!(
1162            size.x > 1.0 && size.x < 4.0,
1163            "długość {size:?} to nie motocykl"
1164        );
1165    }
