12/// Wierzchołek siatki: pozycja, normalna, UV, kolor.
13///
14/// ## Dlaczego UV jest tutaj, a nie osobny bufor
15///
16/// Tekstury wczytujemy z plików modeli (`.obj` z UV z Blendera), więc
17/// UV musi docierać do shadera dla KAŻDEJ siatki. Trzymanie go w tym
18/// samym `Vertex` oznacza jeden bufor wierzchołków i brak dodatkowych
19/// powiązań; cena to 8 B na wierzchołek, co przy 30 tys. wierzchołków
20/// motocykla daje ok. 240 kB — mniej niż jedna tekstura 1k.
21#[repr(C)]
22#[derive(Debug, Clone, Copy, Pod, Zeroable)]
23pub struct Vertex {
24    pub position: [f32; 3],
25    pub normal: [f32; 3],
26    /// Współrzędne tekstury 0..1.
27    pub uv: [f32; 2],
28    pub color: [f32; 3],
29}
30
31impl Vertex {
32    /// Wierzchołek o pozycji i kolorze, bez tekstury.
33    ///
34    /// Normalną trzeba ustawić świadomie — „normalna = pozycja" daje
35    /// złe oświetlenie i cicho psuje wygląd bryły. UV dostaje (0,0),
36    /// czyli brak tekstury: shader i tak użyje koloru wierzchołka.
37    pub fn new(position: Vec3, normal: Vec3, color: [f32; 3]) -> Self {
38        Self {
39            position: position.to_array(),
40            normal: normal.to_array(),
41            uv: [0.0, 0.0],
42            color,
43        }
44    }
45
46    /// Wierzchołek z teksturą — używane przez importer modeli.
47    pub fn new_uv(position: Vec3, normal: Vec3, uv: [f32; 2], color: [f32; 3]) -> Self {
48        Self {
49            position: position.to_array(),
50            normal: normal.to_array(),
51            uv,
52            color,
53        }
54    }
55
56    pub fn pos(&self) -> Vec3 {
57        Vec3::from_array(self.position)
58    }
59
60    pub fn normal(&self) -> Vec3 {
61        Vec3::from_array(self.normal)
62    }
63
64    /// Czy wierzchołek ma niezerowe UV.
65    ///
66    /// Używane przy decyzji, czy siatka w ogóle potrzebuje tekstury —
67    /// geometria budowana w kodzie (tanki, droga) nie ma.
68    pub fn has_uv(&self) -> bool {
69        self.uv != [0.0, 0.0]
70    }
71}

73/// Siatka trzymana po stronie CPU (przed wgraniem na kartę).
74#[derive(Debug, Clone, Default)]
75pub struct Mesh {
76    pub vertices: Vec<Vertex>,
77    /// Indeksy trójkątów. Pusta lista = tryb `draw` zamiast `draw_indexed`.
78    pub indices: Vec<u32>,
79}
80
81impl Mesh {
82    pub fn new() -> Self {
83        Self::default()
84    }
85
86    /// Dodaje wierzchołek i zwraca jego indeks.
87    pub fn push(&mut self, v: Vertex) -> u32 {
88        self.vertices.push(v);
89        (self.vertices.len() - 1) as u32
90    }
91
92    /// Dodaje trójkąt trzema indeksami.
93    pub fn triangle(&mut self, a: u32, b: u32, c: u32) {
94        self.indices.extend_from_slice(&[a, b, c]);
95    }
96
97    /// Dodaje wierzchołek i trójkąt z nim w jednym wywołaniu.
98    pub fn tri(&mut self, a: Vertex, b: Vertex, c: Vertex) {
99        let i = self.vertices.len() as u32;
100        self.vertices.extend_from_slice(&[a, b, c]);
101        self.indices.extend_from_slice(&[i, i + 1, i + 2]);
102    }
103
104    /// Czy bryła ma komplet trójkątów.
105    pub fn is_closed(&self) -> bool {
106        self.indices.len() % 3 == 0 && !self.indices.is_empty()
107    }
108
109    /// Środek bryły średnią wierzchołków.
110    ///
111    /// Używamy średniej, nie AABB — dla symetrycznych brył (czołg, blok)
112    /// to się pokrywa, a dla skrzydeł daje punkt wewnątrz kadłuba.
113    pub fn center(&self) -> Vec3 {
114        if self.vertices.is_empty() {
115            return Vec3::ZERO;
116        }
117        let sum: Vec3 = self.vertices.iter().map(|v| v.pos()).sum();
118        sum / self.vertices.len() as f32
119    }
120
121    /// Wgrywa siatkę na kartę.
122    pub fn upload(&self, device: &wgpu::Device, label: &str) -> GpuMesh {
123        GpuMesh::new(device, label, self)
124    }
125}

127/// Siatka trzymana na GPU.
128pub struct GpuMesh {
129    vertex_buffer: wgpu::Buffer,
130    index_buffer: Option<wgpu::Buffer>,
131    /// `0` = rysuj `draw`, `>0` = rysuj `draw_indexed`.
132    index_count: u32,
133    vertex_count: u32,
134}
135
136impl GpuMesh {
137    pub fn new(device: &wgpu::Device, label: &str, mesh: &Mesh) -> Self {
138        let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
139            label: Some(&format!("{label} (wierzchołki)")),
140            contents: bytemuck::cast_slice(&mesh.vertices),
141            usage: wgpu::BufferUsages::VERTEX,
142        });
143
144        // Pusty bufor indeksów jest błędem walidacji wgpu, więc trzymamy
145        // `Option` i rysujemy albo indeksowo, albo sekwencyjnie.
146        let (index_buffer, index_count) = if mesh.indices.is_empty() {
147            (None, 0)
148        } else {
149            (
150                Some(
151                    device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
152                        label: Some(&format!("{label} (indeksy)")),
153                        contents: bytemuck::cast_slice(&mesh.indices),
154                        usage: wgpu::BufferUsages::INDEX,
155                    }),
156                ),
157                mesh.indices.len() as u32,
158            )
159        };
160
161        Self {
162            vertex_buffer,
163            index_buffer,
164            index_count,
165            vertex_count: mesh.vertices.len() as u32,
166        }
167    }
168
169    pub fn vertex_buffer(&self) -> &wgpu::Buffer {
170        &self.vertex_buffer
171    }
172
173    /// Ustawia slot wierzchołków i rysuje.
174    ///
175    /// `base_instance` musi być ≥ 1, bo WGSL liczy `instance_index` od 1
176    /// przy `first_instance`. Dzięki temu każdy obiekt sięga po własną
177    /// macierz modelu we WSPÓLNYM buforze, zamiast mieć osobny bufor.
178    /// `&'r self` (a nie `&self`) jest tu konieczne: `set_index_buffer`
179    /// wymaga `BufferSlice<'r>`, więc slice musi żyć tak długo jak pass.
180    /// Stąd też pożyczka buforów jest na czas rysowania, a nie na czas
181    /// całej klatki.
182    pub fn draw<'r>(&'r self, pass: &mut wgpu::RenderPass<'r>, slot: u32, base_instance: u32) {
183        pass.set_vertex_buffer(slot, self.vertex_buffer.slice(..));
184        match &self.index_buffer {
185            Some(ib) => {
186                pass.set_index_buffer(ib.slice(..), wgpu::IndexFormat::Uint32);
187                pass.draw_indexed(0..self.index_count, 0, base_instance..base_instance + 1);
188            }
189            // brak indeksów: każda trójka wierzchołków to trójkąt
190            None => pass.draw(0..self.vertex_count, base_instance..base_instance + 1),
191        }
192    }
193}

195/// Uniform przekazywany do shadera 3D w jednej klatce.
196///
197/// UKŁAD MUSI BYĆ ZGODNY Z `Scene` W `s3d.wgsl`. Wszystkie pola
198/// trzymamy w `vec4` także po stronie Rusta, bo WGSL wyrównuje `vec3`
199/// do 16 B — struktura z samymi `vec3` urosłaby i walidacja wgpu
200/// odrzuciłaby bufor.
201///
202/// ## Rozmiar: 400 B
203///
204/// Liczymy pole po polu, bo to jednocześnie kontrakt z shaderem
205/// i asercja kompilacji:
206///
207/// | pole | rozmiar |
208/// |------|---------|
209/// | `view_proj`, `inv_view_proj`, `view`, `light_view_proj` | 4·64 = 256 B |
210/// | `eye`, `light_dir`, `light_color`, `ambient_time` | 4·16 = 64 B |
211/// | `shadow_params`, `shadow_map_info` | 2·16 = 32 B |
212/// | `screen`, `stylize`, `atmos` | 3·16 = 48 B |
213/// | **suma** | **400 B** |
214///
215/// Wszystko jest 16-bajtowo wyrównane, więc `size % 16 == 0` i wgpu
216/// nie zgłosi błędu wyrównania.
217#[repr(C)]
218#[derive(Debug, Clone, Copy, Pod, Zeroable)]
219pub struct SceneUniform {
220    /// Macierz świata -> NDC.
221    pub view_proj: [[f32; 4]; 4],
222    /// Odwrotność `view_proj`: NDC -> świata.
223    ///
224    /// Potrzebna w shaderze nieba do odtworzenia kierunku promienia
225    /// z piksela oraz w post-processingu do odtworzenia pozycji świata
226    /// z samej głębokości (SSR, kontury, SSS, DoF).
227    pub inv_view_proj: [[f32; 4]; 4],
228    /// Macierz świata -> przestrzeń oka.
229    ///
230    /// Normalne G-Bufera trzymamy w przestrzeni oka, bo tam promień
231    /// odbicia SSR jest prosty: odbijamy wektor do oka wokół `N`
232    /// i lecimy wzdłuż niego, bez dodatkowych transformacji.
233    pub view: [[f32; 4]; 4],
234    /// Pozycja oka (do specular), `w` = tan(FOV/2) w pionie.
235    ///
236    /// Tangens jest tu, bo passy pełnoekranowe muszą odtworzyć promień
237    /// przez piksel, a liczenie `tan` w każdym z nich zdradzałoby
238    /// rozszerzenie pola widzenia o niejedno przekształcenie.
239    pub eye: [f32; 4],
240    /// Kierunek światła kierunkowego (normalizowany w shaderze).
241    pub light_dir: [f32; 4],
242    /// Kolor światła.
243    pub light_color: [f32; 4],
244    /// `rgb` = ambient, `a` = czas świata.
245    pub ambient_time: [f32; 4],
246
247    /// --- mapowanie cieni ---
248    ///
249    /// Świat -> NDC tekstury cieni. Osobna macierz, bo kamera cienia
250    /// stoi w zupełnie innym miejscu niż oko gracza: patrzy z pozycji
251    /// słońca, ortograficznie, na całą scenę.
252    pub light_view_proj: [[f32; 4]; 4],
253    /// `x` = bias w głębokości, `y` = odsunięcie wzdłuż normalnej,
254    /// `z` = siła cienia, `w` = promień PCF (w texelach).
255    pub shadow_params: [f32; 4],
256    /// `x` = rozmiar texela w UV, `y` = włącznik (0/1),
257    /// `z`,`w` = wyrównanie na 16 B.
258    pub shadow_map_info: [f32; 4],
259
260    /// --- ekran i atmosfera (wspólne dla shadera sceny i postfx) ---
261    /// `x,y` = rozmiar kadru w px, `z` = bliska płaszczyzna, `w` = daleka.
262    ///
263    /// Near/far są potrzebne do liniaryzacji głębokości: z samego
264    /// `Depth32Float` nie da się odtworzyć odległości w jednostkach
265    /// świata, a bez niej mgła, DoF i kontury miałyby nieliniowy skok
266    /// przy oddaleniu.
267    pub screen: [f32; 4],
268    /// `x` = siła stylizacji (0 = czyste PBR, 1 = mocna rampa anime),
269    /// `y` = miękkość progu terminatora, `z` = siła SSS,
270    /// `w` = wzmocnienie odbicia otoczenia.
271    pub stylize: [f32; 4],
272    /// `x` = gęstość mgły, `y..w` = kolor mgły RGB.
273    ///
274    /// Gęstość jest jedna dla całej mgły, a kolor bierze z gradientu
275    /// w shaderze sceny — mieszanie dwóch barw w samym oświetleniu
276    /// kosztowałoby tyle, ile cały gradient liczony w post-processingu.
277    pub atmos: [f32; 4],
278    /// `x` = wysokość zanikania mgły (m), `y..w` = rezerwa.
279    ///
280    /// ## Po co osobne `vec4`, a nie dołożenie pola do `atmos`
281    ///
282    /// WGSL wymaga wyrównania każdego pola `vec4` na 16 B. Dołożenie
283    /// piątej liczby do `atmos` wymusiłoby albo jej wyrównanie do
284    /// granicy 16 B (marnując 12 B), albo rozbicie `atmos` na dwa
285    /// `vec4` — a to oznaczałoby zmianę deklaracji w `s3d.wgsl`
286    /// i przesunięcie wszystkich dalszych bindingów. Osobne pole
287    /// `vec4` kosztuje 16 B i nie rusza niczego innego.
288    ///
289    /// Gdy `x = 0`, shader traktuje mgłę jako jednolitą (stare
290    /// zachowanie), więc pole jest wstecz kompatybilne.
291    pub fog: [f32; 4],
292}

294/// Rozmiar uniformu jest częścią kontraktu z shaderem: zmiana jednego bez
295/// drugiego kończy się błędem walidacji wgpu dopiero podczas renderowania.
296///
297/// 416 B = 400 B (wcześniejszy rozmiar) + 16 B nowego pola `fog`.
298const _: () = assert!(std::mem::size_of::<SceneUniform>() == 416);

300/// Transformacja i kolor jednego obiektu w scenie.
301///
302/// Zamiast osobnego bufora per obiekt (kilkanaście `write_buffer` na
303/// klatkę przy kilkunastu czołgach) trzymamy JEDNĄ tablicę i sięgamy po
304/// niej indeksem instancji w shaderze.
305#[repr(C)]
306#[derive(Debug, Clone, Copy, Pod, Zeroable)]
307pub struct InstanceModel {
308    /// Macierz modelu (kolumnowa).
309    pub model: [[f32; 4]; 4],
310    /// Kolor podstawowy — mnożymy nim kolor wierzchołka, żeby odróżnić
311    /// drużyny bez osobnych siatek i bez tekstur.
312    pub tint: [f32; 4],
313}
314
315/// Macierz modelu: skalowanie -> obrót -> przesunięcie.
316///
317/// Kolejność `T * R * S` wynika z kolumnowego zapisu macierzy; odwrotna
318/// kolejność przesunęłaby obiekt o złą skalę.
319///
320/// Obrót budujemy jako iloczyn kwaternionów w kolejności Y-X-Z:
321/// najpierw yaw (obrót wokół pionu), potem pitch (pochylenie lufy),
322/// na końcu roll (przechył na boki). Inna kolejność daje „zaplątany"
323/// obrót, którego nie da się opisać trzema kątami.
324pub fn model_matrix(position: Vec3, yaw: f32, pitch: f32, roll: f32, scale: Vec3) -> Mat4 {
325    let rotation =
326        Quat::from_rotation_y(yaw) * Quat::from_rotation_x(pitch) * Quat::from_rotation_z(roll);
327    Mat4::from_translation(position)
328        * Mat4::from_mat3(Mat3::from_quat(rotation))
329        * Mat4::from_scale(scale)
330}

332/// Osiem narożników prostopadłościanu (kolejność: 0-3 spód, 4-7 góra).
333pub fn box_corners(center: Vec3, half: Vec3) -> [Vec3; 8] {
334    [
335        center + Vec3::new(-half.x, -half.y, -half.z),
336        center + Vec3::new(half.x, -half.y, -half.z),
337        center + Vec3::new(half.x, -half.y, half.z),
338        center + Vec3::new(-half.x, -half.y, half.z),
339        center + Vec3::new(-half.x, half.y, -half.z),
340        center + Vec3::new(half.x, half.y, -half.z),
341        center + Vec3::new(half.x, half.y, half.z),
342        center + Vec3::new(-half.x, half.y, half.z),
343    ]
344}

430/// Punkt na trójkącie `a b c` dla współrzędnych barycentrycznych `t`.
431pub fn barycentric(a: Vec3, b: Vec3, c: Vec3, t: Vec2) -> Vec3 {
432    a * (1.0 - t.x - t.y) + b * t.x + c * t.y
433}
