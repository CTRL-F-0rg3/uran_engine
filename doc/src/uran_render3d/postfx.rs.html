1//! Post-processing: AA, bloom, tonemapping, aberracja, winieta, ziarno.
2//!
3//! Celowo ODSEPAROWANY moduł zamiast rozbudowy `Renderer3d`. Trzyma
4//! własne tekstury, potoki i bind grupy, a `Renderer3d` rozmawia z nim
5//! tylko przez trzy metody: `hdr_view`, `composite` i `invalidate`.

7//! Dzięki temu reszta renderera 3D pozostaje nietknięta — warunek,

12/// Parametry post-processingu (UKŁAD ZGODNY Z `Params` W `post.wgsl`).
13///
14/// 11 * `vec4` = 176 B. Wszystko `vec4`, bo WGSL wyrównuje `vec3` do 16 B.
15#[repr(C)]
16#[derive(Debug, Clone, Copy, bytemuck::Pod, bytemuck::Zeroable)]
17pub struct PostParams {
18    /// `x` = ekspozycja, `y` = siła bloom, `z` = winieta, `w` = aberracja.
19    pub grade: [f32; 4],
20    /// `x` = próg bloom, `y` = ziarno, `z` = saturacja, `w` = kontrast.
21    pub fx: [f32; 4],
22    /// `x,y` = rozmiar kadru, `z` = czas, `w` = siła AA.
23    pub screen: [f32; 4],
24    /// `x,y` = słońce w UV, `z` = widoczność (0 = poza kadrem),
25    /// `w` = rezerwa.
26    pub sun: [f32; 4],
27    /// `x` = siła konturu, `y` = grubość px, `z` = próg, `w` = zanikanie.
28    pub outline: [f32; 4],
29    /// `x` = siła SSR, `y` = maks. dystans, `z` = grubość,
30    /// `w` = maks. chropowatość.
31    pub ssr: [f32; 4],
32    /// `x` = siła SSS, `y` = promień px, `z` = siła DoF, `w` = fokus.
33    pub scatter: [f32; 4],
34    /// `x` = apertura, `y` = maks. blur px, `z` = anamorficzna siła,
35    /// `w` = siła flary.
36    pub lens: [f32; 4],
37    /// `x` = siła god rays, `y` = gęstość,
38    /// `z` = odrzucanie próbek AO (patrz `ssao_reject_fadeoff`),
39    /// `w` = rezerwa.
40    pub rays: [f32; 4],
41    /// `x,y` = near, far do liniaryzacji głębokości.
42    pub depth: [f32; 4],
43    /// `x` = siła podziału tonów, `y,z,w` = rezerwa na przyszłe
44    /// krzywe (lift/gamma/gain).
45    pub grade2: [f32; 4],
46    /// `x` = siła SSAO, `y` = promień w jednostkach świata,
47    /// `z` = bias (przesunięcie próbki wzdłuż normalnej),
48    /// `w` = intensywność (potęga zaciennienia).
49    pub ssao: [f32; 4],
50    /// Dane projekcji potrzebne do odtworzenia pozycji w przestrzeni
51    /// oka: `x` = tan(FOV/2) w pionie, `y` = aspect, `z,w` = rezerwa.
52    ///
53    /// Bez tego SSAO nie potrafi zbudować promienia przez piksel, więc
54    /// nie policzyłby odległości w METRACH — dostałby wartości w
55    /// pikselach, czyli takie zacienienie zależałoby od rozdzielczości
56    /// okna i rozjeżdżałoby się przy zmianie rozmiaru.
57    pub proj: [f32; 4],
58    /// Kalibracja obrazu: `x` = ostrość, `y` = lift cieni,
59    /// `z` = korekta chropowatości, `w` = rezerwa.
60    ///
61    /// Osobne `vec4`, a nie doczepianie do istniejących pól: `proj.z/w`
62    /// oraz `grade2.y/w` są już zajęte (SSAO i podział tonów), a
63    /// mieszanie niepowiązanych znaczeń w jednym wektorze sprawia, że
64    /// zmiana jednego efektu cicho rusza drugi.
65    pub clarity: [f32; 4],
66}

68impl PostParams {
69    /// Buduje uniform głównego passu z ustawień i danych kadru.
70    ///
71    /// ## Po co to wydzielone
72    ///
73    /// Funkcja jest czysta — nie dotyka GPU. Wcześniej budowa uniformu
74    /// siedziała w `upload_params`, więc jedyny sposób sprawdzenia, czy
75    /// `ssao_reject_fadeoff` wylądował w `rays.z`, był uruchomienie
76    /// gry i zobaczenie AO. Teraz test robi to bez okna i bez karty.
77    #[allow(clippy::too_many_arguments)]
78    pub fn from_settings(
79        s: &PostSettings,
80        w: f32,
81        h: f32,
82        time: f32,
83        sun: [f32; 4],
84        near: f32,
85        far: f32,
86        focus: f32,
87        tan_half_fov: f32,
88        aspect: f32,
89    ) -> Self {
90        Self {
91            grade: [s.exposure, s.bloom, s.vignette, s.chromatic],
92            fx: [s.bloom_threshold, s.grain, s.saturation, s.contrast],
93            screen: [w, h, time, s.antialias],
94            sun,
95            outline: [
96                s.outline_strength,
97                s.outline_width,
98                s.outline_threshold,
99                s.outline_fade,
100            ],
101            ssr: [
102                s.ssr_strength,
103                // `max_dist` nie jest używany przez shader (promień
104                // kończy się na krawędzi kadru), ale zostawiamy pole,
105                // żeby układ uniformu nie rozjechał się z WGSL.
106                80.0,
107                s.ssr_thickness,
108                s.ssr_roughness,
109            ],
110            scatter: [s.sss_strength, s.sss_radius, 0.0, focus],
111            lens: [s.dof_aperture, s.dof_max_blur, s.anamorphic, s.lens_flare],
112            // `rays.z` niesie `ssao_reject_fadeoff`. Parametr z jednej
113            // grupy (`rays`) trafia do drugiej (`ssao`) — celowo: `rays`
114            // ma wolne `z`, a dopisanie nowego `vec4` do uniformu
115            // przesunęłoby wszystkie późniejsze pola i wymusiłoby zmianę
116            // w WGSL. Jeden przeskok między grupami jest tańszy niż
117            // przesunięcie 9 bindingów.
118            rays: [s.god_rays, s.god_ray_density, s.ssao_reject_fadeoff, 0.0],
119            depth: [near, far, 0.0, 0.0],
120            grade2: [s.split_tone, 0.0, s.split_tone, 0.0],
121            ssao: [
122                s.ssao_strength,
123                s.ssao_radius,
124                s.ssao_bias,
125                s.ssao_intensity,
126            ],
127            // Projekcja: SSAO odtwarza z tego pozycję w metrach.
128            proj: [tan_half_fov, aspect, 0.0, 0.0],
129            clarity: [s.clarity, s.shadow_lift, s.roughness_bias, 0.0],
130        }
131    }
132}

143/// Ustawienia wyglądu, które gra może zmieniać w locie.
144#[derive(Debug, Clone, Copy)]
145pub struct PostSettings {
146    pub exposure: f32,
147    pub bloom: f32,
148    pub bloom_threshold: f32,
149    pub vignette: f32,
150    pub chromatic: f32,
151    pub grain: f32,
152    pub saturation: f32,
153    pub contrast: f32,
154    /// Siła antyaliasingu krawędzi 0..1.
155    pub antialias: f32,
156
157    // --- nowe grupy efektów ---------------------------------------------
158    /// Siła konturów ekranowych 0..1 (`0` = wyłączone).
159    ///
160    /// Uwaga: to NIE jest kontur odwróconej powłoki. Zobacz `fs_composite`
161    /// w `post.wgsl` — kontur liczymy z różnic głębokości i normalnych
162    /// i przygaszamy go w jasnych miejscach.
163    pub outline_strength: f32,
164    /// Grubość konturu w pikselach (przed pomnożeniem przez rozmiar kadru).
165    pub outline_width: f32,
166    /// Próg wykrywania krawędzi — niższy = ciańszy kontur.
167    pub outline_threshold: f32,
168    /// Jak mocno kontur znika w jasnych miejscach 0..1.
169    pub outline_fade: f32,
170
171    /// Siła odbić ekranowych (SSR) 0..1.
172    pub ssr_strength: f32,
173    /// Maksymalna chropowatość, przy której odbicie jeszcze liczymy.
174    /// Powyżej powierzchnia jest zbyt matowa, żeby coś odbić.
175    pub ssr_roughness: f32,
176    /// Grubość tolerancji głębokości dla trafienia promienia.
177    pub ssr_thickness: f32,
178
179    /// Siła podpowierzchniowego rozpraszania w post-processingu 0..1.
180    pub sss_strength: f32,
181    /// Promień SSS w pikselach.
182    pub sss_radius: f32,
183
184    /// Przysłona (mocne rozmycie tła). `0` = cały kadr ostry.
185    pub dof_aperture: f32,
186    /// Ogniskowa odległość w jednostkach świata.
187    ///
188    /// Wartość `0.0` oznacza „auto": renderer ustawia ją na odległość
189    /// od środka kadru, co daje symulację aparatu ustawionego na
190    /// to, na co gracz patrzy.
191    pub dof_focus: f32,
192    /// Maksymalny promień rozmycia w pikselach.
193    pub dof_max_blur: f32,
194
195    /// Siła anamorficznej (poziomej) poświaty 0..1.
196    pub anamorphic: f32,
197    /// Siła flary obiektywu 0..1.
198    pub lens_flare: f32,
199    /// Siła promieni słonecznych (god rays) 0..1.
200    pub god_rays: f32,
201    /// Gęstość smug god rays: dzielnik kroku marszu (`krok / gęstość`).
202    ///
203    /// UWAGA: to NIE jest zasięg smugi w kadrze, tylko jak gęsto są
204    /// próbki wzdłuż promienia. Większa wartość = krótszy krok = krótsze,
205    /// częstsze smugi. Shader liczy `N / gęstość` próbek na piksel.
206    pub god_ray_density: f32,
207    /// Siła podziału tonów (chłodne cienie / ciepłe światła) 0..1.
208    pub split_tone: f32,
209
210    // --- SSAO -----------------------------------------------------------
211    /// Siła cieniowania ekranowego 0..1 (`0` = wyłączone).
212    ///
213    /// SSAO przyciemnia miejsca, w których otoczenie zasłania samo
214    /// siebie: narożniki ścian, styki obiektów z podłogą, wnęki. To
215    /// najtańszy sposób na „doklejenie" obiektów do świata — bez niego
216    /// bryły wiszą w powietrzu i brzmią jak papier.
217    pub ssao_strength: f32,
218    /// Promień próbkowania w jednostkach świata.
219    ///
220    /// Za mały = czarne plamy przy kontaktach, za duży = szare smugi
221    /// rozlewające się po dużych powierzchniach. 0.6 m odpowiada
222    /// człowiekowi stojącemu w drzwiach.
223    pub ssao_radius: f32,
224    /// Przesunięcie próbki wzdłuż normalnej, w jednostkach świata.
225    ///
226    /// Chroni przed samooczytywaniem (*surface acne*), które bez tego
227    /// daje czarne kropki na płaskich powierzchniach.
228    pub ssao_bias: f32,
229    /// Potęga zaciennienia — jak głęboko cienie w klach.
230    ///
231    /// `1.0` = liniowe cieniowanie, `>1` = bardziej dramatyczne narożniki.
232    pub ssao_intensity: f32,
233    /// Stromość odrzucania próbek w teście par (MSAO).
234    ///
235    /// ## Skąd to się wzięło
236    ///
237    /// Test „kulowy" (`dot(V, N) > 0 i |V| < radius`) traktuje próbkę
238    /// leżącą na **tej samej powierzchni** co zasłaniającą, bo formalnie
239    /// mieści się w kuli. W praktyce daje to ciemne plamy tam, gdzie
240    /// powierzchnia zakrzywia się w stronę kamery.
241    ///
242    /// MSAO (WickedEngine, `msaoCS.hlsl`, pochodna MiniEngine) rozwiązuje
243    /// to testem **par**: bierze próbki po obu stronach piksela
244    /// (`+offset` i `-offset`) i porównuje je ze sobą. Obie mierzą to
245    /// samo przenikanie w głąb kuli, więc ich różnica kasuje artefakt
246    /// powierzchni, a zostawia prawdziwe przesłony.
247    ///
248    /// To `xRejectFadeoff` z MSAO. Wyższa = ostrzejsze odrzucanie, ale
249    /// powyżej ~250 próbki na płaskiej powierzchni zaczynają
250    /// „przebijać" i AO znika. Zakres użyteczny: 50..200.
251    pub ssao_reject_fadeoff: f32,
252
253    // --- kalibracja obrazu ----------------------------------------------
254    /// Ostrość obrazu (unsharp mask) — 0 = wyłączona.
255    ///
256    /// ## Dlaczego to potrzebne
257    ///
258    /// Obraz po tonemapowaniu i rozmyciu DoF ma miękkie krawędzie, a
259    /// przy niskim kontraście cieni wygląda jak patrząc przez mętną
260    /// soczewkę. Ostrość wyciąga z powrotem kontur, które tonemap
261    /// spłaszczył.
262    ///
263    /// Implementacja w `post.wgsl` to klasyczny unsharp: różnica
264    /// sąsiadów dodana z powrotem. Wartość 0.5 to ledwie zauważalne
265    /// wyostrzenie, 1.0 wyraźne; powyżej 1.5 pojawia się halo
266    /// (biała obwódka) na kontrastowych krawędziach.
267    pub clarity: f32,
268    /// Podniesienie cieni (lift) 0..1.
269    ///
270    /// ## Po co, skoro ambient już obniżyliśmy
271    ///
272    /// Ambient daje cieniom **kolor** otoczenia, ale nic nie mówi o
273    /// tym, gdzie ma być **czarny punkt**. Lift przesuwa dolny koniec
274    /// krzywej: 0 zostawia absolutną czerń, więc kontakt obiektu
275    /// z podłożem zanika. 0.02–0.04 przywraca czerń w cieniach bez
276    /// szarzenia całego obrazu — to różnica między „cieniem" a
277    /// „dziurą w obrazie".
278    pub shadow_lift: f32,
279    /// Korekta chropowatości materiału, zakres -1..1.
280    ///
281    /// ## Problem, który rozwiązuje
282    ///
283    /// Domyślna chropowatość 0.6 dla każdej powierzchni oznacza, że
284    /// wszystko odbija świat tak samo słabo — obraz traci różnicę
285    /// między matowym betonem a jedwabistym metalu. Ujemna wartość
286    /// wygładza (mniej refleksów, więcej koloru materiału), dodatnia
287    /// pogłębia refleksy.
288    ///
289    /// Nie zastępuje mapy ORM — to korekta globalna, gdy mapa
290    /// chropowatości nie jest dostępna.
291    pub roughness_bias: f32,
292}

294impl Default for PostSettings {
295    fn default() -> Self {
296        Self {
297            // Ekspozycja 1.0: neutralny punkt odniesienia AgX. Wyższa
298            // wartość to już decyzja artystyczna i należy do demo,
299            // nie do domyślnego ustawienia silnika.
300            exposure: 1.0,
301            // Bloom 0.32 → 0.18. Domyślne 0.32 przy progu 1.15 dawało
302            // poświatę na KAŻDEJ jasnej powierzchni powyżej progu, nie
303            // tylko na źródłach — tło nieba i słońce obracały się w
304            // białą plamę. 0.18 zostawia poświatę tam, gdzie jest
305            // źródło, a resztę powierzchni nietkniętą.
306            bloom: 0.18,
307            // Próg wysoko ponad 1.0: poświatę dostaje tylko rdzeń
308            // reaktora, wizjer i tarcza słońca, a nie każda jasna ściana.
309            bloom_threshold: 1.15,
310            // 0.30 → 0.18. Winieta ma delikatnie zamykać kadr; przy 0.30
311            // narożnik był ciemniejszy o ~30% i wyglądał jak defekt
312            // kadru, a nie jak optyka.
313            vignette: 0.18,
314            // Bardzo słaba aberracja: łamie idealną gładkość krawędzi,
315            // ale jest niewidoczna gołym okiem w ruchu.
316            chromatic: 0.0012,
317            // Ziarno 0.014 → 0.007. Jeszcze wystarcza na maskowanie
318            // bandingu w gradiencie nieba, a nie jest widoczną teksturą.
319            grain: 0.007,
320            // Nasycenie 1.04 → 1.12 i kontrast 1.03 → 1.12.
321            //
322            // AgX celowo odbarwia nasycone kolory — to jego zadaniem
323            // (żeby światła nie były kolorowymi plamami). Efekt
324            // uboczny: scena wygląda wyblakła. Podbicie tutaj
325            // przywraca intensywność materiałów tam, gdzie AgX
326            // ją zjadł, ale **po** tonemapie — więc biele i światła
327            // pozostają nietknięte, a kolory odzyskują nasycenie.
328            saturation: 1.10,
329            // 1.12 -> 1.10. Rozciaga zakres tonalny wokół 0.5.
330            // Razem z liftem cieni daje głębię: czarne w
331            // cieniach, biale w swiatlach, wiecej materialu w srodku.
332            contrast: 1.10,
333            // Antyaliasing to NIE jest efekt, tylko redukcja aliasingu.
334            // Obniżanie go dodawałoby schodki zamiast je usuwać, więc
335            // zostaje 0.65.
336            antialias: 0.65,
337
338            // --- kontury ---
339            // 0.55 → 0.02. Kontur praktycznie wyłączony — zostaje
340            // jako najcieńsza akcentacja sylwetki, bo przy 0.02 jego
341            // wkład w obraz jest mniejszy niż szum ziarna.
342            outline_strength: 0.02,
343            // 1.2 → 1.0 px. Przy 1.0 kontur bywa niewidoczny na
344            // krawędziach normalnych, przy 2.0 robi się gruby
345            // i „obłotowy".
346            outline_width: 1.0,
347            // 0.010: niżej kontur łapie szum z głębokości, wyżej
348            // gubi krawędzie obiektów stojących daleko.
349            outline_threshold: 0.010,
350            // 0.75 = kontur prawie znika na słońcu, jest w cieniu.
351            outline_fade: 0.75,
352
353            // --- SSR ---
354            // 0.65 → 0.11. Odbicie ledwie widoczne: odbijają się
355            // tylko naprawdyle gładkie powierzchnie (szyby, woda przy
356            // krawędzi), a metal i mokry beton wyglądają jak suche.
357            ssr_strength: 0.11,
358            // 0.55: powyżej tej chropowatości odbicie jest szumem.
359            ssr_roughness: 0.55,
360            // 1.5 m tolerancji — przy mniejszej odbicie „przepływa"
361            // przez cienkie listwy, przy większej łapie sąsiednie ściany.
362            ssr_thickness: 1.5,
363
364            // --- SSS ---
365            // 0.5 → 0.30: widoczna ciepła poświata na krawędziach skóry,
366            // bez efektu „podświetlonej zielonej główki".
367            sss_strength: 0.30,
368            sss_radius: 2.5,
369
370            // --- DoF ---
371            // Apertura 1.0 → 0.7: miękkie rozmycie tła w rozmowie
372            // lub ataku. Więcej niż 1.0 zamienia obraz w plamę
373            // i utrudnia odczytanie przeciwnika w tle.
374            dof_aperture: 0.7,
375            // 0.0 = autofocus (renderer policzy z kadru).
376            dof_focus: 0.0,
377            // 6.0 → 0.10 px: DoF praktycznie wyłączony. 0.10 px to
378            // mniej niż połowa texela, więc kadr jest praktycznie cały
379            // ostry. Zostawiamy małą wartość zamiast 0.0, bo przy zerze
380            // `1/max(d, 0.001)` w shaderze dawałoby nieskończony blur
381            // na pikselach leżących w płaszczyźnie ogniskowania.
382            dof_max_blur: 0.10,
383
384            // --- obiektyw ---
385            // 0.35 → 0.18: poświata anamorficzna zaznaczona tylko przy
386            // najjaśniejszych źródłach. Powyżej 0.6 każda jasna plama
387            // dostaje poziomą kreskę i obraz wygląda „chory".
388            anamorphic: 0.18,
389            // 0.25 → 0.15: ghosty mają być ledwie zauważalne.
390            lens_flare: 0.15,
391            // 0.4 → 0.25: smugi widoczne, gdy patrzymy w stronę słońca,
392            // bez ciągnięcia się przez środek kadru.
393            god_rays: 0.25,
394            // 1.0 = krótki, gęsty wzór próbek. Większa wartość daje
395            // DŁUŻSZE, rzadsze smugi (krok maleje), mniejsza — krótsze.
396            god_ray_density: 1.0,
397            // Split tone przesuwa cala palete w jeden odcien, wiec
398            // powyzej ~0.15 kazdy kolor w kadrze dostawal ten sam
399            // posmak - i to wlasnie czytalo sie jako "dziwne kolory",
400            // mimo ze paleta materialow byla poprawna.
401            split_tone: 0.10,
402
403            // --- SSAO ---
404            // 0.21 -> 0.10: zacienienie ledwie widoczne. Przy tej
405            // sile AO wciaz dokleja obiekty do podloza, ale nie
406            // przyciemnia calej bryly - zostaje cien styku, nie plama.
407            ssao_strength: 0.10,
408            // 0.35 -> 0.10 m: promien zblizony do 10 cm. Zaciennienie
409            // siega wtedy tylko najblizszego otoczenia piksela, a nie
410            // calego obrysu obiektu.
411            ssao_radius: 0.10,
412            // 0.02 m marginesu grubości kuli: tyle, żeby próbka
413            // nie czytała tej samej powierzchni, którą cieniuje.
414            ssao_bias: 0.02,
415            // 1.4 → 1.2: łagodniejsza krzywa potęgi — słabe zaciennienia
416            // znikają, mocne w szczelinach zostają. Powyżej 2.0 AO
417            // zaczyna wyglądać jak brud na obiektywie.
418            ssao_intensity: 1.2,
419            // 120 = odrzucanie w teście par (MSAO). Wartość z zakresu
420            // użytecznego: zbyt mała zostawia plamy na powierzchniach
421            // zakrzywionych ku kamerze, zbyt duża kasuje AO przy
422            // stykach obiektów. WickedEngine używa tu ~120.
423            ssao_reject_fadeoff: 120.0,
424
425            // --- kalibracja obrazu ---
426            // 0.55 -> 0.12. Wyraźne wyostrzenie, ale bez halo na krawedziach.
427            // Unsharp wzmacnia roznice sasiadow, wiec zbyt duza
428            // wartosc daje biala obwodke tam, gdzie gradient jest
429            // stromy - czyli dokladnie na sylwetkach. Przy 0.55
430            // kazda krawedz dostawala bialy fringes, co odczytywane
431            // bylo jako "dziwne kolory" na obrysach.
432            clarity: 0.12,
433            // 0.03 -> 0.01. Przywraca czern w cieniach. Bez tego AO
434            // i kontakt z podlozem gina w szarosci, bo tonemap AgX
435            // wyplaszcza najciemniejsze wartosci. 0.01 to najmniejsza
436            // wartosc, ktora jeszcze robi roznice - powyzej 0.03 caly
437            // obraz szarzeje.
438            shadow_lift: 0.01,
439            // 0.0: bez korekty. Chropowatosc 0.6 z materialu jest
440            // rozsadnym punktem wyjscia; ustawienie biasu to
441            // decyzja sceny, nie silnika.
442            roughness_bias: 0.0,
443        }
444    }

538        let blur_buffers = [
539            uniform(device, "Uran Blur Bright", POST_PARAMS_SIZE),
540            uniform(device, "Uran Blur H", POST_PARAMS_SIZE),
541            uniform(device, "Uran Blur V", POST_PARAMS_SIZE),
542        ];
543
544        // JEDEN layout obsługuje wszystkie cztery passy: bright/blur/
545        // composite mają ten sam zestaw bindingów, a nieużywane pomija
546        // WGSL. Dzięki temu mamy jeden bind group zamiast trzech layoutów.
547        let layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
548            label: Some("Uran Post BGL"),
549            entries: &[
550                uniform_entry(0, wgpu::ShaderStages::FRAGMENT),
551                texture_entry(1),
552                texture_entry(2),
553                wgpu::BindGroupLayoutEntry {
554                    binding: 3,
555                    visibility: wgpu::ShaderStages::FRAGMENT,
556                    ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
557                    count: None,
558                },
559                texture_entry(4),
560                // Głębokość jest formatem niefiltrowalnym: `Depth` + brak
561                // `filterable` daje `texture_depth_2d` w WGSL. Dla
562                // `textureSample` potrzebny byłby `sampler_comparison`,
563                // ale my czytamy ją przez `textureLoad` (patrz
564                // `view_depth` w `post.wgsl`), więc zwykły sampler
565                // wystarczy i unika odrzucenia layoutu.
566                wgpu::BindGroupLayoutEntry {
567                    binding: 5,
568                    visibility: wgpu::ShaderStages::FRAGMENT,
569                    ty: wgpu::BindingType::Texture {
570                        sample_type: wgpu::TextureSampleType::Depth,
571                        view_dimension: wgpu::TextureViewDimension::D2,
572                        multisampled: false,
573                    },
574                    count: None,
575                },
576                texture_entry(6),
577                texture_entry(7),
578                // Mapa SSAO (R8Unorm). Osobny binding, bo `ssr_tex`
579                // i `rays_tex` to tekstury RGBA16Float, a AO ma inny
580                // format — wgpu nie pozwala zadeklarować jednego slotu
581                // dla dwóch formatów.
582                wgpu::BindGroupLayoutEntry {
583                    binding: 8,
584                    visibility: wgpu::ShaderStages::FRAGMENT,
585                    ty: wgpu::BindingType::Texture {
586                        sample_type: wgpu::TextureSampleType::Float { filterable: true },
587                        view_dimension: wgpu::TextureViewDimension::D2,

590                    count: None,

609

611            pipeline(device, format, samples, &layout, &module, "fs_composite");
612        // cele pośrednie bloom są 1:1 (bez MSAA) — rozmycie i tak gubi
613        // szczegóły, a 4x więcej próbek kosztowałoby bez efektu
614        let bright_pipeline = pipeline(device, HDR_FORMAT, 1, &layout, &module, "fs_bright");
615        let blur_pipeline = pipeline(device, HDR_FORMAT, 1, &layout, &module, "fs_blur");
616        // Pasy ekranowe (SSR, god rays) liczą się w POŁOWIE rozmiaru
617        // kadru: to dobór `ensure_size`. DoF i kontury działają w
618        // kompozycji, w pełnej rozdzielczości.
619        let ssr_pipeline = pipeline(device, HDR_FORMAT, 1, &layout, &module, "fs_ssr");
620        let godray_pipeline = pipeline(device, HDR_FORMAT, 1, &layout, &module, "fs_godray");
621        // SSAO liczymy w POŁOWIE kadru (jak SSR i god rays) i czyta
622        // tylko G-Bufer + głębokość. Ten sam celowy format co reszta
623        // pośrednich, bo `ao_raw` musi dać się czytać jako `texture_2d`.
624        let ssao_pipeline = pipeline(device, AO_FORMAT, 1, &layout, &module, "fs_ssao");
625        let ao_blur_pipeline = pipeline(device, AO_FORMAT, 1, &layout, &module, "fs_ao_blur");
626
627        Self {
628            format,
629            samples,
630            hdr: None,
631            gbuffer: None,
632            depth_copy: None,
633            depth_view_cache: None,
634            bloom_a: None,
635            bloom_b: None,
636            ssr_out: None,
637            rays_out: None,
638            ao_out: None,
639            ao_raw: None,
640            sized: (0, 0),
641            params_buffer,
642            blur_buffers,
643            layout,
644            composite_pipeline,
645            bright_pipeline,
646            blur_pipeline,
647            ssr_pipeline,
648            godray_pipeline,
649            ssao_pipeline,
650            ao_blur_pipeline,
651            sampler,
652            composite_bg: None,
653            bloom_bgs: [None, None, None],
654            effect_bgs: [None, None],
655            ao_bgs: [None, None],
656            settings: PostSettings::default(),

705    pub fn copy_depth(&self, enc: &mut wgpu::CommandEncoder, src: &wgpu::Texture) {
706        let Some(dst) = self.depth_copy.as_ref() else {
707            return;
708        };
709        enc.copy_texture_to_texture(
710            wgpu::ImageCopyTexture {
711                texture: src,
712                mip_level: 0,
713                origin: wgpu::Origin3d::ZERO,
714                aspect: wgpu::TextureAspect::All,
715            },
716            wgpu::ImageCopyTexture {
717                texture: dst,
718                mip_level: 0,
719                origin: wgpu::Origin3d::ZERO,
720                aspect: wgpu::TextureAspect::All,
721            },
722            wgpu::Extent3d {
723                width: self.sized.0.max(1),
724                height: self.sized.1.max(1),
725                depth_or_array_layers: 1,
726            },
727        );
728    }

747impl PostFx {
748    /// Tworzy tekstury i bind grupy dla aktualnego rozmiaru okna.
749    pub fn ensure_size(&mut self, device: &wgpu::Device, w: u32, h: u32) {
750        if self.sized == (w, h) {
751            return;
752        }
753        self.sized = (w, h);
754
755        self.hdr = Some(color_view(
756            device,
757            w,
758            h,
759            "Uran Post HDR",
760            HDR_FORMAT,
761            self.samples,
762        ));
763        // G-Bufer: ten sam format co HDR, bo normalna w 8 bitach daje
764        // widoczne „schodki" na konturach (krawędź normalnej to różnica
765        // rzędu 0.02, a 8 bitów na kanał daje krok 0.004 — na granicy
766        // artefaktu).
767        self.gbuffer = Some(color_view(
768            device,
769            w,
770            h,
771            "Uran Post GBuffer",
772            HDR_FORMAT,
773            self.samples,
774        ));
775        // Kopia głębokości. Format MUSI być identyczny z buforem
776        // głębokości w `scene.rs` — `copy_texture_to_texture` wymaga
777        // zgodności formatów, a przy `Depth24Plus` vs `Depth32Float`
778        // wgpu odrzuci kopię.
779        self.depth_copy = Some(depth_texture(device, w, h, "Uran Post Depth"));
780        self.depth_view_cache = Some(
781            self.depth_copy
782                .as_ref()
783                .expect("właśnie utworzone")
784                .create_view(&wgpu::TextureViewDescriptor::default()),
785        );
786        let (bw, bh) = ((w / 2).max(1), (h / 2).max(1));
787        self.bloom_a = Some(color_view(device, bw, bh, "Uran Bloom A", HDR_FORMAT, 1));
788        self.bloom_b = Some(color_view(device, bw, bh, "Uran Bloom B", HDR_FORMAT, 1));
789        // Efekty ekranowe w POŁOWIE rozmiaru: oba liczą różnice
790        // sąsiednich pikseli (promień SSR, radialne smugi), więc wynik
791        // i tak jest rozmyty przez downsampling.
792        let ssr = Some(color_view(device, bw, bh, "Uran SSR Out", HDR_FORMAT, 1));
793        let rays = Some(color_view(device, bw, bh, "Uran Rays Out", HDR_FORMAT, 1));
794        self.ssr_out = ssr;
795        self.rays_out = rays;
796        // SSAO: dwa bufory w POŁOWIE kadru. `ao_raw` to wynik surowy
797        // (wysoki szum), `ao_out` — po rozmyciu, który czyta kompozycja.
798        // Osobne, bo pass nie może czytać swojego celu.
799        self.ao_raw = Some(color_view(device, bw, bh, "Uran AO Raw", AO_FORMAT, 1));
800        self.ao_out = Some(color_view(device, bw, bh, "Uran AO Out", AO_FORMAT, 1));
801
802        let hdr = self.hdr.as_ref().expect("właśnie utworzone");
803        let ba = self.bloom_a.as_ref().expect("właśnie utworzone");
804        let bb = self.bloom_b.as_ref().expect("właśnie utworzone");
805        let gbuf = self.gbuffer.as_ref().expect("właśnie utworzone");
806        let depth = self.depth_view_cache.as_ref().expect("właśnie utworzone");
807        let ssr_tex = self.ssr_out.as_ref().expect("właśnie utworzone");
808        let rays_tex = self.rays_out.as_ref().expect("właśnie utworzone");
809        let ao_raw = self.ao_raw.as_ref().expect("właśnie utworzone");
810        let ao_out = self.ao_out.as_ref().expect("właśnie utworzone");
811        // w bright/blur binding 2 to samo źródło — nieużywany przez WGSL.
812        // Pasy te nie czytają `ssr_tex` ani `rays_tex`, więc dostają je
813        // tylko po to, by wpis w layout zgadzał się z liczbą wpisów.
814        //
815        // Slot 8 to AO. Passy, które go nie czytają (bloom, SSR, god
816        // rays), dostają `ao_out` — jest neutralny i nigdy nie jest
817        // celem żadnego z tych passów.
818        self.bloom_bgs[0] = Some(self.bind(
819            device,
820            &self.blur_buffers[0],
821            hdr,
822            hdr,
823            gbuf,
824            depth,
825            ssr_tex,
826            rays_tex,
827            ao_out,
828        ));
829        self.bloom_bgs[1] = Some(self.bind(
830            device,
831            &self.blur_buffers[1],
832            ba,
833            ba,
834            gbuf,
835            depth,
836            ssr_tex,
837            rays_tex,
838            ao_out,
839        ));
840        self.bloom_bgs[2] = Some(self.bind(
841            device,
842            &self.blur_buffers[2],
843            bb,
844            bb,
845            gbuf,
846            depth,
847            ssr_tex,
848            rays_tex,
849            ao_out,
850        ));
851        // Kompozycja czyta naraz: HDR, bloom, odbicia, smugi i AO.
852        self.composite_bg = Some(self.bind(
853            device,
854            &self.params_buffer,
855            hdr,
856            ba,
857            gbuf,
858            depth,
859            ssr_tex,
860            rays_tex,
861            ao_out,
862        ));
863        // Każdy pass ekranowy renderuje DO swojego bufora, więc NIE
864        // wiążemy `ssr_out`/`rays_out` w jego własnej grupie: wgpu
865        // odrzuca zasób użyty jednocześnie jako attachment i źródło.
866        // Sloty, których dany shader nie czyta, dostają `hdr` —
867        // jest neutralny i nigdy nie jest celem żadnego z tych passów.
868        self.effect_bgs[0] = Some(self.bind(
869            device,
870            &self.params_buffer,
871            hdr,
872            hdr,
873            gbuf,
874            depth,
875            hdr,
876            hdr,
877            ao_out,
878        ));
879        self.effect_bgs[1] = Some(self.bind(
880            device,
881            &self.params_buffer,
882            hdr,
883            hdr,
884            gbuf,
885            depth,
886            hdr,
887            hdr,
888            ao_out,
889        ));
890        // SSAO czyta gbuffer + depth i pisze do `ao_raw`; więc NIE
891        // wiąże `ao_raw` w jego własnej grupie. `fs_ssao` nie czyta
892        // slotu 8 w ogóle, więc dostaje `ao_out` tylko dla zgodności
893        // liczby wpisów z layoutem.
894        self.ao_bgs[0] = Some(self.bind(
895            device,
896            &self.params_buffer,
897            hdr,
898            hdr,
899            gbuf,
900            depth,
901            hdr,
902            hdr,
903            ao_out,
904        ));
905        // Rozmycie AO: `fs_ao_blur` czyta slot 8, więc to MUSI być
906        // `ao_raw` — celem jest `ao_out`, a wgpu odrzuca zasób użyty
907        // w jednym passie jako źródło i jako color target naraz.
908        self.ao_bgs[1] = Some(self.bind(
909            device,
910            &self.params_buffer,
911            ao_raw,
912            ao_raw,
913            gbuf,
914            depth,
915            hdr,
916            hdr,
917            ao_raw,
918        ));
919    }
920
921    /// Bind grupa: uniform + cztery tekstury + sampler.
922    ///
923    /// Rozdzielenie `aux` (bloom) i `effects` jest konieczne tylko dla
924    /// kompozycji, która czyta jedno i drugie naraz. Pozostałe passy
925    /// dostają `effects = aux`, bo nie używają tego bindingu.
926    #[allow(clippy::too_many_arguments)]
927    fn bind(
928        &self,
929        device: &wgpu::Device,
930        params: &wgpu::Buffer,
931        src: &wgpu::TextureView,
932        aux: &wgpu::TextureView,
933        gbuf: &wgpu::TextureView,
934        depth: &wgpu::TextureView,
935        ssr: &wgpu::TextureView,
936        rays: &wgpu::TextureView,
937        ao: &wgpu::TextureView,
938    ) -> wgpu::BindGroup {
939        device.create_bind_group(&wgpu::BindGroupDescriptor {
940            label: Some("Uran Post BG"),
941            layout: &self.layout,
942            entries: &[
943                wgpu::BindGroupEntry {
944                    binding: 0,
945                    resource: params.as_entire_binding(),
946                },
947                wgpu::BindGroupEntry {
948                    binding: 1,
949                    resource: wgpu::BindingResource::TextureView(src),
950                },
951                wgpu::BindGroupEntry {
952                    binding: 2,
953                    resource: wgpu::BindingResource::TextureView(aux),
954                },
955                wgpu::BindGroupEntry {
956                    binding: 3,
957                    resource: wgpu::BindingResource::Sampler(&self.sampler),
958                },
959                wgpu::BindGroupEntry {
960                    binding: 4,
961                    resource: wgpu::BindingResource::TextureView(gbuf),
962                },
963                wgpu::BindGroupEntry {
964                    binding: 5,
965                    resource: wgpu::BindingResource::TextureView(depth),
966                },
967                // Dwa osobne cele efektów. Nie mogą się dzielić buforem,
968                // bo kompozycja czyta je równocześnie.
969                wgpu::BindGroupEntry {
970                    binding: 6,
971                    resource: wgpu::BindingResource::TextureView(ssr),
972                },
973                wgpu::BindGroupEntry {
974                    binding: 7,
975                    resource: wgpu::BindingResource::TextureView(rays),
976                },
977                wgpu::BindGroupEntry {
978                    binding: 8,
979                    resource: wgpu::BindingResource::TextureView(ao),
980                },
981            ],
982        })
983    }
984
985    /// Wpisuje uniformy na bieżącą klatkę.
986    ///
987    /// Jeden `write_buffer` na pass — niezależnie od liczby obiektów
988    /// w scenie.
989    ///
990    /// `sun_uv` to pozycja słońca w UV (albo `None`, gdy słońce jest poza
991    /// kadrem), a `focus` to ogniskowa odległość DoF. Oba wylicza
992    /// `Renderer3d`, bo wymagają wiedzy o kamerze, której tu nie mamy.
993    pub fn upload_params(
994        &self,
995        queue: &wgpu::Queue,
996        w: u32,
997        h: u32,
998        time: f32,
999        sun_uv: Option<(f32, f32)>,
1000        near: f32,
1001        far: f32,
1002        tan_half_fov: f32,
1003        aspect: f32,
1004    ) {
1005        let s = &self.settings;
1006        // Słońce: `z > 0` to przełącznik widoczności w shaderze. Poza
1007        // kadrem ustawiamy 0, żeby flara i promienie się wyłączyły —
1008        // inaczej pojawiałyby się z krawędzi obrazu.
1009        let sun = match sun_uv {
1010            Some((x, y)) => [x, y, 1.0, 0.0],
1011            None => [0.5, 0.5, 0.0, 0.0],
1012        };
1013        // Autofocus: `0.0` w ustawieniach znaczy „ustaw na to, co
1014        // gracz patrzy". Wartość 1.0 m jest bezpiecznym minimum —
1015        // przy mniejszym promień DoF rósłby do nieskończoności
1016        // przez `1/max(d, 0.001)` w shaderze.
1017        let focus = if s.dof_focus > 0.0 { s.dof_focus } else { 1.0 };
1018
1019        let main = PostParams::from_settings(
1020            s,
1021            w as f32,
1022            h as f32,
1023            time,
1024            sun,
1025            near,
1026            far,
1027            focus,
1028            tan_half_fov,
1029            aspect,
1030        );
1031        queue.write_buffer(&self.params_buffer, 0, bytemuck::bytes_of(&main));
1032
1033        // bright: próg w `fx.x`, kierunek w `grade.x` jest tu nieistotny.
1034        // Pasy bloom nie czytają nowych grup, więc wystarczy wypełnić
1035        // je zerami — jawnie, żeby nie było pól „niezainicjalizowanych".
1036        let bright = PostParams {
1037            grade: [0.0; 4],
1038            fx: [s.bloom_threshold, 0.0, 1.0, 1.0],
1039            screen: [w as f32, h as f32, 0.0, 0.0],
1040            sun: [0.0; 4],
1041            outline: [0.0; 4],
1042            ssr: [0.0; 4],
1043            scatter: [0.0; 4],
1044            lens: [0.0; 4],
1045            rays: [0.0; 4],
1046            depth: [near, far, 0.0, 0.0],
1047            grade2: [0.0; 4],
1048            ssao: [0.0; 4],
1049            proj: [0.0; 4],
1050            // `clarity` zerujemy w passach pomocniczych (bright, blur):
1051            // te passy operują na wycinku obrazu, a nie na całym
1052            // kadrze, więc unsharp liczony na ich danych byłby liczony
1053            // z ułamka sąsiadów i dawałby losowe wyniki.
1054            clarity: [0.0; 4],
1055        };
1056        queue.write_buffer(&self.blur_buffers[0], 0, bytemuck::bytes_of(&bright));
1057
1058        // blur: `grade.x` = kierunek (1 = poziomo, 0 = pionowo),
1059        // `screen.xy` = rozmiar ŹRÓDŁA, bo krok kernela jest w UV.
1060        // Ważne: to rozmiar bufora bloom (połowa kadru), a nie kadru —
1061        // inaczej kernel byłby dwa razy za szeroki.
1062        let mut blur_h = bright;
1063        blur_h.grade[0] = 1.0;
1064        blur_h.screen = [(w / 2).max(1) as f32, (h / 2).max(1) as f32, 0.0, 0.0];
1065        queue.write_buffer(&self.blur_buffers[1], 0, bytemuck::bytes_of(&blur_h));
1066        let mut blur_v = blur_h;
1067        blur_v.grade[0] = 0.0;
1068        queue.write_buffer(&self.blur_buffers[2], 0, bytemuck::bytes_of(&blur_v));
1069    }
1070}

1082impl PostFx {
1083    /// Wykonuje pełny łańcuch post-processingu i kompozycję na powierzchnię.
1084    ///
1085    /// Kolejność:
1086    /// 1. god rays (HDR → `rays_out`)
1087    /// 2. SSR (HDR + G-Bufer + głębokość → `ssr_out`)
1088    /// 3. bloom: bright → blur H → blur V
1089    /// 4. kompozycja na powierzchnię
1090    ///
1091    /// Oba passy ekranowe mają WŁASNE cele, więc ich wyniki nie
1092    /// nadpisują się nawzajem, a kompozycja czyta je równocześnie.
1093    ///
1094    /// Wszystkie passy idą w TYM SAMYM encoderze, więc GPU nie czeka
1095    /// na osobne synchronizacje między nimi.
1096    pub fn composite(&self, enc: &mut wgpu::CommandEncoder, target: &Scene3dTarget<'_>) {
1097        let ba = self.bloom_a.as_ref().expect("ensure_size");
1098        let bb = self.bloom_b.as_ref().expect("ensure_size");
1099        let ssr = self.ssr_out.as_ref().expect("ensure_size");
1100        let rays = self.rays_out.as_ref().expect("ensure_size");
1101        let ao_raw = self.ao_raw.as_ref().expect("ensure_size");
1102        let ao = self.ao_out.as_ref().expect("ensure_size");
1103
1104        // --- SSAO: gbuffer + depth -> ao_raw -> ao_out
1105        //
1106        // PRZED kompozycją, bo to ona czyta AO. Kolejność względem
1107        // SSR i god rays nie ma znaczenia — te zapisują w swoje
1108        // własne cele i czytają surowy HDR, a nie wynik AO.
1109        if let Some(bg) = self.ao_bgs[0].as_ref() {
1110            self.fullscreen(enc, "Uran SSAO", &self.ssao_pipeline, bg, ao_raw);
1111        }
1112        if let Some(bg) = self.ao_bgs[1].as_ref() {
1113            self.fullscreen(enc, "Uran AO Blur", &self.ao_blur_pipeline, bg, ao);
1114        }
1115
1116        // --- god rays: hdr -> rays_out
1117        if let Some(bg) = self.effect_bgs[1].as_ref() {
1118            self.fullscreen(enc, "Uran God Rays", &self.godray_pipeline, bg, rays);
1119        }
1120
1121        // --- SSR: hdr + gbuffer + depth -> ssr_out
1122        if let Some(bg) = self.effect_bgs[0].as_ref() {
1123            self.fullscreen(enc, "Uran SSR", &self.ssr_pipeline, bg, ssr);
1124        }
1125
1126        // --- bright-pass: hdr -> bloom_a
1127        if let Some(bg) = self.bloom_bgs[0].as_ref() {
1128            self.fullscreen(enc, "Uran Bright", &self.bright_pipeline, bg, ba);
1129        }
1130        // --- rozmycie poziome: bloom_a -> bloom_b
1131        if let Some(bg) = self.bloom_bgs[1].as_ref() {
1132            self.fullscreen(enc, "Uran Blur H", &self.blur_pipeline, bg, bb);
1133        }
1134        // --- rozmycie pionowe: bloom_b -> bloom_a
1135        if let Some(bg) = self.bloom_bgs[2].as_ref() {
1136            self.fullscreen(enc, "Uran Blur V", &self.blur_pipeline, bg, ba);
1137        }
1138
1139        // --- kompozycja na powierzchnię (tu widać efekty)
1140        let Some(bg) = self.composite_bg.as_ref() else {
1141            return;
1142        };
1143        let mut p = enc.begin_render_pass(&wgpu::RenderPassDescriptor {
1144            label: Some("Uran Composite"),
1145            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
1146                view: target.color,
1147                resolve_target: target.resolve,
1148                ops: wgpu::Operations {
1149                    load: wgpu::LoadOp::Clear(wgpu::Color::TRANSPARENT),
1150                    store: wgpu::StoreOp::Store,
1151                },
1152            })],
1153            ..Default::default()
1154        });
1155        p.set_pipeline(&self.composite_pipeline);
1156        p.set_bind_group(0, bg, &[]);
1157        p.draw(0..3, 0..1);
1158    }
1159
1160    /// Pełnoekranowy pass pośredni (bez depth, jeden trójkąt).
1161    fn fullscreen(
1162        &self,
1163        enc: &mut wgpu::CommandEncoder,
1164        label: &str,
1165        pipeline: &wgpu::RenderPipeline,
1166        bind_group: &wgpu::BindGroup,
1167        view: &wgpu::TextureView,
1168    ) {
1169        let mut p = enc.begin_render_pass(&wgpu::RenderPassDescriptor {
1170            label: Some(label),
1171            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
1172                view,
1173                resolve_target: None,
1174                ops: wgpu::Operations {
1175                    load: wgpu::LoadOp::Clear(wgpu::Color::TRANSPARENT),
1176                    store: wgpu::StoreOp::Store,
1177                },
1178            })],
1179            ..Default::default()
1180        });
1181        p.set_pipeline(pipeline);
1182        p.set_bind_group(0, bind_group, &[]);
1183        // 3 wierzchołki = pełny prostokąt, bez bufora wierzchołków
1184        p.draw(0..3, 0..1);
1185    }
1186}
