20/// Jedno polecenie rysowania: „narysuj tę siatkę w tej macierzy".
21#[derive(Debug, Clone, Copy)]
22pub struct DrawCmd {
23    pub mesh: MeshId,
24    pub model: Mat4,
25    /// Mnożnik koloru wierzchołka — pozwala jedną siatkę malować na
26    /// różne kolory (gracz / przeciwnik) bez duplikowania geometrii.
27    pub tint: [f32; 4],
28    /// Materiał PBR: tekstury, chropowatość, metaliczność.
29    ///
30    /// Domyślnie `MaterialId(0)` — to zawsze istniejący materiał
31    /// zapasowy (biały, bez map). Dzięki niemu kod, który nie zna
32    /// pojęcia materiału (cały `uran-tanks`), nie musi go udawać
33    /// opcjonalnym i nie psuje się przy zmianie struktury komendy.
34    pub material: MaterialId,
35}
36
37impl DrawCmd {
38    pub fn new(mesh: MeshId, model: Mat4) -> Self {
39        Self {
40            mesh,
41            model,
42            tint: [1.0; 4],
43            material: MaterialId(0),
44        }
45    }
46
47    pub fn tinted(mesh: MeshId, model: Mat4, tint: [f32; 4]) -> Self {
48        Self {
49            mesh,
50            model,
51            tint,
52            material: MaterialId(0),
53        }
54    }
55
56    /// Przypisuje materiał — np. ten, który przyszedł z pliku `.mtl`.
57    pub fn with_material(mut self, material: MaterialId) -> Self {
58        self.material = material;
59        self
60    }
61}

63/// Parametry oświetlenia.
64#[derive(Debug, Clone, Copy)]
65pub struct Lighting {
66    /// Kierunek, w którym świeci (wskazuje OD źródła).
67    pub light_dir: Vec3,
68    pub light_color: [f32; 3],
69    /// Natężenie słońca w przestrzeni LINIOWEJ.
70    ///
71    /// Światło liczymy liniowo, gdzie 1.0 to „biały" — czyli wartość
72    /// bliska 1 daje bardzo słabe słońce. Typowe słońce w bezchmurznym
73    /// dniu to 3..6, a zachód słońca 1.5..2.5.
74    ///
75    /// Dawniej shader mnożył światło przez stałą 1.35, co było
76    /// sprzężone z modelem Phonga; przy oświetleniu PBR i ACES ta
77    /// wartość zmieniała jasność inaczej na każdej powierzchni.
78    pub intensity: f32,
79    pub ambient: [f32; 3],
80}
81
82impl Default for Lighting {
83    fn default() -> Self {
84        Self {
85            // Słońce około 50° nad horyzontem i wyraźnie z boku. Niski kąt
86            // daje długie, czytelne cienie i jasno pokazuje, w którą stronę
87            // pada światło — przy słońcu prosto w głowę bryły są płaskie.
88            light_dir: Vec3::new(0.40, 0.62, 0.68),
89            // Ciepłe, lekko złotawe — słońce niskiego popołudnia.
90            light_color: [1.0, 0.93, 0.80],
91            // 4.0 = jasne popołudniowe słońce; z ACES w post-processingu
92            // daje to biel na białych powierzchniach i nie prześwietla
93            // asfaltu
94            intensity: 4.0,
95            // Ambient NIESIE NIEBO: chłodny, wyraźnie niebieski.
96            //
97            // ## Dlaczego tak mały
98            //
99            // 0.30/0.40/0.58 przy słońcu 4.0 wypełniało cienie na tyle,
100            // że obraz wyglądał jak patrząc przez mętną soczewkę: brak
101            // kontrastu, plamy zamiast brył, każda powierzchnia w tym
102            // samym odcieniu. Stosunek ambient:słońce poniżej ~0.05
103            // oznacza, że strona w cieniu dostaje realny kontrast,
104            // a nie tylko ciemniejszy odcień tego samego koloru.
105            //
106            // 0.10/0.13/0.18 to nadal „niebo z wnętrza" — cienie są
107            // chłodne i czytelne, ale nie zalewają geometrii.
108            ambient: [0.10, 0.13, 0.18],
109        }
110    }
111}

113/// Parametry hybrydy PBR + NPR — miara „rysunkowości" sceny.
114///
115/// To jest globalna skala, do której mnożymy stylizację z materiału
116/// (`Surface::stylize`). Dzięki temu jedna scena może mieć rysunkową
117/// postać i fizyczne otoczenie: materiał mówi „chcę tu rampę", a ten
118/// parametr mówi „na ile mocno".
119///
120/// Przy `0.0` zachowanie jest czyste PBR. Wartość domyślna to `1.0`,
121/// bo to gra decyduje, ile rysunkowości chce — nie renderer.
122#[derive(Debug, Clone, Copy)]
123pub struct Stylization {
124    /// Globalna siła rampy anime 0..1.
125    pub amount: f32,
126    /// Miękkość progu terminatora. Małe = ostrzejszy rysunek.
127    pub softness: f32,
128    /// Globalna siła podpowierzchniowego rozpraszania (mnoży tę z materiału).
129    pub sss: f32,
130}
131
132impl Default for Stylization {
133    fn default() -> Self {
134        Self {
135            // 1.0 = materiał decyduje w pełni. Ustawienie 0.0 byłoby
136            // „czyste PBR zawsze", a to odbierałoby grze narzędzie.
137            amount: 1.0,
138            // 0.12 to rampa wyraźnie widoczna, ale z gradientem szerokości
139            // kilku stopni. Poniżej 0.05 krawędź zaczyna migotać przy
140            // ruchu kamery.
141            softness: 0.12,
142            sss: 1.0,
143        }
144    }
145}

147/// Mgła i niebo — atmosfera sceny.
148#[derive(Debug, Clone, Copy)]
149pub struct Atmosphere {
150    /// Czy rysować fizyczne niebo (Rayleigh + Mie).
151    ///
152    /// Gdy `false`, kadr wypełnia [`Renderer3d::set_clear_color`], a mgła
153    /// nadal działa. To pozwala grze przełączać „wewnętrzna hala" (brak
154    /// nieba) i „pod gołym niebem" jednym wywołaniem.
155    pub sky: bool,
156    /// Gęstość mgły 0..1 — odpowiada za `1 - e^(-d·ρ)`.
157    ///
158    /// 0.004 daje mgłę widoczną od ~150 m; 0.012 zaczyna zasłaniać
159    /// odległe budynki. Powyżej 0.03 scena znika całkowicie.
160    pub fog_density: f32,
161    /// Kolor mgły w sRGB, zakres 0..1.
162    ///
163    /// Niebo wzmacnia go w kierunku słońca (rozpraszanie Mie), więc
164    /// podstawowy kolor można ustawić neutralnie i nieba nie trzeba
165    /// przeliczać ręcznie.
166    pub fog_color: [f32; 3],
167    /// Wysokość (w metrach), na której mgła całkiem zanika.
168    ///
169    /// ## Dlaczego to pole istnieje
170    ///
171    /// Mgła nie jest jednorodna: jej gęstość rośnie w dół i zanika
172    /// powyżej pewnej wysokości. Bez tego parametru mgła zasłaniała
173    /// tak samo kamień leżący na ziemi i dach domu stojącego 30 m
174    /// wyżej — a w realnym świecie dach jest czysty.
175    ///
176    /// Shader liczy analityczny całkowity gęstości wzdłuż promienia
177    /// oka (jak WickedEngine w `fogHF.hlsli`), więc ta wartość wprost
178    /// steruje tym, jak szybko mgła zanika ku górze.
179    ///
180    /// 0 wyłącza ten efekt i daje z powrotem jednolitą mgłę.
181    pub fog_height: f32,
182}
183
184impl Default for Atmosphere {
185    fn default() -> Self {
186        Self {
187            sky: true,
188            // 0.0045 → 0.0016. Wcześniejsza wartość zamieniała w „mętne"
189            // zrzuty: mgła zaczynała jeść już kilkanaście metrów od
190            // kamery, a horyzont znikał w błękitnej ścianie.
191            // 0.0016 daje ledwie widoczne zamglenie na dystansie
192            // ok. 200 m.
193            //
194            // Skala jest mocno nieliniowa w odczuciu, bo to wykładnicza
195            // funkcja dystansu: przy 100 m to `1 - e^(-0.16)` = 15%
196            // zamiast dawnych 36%. Ta różnica 15% vs 36% to dokładnie
197            // granica między „mglisto" a „czytelnie".
198            fog_density: 0.0016,
199            // 0.58/0.70/0.86 -> 0.46/0.55/0.68 -> 0.32/0.38/0.46.
200            //
201            // Kolor mgly musi byc CIEMNIEJSZY od nieba przy
202            // horyzoncie. Inaczej dystans nie zanika, tylko
203            // jasnieje - i caly kadr ciagnie ku blademu błękitowi,
204            // ktory zjada nasycenie calej reszty kadru.
205            //
206            // 0.32/0.38/0.46 to luma ~0.37 wobec nieba 0.46. Mgla
207            // jest tlem dla obiektow, a nie osobna warstwa koloru.
208            fog_color: [0.32, 0.38, 0.46],
209            // 55 m: powyżej wierzchu typowych budynków mgła już dawno
210            // zniknęła, więc dachy i kominy zostają czyste, a dolne
211            // piętra i ulica toną. 0 dałoby z powrotem jednolitą
212            // „mleczną" warstwę na każdej wysokości.
213            fog_height: 55.0,
214        }
215    }
216}

218/// Renderer 3D — implementacja [`Scene3d`].
219///
220/// Świadomie NIE trzymamy `Device`/`Queue` w polach. W wgpu 0.19 oba typy
221/// nie implementują `Clone`, a urządzenie należy do renderera 2D. Zamiast
222/// tego każdą operację na GPU wykonujemy przez `Scene3dTarget::gpu`, które
223/// `uran-render` przekazuje w `draw()`. Jedyny wyjątek to konstrukcja,
224/// gdzie `&GpuContext` mamy z parametru.
225pub struct Renderer3d {
226    format: wgpu::TextureFormat,
227    samples: u32,
228
229    /// Depth buffer 3D. 2D go nie ma w ogóle (rysowanie płaskie), więc
230    /// każdy z rendererów ma swój.
231    depth: Option<wgpu::TextureView>,
232    /// Sama tekstura głębokości — trzymana osobno od widoku, bo
233    /// post-processing potrzebuje jej jako ŹRÓDŁA do kopii
234    /// (`PostFx::copy_depth`). Sam `TextureView` nie wystarczy:
235    /// `copy_texture_to_texture` przyjmuje `&Texture`.
236    depth_tex: Option<wgpu::Texture>,
237    depth_size: (u32, u32),
238
239    scene_buffer: wgpu::Buffer,
240    models_buffer: wgpu::Buffer,
241    models_capacity: usize,
242    bind_group: wgpu::BindGroup,
243    /// Layout bind grupy trzymany osobno: `ensure_capacity` musi przebudować
244    /// bind grupę po zmianie bufora modeli, a layout musi zostać ten sam.
245    models_bind_layout: wgpu::BindGroupLayout,
246
247    /// Post-processing (AA, bloom, tonemapping) — własny moduł.
248    postfx: PostFx,
249
250    pipeline: wgpu::RenderPipeline,
251    /// Pipeline nieba: pełnoekranowy trójkąt z fizycznym rozpraszaniem
252    /// atmosferycznym. Rysowany jako pierwszy w passie głównym.
253    ///
254    /// Niebo opcjonalne — patrz [`Atmosphere`]. Gdy `enabled` jest wyłączone,
255    /// pipeline w ogóle nie jest rysowany, a kadr wypełnia kolor czyszczenia.
256    sky_pipeline: wgpu::RenderPipeline,
257    /// Pipeline passu cieni: ten sam shader, inny punkt wejścia
258    /// (`vs_shadow`) i brak fragment shadera. Rysuje wyłącznie
259    /// głębokość do mapy cieni.
260    shadow_pipeline: wgpu::RenderPipeline,
261    meshes: Vec<GpuMesh>,
262    /// Materiały PBR i ich bind grupy (bind group 1 w potoku).
263    pub materials: MaterialBank,
264
265    /// Mapa cieni kierunkowych (tekstura głębokości + próbnik porównawczy).
266    pub shadow: ShadowMap,
267    /// Layout bind grupy samego passu cieni.
268    ///
269    /// OSOBNY layout, bo wgpu zabrania wiązania zasobu, który w tym
270    /// samym passie jest celem renderowania. Gdybyśmy użyli tu tej
271    /// samej grupy co w passie głównym, mapa cieni byłaby jednocześnie
272    /// źródłem i celem — walidacja odrzuci pass.
273    shadow_bind_layout: wgpu::BindGroupLayout,
274    /// Bind grupy passu cieni: uniform sceny + tablica modeli.
275    shadow_bind_group: wgpu::BindGroup,
276
277    camera: Camera3d,
278    lighting: Lighting,
279    /// Globalna skala rysunkowości (rampa anime na dyfuzji).
280    stylization: Stylization,
281    /// Niebo i mgła.
282    atmosphere: Atmosphere,
283    clear_color: [f32; 4],
284    time: f32,
285    commands: Vec<DrawCmd>,
286    /// Ile obiektów narysowano w ostatniej klatce (do HUD-a i diagnostyki).
287    pub last_draw_count: usize,
288}

290impl Renderer3d {
291    /// Tworzy renderer 3D na tym samym urządzeniu, co renderer 2D.
292    pub fn new(gpu: &uran_render::GpuContext, format: wgpu::TextureFormat, samples: u32) -> Self {
293        let device = &gpu.device;
294
295        let scene_buffer = device.create_buffer(&wgpu::BufferDescriptor {
296            label: Some("Uran 3D Scene Uniform"),
297            size: std::mem::size_of::<SceneUniform>() as u64,
298            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
299            mapped_at_creation: false,
300        });
301
302        // Tablica modeli. Rośnie potęgami 2, a bind grupa przebudowywana
303        // jest raz przy zmianie rozmiaru (zdarzenie rzadkie).
304        let models_capacity = 256usize;
305        let models_buffer = create_models_buffer(&device, models_capacity);
306
307        let layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
308            label: Some("Uran 3D BGL"),
309            entries: &[
310                wgpu::BindGroupLayoutEntry {
311                    binding: 0,
312                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
313                    ty: wgpu::BindingType::Buffer {
314                        ty: wgpu::BufferBindingType::Uniform,
315                        has_dynamic_offset: false,
316                        min_binding_size: None,
317                    },
318                    count: None,
319                },
320                wgpu::BindGroupLayoutEntry {
321                    binding: 1,
322                    // modele czyta TYLKO shader wierzchołkowy
323                    visibility: wgpu::ShaderStages::VERTEX,
324                    ty: wgpu::BindingType::Buffer {
325                        ty: wgpu::BufferBindingType::Storage { read_only: true },
326                        has_dynamic_offset: false,
327                        min_binding_size: None,
328                    },
329                    count: None,
330                },
331                wgpu::BindGroupLayoutEntry {
332                    binding: 2,
333                    // mapę cieni czyta shader FRAGMENT
334                    visibility: wgpu::ShaderStages::FRAGMENT,
335                    ty: wgpu::BindingType::Texture {
336                        // `Depth` wymusza `texture_depth_2d` w WGSL, co
337                        // jest warunkiem użycia `textureSampleCompare`.
338                        sample_type: wgpu::TextureSampleType::Depth,
339                        view_dimension: wgpu::TextureViewDimension::D2,
340                        multisampled: false,
341                    },
342                    count: None,
343                },
344                wgpu::BindGroupLayoutEntry {
345                    binding: 3,
346                    visibility: wgpu::ShaderStages::FRAGMENT,
347                    // `Comparison` daje `sampler_comparison` w WGSL.
348                    ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Comparison),
349                    count: None,
350                },
351            ],
352        });
353
354        // Mapa cieni musi istnieć PRZED bind grupą: binding 2 w grupie 0
355        // wskazuje na jej widok, a binding 3 na jej próbnik.
356        let shadow = ShadowMap::new(&device, ShadowSettings::default());
357
358        let bind_group = make_bind_group(
359            &device,
360            &layout,
361            &scene_buffer,
362            &models_buffer,
363            Some(shadow.view()),
364            Some(shadow.sampler()),
365        );
366
367        // Layout passu cieni: TYLKO uniform sceny i tablica modeli.
368        // Świadomie bez mapy cieni — w trakcie tego passu jest ona
369        // celem renderowania, a wgpu odrzuca zasób użyty w jednym
370        // passie jako źródło i jako attachment.
371        let shadow_bind_layout =
372            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
373                label: Some("Uran 3D Shadow BGL"),
374                entries: &[
375                    wgpu::BindGroupLayoutEntry {
376                        binding: 0,
377                        visibility: wgpu::ShaderStages::VERTEX,
378                        ty: wgpu::BindingType::Buffer {
379                            ty: wgpu::BufferBindingType::Uniform,
380                            has_dynamic_offset: false,
381                            min_binding_size: None,
382                        },
383                        count: None,
384                    },
385                    wgpu::BindGroupLayoutEntry {
386                        binding: 1,
387                        visibility: wgpu::ShaderStages::VERTEX,
388                        ty: wgpu::BindingType::Buffer {
389                            ty: wgpu::BufferBindingType::Storage { read_only: true },
390                            has_dynamic_offset: false,
391                            min_binding_size: None,
392                        },
393                        count: None,
394                    },
395                ],
396            });
397        // Layout passu cienia deklaruje TYLKO bindingi 0 i 1, wiec
398        // przekazujemy `None`/`None` — `create_bind_group` wymaga
399        // dokladnie tylu wpisow, ile jest w layoutcie.
400        let shadow_bind_group = make_bind_group(
401            &device,
402            &shadow_bind_layout,
403            &scene_buffer,
404            &models_buffer,
405            None,
406            None,
407        );
408
409        // Materialy PBR: layout z grupy 1. Bank tworzymy tu, bo
410        // `pipeline_layout` musi znać jego layout, a sam bank potrzebuje
411        // `device` i `queue`.
412        // Layout trzymamy w `Arc`, bo `wgpu::BindGroupLayout` nie
413        // implementuje `Clone` w wgpu 0.19.
414        let materials = MaterialBank::new(&device, &gpu.queue);
415        let materials_layout = std::sync::Arc::clone(&materials.layout);
416
417        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
418            label: Some("Uran 3D Shader"),
419            source: wgpu::ShaderSource::Wgsl(include_str!("s3d.wgsl").into()),
420        });
421
422        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
423            label: Some("Uran 3D Pipeline Layout"),
424            bind_group_layouts: &[&layout, materials_layout.as_ref()],
425            push_constant_ranges: &[],
426        });
427
428        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
429            label: Some("Uran 3D Pipeline"),
430            layout: Some(&pipeline_layout),
431            vertex: wgpu::VertexState {
432                module: &shader,
433                entry_point: "vs_main",
434                buffers: &[wgpu::VertexBufferLayout {
435                    array_stride: std::mem::size_of::<crate::mesh::Vertex>() as u64,
436                    step_mode: wgpu::VertexStepMode::Vertex,
437                    // Kolejność musi być identyczna z `Vertex`: pozycja,
438                    // normalna, UV, kolor. Offsety liczymy ręcznie, bo
439                    // `#[repr(C)]` daje 44 B bez wypełnienia.
440                    attributes: &[
441                        wgpu::VertexAttribute {
442                            format: wgpu::VertexFormat::Float32x3,
443                            offset: 0,
444                            shader_location: 0,
445                        },
446                        wgpu::VertexAttribute {
447                            format: wgpu::VertexFormat::Float32x3,
448                            offset: 12,
449                            shader_location: 1,
450                        },
451                        wgpu::VertexAttribute {
452                            format: wgpu::VertexFormat::Float32x2,
453                            offset: 24,
454                            shader_location: 3,
455                        },
456                        wgpu::VertexAttribute {
457                            format: wgpu::VertexFormat::Float32x3,
458                            offset: 32,
459                            shader_location: 2,
460                        },
461                    ],
462                }],
463            },
464            fragment: Some(wgpu::FragmentState {
465                module: &shader,
466                entry_point: "fs_main",
467                // DWA cele kolorowe (G-Buffer):
468                //   [0] HDR — to, co widzi gracz,
469                //   [1] normalna w przestrzeni oka + chropowatość, z czego
470                //       post-processing bierze SSR, kontury, SSS i DoF.
471                // Oba muszą mieć format zgodny z załącznikami render passu
472                // (patrz `PostFx::gbuffer_view`), inaczej walidacja wgpu
473                // odrzuci potok.
474                targets: &[
475                    Some(wgpu::ColorTargetState {
476                        // Scena rysuje do celu HDR (Rgba16Float), a NIE na
477                        // powierzchnię. Format musi się zgadzać z załącznikiem
478                        // render passu, inaczej walidacja wgpu to odrzuci.
479                        format: wgpu::TextureFormat::Rgba16Float,
480                        blend: Some(wgpu::BlendState::REPLACE),
481                        write_mask: wgpu::ColorWrites::ALL,
482                    }),
483                    Some(wgpu::ColorTargetState {
484                        format: wgpu::TextureFormat::Rgba16Float,
485                        blend: Some(wgpu::BlendState::REPLACE),
486                        write_mask: wgpu::ColorWrites::ALL,
487                    }),
488                ],
489            }),
490            primitive: wgpu::PrimitiveState {
491                topology: wgpu::PrimitiveTopology::TriangleList,
492                // tyłna ściana musi być rysowana, inaczej wnętrze czołgu
493                // wygląda dziurawie
494                cull_mode: Some(wgpu::Face::Back),
495                ..Default::default()
496            },
497            depth_stencil: Some(wgpu::DepthStencilState {
498                format: wgpu::TextureFormat::Depth32Float,
499                depth_write_enabled: true,
500                depth_compare: wgpu::CompareFunction::Less,
501                stencil: wgpu::StencilState::default(),
502                bias: wgpu::DepthBiasState::default(),
503            }),
504            multisample: wgpu::MultisampleState {
505                count: samples,
506                mask: !0,
507                alpha_to_coverage_enabled: false,
508            },
509            multiview: None,
510        });
511
512        // --- pipeline nieba ---
513        //
514        // Pełnoekranowy trójkąt rysowany PRZED bryłami, z wyłączonym
515        // zapisem głębokości. Dzięki temu:
516        //   * niebo wypełnia cały kadru bez geometrii,
517        //   * każda bryła go zasłania, bo ma mniejszą głębokość.
518        //
519        // Layout ma TYLKO grupę 0 (uniform sceny) i to samo, co główny
520        // potok — niebo potrzebuje kierunku słońca, a nic więcej.
521        // Nie ustawiamy grupy 1 (materiały), bo shader nieba nie ma
522        // żadnego `@group(1)`.
523        // Layout ma TYLKO grupę 0 (uniform sceny) — i to wymaga OSOBNEGO
524        // `PipelineLayout`, bo `pipeline_layout` głównego potoku zawiera
525        // również grupę 1 (materiały). Gdybyśmy podali go tutaj, wgpu
526        // wymagałby powiązania grupy 1 przy każdym rysowaniu nieba,
527        // a `draw` nie ustawia żadnej — błąd walidacji wyskakiwałby
528        // dopiero w trakcie renderowania, mimo że shader nieba nie ma
529        // ani jednego `@group(1)`.
530        let sky_pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
531            label: Some("Uran 3D Layout Nieba"),
532            bind_group_layouts: &[&layout],
533            push_constant_ranges: &[],
534        });
535        let sky_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
536            label: Some("Uran 3D Sky Pipeline"),
537            layout: Some(&sky_pipeline_layout),
538            vertex: wgpu::VertexState {
539                module: &shader,
540                entry_point: "vs_sky",
541                // Pusty bufor wierzchołków: pozycje generuje sam shader
542                // z `vertex_index`, więc geometria nie jest potrzebna.
543                buffers: &[],
544            },
545            fragment: Some(wgpu::FragmentState {
546                module: &shader,
547                entry_point: "fs_sky",
548                // Drugi cel MUSI tu być, nawet jeśli nic do niego nie
549                // zapisujemy: render pass ma DWA załączniki kolorowe
550                // (HDR + G-Buffer), a wgpu wymaga, żeby potok deklarował
551                // dokładnie tyle samo targetów, ile pass ma załączników.
552                // Bez tego walidacja odrzuca `set_pipeline` komunikatem
553                // „targets are incompatible" — i to dopiero w trakcie
554                // renderowania, czyli po zbudowaniu okna i wczytaniu
555                // assetów, nie na `cargo test`.
556                targets: &[
557                    Some(wgpu::ColorTargetState {
558                        format: wgpu::TextureFormat::Rgba16Float,
559                        blend: Some(wgpu::BlendState::REPLACE),
560                        write_mask: wgpu::ColorWrites::ALL,
561                    }),
562                    Some(wgpu::ColorTargetState {
563                        format: wgpu::TextureFormat::Rgba16Float,
564                        blend: Some(wgpu::BlendState::REPLACE),
565                        // Niebo nie ma normalnej ani chropowatości, więc
566                        // G-Bufera nie ruszamy. `empty()` mówi
567                        // „załącznik jest, ale nic do niego nie piszemy" —
568                        // taniej niż generowanie zerowego wektora
569                        // normalnej w shaderze.
570                        write_mask: wgpu::ColorWrites::empty(),
571                    }),
572                ],
573            }),
574            primitive: wgpu::PrimitiveState {
575                topology: wgpu::PrimitiveTopology::TriangleList,
576                ..Default::default()
577            },
578            depth_stencil: Some(wgpu::DepthStencilState {
579                format: wgpu::TextureFormat::Depth32Float,
580                // `false` jest tu kluczowe: gdyby niebo zapisywało
581                // głębokość 1.0, zamalowałoby nim bryły narysowane
582                // wcześniej (przy `Less` nie, ale przy zmianie `LessEqual`
583                // w przyszłości — tak).
584                depth_write_enabled: false,
585                depth_compare: wgpu::CompareFunction::LessEqual,
586                stencil: wgpu::StencilState::default(),
587                bias: wgpu::DepthBiasState::default(),
588            }),
589            multisample: wgpu::MultisampleState {
590                count: samples,
591                mask: !0,
592                alpha_to_coverage_enabled: false,
593            },
594            multiview: None,
595        });
596
597        // --- pipeline passu cieni ---
598        //
599        // Różni się od głównego w trzech miejscach:
600        //   1. entry point `vs_shadow` zamiast `vs_main`,
601        //   2. `fragment: None` — brak color attachmentu,
602        //   3. inny layout bind grup (tylko uniform + modele).
603        //
604        // Układ wierzchołków MUSI być identyczny z głównym potokiem:
605        // oba czytają ten sam `GpuMesh` z tym samym buforem, a layout
606        // bufora jest częścią kontraktu z potokiem.
607        let shadow_pipeline_layout =
608            device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
609                label: Some("Uran 3D Shadow Pipeline Layout"),
610                bind_group_layouts: &[&shadow_bind_layout],
611                push_constant_ranges: &[],
612            });
613
614        let shadow_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
615            label: Some("Uran 3D Shadow Pipeline"),
616            layout: Some(&shadow_pipeline_layout),
617            vertex: wgpu::VertexState {
618                module: &shader,
619                entry_point: "vs_shadow",
620                buffers: &[wgpu::VertexBufferLayout {
621                    array_stride: std::mem::size_of::<crate::mesh::Vertex>() as u64,
622                    step_mode: wgpu::VertexStepMode::Vertex,
623                    attributes: &[
624                        wgpu::VertexAttribute {
625                            format: wgpu::VertexFormat::Float32x3,
626                            offset: 0,
627                            shader_location: 0,
628                        },
629                        wgpu::VertexAttribute {
630                            format: wgpu::VertexFormat::Float32x3,
631                            offset: 12,
632                            shader_location: 1,
633                        },
634                        wgpu::VertexAttribute {
635                            format: wgpu::VertexFormat::Float32x2,
636                            offset: 24,
637                            shader_location: 3,
638                        },
639                        wgpu::VertexAttribute {
640                            format: wgpu::VertexFormat::Float32x3,
641                            offset: 32,
642                            shader_location: 2,
643                        },
644                    ],
645                }],
646            },
647            // `None` = brak stage'u fragmentowego. Wymagane przez
648            // specyfikację, gdy nie ma color attachmentu.
649            fragment: None,
650            primitive: wgpu::PrimitiveState {
651                topology: wgpu::PrimitiveTopology::TriangleList,
652                // Culling musi być TAKIE SAME jak w głównym potoku.
653                // Różnica dałaby cienie „od tyłu" dla obiektów, których
654                // ściany są jednorodne — np. płoty i ściany stodoła.
655                cull_mode: Some(wgpu::Face::Back),
656                ..Default::default()
657            },
658            depth_stencil: Some(wgpu::DepthStencilState {
659                format: wgpu::TextureFormat::Depth32Float,
660                depth_write_enabled: true,
661                depth_compare: wgpu::CompareFunction::Less,
662                stencil: wgpu::StencilState::default(),
663                // Depth bias na poziomie potoku: zależny od nachylenia
664                // powierzchni, więc działa tam, gdzie statyczny bias
665                // w shaderze nie wystarcza (skośne dachy, pochyłe ściany).
666                bias: wgpu::DepthBiasState {
667                    constant: 2,
668                    slope_scale: 2.0,
669                    clamp: 0.0,
670                },
671            }),
672            // 1: pass cienia nie używa MSAA — rysujemy samą głębokość.
673            multisample: wgpu::MultisampleState {
674                count: 1,
675                mask: !0,
676                alpha_to_coverage_enabled: false,
677            },
678            multiview: None,
679        });
680
681        let (w, h) = gpu.size();
682        Self {
683            format,
684            samples,
685            depth: None,
686            depth_tex: None,
687            depth_size: (0, 0),
688            scene_buffer,
689            models_buffer,
690            models_capacity,
691            bind_group,
692            models_bind_layout: layout,
693            postfx: PostFx::new(device, format, samples),
694            pipeline,
695            sky_pipeline,
696            shadow_pipeline,
697            meshes: Vec::new(),
698            materials,
699            shadow,
700            shadow_bind_layout,
701            shadow_bind_group,
702            camera: Camera3d::default(),
703            lighting: Lighting::default(),
704            stylization: Stylization::default(),
705            atmosphere: Atmosphere::default(),
706            clear_color: [0.45, 0.55, 0.68, 1.0],
707            time: 0.0,
708            commands: Vec::new(),
709            last_draw_count: 0,
710        }
711        .with_depth(device, w, h)
712    }

815    /// Upewnia się, że tablica modeli pomieści `count` obiektów.
816    ///
817    /// Zmiana pojemności oznacza nowy bufor I nową bind grupę, dlatego
818    /// trzymamy tu też `layout`, zamiast odtwarzać go przy każdym wzroście.
819    fn ensure_capacity(&mut self, device: &wgpu::Device, count: usize) {
820        if count <= self.models_capacity {
821            return;
822        }
823        let mut capacity = self.models_capacity.max(1);
824        while capacity < count {
825            capacity *= 2;
826        }
827        self.models_buffer = create_models_buffer(device, capacity);
828        self.models_capacity = capacity;
829        self.bind_group = make_bind_group(
830            device,
831            &self.models_bind_layout,
832            &self.scene_buffer,
833            &self.models_buffer,
834            Some(self.shadow.view()),
835            Some(self.shadow.sampler()),
836        );
837        // Pass cienia ma OSOBNĄ bind grupę wskazującą na tę samą tablicę
838        // modeli — inaczej po powiększeniu bufora rysowałby z dziurą.
839        self.shadow_bind_group = make_bind_group(
840            device,
841            &self.shadow_bind_layout,
842            &self.scene_buffer,
843            &self.models_buffer,
844            None,
845            None,
846        );
847    }

849    /// AABB sceny liczone z macierzy modeli wszystkich komend.
850    ///
851    /// Bierzemy osiem narożników jednostkowego sześcianu (-1..1) i
852    /// przekształcamy je każdą macierzą. `GpuMesh` NIE trzyma już
853    /// geometrii na CPU (`upload` zostawia ją tylko na karcie), więc
854    /// AABB liczymy z transformacji, a nie z wierzchołków.
855    ///
856    /// Jednostkowy sześcian zamiast bryły modelu daje **nadmiarowy**
857    /// kadr: dopasowany do sześcianu, a nie do rzeczywistej bryły.
858    /// To kosztuje rozdzielczość cienia (większy kadr na tę samą mapę),
859    /// ale jest bezpieczne — nigdy nie wytniemy obiektu.
860    fn bounds_from_cache(&self) -> SceneBounds {
861        let mut b = SceneBounds::empty();
862        for cmd in &self.commands {
863            for i in 0..8 {
864                let corner = Vec3::new(
865                    if i & 1 == 0 { -1.0 } else { 1.0 },
866                    if i & 2 == 0 { -1.0 } else { 1.0 },
867                    if i & 4 == 0 { -1.0 } else { 1.0 },
868                );
869                b.expand(cmd.model.transform_point3(corner));
870            }
871        }
872        b
873    }

875    /// Pozycja tarczy słońca w UV albo `None`, gdy jest poza kadrem.
876    ///
877    /// Promień do słońca rzutujemy tą samą macierzą `view_proj`, co
878    /// geometrię, więc wynik jest zgodny z tym, co widać. Dodatkowo
879    /// sprawdzamy, czy punkt leży PRZED kamerą (po stronie -Z): słońce
880    /// za plecami dawałoby smugi wychodzące z krawędzi obrazu, bo
881    /// `to_sun` w shaderze wskazywałoby w stronę odwrotną.
882    fn sun_screen_uv(&self) -> Option<(f32, f32)> {
883        // `light_dir` wskazuje OD źródła, więc pozycja tarczy leży
884        // w kierunku `-light_dir`.
885        let to_sun = -self.lighting.light_dir.normalize_or_zero();
886        let world = self.camera.position + to_sun * 1000.0;
887        let clip = self.camera.view_proj() * uran_math::Vec4::new(world.x, world.y, world.z, 1.0);
888        // `w <= 0` = punkt za kamerą (po stronie +Z).
889        if clip.w <= 1e-4 {
890            return None;
891        }
892        let ndc_x = clip.x / clip.w;
893        let ndc_y = clip.y / clip.w;
894        // Trzymamy się tu+1 zapasu: flara i promienie są szerokie,
895        // więc tarcza tuż za krawędzią wciąż powinna dawać poświat.
896        if !(-1.2..=1.2).contains(&ndc_x) || !(-1.2..=1.2).contains(&ndc_y) {
897            return None;
898        }
899        // NDC -> UV. Oś Y odwrócona, bo w NDC rośnie w górę.
900        Some(((ndc_x + 1.0) * 0.5, (1.0 - ndc_y) * 0.5))
901    }

903    /// Renderuje scenę: czyści kolor i głębokość, rysuje wszystkie obiekty.
904    fn render_pass(&mut self, target: &Scene3dTarget<'_>) {
905        let device = &target.gpu.device;
906        let queue = &target.gpu.queue;
907        let w = target.gpu.config.width.max(1);
908        let h = target.gpu.config.height.max(1);
909        if self.depth.is_none() || self.depth_size != (w, h) {
910            self.depth_size = (w, h);
911            let texture = depth_texture(device, w, h, self.samples);
912            self.depth = Some(texture.create_view(&wgpu::TextureViewDescriptor::default()));
913            self.depth_tex = Some(texture);
914            self.camera.set_aspect(w as f32 / h as f32);
915        }
916
917        // Kadr cienia dobieramy do AABB CAŁEJ sceny (liczy go
918        // `bounds_from_cache`). Bez tego obiekty poza kadrem nie
919        // rzucałyby cienia wcale — wygląda to jak wycięcie budynku.
920        let bounds = self.bounds_from_cache();
921        self.shadow.fit(bounds, self.lighting.light_dir);
922
923        let c = self.lighting.ambient;
924        let s = &self.shadow.settings;
925        let shadow_on = if s.enabled { 1.0 } else { 0.0 };
926        // `view_proj` liczymy RAZ i dzielimy na trzy macierze, które
927        // potrzebują różnych passów. Trzy osobne wywołania `view_proj()`
928        // dałyby te same liczby, ale kosztowałyby trzy pełne mnożenia
929        // macierzy na klatkę i groziłyby rozjazdem o 1 ULP.
930        let view_proj = self.camera.view_proj();
931        let uniform = SceneUniform {
932            view_proj: view_proj.to_cols_array_2d(),
933            // Odwrotność potrzebna niebu (odtworzenie promienia) i
934            // post-processingu (odtworzenie pozycji z głębokości).
935            // Przy poprawnej kamerze macierz jest zawsze odwracalna;
936            // `glam` zwraca zera dla osobliwości, a wtedy `w` w shaderze
937            // daje 0 i dzielenie zostawia NaN — dlatego shader nieba
938            // normalizuje różnicę punktów, nie ich współrzędnych.
939            inv_view_proj: view_proj.inverse().to_cols_array_2d(),
940            view: self.camera.view().to_cols_array_2d(),
941            eye: [
942                self.camera.position.x,
943                self.camera.position.y,
944                self.camera.position.z,
945                self.camera.tan_half_fov(),
946            ],
947            light_dir: [
948                self.lighting.light_dir.x,
949                self.lighting.light_dir.y,
950                self.lighting.light_dir.z,
951                0.0,
952            ],
953            light_color: [
954                self.lighting.light_color[0],
955                self.lighting.light_color[1],
956                self.lighting.light_color[2],
957                // `w` to natężenie — wcześniej było tu 0.0, przez co
958                // shader PBR liczył słońce o natężeniu zerowym
959                self.lighting.intensity,
960            ],
961            ambient_time: [c[0], c[1], c[2], self.time],
962            light_view_proj: self.shadow.light_view_proj().to_cols_array_2d(),
963            shadow_params: [s.depth_bias, s.normal_offset, s.strength, s.radius],
964            shadow_map_info: [self.shadow.texel_uv(), shadow_on, 0.0, 0.0],
965            // Near/far w jednym miejscu: od nich zależą liniaryzacja
966            // głębokości w postfx i rekonstrukcja pozycji w SSR.
967            screen: [w as f32, h as f32, self.camera.near, self.camera.far],
968            stylize: [
969                self.stylization.amount,
970                self.stylization.softness,
971                self.stylization.sss,
972                0.0,
973            ],
974            atmos: [
975                self.atmosphere.fog_density,
976                self.atmosphere.fog_color[0],
977                self.atmosphere.fog_color[1],
978                self.atmosphere.fog_color[2],
979            ],
980            // `x` steruje zanikaniem mgły ku górze. Gdy 0, shader
981            // używa jednolitej gęstości i zachowuje się jak wcześniej.
982            fog: [self.atmosphere.fog_height, 0.0, 0.0, 0.0],
983        };
984        queue.write_buffer(&self.scene_buffer, 0, bytemuck::bytes_of(&uniform));
985        // tablica modeli: jeden wpis na obiekt
986        self.ensure_capacity(device, self.commands.len());
987        if !self.commands.is_empty() {
988            let models: Vec<InstanceModel> = self
989                .commands
990                .iter()
991                .map(|c| InstanceModel {
992                    model: c.model.to_cols_array_2d(),
993                    tint: c.tint,
994                })
995                .collect();
996            queue.write_buffer(&self.models_buffer, 0, bytemuck::cast_slice(&models));
997        }
998
999        let clear = self.clear_color;
1000        let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
1001            label: Some("Uran 3D"),
1002        });
1003        // post-processing: cel HDR i tekstury zależne od rozmiaru okna
1004        self.postfx.ensure_size(device, w, h);
1005        self.postfx.upload_params(
1006            queue,
1007            w,
1008            h,
1009            self.time,
1010            self.sun_screen_uv(),
1011            self.camera.near,
1012            self.camera.far,
1013            self.camera.tan_half_fov(),
1014            self.camera.aspect,
1015        );
1016
1017        // --- pass cieni: sama głębokość z pozycji słońca ---
1018        //
1019        // Musi być PRZED passem głównym, bo shader sceny czyta tę mapę.
1020        // Bind grupy jest OSOBNA (bez mapy cieni), bo wgpu nie pozwala
1021        // w tym samym passie czytać zasobu, który jest jego celem.
1022        if self.shadow.settings.enabled {
1023            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
1024                label: Some("Uran 3D Shadow Pass"),
1025                // Pusty kolor: `color_attachments` musi być pustą listą,
1026                // nie listą z `None` — pass bez color attachmentu.
1027                color_attachments: &[],
1028                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
1029                    view: self.shadow.view(),
1030                    // 1.0 = „nic nie blokuje" — dalej od słońca znaczy
1031                    // większa głębokość, więc 1 to maksimum.
1032                    depth_ops: Some(wgpu::Operations {
1033                        load: wgpu::LoadOp::Clear(1.0),
1034                        store: wgpu::StoreOp::Store,
1035                    }),
1036                    stencil_ops: None,
1037                }),
1038                timestamp_writes: None,
1039                occlusion_query_set: None,
1040            });
1041            pass.set_pipeline(&self.shadow_pipeline);
1042            pass.set_bind_group(0, &self.shadow_bind_group, &[]);
1043
1044            // Ten sam bufor modeli co w passie głównym, więc indeks
1045            // instancji (`base_instance`) musi się zgadzać: `instance_index`
1046            // w WGSL liczy od 1, bo `first_instance` to i+1.
1047            for (i, cmd) in self.commands.iter().enumerate() {
1048                let Some(mesh) = self.meshes.get(cmd.mesh.0 as usize) else {
1049                    continue;
1050                };
1051                mesh.draw(&mut pass, 0, i as u32 + 1);
1052            }
1053        }
1054
1055        {
1056            // klonujemy widok głębokości, żeby pożyczka `&mut pass` nie
1057            // kolidowała z pożyczką `&mut self` w pętli rysowania
1058            // pożyczka widoku głębokości trwa tylko do końca passu
1059            let depth_view = self.depth.as_ref().expect("depth utworzony wyżej");
1060            // Dwóch celów kolorowych w kolejności zgodnej z `targets`
1061            // w pipeline głównym. Nie wolno ich zamienić miejscami —
1062            // wgpu odrzuci pass, a komunikat błędu nie mówi wprost,
1063            // o które chodzi.
1064            let hdr_view = self.postfx.hdr_view();
1065            let gbuf_view = self.postfx.gbuffer_view();
1066            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
1067                label: Some("Uran 3D Pass"),
1068                color_attachments: &[
1069                    Some(wgpu::RenderPassColorAttachment {
1070                        // Rysujemy do celu HDR, NIE na powierzchnię. Na ekran
1071                        // wrzuca nas dopiero `postfx.composite` (tonemapping,
1072                        // bloom, AA) — bez tego poświaty obciąłoby 8 bitów.
1073                        view: hdr_view,
1074                        resolve_target: None,
1075                        ops: wgpu::Operations {
1076                            // 3D czyści sam: 2D dostanie potem LoadOp::Load
1077                            load: wgpu::LoadOp::Clear(wgpu::Color {
1078                                r: clear[0] as f64,
1079                                g: clear[1] as f64,
1080                                b: clear[2] as f64,
1081                                a: clear[3] as f64,
1082                            }),
1083                            store: wgpu::StoreOp::Store,
1084                        },
1085                    }),
1086                    Some(wgpu::RenderPassColorAttachment {
1087                        // G-Bufer: normalna w przestrzeni oka + chropowatość.
1088                        // Czyścimy na czarno, co po liniaryzacji głębokości
1089                        // daje „tło nieba" (normalna = 0, chropowatość = 0,
1090                        // czyli najgładziej) — a niebo i tak nadpisuje ten
1091                        // cel swoim własnym kolorem, więc w praktyce
1092                        // wartość czyszczenia nie jest tu krytyczna.
1093                        view: gbuf_view,
1094                        resolve_target: None,
1095                        ops: wgpu::Operations {
1096                            load: wgpu::LoadOp::Clear(wgpu::Color {
1097                                r: 0.0,
1098                                g: 0.0,
1099                                b: 0.0,
1100                                a: 1.0,
1101                            }),
1102                            store: wgpu::StoreOp::Store,
1103                        },
1104                    }),
1105                ],
1106                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
1107                    view: depth_view,
1108                    depth_ops: Some(wgpu::Operations {
1109                        load: wgpu::LoadOp::Clear(1.0),
1110                        store: wgpu::StoreOp::Store,
1111                    }),
1112                    stencil_ops: None,
1113                }),
1114                timestamp_writes: None,
1115                occlusion_query_set: None,
1116            });
1117
1118            // --- niebo: PIERWSZE, zanim jakakolwiek bryła ---
1119            //
1120            // Kolejność jest tu istotna. Niebo ma wyłączony zapis
1121            // głębokości i porównanie `LessEqual`, więc rysowane później
1122            // nadpisałoby kolor na bryłach (wygrałoby, bo 1.0 <= głębokość
1123            // bryły). Rysowane wcześniej — wypełnia tło i zostaje
1124            // zasłonięte przez bryły, które mają mniejszą głębokość.
1125            if self.atmosphere.sky {
1126                pass.set_pipeline(&self.sky_pipeline);
1127                // Ta sama grupa 0 co dla brył: uniform sceny z kierunkiem
1128                // słońca i macierzą `inv_view_proj`.
1129                pass.set_bind_group(0, &self.bind_group, &[]);
1130                // 3 wierzchołki = pełny prostokąt, bez bufora wierzchołków.
1131                pass.draw(0..3, 0..1);
1132            }
1133
1134            pass.set_pipeline(&self.pipeline);
1135            pass.set_bind_group(0, &self.bind_group, &[]);
1136
1137            let mut drawn = 0usize;
1138            for (i, cmd) in self.commands.iter().enumerate() {
1139                let Some(mesh) = self.meshes.get(cmd.mesh.0 as usize) else {
1140                    // zły identyfikator — cicho pomijamy, bo walidacja
1141                    // wgpu nie zna indeksów naszych siatek
1142                    continue;
1143                };
1144                // Bind grupy 1 = materiał. Musi być ustawiona PRZED
1145                // draw, bo to osobny stan potoku: poprzednia komenda
1146                // zostawiłaby tu swój materiał i obiekt zostałby
1147                // pomalowany cudzym kolorem.
1148                if let Some(bg) = self.materials.bind_group(cmd.material) {
1149                    pass.set_bind_group(1, bg, &[]);
1150                }
1151                // `base_instance` = i+1, bo WGSL liczy `instance_index` od 1
1152                mesh.draw(&mut pass, 0, i as u32 + 1);
1153                drawn += 1;
1154            }
1155            self.last_draw_count = drawn;
1156        }
1157
1158        // --- kopia głębokości dla post-processingu ---
1159        //
1160        // MUSI być poza blokiem powyżej: w tym samym passie nie wolno
1161        // czytać tekstury, która jest jego celem. Kopia do osobnej
1162        // tekstury (z `COPY_DST`) jest już legalna, bo tamten pass
1163        // się skończył.
1164        //
1165        // Głębokość jest potrzebna dla: konturów (krawędzie obiektów),
1166        // DoF (odległość), SSR (kolidencja promienia) i mgły w passie
1167        // kompozycji. Bez niej te efekty miałyby zgadywać geometrię
1168        // z samego koloru.
1169        if self.samples == 1 {
1170            // Przy MSAA kopia byłaby niemożliwa (głębokość
1171            // multisamplingowa nie da się skopiować jako 2D), więc
1172            // efekty ekranowe po prostu nie włączamy. Gry 3D ustawiają
1173            // `.samples(1)`, więc ścieżka jest domyślna.
1174            if let Some(tex) = self.depth_tex.as_ref() {
1175                self.postfx.copy_depth(&mut encoder, tex);
1176            }
1177        }
1178
1179        // post-processing na powierzchnię: bloom (3 passy) + kompozycja
1180        self.postfx.composite(&mut encoder, target);
1181
1182        queue.submit(Some(encoder.finish()));
1183    }
1184}

1186/// Bufor głębokości sceny.
1187///
1188/// Osobna funkcja, bo głębokość tworzymy w dwóch miejscach
1189/// (`with_depth` przy starcie i `render_pass` przy resize) i obie
1190/// ścieżki muszą zgadzać się co do formatu oraz `sample_count` —
1191/// rozjazd kończy się odrzuceniem potoku przez walidację, bez
1192/// wskazania, która z dwóch ścieżek się rozjechała.
1193fn depth_texture(device: &wgpu::Device, w: u32, h: u32, samples: u32) -> wgpu::Texture {
1194    device.create_texture(&wgpu::TextureDescriptor {
1195        label: Some("Uran 3D Depth"),
1196        size: wgpu::Extent3d {
1197            width: w,
1198            height: h,
1199            depth_or_array_layers: 1,
1200        },
1201        mip_level_count: 1,
1202        sample_count: samples,
1203        dimension: wgpu::TextureDimension::D2,
1204        format: wgpu::TextureFormat::Depth32Float,
1205        // `COPY_SRC` jest potrzebne do skopiowania głębokości dla
1206        // post-processingu (`PostFx::copy_depth`). Bez tego flagi
1207        // wgpu odrzuci `copy_texture_to_texture` dopiero w trakcie
1208        // renderowania.
1209        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::COPY_SRC,
1210        view_formats: &[],
1211    })
1212}

1214/// Bufor tablicy modeli o zadanej pojemności.
1215fn create_models_buffer(device: &wgpu::Device, capacity: usize) -> wgpu::Buffer {
1216    device.create_buffer(&wgpu::BufferDescriptor {
1217        label: Some("Uran 3D Models"),
1218        size: (std::mem::size_of::<InstanceModel>() * capacity.max(1)) as u64,
1219        usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
1220        mapped_at_creation: false,
1221    })
1222}

1224/// Bind grupa grupy 0: uniform sceny (0) + tablica modeli (1) +
1225/// mapa cieni (2) + próbnik porównawczy (3).
1226///
1227/// Budujemy ją z **nazwy** layoutu zamiast z pozycji wpisów, bo
1228/// `create_bind_group` wymaga tylu wpisów, ile jest w layoutcie. Layout
1229/// passu cienia ma tylko 2, więc przekazanie pustego `TextureView`
1230/// w `Option` pozwala obsłużyć oba przypadki jedną funkcją.
1231fn make_bind_group(
1232    device: &wgpu::Device,
1233    layout: &wgpu::BindGroupLayout,
1234    scene: &wgpu::Buffer,
1235    models: &wgpu::Buffer,
1236    shadow_tex: Option<&wgpu::TextureView>,
1237    shadow_samp: Option<&wgpu::Sampler>,
1238) -> wgpu::BindGroup {
1239    let mut entries = vec![
1240        wgpu::BindGroupEntry {
1241            binding: 0,
1242            resource: scene.as_entire_binding(),
1243        },
1244        wgpu::BindGroupEntry {
1245            binding: 1,
1246            resource: models.as_entire_binding(),
1247        },
1248    ];
1249    // Dodajemy 2 i 3 TYLKO gdy layout je deklaruje. Wpisania pustego
1250    // `Option` do bind grupy wgpu odrzuci z „binding not found".
1251    if let (Some(tex), Some(samp)) = (shadow_tex, shadow_samp) {
1252        entries.push(wgpu::BindGroupEntry {
1253            binding: 2,
1254            resource: wgpu::BindingResource::TextureView(tex),
1255        });
1256        entries.push(wgpu::BindGroupEntry {
1257            binding: 3,
1258            resource: wgpu::BindingResource::Sampler(samp),
1259        });
1260    }
1261    device.create_bind_group(&wgpu::BindGroupDescriptor {
1262        label: Some("Uran 3D Bind Group"),
1263        layout,
1264        entries: &entries,
1265    })
1266}
