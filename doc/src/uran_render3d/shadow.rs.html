1//! Mapowanie cieni kierunkowych: macierz światła i mapa głębokości.
2//!
3//! ## Jak to działa
4//!
5//! Cienie są **kierunkowe** (jedno słońce), bo taki model pasuje do
6//! [`crate::scene::Lighting`], który trzyma jeden `light_dir`. Światło
7//! kierunkowe zamienia się w punktowe tylko wtedy, gdy umieścimy je
8//! bardzo daleko — a robimy właśnie to.
9//!
10//! 1. **Dobieramy kadr.** AABB całej sceny dostaje kwadratowy obszar
11//!    ortograficzny liczony wokół jego środka. Kwadrat, bo mapa ma
12//!    kwadratowy format — prostokąt dałby niejednakową rozdzielczość
13//!    na piksel w obu osiach, czyli rozjeżdżające cienie.
14//! 2. **Rysujemy głębokość z pozycji słońca** do tekstury `Depth32Float`.
15//!    Pass zapamiętuje WYŁĄCZNIE głębokość, bez color attachmentu.
16//! 3. **Porównujemy** głębokość piksela z mapą, z 9 próbek (PCF 3×3)
17//!    wyliczamy uśredniony wynik jasny/uciemniony.
18//!
19//! ## Dlaczego `Depth32Float`, a nie `Depth24Plus`
20//!
21//! Tekstura głębokości jest jednocześnie buforem depth i źródłem do
22//! próbkowania. `Depth32Float` daje pełną precyzję float, więc
23//! porównanie z bias-em daje stabilne krawędzie cienia. `Depth24Plus`
24//! po zapisie i odczycie potrafi dawać „schodki" na płaskich
25//! powierzchniach równoległych do kierunku światła.
26//!
27//! ## Bias — najczęstsza przyczyna zepsutych cieni
28//!
29//! Bez biasu powierzchnia „oświetla sama siebie" (*shadow acne*): piksel
30//! porównuje głębokość z samą sobą, wynosi 0, a cała płaszczyzna
31//! dostaje czarne linie. Rozwiązujemy to na dwa sposoby naraz:
32//!
33//! * **offset w przestrzeni światła** — przesuwamy próbkę wzdłuż
34//!   kierunku do słońca o `depth_bias`,
35//! * **normal-offset** — odsuwamy próbkę o `normal * normal_offset`,
36//!   co pomaga przy powierzchniach odchylonych od osi światła.
37//!
38//! ## Koszt
39//!
40//! Jeden dodatkowy pass rysujący tę samą geometrię — dla 300 obiektów
41//! to kilka tysięcy trójkątów ponownie, na GPU praktycznie za darmo.
42

49/// Rozdzielczość mapy cieni w pikselach na krawędź.
50///
51/// 2048² to standardowy kompromis: 512 daje widoczne schodki na
52/// krawędziach, 4096 kosztuje cztery razy więcej pamięci i czasu
53/// bez zauważalnej różnicy przy kadrze typowym dla gry.
54pub const SHADOW_RES: u32 = 2048;
55
56/// Odległość od obiektu, na której stawiamy oko cienia.
57///
58/// Światło jest **kierunkowe**, więc jego pozycja nie ma znaczenia —
59/// liczy się tylko kierunek. Ten dystans daje tyle, że scena z
60/// odległości setek metrów nadal mieści się między płaszczyznami nożyc.
61const EYE_BACKOFF: f32 = 400.0;
62
63/// Jak mocno cienie przyciemniają światło bezpośrednie.
64#[derive(Debug, Clone, Copy)]
65pub struct ShadowSettings {
66    /// Przesunięcie próbki w głębokości (0 = wyłączony).
67    pub depth_bias: f32,
68    /// Przesunięcie próbki wzdłuż normalnej, w jednostkach świata.
69    pub normal_offset: f32,
70    /// 0.0 = brak cienia, 1.0 = pełny cień.
71    pub strength: f32,
72    /// Promień PCF w texelach.
73    pub radius: f32,
74    /// Czy w ogóle rysujemy cienie.
75    pub enabled: bool,
76}
77
78impl Default for ShadowSettings {
79    fn default() -> Self {
80        Self {
81            // 0.0016 w jednostkach NDC z głębokości. Przy `Depth32Float`
82            // zakres nożyc to 0..1, więc bias musi być wyrażony w tych
83            // samych jednostkach — za mały daje acne, za duży „odkleja"
84            // cień od obiektu i robi ciemną obwódkę.
85            depth_bias: 0.0016,
86            // 0.035 m to grubość „skórki" cienia dla obiektów tej skali
87            // (farma, budynki, postać). Dla 2-metrowego domu to 1,7%
88            // wysokości — niewidoczne, ale wystarczające.
89            normal_offset: 0.035,
90            // Pełny cień, ale shader miesza go ze światłem rozproszonym,
91            // więc w cieniu widać niebieskawe otoczenie, nie czerń.
92            strength: 0.78,
93            radius: 1.5,
94            enabled: true,
95        }
96    }
97}

99/// AABB świata, z którego liczymy kadr cienia.
100#[derive(Debug, Clone, Copy)]
101pub struct SceneBounds {
102    pub lo: Vec3,
103    pub hi: Vec3,
104}
105
106impl SceneBounds {
107    /// Pusty zakres gotowy do [`Self::expand`].
108    ///
109    /// `lo`/`hi` z przeciwnymi nieskończonościami: pierwszy punkt
110    /// rozszerzy zakres do samego siebie, a nie zostawi zerowy.
111    pub fn empty() -> Self {
112        Self {
113            lo: Vec3::splat(f32::INFINITY),
114            hi: Vec3::splat(f32::NEG_INFINITY),
115        }
116    }
117
118    /// Wstawia punkt do AABB.
119    pub fn expand(&mut self, p: Vec3) {
120        self.lo = self.lo.min(p);
121        self.hi = self.hi.max(p);
122    }
123
124    /// Czy zakres ma sens (co najmniej jeden punkt).
125    pub fn is_valid(&self) -> bool {
126        self.lo.x.is_finite() && self.lo.x <= self.hi.x
127    }
128
129    /// Środek AABB.
130    pub fn center(&self) -> Vec3 {
131        (self.lo + self.hi) * 0.5
132    }
133
134    /// Największa z trzech długości.
135    pub fn max_extent(&self) -> f32 {
136        (self.hi - self.lo).max_element().max(0.001)
137    }
138
139    /// Promień kuli opisanej na AABB: połowa długości przekątnej.
140    ///
141    /// Kadr cienia to KWADRAT obrócony dookoła kierunku światła, więc
142    /// musi objąć całą kulę opisaną na AABB — a nie połowę
143    /// najdłuższego boku. Różnica jest bardzo widoczna: dla AABB
144    /// 20×12×18 m `max_extent/2` to 10 m, a promień kuli ~14,7 m.
145    /// Przy zbyt małym kadrze narożniki bryły wypadają poza mapę,
146    /// ich `in_range` daje fałsz i fragmenty są bez cienia.
147    pub fn radius(&self) -> f32 {
148        ((self.hi - self.lo).length() * 0.5).max(0.001)
149    }
150}

152/// Tekstura + próbnik + macierz światła.
153pub struct ShadowMap {
154    /// Widok mapy głębokości. Trzymamy go w strukturze, bo `bind_group`
155    /// wskazuje na tę teksturę — wgpu nie współdzieli zasobów.
156    view: wgpu::TextureView,
157    /// Prównywarka (`LessEqual`), nie zwykły sampler.
158    sampler: wgpu::Sampler,
159    pub settings: ShadowSettings,
160    /// Obliczona w klatce, dla debugu i testów.
161    light_view_proj: Mat4,
162}
163
164impl ShadowMap {
165    /// Tworzy teksturę i próbnik porównujący.
166    pub fn new(device: &wgpu::Device, settings: ShadowSettings) -> Self {
167        let texture = device.create_texture(&wgpu::TextureDescriptor {
168            label: Some("Uran 3D Shadow Map"),
169            size: wgpu::Extent3d {
170                width: SHADOW_RES,
171                height: SHADOW_RES,
172                depth_or_array_layers: 1,
173            },
174            mip_level_count: 1,
175            // 1, nie `samples` z renderera głównego: pass cienia nie ma
176            // MSAA, bo rysujemy wyłącznie głębokość. MSAA na samej
177            // głębokości nic by nie dało poza kosztem.
178            sample_count: 1,
179            dimension: wgpu::TextureDimension::D2,
180            format: wgpu::TextureFormat::Depth32Float,
181            // `TEXTURE_BINDING` jest konieczne: tę samą teksturę czyta
182            // shader sceny. Bez tej flagi walidacja wgpu odrzuci bind group.
183            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
184            view_formats: &[],
185        });
186
187        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
188            label: Some("Uran 3D Shadow Sampler"),
189            // `compare` czyni z samplera PRÓBOWNIK PORÓWNAWCZY.
190            // `LessEqual` znaczy: „oświetlony, jeżeli moja głębokość
191            // jest bliższa niż zapisana" — standardowa konwencja shadow
192            // map. Bez `compare` tekstury depth nie da się próbkować w WGSL.
193            compare: Some(wgpu::CompareFunction::LessEqual),
194            mag_filter: wgpu::FilterMode::Linear,
195            min_filter: wgpu::FilterMode::Linear,
196            // `ClampToEdge` zamiast `Repeat`: poza kadrzem ma być
197            // oświetlenie (kadr dobieramy do całej sceny), a nie wzór.
198            address_mode_u: wgpu::AddressMode::ClampToEdge,
199            address_mode_v: wgpu::AddressMode::ClampToEdge,
200            ..Default::default()
201        });
202
203        Self {
204            view: texture.create_view(&Default::default()),
205            sampler,
206            settings,
207            light_view_proj: Mat4::IDENTITY,
208        }
209    }
210
211    pub fn view(&self) -> &wgpu::TextureView {
212        &self.view
213    }
214
215    pub fn sampler(&self) -> &wgpu::Sampler {
216        &self.sampler
217    }
218
219    /// Macierz świata -> przestrzeń cienia, liczona w [`Self::fit`].
220    pub fn light_view_proj(&self) -> Mat4 {
221        self.light_view_proj
222    }
223
224    /// Rozmiar jednego texela w skali UV — do skalowania promienia PCF.
225    pub fn texel_uv(&self) -> f32 {
226        1.0 / SHADOW_RES as f32
227    }
228
229    /// Liczy macierz `light_view_proj` dla podanych granic sceny.
230    ///
231    /// Światło jest kierunkowe, więc stawiamy oko daleko wzdłuż
232    /// `light_dir` (który wskazuje OD źródła) i patrzymy na środek kadru.
233    pub fn fit(&mut self, bounds: SceneBounds, light_dir: Vec3) {
234        if !bounds.is_valid() {
235            // Pusta scena: trzymamy poprzednią macierz, żeby nie mrugać.
236            return;
237        }
238        self.light_view_proj = light_matrix(bounds, light_dir);
239    }
240}

242/// Buduje macierz projekcji cienia dla kierunku `light_dir`.
243///
244/// `light_dir` wskazuje OD źródła (tak jak [`crate::scene::Lighting`]),
245/// więc oko stawiamy w `center + light_dir * EYE_BACKOFF` i patrzymy
246/// na środek kadru.
247///
248/// ## Dlaczego kwadratowy kadr
249///
250/// Używamy jednego `half` dla obu osi. Gdyby każda oś dostała swój
251/// promień, obraz postaciującej (`half / (far - near)`) różniłby się
252/// w osi X i Y — ten sam obiekt dawałby inne zniekształcenie zależnie
253/// od tego, w którą stronę na niego patrzymy. Kwadrat eliminuje ten
254/// problem za cenę trochę zmarnowanych texeli.
255pub fn light_matrix(bounds: SceneBounds, light_dir: Vec3) -> Mat4 {
256    let center = bounds.center();
257    // Kadr to KWADRAT obrócony dookoła osi światła, więc bierzemy
258    // promień kuli opisanej na AABB, a nie połowę najdłuższego boku.
259    // +1 m marginesu na błąd precyzji głębokości przy samej krawędzi.
260    let half = bounds.radius() + 1.0;
261
262    // Światło kierunkowe: kierunek musi być jednostkowy, bo `look_at`
263    // normalizuje różnicę oczu i celu. Zerowy wektor daje NaN, a ten
264    // NaN trafia do macierzy i psuje CAŁĄ scenę, nie tylko cień.
265    let dir = light_dir.normalize_or_zero();
266    let dir = if dir == Vec3::ZERO {
267        Vec3::new(0.0, 1.0, 0.0)
268    } else {
269        dir
270    };
271
272    // Wektor „w górę" musi być NIErównoległy do kierunku patrzenia.
273    // Przy słońcu prosto w dół (`dir` = +Y) wektor `up` = +Y jest
274    // równoległy do osi oka i `look_at` zwraca macierz pełną NaN —
275    // wtedy znika CAŁA scena, nie tylko cień. Dlatego wybieramy
276    // dowolną oś nieprostopadłą: +Z, gdy świeci z góry.
277    let up = if dir.dot(Vec3::Y).abs() > 0.999 {
278        Vec3::Z
279    } else {
280        Vec3::Y
281    };
282
283    let eye = center + dir * EYE_BACKOFF;
284    let view = Mat4::look_at_rh(eye, center, up);
285
286    // Nożyce: od oko do `far`. `near` dodatnie i mniejsze niż
287    // odległość, żeby scena nie wyszła za płaszczyznę.
288    let near = 0.1;
289    let far = EYE_BACKOFF + half * 2.0;
290    let proj = Mat4::orthographic_rh(-half, half, -half, half, near, far);
291    proj * view
292}

298    fn scene_at(center: Vec3, size: Vec3) -> SceneBounds {
299        let mut b = SceneBounds::empty();
300        b.expand(center - size * 0.5);
301        b.expand(center + size * 0.5);
302        b
303    }

305    /// Wszystkie 8 narożników AABB jako wektory jednorodne.
306    fn corners(b: SceneBounds) -> Vec<Vec4> {
307        (0..8)
308            .map(|i| {
309                let p = Vec3::new(
310                    if i & 1 == 0 { b.lo.x } else { b.hi.x },
311                    if i & 2 == 0 { b.lo.y } else { b.hi.y },
312                    if i & 4 == 0 { b.lo.z } else { b.hi.z },
313                );
314                Vec4::new(p.x, p.y, p.z, 1.0)
315            })
316            .collect()
317    }

319    #[test]
320    fn expanding_once_makes_bounds_valid() {
321        let mut b = SceneBounds::empty();
322        assert!(!b.is_valid());
323        b.expand(Vec3::new(3.0, 4.0, 5.0));
324        assert!(b.is_valid());
325        assert_eq!(b.lo, Vec3::new(3.0, 4.0, 5.0));
326        assert_eq!(b.hi, Vec3::new(3.0, 4.0, 5.0));
327    }

337    /// Najważniejszy test: CAŁA scena musi zmieścić się w kadrze.
338    ///
339    /// Obiekt wychodzący poza mapę cienia albo znika (brak shadow map),
340    /// albo rzuca cień, którego nie ma — w obu przypadkach wygląda to
341    /// jak dziura w ścianach budynku.
342    #[test]
343    fn whole_scene_fits_into_shadow_frustum() {
344        let bounds = scene_at(Vec3::new(5.0, 0.0, -3.0), Vec3::new(20.0, 12.0, 18.0));
345        let m = light_matrix(bounds, Vec3::new(0.4, 0.6, 0.7));
346
347        for (i, p) in corners(bounds).into_iter().enumerate() {
348            let clip = m * p;
349            // `w` dodatnie — za nim kryją się nożyce, a ujemne
350            // oznaczałoby punkt za kamerą cienia.
351            assert!(clip.w > 0.0, "narożnik {i} jest za kamerą cienia: {clip:?}");
352            let ndc = clip.truncate() / clip.w;
353            assert!(
354                ndc.x.abs() <= 1.0 && ndc.y.abs() <= 1.0,
355                "narożnik {i} wypadł z kadru: {ndc:?}"
356            );
357            assert!(
358                (-1.0..=1.0).contains(&ndc.z),
359                "narożnik {i} poza nożycami: {ndc:?}"
360            );
361        }
362    }
