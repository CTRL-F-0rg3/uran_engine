11/// Półrozmiar mapy w jednostkach świata.
12pub const ARENA_HALF: f32 = 90.0;
13
14/// Generator xorshift32 — powtarzalny, żeby demo wyglądało tak samo
15/// po każdym uruchomieniu (ułatwia porównywanie zrzutów).
16#[derive(Debug, Clone)]
17pub struct Rng(u32);
18
19impl Rng {
20    pub fn new(seed: u32) -> Self {
21        Rng(seed | 1) // 0 to punkt stały — xorshift z zerem nie rusza
22    }
23
24    pub fn next_u32(&mut self) -> u32 {
25        let mut x = self.0;
26        x ^= x << 13;
27        x ^= x >> 17;
28        x ^= x << 5;
29        self.0 = x;
30        x
31    }
32
33    /// Liczba z przedziału `[0, 1)`.
34    pub fn f32(&mut self) -> f32 {
35        (self.next_u32() & 0x00FF_FFFF) as f32 / 0x00FF_FFFF as f32
36    }
37
38    /// Liczba z przedziału `[min, max)`.
39    pub fn range(&mut self, min: f32, max: f32) -> f32 {
40        min + self.f32() * (max - min)
41    }
42}

44/// Teren analityczny — wysokość z kilku fal, bez siatki wierzchołków.
45///
46/// Dzięki temu czołg stoi dokładnie na gruncie, a pobranie wysokości
47/// to jedno mnożenie i suma sinusoid zamiast przeszukiwania siatki.
48#[derive(Debug, Clone, Copy)]
49pub struct Terrain {
50    pub phase_x: f32,
51    pub phase_z: f32,
52}
53
54impl Terrain {
55    pub fn new(rng: &mut Rng) -> Self {
56        Self {
57            phase_x: rng.range(0.0, std::f32::consts::TAU),
58            phase_z: rng.range(0.0, std::f32::consts::TAU),
59        }
60    }
61
62    /// Wysokość gruntu. Zawsze `>= 0` — czołgi nie potrafią pływać.
63    pub fn height(&self, x: f32, z: f32) -> f32 {
64        let a = (x * 0.045 + self.phase_x).sin();
65        let b = (z * 0.037 + self.phase_z).sin();
66        let c = ((x + z) * 0.021).sin();
67        // kombinacja fal o różnych długościach: niska i wysoka
68        let h = 1.5 * a + 0.9 * b + 0.6 * c;
69        h.max(0.0)
70    }
71
72    /// Nachylenie gruntu (normalna powierzchni) w punkcie.
73    ///
74    /// Różniczka centralna — taniej niż wzór analityczny, a przy tej
75    /// skali terenu dokładność w zupełności wystarcza.
76    pub fn normal(&self, x: f32, z: f32) -> Vec3 {
77        let e = 0.5;
78        let hx = self.height(x + e, z) - self.height(x - e, z);
79        let hz = self.height(x, z + e) - self.height(x, z - e);
80        Vec3::new(-hx, 2.0 * e, -hz).normalize_or_zero()
81    }
82}

84/// Cały świat gry.
85pub struct World {
86    pub terrain: Terrain,
87    pub player: Tank,
88    pub enemies: Vec<Tank>,
89    pub shells: Vec<Shell>,
90    /// Iskry po trafieniu: `(pozycja, prędkość, czas życia)`.
91    pub sparks: Vec<(Vec3, Vec3, f32)>,
92    pub player_hits: u32,
93    pub enemy_kills: u32,
94    pub rng: Rng,
95}
96
97impl World {
98    /// Tworzy świat: gracz na środku, `enemy_count` przeciwników
99    /// rozsianych na pierścieniu (nie w jednym miejscu).
100    pub fn new(enemy_count: usize) -> Self {
101        let mut rng = Rng::new(0xC0FFEE);
102        let terrain = Terrain::new(&mut rng);
103
104        let player_pos = Vec3::new(0.0, terrain.height(0.0, 0.0), 30.0);
105        let mut player = Tank::new(player_pos, Team::Player);
106        // twarzą w stronę środka pierścienia, gdzie są przeciwnicy
107        player.heading = std::f32::consts::PI;
108        // lufa wyrównana z kadłubem — inaczej czołg stałby bokiem do
109        // kierunku jazdy i kamera patrzyłaby na jego bok
110        player.turret_yaw = player.heading;
111
112        let mut enemies = Vec::with_capacity(enemy_count);
113        for i in 0..enemy_count {
114            let angle = i as f32 / enemy_count.max(1) as f32 * std::f32::consts::TAU;
115            let radius = rng.range(45.0, ARENA_HALF - 15.0);
116            let x = angle.cos() * radius;
117            let z = angle.sin() * radius;
118            let mut e = Tank::new(Vec3::new(x, terrain.height(x, z), z), Team::Enemy);
119            e.heading = (player_pos.x - x).atan2(player_pos.z - z);
120            enemies.push(e);
121        }
122
123        Self {
124            terrain,
125            player,
126            enemies,
127            shells: Vec::new(),
128            sparks: Vec::new(),
129            player_hits: 0,
130            enemy_kills: 0,
131            rng,
132        }
133    }
134
135    /// Przyciąga czołg do terenu (tylko `World` zna teren).
136    pub fn snap_to_ground(&self, tank: &mut Tank) {
137        snap_to_ground(&self.terrain, tank);
138    }
139
140    /// Czy gra się skończyła.
141    pub fn is_over(&self) -> bool {
142        !self.player.is_alive()
143    }
144
145    /// Ile przeciwników zostało.
146    pub fn enemies_left(&self) -> usize {
147        self.enemies.iter().filter(|e| e.is_alive()).count()
148    }
149
150    /// Odpala pocisk gracza (przy naciśnięciu LPM).
151    pub fn player_fire(&mut self) {
152        if let Some(s) = self.player.fire() {
153            self.shells.push(s);
154        }
155    }
156}

158/// Przyciąga czołg do powierzchni gruntu.
159///
160/// WOLNA funkcja, a nie metoda `World`, bo wywołujemy ją w pętli
161/// `self.enemies.iter_mut()` — metoda brałaby `&self` i kolidowała
162/// z pożyczką `&mut self.enemies`.
163fn snap_to_ground(terrain: &Terrain, tank: &mut Tank) {
164    tank.pos.y = terrain.height(tank.pos.x, tank.pos.z);
165}
166
167/// Trzyma czołg w granicach areny (ta sama uwaga co wyżej).
168fn clamp_to_arena(tank: &mut Tank) {
169    let r = tank.radius() * 0.6;
170    tank.pos.x = tank.pos.x.clamp(-ARENA_HALF + r, ARENA_HALF - r);
171    tank.pos.z = tank.pos.z.clamp(-ARENA_HALF + r, ARENA_HALF - r);
172}
173
174/// Zawijanie różnicy kątów do `[-PI, PI]`.
175///
176/// Bez tego czołg skręcałby w drugą stronę, gdy przeciwnik jest „za
177/// plecami" — obrót o 350° zamiast o -10°.
178fn wrap_angle(mut a: f32) -> f32 {
179    while a > std::f32::consts::PI {
180        a -= std::f32::consts::TAU;
181    }
182    while a < -std::f32::consts::PI {
183        a += std::f32::consts::TAU;
184    }
185    a
186}
187
188/// Długość wektora w płaszczyźnie XZ.
189fn flat_len(x: f32, z: f32) -> f32 {
190    (x * x + z * z).sqrt()
191}
192
193/// Test trafienia punktu w kulę.
194fn sphere_hit(point: Vec3, center: Vec3, radius: f32) -> bool {
195    (point - center).length() <= radius
196}

198impl World {
199    /// AI przeciwników: podjeżdża do gracza, obraca wieżę, strzela.
200    fn update_ai(&mut self, dt: f32) {
201        // Stan świata kopiujemy do zmiennych PRZED pożyczką `enemies`.
202        // `self.enemies.iter_mut()` trzyma `&mut self.enemies`, więc
203        // każde wywołanie `self.shells.push(..)` w pętli byłoby drugą
204        // pożyczką tego samego `&mut self` — kompilator to odrzuci.
205        let player_pos = self.player.pos;
206        let player_turret = self.player.turret_position();
207        let player_alive = self.player.is_alive();
208        let mut shells: Vec<Shell> = Vec::new();
209
210        for e in self.enemies.iter_mut() {
211            if !e.is_alive() {
212                continue;
213            }
214            let to_player = player_pos - e.pos;
215            let dist = flat_len(to_player.x, to_player.z);
216
217            // --- kadłub: skręcamy w stronę gracza
218            let steer = wrap_angle(to_player.x.atan2(to_player.z) - e.heading).clamp(-1.0, 1.0);
219            // podjeżdżamy z daleka, cofamy się w zwarciu (inaczej wbijają
220            // się w gracza i blokują mu strzał)
221            let throttle = if dist > 25.0 { 0.55 } else { -0.25 };
222            e.update(dt, throttle, steer);
223
224            // --- wieża: celujemy w gracza z ograniczoną prędkością obrotu
225            let aim = player_turret - e.turret_position();
226            let tdiff = wrap_angle(aim.x.atan2(aim.z) - e.turret_yaw);
227            // maks. 2 rad/s — daje graczowi czas na reakcję i celowanie
228            e.aim(tdiff.clamp(-2.0 * dt, 2.0 * dt), 0.0);
229            // celujemy w kadłub (niżej niż wieża), więc lekki pitch w dół
230            let want_pitch = (aim.y - 0.4).atan2(dist.max(0.1));
231            e.aim(0.0, (want_pitch - e.gun_pitch).clamp(-0.8 * dt, 0.8 * dt));
232
233            // --- ogień: w zasięgu, wycelowany i gotowy
234            if dist < e.config.range && tdiff.abs() < 0.12 && player_alive && e.can_fire() {
235                if let Some(s) = e.fire() {
236                    shells.push(s);
237                }
238            }
239        }
240        self.shells.append(&mut shells);
241
242        // przyciąganie do terenu dopiero po zwolnieniu pożyczki `enemies`
243        let terrain = self.terrain;
244        for e in self.enemies.iter_mut() {
245            clamp_to_arena(e);
246            snap_to_ground(&terrain, e);
247        }
248    }
249
250    /// Krok pocisków: ruch, kolizje z czołgami i ziemią.
251    fn update_shells(&mut self, dt: f32) {
252        // Całą listę zabieramy przez `std::mem::take`, żeby pętla nie
253        // pożyczała `&mut self.shells` w chwili, gdy wołamy `add_spark`.
254        let mut incoming: Vec<Shell> = std::mem::take(&mut self.shells);
255        let mut alive: Vec<Shell> = Vec::with_capacity(incoming.len());
256
257        for mut s in incoming.drain(..) {
258            s.ttl -= dt;
259            if s.ttl <= 0.0 {
260                continue;
261            }
262            s.pos += s.vel * dt;
263
264            // --- kolizja z ziemią
265            if s.pos.y <= self.terrain.height(s.pos.x, s.pos.z) {
266                self.add_spark(s.pos, 0.55);
267                continue;
268            }
269
270            // --- pocisk gracza: szukamy trafienia w przeciwnika
271            if s.from_player {
272                let mut hit = false;
273                for e in self.enemies.iter_mut() {
274                    if !e.is_alive() {
275                        continue;
276                    }
277                    if sphere_hit(s.pos, e.pos, e.radius() * 0.9) {
278                        let dmg = e.config.damage;
279                        let destroyed = e.take_damage(dmg);
280                        if destroyed {
281                            self.enemy_kills += 1;
282                        } else {
283                            self.player_hits += 1;
284                        }
285                        hit = true;
286                        break;
287                    }
288                }
289                if hit {
290                    self.add_spark(s.pos, 1.0);
291                    continue;
292                }
293            } else if self.player.is_alive()
294                && sphere_hit(s.pos, self.player.pos, self.player.radius() * 0.9)
295            {
296                // --- pocisk przeciwnika trafia gracza
297                let dmg = self.player.config.damage;
298                self.player.take_damage(dmg);
299                self.add_spark(s.pos, 1.0);
300                continue;
301            }
302
303            alive.push(s);
304        }
305
306        self.shells = alive;
307    }
308
309    /// Iskry po trafieniu: kilka cząstek w losowych kierunkach.
310    fn add_spark(&mut self, pos: Vec3, intensity: f32) {
311        for _ in 0..6 {
312            let v = Vec3::new(
313                self.rng.range(-1.0, 1.0),
314                self.rng.range(0.2, 1.4),
315                self.rng.range(-1.0, 1.0),
316            ) * (2.0 * intensity);
317            self.sparks.push((pos, v, 0.45 + 0.3 * self.rng.f32()));
318        }
319    }
320
321    fn update_sparks(&mut self, dt: f32) {
322        self.sparks.retain_mut(|(pos, vel, life)| {
323            *life -= dt;
324            *pos += *vel * dt;
325            vel.y -= 9.0 * dt; // grawitacja
326            *life > 0.0
327        });
328    }
329
330    /// Pełny klatka świata. `throttle`/`steer` dotyczą gracza.
331    pub fn update(&mut self, dt: f32, throttle: f32, steer: f32) {
332        self.player.update(dt, throttle, steer);
333        clamp_to_arena(&mut self.player);
334        snap_to_ground(&self.terrain, &mut self.player);
335        self.update_ai(dt);
336        self.update_shells(dt);
337        self.update_sparks(dt);
338    }
339}
