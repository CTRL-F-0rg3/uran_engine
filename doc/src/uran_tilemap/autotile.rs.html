1//! Autotiling: dobieranie kafla brzegowego do otoczenia.
2//!
3//! Bez autotilingu woda jest prostokątem z widocznymi narożnikami. Klasyczne
4//! rozwiązanie („blob", „terrain autotile”) to **16 wariantów** opisanych
5//! czterema bitami: czy sąsiedzi góra/dół/lewo/prawa należą do tego samego
6//! terenu. Kafel środkowy to 0b1111, narożnik zewnętrzny to 0b0101 itd.
7//!
8//! ## Dlaczego 4 bity, a nie 8
9//!
10//! Pełne 256 wariantów liczyłoby też narożniki „przekątne", których w
11//! klasycznym stylu pixel art w ogóle się nie rysuje. Cztery bity dają 16
12//! kafli na teren — tyle właśnie zawiera arkusz `terrains` z assets gry.
13
14use crate::layer::TileLayer;
15
16/// Sposób liczenia maski sąsiadów.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum MatchMode {
19    /// Sąsiad musi mieć **dokładnie** ten sam indeks kafla.
20    Exact,
21    /// Liczy się tylko, czy sąsiad jest pusty — pozwala łączyć różne kafle
22    /// tego samego terenu (np. dwa warianty trawy) w jeden spójny brzeg.
23    NonEmpty,
24}
25
26/// Opis zestawu 16 kafli tworzących jeden teren.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub struct AutoTile {
29    /// Indeks pierwszego kafla wariantów; kolejne 16 leżą w kolejności
30    /// `mask = 0, 1, 2, ...` (jak `terrains` w assets).
31    pub base: u32,
32    /// Czy środek terenu ma rysować wariant 0b1111, czy bazowy kafel.
33    pub include_self: bool,
34    /// Jak porównywać sąsiadów.
35    pub mode: MatchMode,
36}
37
38impl AutoTile {
39    /// Autotiling po identycznym indeksie kafla.
40    pub const fn exact(base: u32) -> Self {
41        Self {
42            base,
43            include_self: true,
44            mode: MatchMode::Exact,
45        }
46    }
47
48    /// Autotiling po „kafel pusty / niepusty".
49    pub const fn by_presence(base: u32) -> Self {
50        Self {
51            base,
52            include_self: true,
53            mode: MatchMode::NonEmpty,
54        }
55    }
56
57    /// Maska sąsiedów wokół kafla (x, y) w warstwie.
58    ///
59    /// Bity, od najmniej do najbardziej znaczącego:
60    /// `1` góra, `2` prawa, `4` dół, `8` lewa.
61    pub fn mask_at(&self, layer: &TileLayer, x: i32, y: i32) -> u8 {
62        let center = layer.get(x, y);
63        let up = self.matches(layer, x, y + 1, center);
64        let right = self.matches(layer, x + 1, y, center);
65        let down = self.matches(layer, x, y - 1, center);
66        let left = self.matches(layer, x - 1, y, center);
67        u8::from(up) | (u8::from(right) << 1) | (u8::from(down) << 2) | (u8::from(left) << 3)
68    }
69
70    /// Czy sąsiad na (nx, ny) należy do tego samego terenu.
71    fn matches(&self, layer: &TileLayer, nx: i32, ny: i32, center: u32) -> bool {
72        let t = layer.get(nx, ny);
73        match self.mode {
74            // `TILE_EMPTY` poza mapą to ta sama wartość co „brak kafla",
75            // więc porównanie musi odrzucić pusty środek inaczej każdy
76            // pusty kafel na krawędzi mapy dostałby maskę 0b1111.
77            MatchMode::Exact => center != crate::layer::TILE_EMPTY && t == center,
78            MatchMode::NonEmpty => t != crate::layer::TILE_EMPTY,
79        }
80    }
81
82    /// Indeks kafla wariantu dla maski sąsiadów.
83    pub fn index_for_mask(&self, mask: u8) -> u32 {
84        // Maska 15 oznacza środek terenu. Jeśli `include_self` jest wyłączone,
85        // środek ma być rysowany jako zwykły kafel (indeks bazowy), a nie
86        // wariant — inaczej woda w środku jeziorka dostałaby brzeg ze środka.
87        let slot = if mask == 0b1111 && !self.include_self {
88            0
89        } else {
90            mask as u32 % 16
91        };
92        self.base + slot
93    }
94
95    /// Przelicza warianty w podanym miejscu i w promieniu `radius`.
96    ///
97    /// Po zmianie jednego kafla trzeba przeliczyć też jego sąsiadów, bo ich
98    /// maski też się zmieniły — inaczej wzdłuż wykopanego rowu zostawałby
99    /// stary, nieaktualny brzeg.
100    pub fn refresh(&self, layer: &mut TileLayer, changed: (i32, i32), radius: i32) {
101        let (cx, cy) = changed;
102        let mut updates = Vec::new();
103        for dy in -radius..=radius {
104            for dx in -radius..=radius {
105                let (x, y) = (cx + dx, cy + dy);
106                if !layer.in_bounds(x, y) {
107                    continue;
108                }
109                let mask = self.mask_at(layer, x, y);
110                updates.push((x, y, self.index_for_mask(mask)));
111            }
112        }
113        for (x, y, index) in updates {
114            layer.set(x, y, index);
115        }
116    }
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122    use crate::layer::{TileLayer, TILE_EMPTY};
123
124    /// Warstwa 5x5 z polem 3x3 pośrodku, otoczonym pustkiem.
125    fn field() -> TileLayer {
126        let mut l = TileLayer::new("t", 5, 5);
127        l.fill_rect(1, 1, 3, 3, 10);
128        l
129    }
130
131    #[test]
132    fn corner_tile_has_two_neighbours() {
133        let l = field();
134        let at = AutoTile::exact(0);
135        // (1,1) to róg pola 3x3: sąsieduje góra (1,2) i prawa (2,1);
136        // dołu (1,0) i lewej (0,1) nie ma, bo tam pusto.
137        assert_eq!(at.mask_at(&l, 1, 1), 0b0011);
138    }
139
140    #[test]
141    fn interior_tile_has_all_neighbours() {
142        let l = field();
143        let at = AutoTile::exact(0);
144        assert_eq!(at.mask_at(&l, 2, 2), 0b1111, "centrum pola 3x3");
145    }
146
147    #[test]
148    fn bit_order_is_up_right_down_left() {
149        let mut l = TileLayer::new("t", 3, 3);
150        l.set(1, 1, 7);
151        let at = AutoTile::exact(0);
152
153        l.set(1, 2, 7);
154        assert_eq!(at.mask_at(&l, 1, 1) & 0b0001, 0b0001, "góra");
155        l.set(1, 2, TILE_EMPTY);
156
157        l.set(2, 1, 7);
158        assert_eq!(at.mask_at(&l, 1, 1) & 0b0010, 0b0010, "prawa");
159        l.set(2, 1, TILE_EMPTY);
160
161        l.set(1, 0, 7);
162        assert_eq!(at.mask_at(&l, 1, 1) & 0b0100, 0b0100, "dół");
163        l.set(1, 0, TILE_EMPTY);
164
165        l.set(0, 1, 7);
166        assert_eq!(at.mask_at(&l, 1, 1) & 0b1000, 0b1000, "lewa");
167    }
168
169    #[test]
170    fn exact_mode_ignores_other_tiles() {
171        let mut l = TileLayer::new("t", 3, 3);
172        l.fill_rect(0, 0, 3, 3, 1);
173        l.set(1, 1, 2);
174        let at = AutoTile::exact(0);
175        // Sąsiad (2,1) to kafel 1, a środek to 2 -> brak dopasowania.
176        assert_eq!(at.mask_at(&l, 1, 1) & 0b0010, 0);
177    }
178
179    #[test]
180    fn presence_mode_joins_different_tiles() {
181        let mut l = TileLayer::new("t", 3, 3);
182        l.fill_rect(0, 0, 3, 3, 1);
183        l.set(1, 1, 2);
184        let at = AutoTile::by_presence(0);
185        assert_eq!(at.mask_at(&l, 1, 1), 0b1111, "każdy sąsiad jest niepusty");
186    }
187
188    #[test]
189    fn out_of_bounds_counts_as_no_neighbour() {
190        let mut l = TileLayer::new("t", 2, 2);
191        // Tylko narożny kafel (0,0); poza mapą nie ma sąsiadów.
192        l.set(0, 0, 3);
193        let at = AutoTile::exact(0);
194        // (0,0): góra (0,1) i prawa (1,0) są puste, dół i lewa poza mapą.
195        assert_eq!(at.mask_at(&l, 0, 0), 0, "dół i lewa są poza mapą");
196        // (1,1) jest pusty, więc w ogóle nie ma czego liczyć — regresja na
197        // `TILE_EMPTY` poza mapą, który w `Exact` wyglądałby jak sąsiad.
198        assert_eq!(at.mask_at(&l, 1, 1), 0);
199    }
200
201    #[test]
202    fn presence_mode_ignores_out_of_bounds() {
203        let mut l = TileLayer::new("t", 2, 2);
204        l.set(1, 1, 3);
205        let at = AutoTile::by_presence(0);
206        // (1,1) nie ma niepustych sąsiadów w obrębie mapy: (0,1) i (1,0)
207        // są puste, a góra i prawa wychodzą poza mapę.
208        assert_eq!(at.mask_at(&l, 1, 1), 0);
209    }
210
211    #[test]
212    fn include_self_off_keeps_base_tile_in_the_middle() {
213        let at = AutoTile {
214            base: 100,
215            include_self: false,
216            mode: MatchMode::NonEmpty,
217        };
218        assert_eq!(at.index_for_mask(0b1111), 100, "środek wraca do bazowego");
219        assert_ne!(at.index_for_mask(0b1010), 100, "brzeg to wariant");
220    }
221
222    #[test]
223    fn refresh_updates_neighbourhood() {
224        let mut l = field();
225        AutoTile::exact(0).refresh(&mut l, (2, 2), 1);
226        assert_eq!(l.get(2, 2), 15, "środek -> maska 0b1111");
227        assert_eq!(l.get(1, 1), 0b0011, "narożnik: góra + prawa");
228    }

230    #[test]
231    fn refresh_ignores_out_of_bounds() {
232        let mut l = field();
233        AutoTile::exact(0).refresh(&mut l, (0, 0), 2);
234        assert_eq!((l.width(), l.height()), (5, 5));
235        assert_eq!(l.get(-1, -1), TILE_EMPTY, "nic nie przyrosło poza mapą");
