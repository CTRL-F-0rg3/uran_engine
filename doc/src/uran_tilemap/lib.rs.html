1//! Tilemapa 2D dla silnika Uran.
2//!
3//! Moduł daje grze komplet do budowy i rysowania świata z kafli:
4//!
5//! * [`tileset::Tileset`] — arkusz kafli: margines, spacing, flipping,
6//!   przeliczanie UV,
7//! * [`layer::TileLayer`] — jedna warstwa siatki kafli z cullingiem
8//!   i zapytaniami geometrycznymi,
9//! * [`autotile::AutoTile`] — dobieranie kafla brzegowego do otoczenia
10//!   (16 wariantów „blob"),
11//! * [`map::TileMap`] — cała mapa: warstwy, kolizje, obszar edytowalny,
12//! * [`xml`] — wczytywanie i zapisywanie mapy w XML (format `*.xml`).
13//!
14//! ## Przykład
15//!
16//! ```ignore
17//! use uran_tilemap::prelude::*;
18//!
19//! let mut map = TileMap::load_xml(assets, "maps/farm.xml")?;
20//!
21//! // rysowanie: kamera pokazuje prostokąt `view` w świecie
22//! map.draw(&mut ctx.gfx, ctx.camera.visible_rect(ctx.window.size));
23//!
24//! // kolizja gracza
25//! if map.overlaps_solid(player_rect) { /* cofnij ruch */ }
26//!
27//! // edycja w ograniczonym zakresie (reszta mapy jest zablokowana)
28//! if map.can_edit(4, 7) { map.set_tile(4, 7, TileId::new(12)); }
29//! ```
