16/// Error validating a draw call.
17#[derive(Clone, Debug, Error, Eq, PartialEq)]
18#[non_exhaustive]
19pub enum DrawError {
20    #[error("Blend constant needs to be set")]
21    MissingBlendConstant,
22    #[error("Render pipeline must be set")]
23    MissingPipeline,
24    #[error("Vertex buffer {index} must be set")]
25    MissingVertexBuffer { index: u32 },
26    #[error("Index buffer must be set")]
27    MissingIndexBuffer,
28    #[error("Incompatible bind group at index {index} in the current render pipeline")]
29    IncompatibleBindGroup { index: u32, diff: Vec<String> },
30    #[error("Vertex {last_vertex} extends beyond limit {vertex_limit} imposed by the buffer in slot {slot}. Did you bind the correct `Vertex` step-rate vertex buffer?")]
31    VertexBeyondLimit {
32        last_vertex: u64,
33        vertex_limit: u64,
34        slot: u32,
35    },
36    #[error("Instance {last_instance} extends beyond limit {instance_limit} imposed by the buffer in slot {slot}. Did you bind the correct `Instance` step-rate vertex buffer?")]
37    InstanceBeyondLimit {
38        last_instance: u64,
39        instance_limit: u64,
40        slot: u32,
41    },
42    #[error("Index {last_index} extends beyond limit {index_limit}. Did you bind the correct index buffer?")]
43    IndexBeyondLimit { last_index: u64, index_limit: u64 },
44    #[error(
45        "Pipeline index format ({pipeline:?}) and buffer index format ({buffer:?}) do not match"
46    )]
47    UnmatchedIndexFormats {
48        pipeline: wgt::IndexFormat,
49        buffer: wgt::IndexFormat,
50    },
51    #[error(transparent)]
52    BindingSizeTooSmall(#[from] LateMinBufferBindingSizeMismatch),
53}

55/// Error encountered when encoding a render command.
56/// This is the shared error set between render bundles and passes.
57#[derive(Clone, Debug, Error)]
58#[non_exhaustive]
59pub enum RenderCommandError {
60    #[error("Bind group {0:?} is invalid")]
61    InvalidBindGroup(id::BindGroupId),
62    #[error("Render bundle {0:?} is invalid")]
63    InvalidRenderBundle(id::RenderBundleId),
64    #[error("Bind group index {index} is greater than the device's requested `max_bind_group` limit {max}")]
65    BindGroupIndexOutOfRange { index: u32, max: u32 },
66    #[error("Vertex buffer index {index} is greater than the device's requested `max_vertex_buffers` limit {max}")]
67    VertexBufferIndexOutOfRange { index: u32, max: u32 },
68    #[error("Dynamic buffer offset {0} does not respect device's requested `{1}` limit {2}")]
69    UnalignedBufferOffset(u64, &'static str, u32),
70    #[error("Number of buffer offsets ({actual}) does not match the number of dynamic bindings ({expected})")]
71    InvalidDynamicOffsetCount { actual: usize, expected: usize },
72    #[error("Render pipeline {0:?} is invalid")]
73    InvalidPipeline(id::RenderPipelineId),
74    #[error("QuerySet {0:?} is invalid")]
75    InvalidQuerySet(id::QuerySetId),
76    #[error("Render pipeline targets are incompatible with render pass")]
77    IncompatiblePipelineTargets(#[from] crate::device::RenderPassCompatibilityError),
78    #[error("Pipeline writes to depth/stencil, while the pass has read-only depth/stencil")]
79    IncompatiblePipelineRods,
80    #[error(transparent)]
81    UsageConflict(#[from] UsageConflict),
82    #[error("Buffer {0:?} is destroyed")]
83    DestroyedBuffer(id::BufferId),
84    #[error(transparent)]
85    MissingBufferUsage(#[from] MissingBufferUsageError),
86    #[error(transparent)]
87    MissingTextureUsage(#[from] MissingTextureUsageError),
88    #[error(transparent)]
89    PushConstants(#[from] PushConstantUploadError),
90    #[error("Viewport has invalid rect {0:?}; origin and/or size is less than or equal to 0, and/or is not contained in the render target {1:?}")]
91    InvalidViewportRect(Rect<f32>, wgt::Extent3d),
92    #[error("Viewport minDepth {0} and/or maxDepth {1} are not in [0, 1]")]
93    InvalidViewportDepth(f32, f32),
94    #[error("Scissor {0:?} is not contained in the render target {1:?}")]
95    InvalidScissorRect(Rect<u32>, wgt::Extent3d),
96    #[error("Support for {0} is not implemented yet")]
97    Unimplemented(&'static str),
98}
99impl crate::error::PrettyError for RenderCommandError {
100    fn fmt_pretty(&self, fmt: &mut ErrorFormatter) {
101        fmt.error(self);
102        match *self {
103            Self::InvalidBindGroup(id) => {
104                fmt.bind_group_label(&id);
105            }
106            Self::InvalidPipeline(id) => {
107                fmt.render_pipeline_label(&id);
108            }
109            Self::UsageConflict(UsageConflict::TextureInvalid { id }) => {
110                fmt.texture_label(&id);
111            }
112            Self::UsageConflict(UsageConflict::BufferInvalid { id })
113            | Self::DestroyedBuffer(id) => {
114                fmt.buffer_label(&id);
115            }
116            _ => {}
117        };
118    }
119}
