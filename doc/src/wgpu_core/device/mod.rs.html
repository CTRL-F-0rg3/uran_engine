119impl RenderPassContext {
120    // Assumes the renderpass only contains one subpass
121    pub(crate) fn check_compatible(
122        &self,
123        other: &Self,
124        ty: RenderPassCompatibilityCheckType,
125    ) -> Result<(), RenderPassCompatibilityError> {
126        if self.attachments.colors != other.attachments.colors {
127            let indices = self
128                .attachments
129                .colors
130                .iter()
131                .zip(&other.attachments.colors)
132                .enumerate()
133                .filter_map(|(idx, (left, right))| (left != right).then_some(idx))
134                .collect();
135            return Err(RenderPassCompatibilityError::IncompatibleColorAttachment {
136                indices,
137                expected: self.attachments.colors.iter().cloned().collect(),
138                actual: other.attachments.colors.iter().cloned().collect(),
139                ty,
140            });
141        }
142        if self.attachments.depth_stencil != other.attachments.depth_stencil {
143            return Err(
144                RenderPassCompatibilityError::IncompatibleDepthStencilAttachment {
145                    expected: self.attachments.depth_stencil,
146                    actual: other.attachments.depth_stencil,
147                    ty,
148                },
149            );
150        }
151        if self.sample_count != other.sample_count {
152            return Err(RenderPassCompatibilityError::IncompatibleSampleCount {
153                expected: self.sample_count,
154                actual: other.sample_count,
155                ty,
156            });
157        }
158        if self.multiview != other.multiview {
159            return Err(RenderPassCompatibilityError::IncompatibleMultiview {
160                expected: self.multiview,
161                actual: other.multiview,
162                ty,
163            });
164        }
165        Ok(())
166    }
167}

171#[derive(Default)]
172pub struct UserClosures {
173    pub mappings: Vec<BufferMapPendingClosure>,
174    pub submissions: SmallVec<[queue::SubmittedWorkDoneClosure; 1]>,
175    pub device_lost_invocations: SmallVec<[DeviceLostInvocation; 1]>,
176}
177
178impl UserClosures {
179    fn extend(&mut self, other: Self) {
180        self.mappings.extend(other.mappings);
181        self.submissions.extend(other.submissions);
182        self.device_lost_invocations
183            .extend(other.device_lost_invocations);
184    }
185
186    fn fire(self) {
187        // Note: this logic is specifically moved out of `handle_mapping()` in order to
188        // have nothing locked by the time we execute users callback code.
189
190        // Mappings _must_ be fired before submissions, as the spec requires all mapping callbacks that are registered before
191        // a on_submitted_work_done callback to be fired before the on_submitted_work_done callback.
192        for (mut operation, status) in self.mappings {
193            if let Some(callback) = operation.callback.take() {
194                callback.call(status);
195            }
196        }
197        for closure in self.submissions {
198            closure.call();
199        }
200        for invocation in self.device_lost_invocations {
201            invocation
202                .closure
203                .call(invocation.reason, invocation.message);
204        }
205    }
206}

244pub struct DeviceLostClosure {
245    // We wrap this so creating the enum in the C variant can be unsafe,
246    // allowing our call function to be safe.
247    inner: DeviceLostClosureInner,
248}

250pub struct DeviceLostInvocation {
251    closure: DeviceLostClosure,
252    reason: DeviceLostReason,
253    message: String,
254}
255
256enum DeviceLostClosureInner {
257    Rust { inner: DeviceLostClosureRust },
258    C { inner: DeviceLostClosureC },
259}
260
261impl DeviceLostClosure {
262    pub fn from_rust(callback: DeviceLostCallback) -> Self {
263        let inner = DeviceLostClosureRust {
264            callback,
265            consumed: false,
266        };
267        Self {
268            inner: DeviceLostClosureInner::Rust { inner },
269        }
270    }
271
272    /// # Safety
273    ///
274    /// - The callback pointer must be valid to call with the provided `user_data`
275    ///   pointer.
276    ///
277    /// - Both pointers must point to `'static` data, as the callback may happen at
278    ///   an unspecified time.
279    pub unsafe fn from_c(mut closure: DeviceLostClosureC) -> Self {
280        // Build an inner with the values from closure, ensuring that
281        // inner.consumed is false.
282        let inner = DeviceLostClosureC {
283            callback: closure.callback,
284            user_data: closure.user_data,
285            consumed: false,
286        };
287
288        // Mark the original closure as consumed, so we can safely drop it.
289        closure.consumed = true;
290
291        Self {
292            inner: DeviceLostClosureInner::C { inner },
293        }
294    }
295
296    pub(crate) fn call(self, reason: DeviceLostReason, message: String) {
297        match self.inner {
298            DeviceLostClosureInner::Rust { mut inner } => {
299                inner.consumed = true;
300
301                (inner.callback)(reason, message)
302            }
303            // SAFETY: the contract of the call to from_c says that this unsafe is sound.
304            DeviceLostClosureInner::C { mut inner } => unsafe {
305                inner.consumed = true;
306
307                // Ensure message is structured as a null-terminated C string. It only
308                // needs to live as long as the callback invocation.
309                let message = std::ffi::CString::new(message).unwrap();
310                (inner.callback)(inner.user_data, reason as u8, message.as_ptr())
311            },
312        }
313    }
314}

316fn map_buffer<A: HalApi>(
317    raw: &A::Device,
318    buffer: &Buffer<A>,
319    offset: BufferAddress,
320    size: BufferAddress,
321    kind: HostMap,
322) -> Result<ptr::NonNull<u8>, BufferAccessError> {
323    let snatch_guard = buffer.device.snatchable_lock.read();
324    let raw_buffer = buffer
325        .raw(&snatch_guard)
326        .ok_or(BufferAccessError::Destroyed)?;
327    let mapping = unsafe {
328        raw.map_buffer(raw_buffer, offset..offset + size)
329            .map_err(DeviceError::from)?
330    };
331
332    *buffer.sync_mapped_writes.lock() = match kind {
333        HostMap::Read if !mapping.is_coherent => unsafe {
334            raw.invalidate_mapped_ranges(raw_buffer, iter::once(offset..offset + size));
335            None
336        },
337        HostMap::Write if !mapping.is_coherent => Some(offset..offset + size),
338        _ => None,
339    };
340
341    assert_eq!(offset % wgt::COPY_BUFFER_ALIGNMENT, 0);
342    assert_eq!(size % wgt::COPY_BUFFER_ALIGNMENT, 0);
343    // Zero out uninitialized parts of the mapping. (Spec dictates all resources
344    // behave as if they were initialized with zero)
345    //
346    // If this is a read mapping, ideally we would use a `clear_buffer` command
347    // before reading the data from GPU (i.e. `invalidate_range`). However, this
348    // would require us to kick off and wait for a command buffer or piggy back
349    // on an existing one (the later is likely the only worthwhile option). As
350    // reading uninitialized memory isn't a particular important path to
351    // support, we instead just initialize the memory here and make sure it is
352    // GPU visible, so this happens at max only once for every buffer region.
353    //
354    // If this is a write mapping zeroing out the memory here is the only
355    // reasonable way as all data is pushed to GPU anyways.
356
357    // No need to flush if it is flushed later anyways.
358    let zero_init_needs_flush_now =
359        mapping.is_coherent && buffer.sync_mapped_writes.lock().is_none();
360    let mapped = unsafe { std::slice::from_raw_parts_mut(mapping.ptr.as_ptr(), size as usize) };
361
362    for uninitialized in buffer
363        .initialization_status
364        .write()
365        .drain(offset..(size + offset))
366    {
367        // The mapping's pointer is already offset, however we track the
368        // uninitialized range relative to the buffer's start.
369        let fill_range =
370            (uninitialized.start - offset) as usize..(uninitialized.end - offset) as usize;
371        mapped[fill_range].fill(0);
372
373        if zero_init_needs_flush_now {
374            unsafe { raw.flush_mapped_ranges(raw_buffer, iter::once(uninitialized)) };
375        }
376    }
377
378    Ok(mapping.ptr)
379}

381pub(crate) struct CommandAllocator<A: HalApi> {
382    free_encoders: Vec<A::CommandEncoder>,
383}
384
385impl<A: HalApi> CommandAllocator<A> {
386    fn acquire_encoder(
387        &mut self,
388        device: &A::Device,
389        queue: &A::Queue,
390    ) -> Result<A::CommandEncoder, hal::DeviceError> {
391        match self.free_encoders.pop() {
392            Some(encoder) => Ok(encoder),
393            None => unsafe {
394                let hal_desc = hal::CommandEncoderDescriptor { label: None, queue };
395                device.create_command_encoder(&hal_desc)
396            },
397        }
398    }
399
400    fn release_encoder(&mut self, encoder: A::CommandEncoder) {
401        self.free_encoders.push(encoder);
402    }
403
404    fn dispose(self, device: &A::Device) {
405        resource_log!(
406            "CommandAllocator::dispose encoders {}",
407            self.free_encoders.len()
408        );
409        for cmd_encoder in self.free_encoders {
410            unsafe {
411                device.destroy_command_encoder(cmd_encoder);
412            }
413        }
414    }
415}

423pub enum DeviceError {
424    #[error("Parent device is invalid.")]
425    Invalid,
426    #[error("Parent device is lost")]
427    Lost,
428    #[error("Not enough memory left.")]
429    OutOfMemory,
430    #[error("Creation of a resource failed for a reason other than running out of memory.")]
431    ResourceCreationFailed,
432    #[error("QueueId is invalid")]
433    InvalidQueueId,
434    #[error("Attempt to use a resource with a different device from the one that created it")]
435    WrongDevice,
436}
437
438impl From<hal::DeviceError> for DeviceError {
439    fn from(error: hal::DeviceError) -> Self {
440        match error {
441            hal::DeviceError::Lost => DeviceError::Lost,
442            hal::DeviceError::OutOfMemory => DeviceError::OutOfMemory,
443            hal::DeviceError::ResourceCreationFailed => DeviceError::ResourceCreationFailed,
444        }
445    }
446}
