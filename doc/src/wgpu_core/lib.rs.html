1//! This library safely implements WebGPU on native platforms.
2//! It is designed for integration into browsers, as well as wrapping
3//! into other language-specific user-friendly libraries.
4//!
5//! ## Feature flags
6// NOTE: feature docs. below should be kept in sync. with `Cargo.toml`!
7//!
8//! - **`api_log_info`** --- Log all API entry points at info instead of trace level.
9//! - **`resource_log_info`** --- Log resource lifecycle management at info instead of trace level.
10//! - **`link`** _(enabled by default)_ --- Use static linking for libraries. Disale to manually
11//!   link. Enabled by default.
12//! - **`renderdoc`** --- Support the Renderdoc graphics debugger:
13//!   [https://renderdoc.org/](https://renderdoc.org/)
14//! - **`strict_asserts`** --- Apply run-time checks, even in release builds. These are in addition
15//!   to the validation carried out at public APIs in all builds.
16//! - **`trace`** --- Enable API tracing.
17//! - **`replay`** --- Enable API replaying
18//! - **`serial-pass`** --- Enable serializable compute/render passes, and bundle encoders.
19//! - **`wgsl`** --- Enable `ShaderModuleSource::Wgsl`
20//! - **`fragile-send-sync-non-atomic-wasm`** --- Implement `Send` and `Sync` on Wasm, but only if
21//!   atomics are not enabled.
22//!
23//!   WebGL/WebGPU objects can not be shared between threads. However, it can be useful to
24//!   artificially mark them as `Send` and `Sync` anyways to make it easier to write cross-platform
25//!   code. This is technically _very_ unsafe in a multithreaded environment, but on a wasm binary
26//!   compiled without atomics we know we are definitely not in a multithreaded environment.
27//!
28//!  ### Backends, passed through to wgpu-hal
29//!
30//!  - **`metal`** --- Enable the `metal` backend.
31//!  - **`vulkan`** --- Enable the `vulkan` backend.
32//!  - **`gles`** --- Enable the `GLES` backend.
33//!
34//!    This is used for all of GLES, OpenGL, and WebGL.
35//!  - **`dx12`** --- Enable the `dx12` backend.
36

125trait LabelHelpers<'a> {
126    fn borrow_option(&'a self) -> Option<&'a str>;
127    fn to_hal(&'a self, flags: wgt::InstanceFlags) -> Option<&'a str>;
128    fn borrow_or_default(&'a self) -> &'a str;
129}
130impl<'a> LabelHelpers<'a> for Label<'a> {
131    fn borrow_option(&'a self) -> Option<&'a str> {
132        self.as_ref().map(|cow| cow.as_ref())
133    }
134    fn to_hal(&'a self, flags: wgt::InstanceFlags) -> Option<&'a str> {
135        if flags.contains(wgt::InstanceFlags::DISCARD_HAL_LABELS) {
136            return None;
137        }
138
139        self.as_ref().map(|cow| cow.as_ref())
140    }
141    fn borrow_or_default(&'a self) -> &'a str {
142        self.borrow_option().unwrap_or_default()
143    }
144}

147    if flags.contains(wgt::InstanceFlags::DISCARD_HAL_LABELS) {
148        return None;
149    }
150
151    opt
152}

165// #[cfg] attributes in exported macros are interesting!
166//
167// The #[cfg] conditions in a macro's expansion are evaluated using the
168// configuration options (features, target architecture and os, etc.) in force
169// where the macro is *used*, not where it is *defined*. That is, if crate A
170// defines a macro like this:
171//
172//     #[macro_export]
173//     macro_rules! if_bleep {
174//         { } => {
175//             #[cfg(feature = "bleep")]
176//             bleep();
177//         }
178//     }
179//
180// and then crate B uses it like this:
181//
182//     fn f() {
183//         if_bleep! { }
184//     }
185//
186// then it is crate B's `"bleep"` feature, not crate A's, that determines
187// whether the macro expands to a function call or an empty statement. The
188// entire configuration predicate is evaluated in the use's context, not the
189// definition's.
190//
191// Since `wgpu-core` selects back ends using features, we need to make sure the
192// arms of the `gfx_select!` macro are pruned according to `wgpu-core`'s
193// features, not those of whatever crate happens to be using `gfx_select!`. This
194// means we can't use `#[cfg]` attributes in `gfx_select!`s definition itself.
195// Instead, for each backend, `gfx_select!` must use a macro whose definition is
196// selected by `#[cfg]` in `wgpu-core`. The configuration predicate is still
197// evaluated when the macro is used; we've just moved the `#[cfg]` into a macro
198// used by `wgpu-core` itself.
199
200/// Define an exported macro named `$public` that expands to an expression if
201/// the feature `$feature` is enabled, or to a panic otherwise.
202///
203/// This is used in the definition of `gfx_select!`, to dispatch the
204/// call to the appropriate backend, but panic if that backend was not
205/// compiled in.
206///
207/// For a call like this:
208///
209/// ```ignore
210/// define_backend_caller! { name, private, "feature" if cfg_condition }
211/// ```
212///
213/// define a macro `name`, used like this:
214///
215/// ```ignore
216/// name!(expr)
217/// ```
218///
219/// that expands to `expr` if `#[cfg(cfg_condition)]` is enabled, or a
220/// panic otherwise. The panic message complains that `"feature"` is
221/// not enabled.
222///
223/// Because of odd technical limitations on exporting macros expanded
224/// by other macros, you must supply both a public-facing name for the
225/// macro and a private name, `$private`, which is never used
226/// outside this macro. For details:
227/// <https://github.com/rust-lang/rust/pull/52234#issuecomment-976702997>
228macro_rules! define_backend_caller {
229    { $public:ident, $private:ident, $feature:literal if $cfg:meta } => {
230        #[cfg($cfg)]
231        #[macro_export]
232        macro_rules! $private {
233            ( $call:expr ) => ( $call )
234        }
235
236        #[cfg(not($cfg))]
237        #[macro_export]
238        macro_rules! $private {
239            ( $call:expr ) => (
240                panic!("Identifier refers to disabled backend feature {:?}", $feature)
241            )
242        }
243
244        // See note about rust-lang#52234 above.
245        #[doc(hidden)] pub use $private as $public;
246    }
247}

249// Define a macro for each `gfx_select!` match arm. For example,
250//
251//     gfx_if_vulkan!(expr)
252//
253// expands to `expr` if the `"vulkan"` feature is enabled, or to a panic
254// otherwise.
255define_backend_caller! { gfx_if_vulkan, gfx_if_vulkan_hidden, "vulkan" if all(feature = "vulkan", not(target_arch = "wasm32")) }
256define_backend_caller! { gfx_if_metal, gfx_if_metal_hidden, "metal" if all(feature = "metal", any(target_os = "macos", target_os = "ios")) }
257define_backend_caller! { gfx_if_dx12, gfx_if_dx12_hidden, "dx12" if all(feature = "dx12", windows) }
258define_backend_caller! { gfx_if_gles, gfx_if_gles_hidden, "gles" if feature = "gles" }
259define_backend_caller! { gfx_if_empty, gfx_if_empty_hidden, "empty" if all(
260    not(any(feature = "metal", feature = "vulkan", feature = "gles")),
261    any(target_os = "macos", target_os = "ios"),
262) }

264/// Dispatch on an [`Id`]'s backend to a backend-generic method.
265///
266/// Uses of this macro have the form:
267///
268/// ```ignore
269///
270///     gfx_select!(id => value.method(args...))
271///
272/// ```
273///
274/// This expands to an expression that calls `value.method::<A>(args...)` for
275/// the backend `A` selected by `id`. The expansion matches on `id.backend()`,
276/// with an arm for each backend type in [`wgpu_types::Backend`] which calls the
277/// specialization of `method` for the given backend. This allows resource
278/// identifiers to select backends dynamically, even though many `wgpu_core`
279/// methods are compiled and optimized for a specific back end.
280///
281/// This macro is typically used to call methods on [`wgpu_core::global::Global`],
282/// many of which take a single `hal::Api` type parameter. For example, to
283/// create a new buffer on the device indicated by `device_id`, one would say:
284///
285/// ```ignore
286/// gfx_select!(device_id => global.device_create_buffer(device_id, ...))
287/// ```
288///
289/// where the `device_create_buffer` method is defined like this:
290///
291/// ```ignore
292/// impl<...> Global<...> {
293///    pub fn device_create_buffer<A: HalApi>(&self, ...) -> ...
294///    { ... }
295/// }
296/// ```
297///
298/// That `gfx_select!` call uses `device_id`'s backend to select the right
299/// backend type `A` for a call to `Global::device_create_buffer<A>`.
300///
301/// However, there's nothing about this macro that is specific to `hub::Global`.
302/// For example, Firefox's embedding of `wgpu_core` defines its own types with
303/// methods that take `hal::Api` type parameters. Firefox uses `gfx_select!` to
304/// dynamically dispatch to the right specialization based on the resource's id.
305///
306/// [`wgpu_types::Backend`]: wgt::Backend
307/// [`wgpu_core::global::Global`]: crate::global::Global
308/// [`Id`]: id::Id
309#[macro_export]
310macro_rules! gfx_select {
311    // Simple two-component expression, like `self.0.method(..)`.
312    ($id:expr => $c0:ident.$c1:tt.$method:ident $params:tt) => {
313        $crate::gfx_select!($id => {$c0.$c1}, $method $params)
314    };
315
316    // Simple identifier-only expression, like `global.method(..)`.
317    ($id:expr => $c0:ident.$method:ident $params:tt) => {
318        $crate::gfx_select!($id => {$c0}, $method $params)
319    };
320
321    ($id:expr => {$($c:tt)*}, $method:ident $params:tt) => {
322        match $id.backend() {
323            wgt::Backend::Vulkan => $crate::gfx_if_vulkan!($($c)*.$method::<$crate::api::Vulkan> $params),
324            wgt::Backend::Metal => $crate::gfx_if_metal!($($c)*.$method::<$crate::api::Metal> $params),
325            wgt::Backend::Dx12 => $crate::gfx_if_dx12!($($c)*.$method::<$crate::api::Dx12> $params),
326            wgt::Backend::Gl => $crate::gfx_if_gles!($($c)*.$method::<$crate::api::Gles> $params),
327            wgt::Backend::Empty => $crate::gfx_if_empty!($($c)*.$method::<$crate::api::Empty> $params),
328            other => panic!("Unexpected backend {:?}", other),
329        }
330    };
331}

353#[inline]
354pub(crate) fn get_lowest_common_denom(a: u32, b: u32) -> u32 {
355    let gcd = if a >= b {
356        get_greatest_common_divisor(a, b)
357    } else {
358        get_greatest_common_divisor(b, a)
359    };
360    a * b / gcd
361}
362
363#[inline]
364pub(crate) fn get_greatest_common_divisor(mut a: u32, mut b: u32) -> u32 {
365    assert!(a >= b);
366    loop {
367        let c = a % b;
368        if c == 0 {
369            return b;
370        } else {
371            a = b;
372            b = c;
373        }
374    }
375}
