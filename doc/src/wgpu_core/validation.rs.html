386impl Resource {
387    fn check_binding_use(&self, entry: &BindGroupLayoutEntry) -> Result<(), BindingError> {
388        match self.ty {
389            ResourceType::Buffer { size } => {
390                let min_size = match entry.ty {
391                    BindingType::Buffer {
392                        ty,
393                        has_dynamic_offset: _,
394                        min_binding_size,
395                    } => {
396                        let class = match ty {
397                            wgt::BufferBindingType::Uniform => naga::AddressSpace::Uniform,
398                            wgt::BufferBindingType::Storage { read_only } => {
399                                let mut naga_access = naga::StorageAccess::LOAD;
400                                naga_access.set(naga::StorageAccess::STORE, !read_only);
401                                naga::AddressSpace::Storage {
402                                    access: naga_access,
403                                }
404                            }
405                        };
406                        if self.class != class {
407                            return Err(BindingError::WrongAddressSpace {
408                                binding: class,
409                                shader: self.class,
410                            });
411                        }
412                        min_binding_size
413                    }
414                    _ => return Err(BindingError::WrongType),
415                };
416                match min_size {
417                    Some(non_zero) if non_zero < size => {
418                        return Err(BindingError::WrongBufferSize(size))
419                    }
420                    _ => (),
421                }
422            }
423            ResourceType::Sampler { comparison } => match entry.ty {
424                BindingType::Sampler(ty) => {
425                    if (ty == wgt::SamplerBindingType::Comparison) != comparison {
426                        return Err(BindingError::WrongSamplerComparison);
427                    }
428                }
429                _ => return Err(BindingError::WrongType),
430            },
431            ResourceType::Texture {
432                dim,
433                arrayed,
434                class,
435            } => {
436                let view_dimension = match entry.ty {
437                    BindingType::Texture { view_dimension, .. }
438                    | BindingType::StorageTexture { view_dimension, .. } => view_dimension,
439                    _ => {
440                        return Err(BindingError::WrongTextureViewDimension {
441                            dim,
442                            is_array: false,
443                            binding: entry.ty,
444                        })
445                    }
446                };
447                if arrayed {
448                    match (dim, view_dimension) {
449                        (naga::ImageDimension::D2, wgt::TextureViewDimension::D2Array) => (),
450                        (naga::ImageDimension::Cube, wgt::TextureViewDimension::CubeArray) => (),
451                        _ => {
452                            return Err(BindingError::WrongTextureViewDimension {
453                                dim,
454                                is_array: true,
455                                binding: entry.ty,
456                            })
457                        }
458                    }
459                } else {
460                    match (dim, view_dimension) {
461                        (naga::ImageDimension::D1, wgt::TextureViewDimension::D1) => (),
462                        (naga::ImageDimension::D2, wgt::TextureViewDimension::D2) => (),
463                        (naga::ImageDimension::D3, wgt::TextureViewDimension::D3) => (),
464                        (naga::ImageDimension::Cube, wgt::TextureViewDimension::Cube) => (),
465                        _ => {
466                            return Err(BindingError::WrongTextureViewDimension {
467                                dim,
468                                is_array: false,
469                                binding: entry.ty,
470                            })
471                        }
472                    }
473                }
474                let expected_class = match entry.ty {
475                    BindingType::Texture {
476                        sample_type,
477                        view_dimension: _,
478                        multisampled: multi,
479                    } => match sample_type {
480                        wgt::TextureSampleType::Float { .. } => naga::ImageClass::Sampled {
481                            kind: naga::ScalarKind::Float,
482                            multi,
483                        },
484                        wgt::TextureSampleType::Sint => naga::ImageClass::Sampled {
485                            kind: naga::ScalarKind::Sint,
486                            multi,
487                        },
488                        wgt::TextureSampleType::Uint => naga::ImageClass::Sampled {
489                            kind: naga::ScalarKind::Uint,
490                            multi,
491                        },
492                        wgt::TextureSampleType::Depth => naga::ImageClass::Depth { multi },
493                    },
494                    BindingType::StorageTexture {
495                        access,
496                        format,
497                        view_dimension: _,
498                    } => {
499                        let naga_format = map_storage_format_to_naga(format)
500                            .ok_or(BindingError::BadStorageFormat(format))?;
501                        let naga_access = match access {
502                            wgt::StorageTextureAccess::ReadOnly => naga::StorageAccess::LOAD,
503                            wgt::StorageTextureAccess::WriteOnly => naga::StorageAccess::STORE,
504                            wgt::StorageTextureAccess::ReadWrite => naga::StorageAccess::all(),
505                        };
506                        naga::ImageClass::Storage {
507                            format: naga_format,
508                            access: naga_access,
509                        }
510                    }
511                    _ => return Err(BindingError::WrongType),
512                };
513                if class != expected_class {
514                    return Err(BindingError::WrongTextureClass {
515                        binding: expected_class,
516                        shader: class,
517                    });
518                }
519            }
520        };
521
522        Ok(())
523    }

525    fn derive_binding_type(&self) -> Result<BindingType, BindingError> {
526        Ok(match self.ty {
527            ResourceType::Buffer { size } => BindingType::Buffer {
528                ty: match self.class {
529                    naga::AddressSpace::Uniform => wgt::BufferBindingType::Uniform,
530                    naga::AddressSpace::Storage { access } => wgt::BufferBindingType::Storage {
531                        read_only: access == naga::StorageAccess::LOAD,
532                    },
533                    _ => return Err(BindingError::WrongType),
534                },
535                has_dynamic_offset: false,
536                min_binding_size: Some(size),
537            },
538            ResourceType::Sampler { comparison } => BindingType::Sampler(if comparison {
539                wgt::SamplerBindingType::Comparison
540            } else {
541                wgt::SamplerBindingType::Filtering
542            }),
543            ResourceType::Texture {
544                dim,
545                arrayed,
546                class,
547            } => {
548                let view_dimension = match dim {
549                    naga::ImageDimension::D1 => wgt::TextureViewDimension::D1,
550                    naga::ImageDimension::D2 if arrayed => wgt::TextureViewDimension::D2Array,
551                    naga::ImageDimension::D2 => wgt::TextureViewDimension::D2,
552                    naga::ImageDimension::D3 => wgt::TextureViewDimension::D3,
553                    naga::ImageDimension::Cube if arrayed => wgt::TextureViewDimension::CubeArray,
554                    naga::ImageDimension::Cube => wgt::TextureViewDimension::Cube,
555                };
556                match class {
557                    naga::ImageClass::Sampled { multi, kind } => BindingType::Texture {
558                        sample_type: match kind {
559                            naga::ScalarKind::Float => {
560                                wgt::TextureSampleType::Float { filterable: true }
561                            }
562                            naga::ScalarKind::Sint => wgt::TextureSampleType::Sint,
563                            naga::ScalarKind::Uint => wgt::TextureSampleType::Uint,
564                            naga::ScalarKind::AbstractInt
565                            | naga::ScalarKind::AbstractFloat
566                            | naga::ScalarKind::Bool => unreachable!(),
567                        },
568                        view_dimension,
569                        multisampled: multi,
570                    },
571                    naga::ImageClass::Depth { multi } => BindingType::Texture {
572                        sample_type: wgt::TextureSampleType::Depth,
573                        view_dimension,
574                        multisampled: multi,
575                    },
576                    naga::ImageClass::Storage { format, access } => BindingType::StorageTexture {
577                        access: {
578                            const LOAD_STORE: naga::StorageAccess = naga::StorageAccess::all();
579                            match access {
580                                naga::StorageAccess::LOAD => wgt::StorageTextureAccess::ReadOnly,
581                                naga::StorageAccess::STORE => wgt::StorageTextureAccess::WriteOnly,
582                                LOAD_STORE => wgt::StorageTextureAccess::ReadWrite,
583                                _ => unreachable!(),
584                            }
585                        },
586                        view_dimension,
587                        format: {
588                            let f = map_storage_format_from_naga(format);
589                            let original = map_storage_format_to_naga(f)
590                                .ok_or(BindingError::BadStorageFormat(f))?;
591                            debug_assert_eq!(format, original);
592                            f
593                        },
594                    },
595                }
596            }
597        })
598    }
599}

601impl NumericType {
602    fn from_vertex_format(format: wgt::VertexFormat) -> Self {
603        use naga::{Scalar, VectorSize as Vs};
604        use wgt::VertexFormat as Vf;
605
606        let (dim, scalar) = match format {
607            Vf::Uint32 => (NumericDimension::Scalar, Scalar::U32),
608            Vf::Uint8x2 | Vf::Uint16x2 | Vf::Uint32x2 => {
609                (NumericDimension::Vector(Vs::Bi), Scalar::U32)
610            }
611            Vf::Uint32x3 => (NumericDimension::Vector(Vs::Tri), Scalar::U32),
612            Vf::Uint8x4 | Vf::Uint16x4 | Vf::Uint32x4 => {
613                (NumericDimension::Vector(Vs::Quad), Scalar::U32)
614            }
615            Vf::Sint32 => (NumericDimension::Scalar, Scalar::I32),
616            Vf::Sint8x2 | Vf::Sint16x2 | Vf::Sint32x2 => {
617                (NumericDimension::Vector(Vs::Bi), Scalar::I32)
618            }
619            Vf::Sint32x3 => (NumericDimension::Vector(Vs::Tri), Scalar::I32),
620            Vf::Sint8x4 | Vf::Sint16x4 | Vf::Sint32x4 => {
621                (NumericDimension::Vector(Vs::Quad), Scalar::I32)
622            }
623            Vf::Float32 => (NumericDimension::Scalar, Scalar::F32),
624            Vf::Unorm8x2
625            | Vf::Snorm8x2
626            | Vf::Unorm16x2
627            | Vf::Snorm16x2
628            | Vf::Float16x2
629            | Vf::Float32x2 => (NumericDimension::Vector(Vs::Bi), Scalar::F32),
630            Vf::Float32x3 => (NumericDimension::Vector(Vs::Tri), Scalar::F32),
631            Vf::Unorm8x4
632            | Vf::Snorm8x4
633            | Vf::Unorm16x4
634            | Vf::Snorm16x4
635            | Vf::Float16x4
636            | Vf::Float32x4 => (NumericDimension::Vector(Vs::Quad), Scalar::F32),
637            Vf::Float64 => (NumericDimension::Scalar, Scalar::F64),
638            Vf::Float64x2 => (NumericDimension::Vector(Vs::Bi), Scalar::F64),
639            Vf::Float64x3 => (NumericDimension::Vector(Vs::Tri), Scalar::F64),
640            Vf::Float64x4 => (NumericDimension::Vector(Vs::Quad), Scalar::F64),
641        };
642
643        NumericType {
644            dim,
645            //Note: Shader always sees data as int, uint, or float.
646            // It doesn't know if the original is normalized in a tighter form.
647            scalar,
648        }
649    }
650
651    fn from_texture_format(format: wgt::TextureFormat) -> Self {
652        use naga::{Scalar, VectorSize as Vs};
653        use wgt::TextureFormat as Tf;
654
655        let (dim, scalar) = match format {
656            Tf::R8Unorm | Tf::R8Snorm | Tf::R16Float | Tf::R32Float => {
657                (NumericDimension::Scalar, Scalar::F32)
658            }
659            Tf::R8Uint | Tf::R16Uint | Tf::R32Uint => (NumericDimension::Scalar, Scalar::U32),
660            Tf::R8Sint | Tf::R16Sint | Tf::R32Sint => (NumericDimension::Scalar, Scalar::I32),
661            Tf::Rg8Unorm | Tf::Rg8Snorm | Tf::Rg16Float | Tf::Rg32Float => {
662                (NumericDimension::Vector(Vs::Bi), Scalar::F32)
663            }
664            Tf::Rg8Uint | Tf::Rg16Uint | Tf::Rg32Uint => {
665                (NumericDimension::Vector(Vs::Bi), Scalar::U32)
666            }
667            Tf::Rg8Sint | Tf::Rg16Sint | Tf::Rg32Sint => {
668                (NumericDimension::Vector(Vs::Bi), Scalar::I32)
669            }
670            Tf::R16Snorm | Tf::R16Unorm => (NumericDimension::Scalar, Scalar::F32),
671            Tf::Rg16Snorm | Tf::Rg16Unorm => (NumericDimension::Vector(Vs::Bi), Scalar::F32),
672            Tf::Rgba16Snorm | Tf::Rgba16Unorm => (NumericDimension::Vector(Vs::Quad), Scalar::F32),
673            Tf::Rgba8Unorm
674            | Tf::Rgba8UnormSrgb
675            | Tf::Rgba8Snorm
676            | Tf::Bgra8Unorm
677            | Tf::Bgra8UnormSrgb
678            | Tf::Rgb10a2Unorm
679            | Tf::Rgba16Float
680            | Tf::Rgba32Float => (NumericDimension::Vector(Vs::Quad), Scalar::F32),
681            Tf::Rgba8Uint | Tf::Rgba16Uint | Tf::Rgba32Uint | Tf::Rgb10a2Uint => {
682                (NumericDimension::Vector(Vs::Quad), Scalar::U32)
683            }
684            Tf::Rgba8Sint | Tf::Rgba16Sint | Tf::Rgba32Sint => {
685                (NumericDimension::Vector(Vs::Quad), Scalar::I32)
686            }
687            Tf::Rg11b10Float => (NumericDimension::Vector(Vs::Tri), Scalar::F32),
688            Tf::Stencil8
689            | Tf::Depth16Unorm
690            | Tf::Depth32Float
691            | Tf::Depth32FloatStencil8
692            | Tf::Depth24Plus
693            | Tf::Depth24PlusStencil8 => {
694                panic!("Unexpected depth format")
695            }
696            Tf::NV12 => panic!("Unexpected nv12 format"),
697            Tf::Rgb9e5Ufloat => (NumericDimension::Vector(Vs::Tri), Scalar::F32),
698            Tf::Bc1RgbaUnorm
699            | Tf::Bc1RgbaUnormSrgb
700            | Tf::Bc2RgbaUnorm
701            | Tf::Bc2RgbaUnormSrgb
702            | Tf::Bc3RgbaUnorm
703            | Tf::Bc3RgbaUnormSrgb
704            | Tf::Bc7RgbaUnorm
705            | Tf::Bc7RgbaUnormSrgb
706            | Tf::Etc2Rgb8A1Unorm
707            | Tf::Etc2Rgb8A1UnormSrgb
708            | Tf::Etc2Rgba8Unorm
709            | Tf::Etc2Rgba8UnormSrgb => (NumericDimension::Vector(Vs::Quad), Scalar::F32),
710            Tf::Bc4RUnorm | Tf::Bc4RSnorm | Tf::EacR11Unorm | Tf::EacR11Snorm => {
711                (NumericDimension::Scalar, Scalar::F32)
712            }
713            Tf::Bc5RgUnorm | Tf::Bc5RgSnorm | Tf::EacRg11Unorm | Tf::EacRg11Snorm => {
714                (NumericDimension::Vector(Vs::Bi), Scalar::F32)
715            }
716            Tf::Bc6hRgbUfloat | Tf::Bc6hRgbFloat | Tf::Etc2Rgb8Unorm | Tf::Etc2Rgb8UnormSrgb => {
717                (NumericDimension::Vector(Vs::Tri), Scalar::F32)
718            }
719            Tf::Astc {
720                block: _,
721                channel: _,
722            } => (NumericDimension::Vector(Vs::Quad), Scalar::F32),
723        };
724
725        NumericType {
726            dim,
727            //Note: Shader always sees data as int, uint, or float.
728            // It doesn't know if the original is normalized in a tighter form.
729            scalar,
730        }
731    }
732
733    fn is_subtype_of(&self, other: &NumericType) -> bool {
734        if self.scalar.width > other.scalar.width {
735            return false;
736        }
737        if self.scalar.kind != other.scalar.kind {
738            return false;
739        }
740        match (self.dim, other.dim) {
741            (NumericDimension::Scalar, NumericDimension::Scalar) => true,
742            (NumericDimension::Scalar, NumericDimension::Vector(_)) => true,
743            (NumericDimension::Vector(s0), NumericDimension::Vector(s1)) => s0 <= s1,
744            (NumericDimension::Matrix(c0, r0), NumericDimension::Matrix(c1, r1)) => {
745                c0 == c1 && r0 == r1
746            }
747            _ => false,
748        }
749    }
750
751    fn is_compatible_with(&self, other: &NumericType) -> bool {
752        if self.scalar.kind != other.scalar.kind {
753            return false;
754        }
755        match (self.dim, other.dim) {
756            (NumericDimension::Scalar, NumericDimension::Scalar) => true,
757            (NumericDimension::Scalar, NumericDimension::Vector(_)) => true,
758            (NumericDimension::Vector(_), NumericDimension::Vector(_)) => true,
759            (NumericDimension::Matrix(..), NumericDimension::Matrix(..)) => true,
760            _ => false,
761        }
762    }
763}

765/// Return true if the fragment `format` is covered by the provided `output`.
766pub fn check_texture_format(
767    format: wgt::TextureFormat,
768    output: &NumericType,
769) -> Result<(), NumericType> {
770    let nt = NumericType::from_texture_format(format);
771    if nt.is_subtype_of(output) {
772        Ok(())
773    } else {
774        Err(nt)
775    }
776}

799pub type StageIo = FastHashMap<wgt::ShaderLocation, InterfaceVar>;
800
801impl Interface {
802    fn populate(
803        list: &mut Vec<Varying>,
804        binding: Option<&naga::Binding>,
805        ty: naga::Handle<naga::Type>,
806        arena: &naga::UniqueArena<naga::Type>,
807    ) {
808        let numeric_ty = match arena[ty].inner {
809            naga::TypeInner::Scalar(scalar) => NumericType {
810                dim: NumericDimension::Scalar,
811                scalar,
812            },
813            naga::TypeInner::Vector { size, scalar } => NumericType {
814                dim: NumericDimension::Vector(size),
815                scalar,
816            },
817            naga::TypeInner::Matrix {
818                columns,
819                rows,
820                scalar,
821            } => NumericType {
822                dim: NumericDimension::Matrix(columns, rows),
823                scalar,
824            },
825            naga::TypeInner::Struct { ref members, .. } => {
826                for member in members {
827                    Self::populate(list, member.binding.as_ref(), member.ty, arena);
828                }
829                return;
830            }
831            ref other => {
832                //Note: technically this should be at least `log::error`, but
833                // the reality is - every shader coming from `glslc` outputs an array
834                // of clip distances and hits this path :(
835                // So we lower it to `log::warn` to be less annoying.
836                log::warn!("Unexpected varying type: {:?}", other);
837                return;
838            }
839        };
840
841        let varying = match binding {
842            Some(&naga::Binding::Location {
843                location,
844                interpolation,
845                sampling,
846                .. // second_blend_source
847            }) => Varying::Local {
848                location,
849                iv: InterfaceVar {
850                    ty: numeric_ty,
851                    interpolation,
852                    sampling,
853                },
854            },
855            Some(&naga::Binding::BuiltIn(built_in)) => Varying::BuiltIn(built_in),
856            None => {
857                log::error!("Missing binding for a varying");
858                return;
859            }
860        };
861        list.push(varying);
862    }
863
864    pub fn new(
865        module: &naga::Module,
866        info: &naga::valid::ModuleInfo,
867        limits: wgt::Limits,
868        features: wgt::Features,
869    ) -> Self {
870        let mut resources = naga::Arena::new();
871        let mut resource_mapping = FastHashMap::default();
872        for (var_handle, var) in module.global_variables.iter() {
873            let bind = match var.binding {
874                Some(ref br) => br.clone(),
875                _ => continue,
876            };
877            let naga_ty = &module.types[var.ty].inner;
878
879            let inner_ty = match *naga_ty {
880                naga::TypeInner::BindingArray { base, .. } => &module.types[base].inner,
881                ref ty => ty,
882            };
883
884            let ty = match *inner_ty {
885                naga::TypeInner::Image {
886                    dim,
887                    arrayed,
888                    class,
889                } => ResourceType::Texture {
890                    dim,
891                    arrayed,
892                    class,
893                },
894                naga::TypeInner::Sampler { comparison } => ResourceType::Sampler { comparison },
895                naga::TypeInner::Array { stride, .. } => ResourceType::Buffer {
896                    size: wgt::BufferSize::new(stride as u64).unwrap(),
897                },
898                ref other => ResourceType::Buffer {
899                    size: wgt::BufferSize::new(other.size(module.to_ctx()) as u64).unwrap(),
900                },
901            };
902            let handle = resources.append(
903                Resource {
904                    name: var.name.clone(),
905                    bind,
906                    ty,
907                    class: var.space,
908                },
909                Default::default(),
910            );
911            resource_mapping.insert(var_handle, handle);
912        }
913
914        let mut entry_points = FastHashMap::default();
915        entry_points.reserve(module.entry_points.len());
916        for (index, entry_point) in module.entry_points.iter().enumerate() {
917            let info = info.get_entry_point(index);
918            let mut ep = EntryPoint::default();
919            for arg in entry_point.function.arguments.iter() {
920                Self::populate(&mut ep.inputs, arg.binding.as_ref(), arg.ty, &module.types);
921            }
922            if let Some(ref result) = entry_point.function.result {
923                Self::populate(
924                    &mut ep.outputs,
925                    result.binding.as_ref(),
926                    result.ty,
927                    &module.types,
928                );
929            }
930
931            for (var_handle, var) in module.global_variables.iter() {
932                let usage = info[var_handle];
933                if !usage.is_empty() && var.binding.is_some() {
934                    ep.resources.push(resource_mapping[&var_handle]);
935                }
936            }
937
938            for key in info.sampling_set.iter() {
939                ep.sampling_pairs
940                    .insert((resource_mapping[&key.image], resource_mapping[&key.sampler]));
941            }
942            ep.dual_source_blending = info.dual_source_blending;
943            ep.workgroup_size = entry_point.workgroup_size;
944
945            entry_points.insert((entry_point.stage, entry_point.name.clone()), ep);
946        }
947
948        Self {
949            limits,
950            features,
951            resources,
952            entry_points,
953        }
954    }

956    pub fn check_stage(
957        &self,
958        layouts: &mut BindingLayoutSource<'_>,
959        shader_binding_sizes: &mut FastHashMap<naga::ResourceBinding, wgt::BufferSize>,
960        entry_point_name: &str,
961        stage_bit: wgt::ShaderStages,
962        inputs: StageIo,
963        compare_function: Option<wgt::CompareFunction>,
964    ) -> Result<StageIo, StageError> {
965        // Since a shader module can have multiple entry points with the same name,
966        // we need to look for one with the right execution model.
967        let shader_stage = match stage_bit {
968            wgt::ShaderStages::VERTEX => naga::ShaderStage::Vertex,
969            wgt::ShaderStages::FRAGMENT => naga::ShaderStage::Fragment,
970            wgt::ShaderStages::COMPUTE => naga::ShaderStage::Compute,
971            _ => unreachable!(),
972        };
973        let pair = (shader_stage, entry_point_name.to_string());
974        let entry_point = self
975            .entry_points
976            .get(&pair)
977            .ok_or(StageError::MissingEntryPoint(pair.1))?;
978
979        // check resources visibility
980        for &handle in entry_point.resources.iter() {
981            let res = &self.resources[handle];
982            let result = 'err: {
983                match layouts {
984                    BindingLayoutSource::Provided(layouts) => {
985                        // update the required binding size for this buffer
986                        if let ResourceType::Buffer { size } = res.ty {
987                            match shader_binding_sizes.entry(res.bind.clone()) {
988                                Entry::Occupied(e) => {
989                                    *e.into_mut() = size.max(*e.get());
990                                }
991                                Entry::Vacant(e) => {
992                                    e.insert(size);
993                                }
994                            }
995                        }
996
997                        let Some(map) = layouts.get(res.bind.group as usize) else {
998                            break 'err Err(BindingError::Missing);
999                        };
1000
1001                        let Some(entry) = map.get(res.bind.binding) else {
1002                            break 'err Err(BindingError::Missing);
1003                        };
1004
1005                        if !entry.visibility.contains(stage_bit) {
1006                            break 'err Err(BindingError::Invisible);
1007                        }
1008
1009                        res.check_binding_use(entry)
1010                    }
1011                    BindingLayoutSource::Derived(layouts) => {
1012                        let Some(map) = layouts.get_mut(res.bind.group as usize) else {
1013                            break 'err Err(BindingError::Missing);
1014                        };
1015
1016                        let ty = match res.derive_binding_type() {
1017                            Ok(ty) => ty,
1018                            Err(error) => break 'err Err(error),
1019                        };
1020
1021                        match map.entry(res.bind.binding) {
1022                            indexmap::map::Entry::Occupied(e) if e.get().ty != ty => {
1023                                break 'err Err(BindingError::InconsistentlyDerivedType)
1024                            }
1025                            indexmap::map::Entry::Occupied(e) => {
1026                                e.into_mut().visibility |= stage_bit;
1027                            }
1028                            indexmap::map::Entry::Vacant(e) => {
1029                                e.insert(BindGroupLayoutEntry {
1030                                    binding: res.bind.binding,
1031                                    ty,
1032                                    visibility: stage_bit,
1033                                    count: None,
1034                                });
1035                            }
1036                        }
1037                        Ok(())
1038                    }
1039                }
1040            };
1041            if let Err(error) = result {
1042                return Err(StageError::Binding(res.bind.clone(), error));
1043            }
1044        }
1045
1046        // Check the compatibility between textures and samplers
1047        //
1048        // We only need to do this if the binding layout is provided by the user, as derived
1049        // layouts will inherently be correctly tagged.
1050        if let BindingLayoutSource::Provided(layouts) = layouts {
1051            for &(texture_handle, sampler_handle) in entry_point.sampling_pairs.iter() {
1052                let texture_bind = &self.resources[texture_handle].bind;
1053                let sampler_bind = &self.resources[sampler_handle].bind;
1054                let texture_layout = layouts[texture_bind.group as usize]
1055                    .get(texture_bind.binding)
1056                    .unwrap();
1057                let sampler_layout = layouts[sampler_bind.group as usize]
1058                    .get(sampler_bind.binding)
1059                    .unwrap();
1060                assert!(texture_layout.visibility.contains(stage_bit));
1061                assert!(sampler_layout.visibility.contains(stage_bit));
1062
1063                let sampler_filtering = matches!(
1064                    sampler_layout.ty,
1065                    wgt::BindingType::Sampler(wgt::SamplerBindingType::Filtering)
1066                );
1067                let texture_sample_type = match texture_layout.ty {
1068                    BindingType::Texture { sample_type, .. } => sample_type,
1069                    _ => unreachable!(),
1070                };
1071
1072                let error = match (sampler_filtering, texture_sample_type) {
1073                    (true, wgt::TextureSampleType::Float { filterable: false }) => {
1074                        Some(FilteringError::Float)
1075                    }
1076                    (true, wgt::TextureSampleType::Sint) => Some(FilteringError::Integer),
1077                    (true, wgt::TextureSampleType::Uint) => Some(FilteringError::Integer),
1078                    _ => None,
1079                };
1080
1081                if let Some(error) = error {
1082                    return Err(StageError::Filtering {
1083                        texture: texture_bind.clone(),
1084                        sampler: sampler_bind.clone(),
1085                        error,
1086                    });
1087                }
1088            }
1089        }
1090
1091        // check workgroup size limits
1092        if shader_stage == naga::ShaderStage::Compute {
1093            let max_workgroup_size_limits = [
1094                self.limits.max_compute_workgroup_size_x,
1095                self.limits.max_compute_workgroup_size_y,
1096                self.limits.max_compute_workgroup_size_z,
1097            ];
1098            let total_invocations = entry_point.workgroup_size.iter().product::<u32>();
1099
1100            if entry_point.workgroup_size.iter().any(|&s| s == 0)
1101                || total_invocations > self.limits.max_compute_invocations_per_workgroup
1102                || entry_point.workgroup_size[0] > max_workgroup_size_limits[0]
1103                || entry_point.workgroup_size[1] > max_workgroup_size_limits[1]
1104                || entry_point.workgroup_size[2] > max_workgroup_size_limits[2]
1105            {
1106                return Err(StageError::InvalidWorkgroupSize {
1107                    current: entry_point.workgroup_size,
1108                    current_total: total_invocations,
1109                    limit: max_workgroup_size_limits,
1110                    total: self.limits.max_compute_invocations_per_workgroup,
1111                });
1112            }
1113        }
1114
1115        let mut inter_stage_components = 0;
1116
1117        // check inputs compatibility
1118        for input in entry_point.inputs.iter() {
1119            match *input {
1120                Varying::Local { location, ref iv } => {
1121                    let result =
1122                        inputs
1123                            .get(&location)
1124                            .ok_or(InputError::Missing)
1125                            .and_then(|provided| {
1126                                let (compatible, num_components) = match shader_stage {
1127                                    // For vertex attributes, there are defaults filled out
1128                                    // by the driver if data is not provided.
1129                                    naga::ShaderStage::Vertex => {
1130                                        // vertex inputs don't count towards inter-stage
1131                                        (iv.ty.is_compatible_with(&provided.ty), 0)
1132                                    }
1133                                    naga::ShaderStage::Fragment => {
1134                                        if iv.interpolation != provided.interpolation {
1135                                            return Err(InputError::InterpolationMismatch(
1136                                                provided.interpolation,
1137                                            ));
1138                                        }
1139                                        if iv.sampling != provided.sampling {
1140                                            return Err(InputError::SamplingMismatch(
1141                                                provided.sampling,
1142                                            ));
1143                                        }
1144                                        (
1145                                            iv.ty.is_subtype_of(&provided.ty),
1146                                            iv.ty.dim.num_components(),
1147                                        )
1148                                    }
1149                                    naga::ShaderStage::Compute => (false, 0),
1150                                };
1151                                if compatible {
1152                                    Ok(num_components)
1153                                } else {
1154                                    Err(InputError::WrongType(provided.ty))
1155                                }
1156                            });
1157                    match result {
1158                        Ok(num_components) => {
1159                            inter_stage_components += num_components;
1160                        }
1161                        Err(error) => {
1162                            return Err(StageError::Input {
1163                                location,
1164                                var: iv.clone(),
1165                                error,
1166                            })
1167                        }
1168                    }
1169                }
1170                Varying::BuiltIn(_) => {}
1171            }
1172        }
1173
1174        // Check all vertex outputs and make sure the fragment shader consumes them.
1175        // This requirement is removed if the `SHADER_UNUSED_VERTEX_OUTPUT` feature is enabled.
1176        if shader_stage == naga::ShaderStage::Fragment
1177            && !self
1178                .features
1179                .contains(wgt::Features::SHADER_UNUSED_VERTEX_OUTPUT)
1180        {
1181            for &index in inputs.keys() {
1182                // This is a linear scan, but the count should be low enough
1183                // that this should be fine.
1184                let found = entry_point.inputs.iter().any(|v| match *v {
1185                    Varying::Local { location, .. } => location == index,
1186                    Varying::BuiltIn(_) => false,
1187                });
1188
1189                if !found {
1190                    return Err(StageError::InputNotConsumed { location: index });
1191                }
1192            }
1193        }
1194
1195        if shader_stage == naga::ShaderStage::Vertex {
1196            for output in entry_point.outputs.iter() {
1197                //TODO: count builtins towards the limit?
1198                inter_stage_components += match *output {
1199                    Varying::Local { ref iv, .. } => iv.ty.dim.num_components(),
1200                    Varying::BuiltIn(_) => 0,
1201                };
1202
1203                if let Some(
1204                    cmp @ wgt::CompareFunction::Equal | cmp @ wgt::CompareFunction::NotEqual,
1205                ) = compare_function
1206                {
1207                    if let Varying::BuiltIn(naga::BuiltIn::Position { invariant: false }) = *output
1208                    {
1209                        log::warn!(
1210                            "Vertex shader with entry point {entry_point_name} outputs a @builtin(position) without the @invariant \
1211                            attribute and is used in a pipeline with {cmp:?}. On some machines, this can cause bad artifacting as {cmp:?} assumes \
1212                            the values output from the vertex shader exactly match the value in the depth buffer. The @invariant attribute on the \
1213                            @builtin(position) vertex output ensures that the exact same pixel depths are used every render."
1214                        );
1215                    }
1216                }
1217            }
1218        }
1219
1220        if inter_stage_components > self.limits.max_inter_stage_shader_components {
1221            return Err(StageError::TooManyVaryings {
1222                used: inter_stage_components,
1223                limit: self.limits.max_inter_stage_shader_components,
1224            });
1225        }
1226
1227        let outputs = entry_point
1228            .outputs
1229            .iter()
1230            .filter_map(|output| match *output {
1231                Varying::Local { location, ref iv } => Some((location, iv.clone())),
1232                Varying::BuiltIn(_) => None,
1233            })
1234            .collect();
1235        Ok(outputs)
1236    }
