1/*! This library describes the internal unsafe graphics abstraction API.
2 *  It follows WebGPU for the most part, re-using wgpu-types,
3 *  with the following deviations:
4 *  - Fully unsafe: zero overhead, zero validation.
5 *  - Compile-time backend selection via traits.
6 *  - Objects are passed by references and returned by value. No IDs.
7 *  - Mapping is persistent, with explicit synchronization.
8 *  - Resource transitions are explicit.
9 *  - All layouts are explicit. Binding model has compatibility.
10 *
11 *  General design direction is to follow the majority by the following weights:
12 *  - wgpu-core: 1.5
13 *  - primary backends (Vulkan/Metal/DX12): 1.0 each
14 *  - secondary backend (GLES): 0.5
15 */

154/// Error occurring while trying to create an instance, or create a surface from an instance;
155/// typically relating to the state of the underlying graphics API or hardware.
156#[derive(Clone, Debug, Error)]
157#[error("{message}")]
158pub struct InstanceError {
159    /// These errors are very platform specific, so do not attempt to encode them as an enum.
160    ///
161    /// This message should describe the problem in sufficient detail to be useful for a
162    /// user-to-developer “why won't this work on my machine” bug report, and otherwise follow
163    /// <https://rust-lang.github.io/api-guidelines/interoperability.html#error-types-are-meaningful-and-well-behaved-c-good-err>.
164    message: String,
165
166    /// Underlying error value, if any is available.
167    #[source]
168    source: Option<Arc<dyn std::error::Error + Send + Sync + 'static>>,
169}
170
171impl InstanceError {
172    #[allow(dead_code)] // may be unused on some platforms
173    pub(crate) fn new(message: String) -> Self {
174        Self {
175            message,
176            source: None,
177        }
178    }
179    #[allow(dead_code)] // may be unused on some platforms
180    pub(crate) fn with_source(
181        message: String,
182        source: impl std::error::Error + Send + Sync + 'static,
183    ) -> Self {
184        Self {
185            message,
186            source: Some(Arc::new(source)),
187        }
188    }
189}

230pub trait Surface<A: Api>: WasmNotSendSync {
231    /// Configures the surface to use the given device.
232    ///
233    /// # Safety
234    ///
235    /// - All gpu work that uses the surface must have been completed.
236    /// - All [`AcquiredSurfaceTexture`]s must have been destroyed.
237    /// - All [`Api::TextureView`]s derived from the [`AcquiredSurfaceTexture`]s must have been destroyed.
238    /// - All surfaces created using other devices must have been unconfigured before this call.
239    unsafe fn configure(
240        &self,
241        device: &A::Device,
242        config: &SurfaceConfiguration,
243    ) -> Result<(), SurfaceError>;
244
245    /// Unconfigures the surface on the given device.
246    ///
247    /// # Safety
248    ///
249    /// - All gpu work that uses the surface must have been completed.
250    /// - All [`AcquiredSurfaceTexture`]s must have been destroyed.
251    /// - All [`Api::TextureView`]s derived from the [`AcquiredSurfaceTexture`]s must have been destroyed.
252    /// - The surface must have been configured on the given device.
253    unsafe fn unconfigure(&self, device: &A::Device);
254
255    /// Returns the next texture to be presented by the swapchain for drawing
256    ///
257    /// A `timeout` of `None` means to wait indefinitely, with no timeout.
258    ///
259    /// # Portability
260    ///
261    /// Some backends can't support a timeout when acquiring a texture and
262    /// the timeout will be ignored.
263    ///
264    /// Returns `None` on timing out.
265    unsafe fn acquire_texture(
266        &self,
267        timeout: Option<std::time::Duration>,
268    ) -> Result<Option<AcquiredSurfaceTexture<A>>, SurfaceError>;
269    unsafe fn discard_texture(&self, texture: A::SurfaceTexture);
270}

721bitflags!(
722    /// Texture format capability flags.
723    #[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
724    pub struct FormatAspects: u8 {
725        const COLOR = 1 << 0;
726        const DEPTH = 1 << 1;
727        const STENCIL = 1 << 2;
728        const PLANE_0 = 1 << 3;
729        const PLANE_1 = 1 << 4;
730        const PLANE_2 = 1 << 5;
731
732        const DEPTH_STENCIL = Self::DEPTH.bits() | Self::STENCIL.bits();
733    }
734);
735
736impl FormatAspects {
737    pub fn new(format: wgt::TextureFormat, aspect: wgt::TextureAspect) -> Self {
738        let aspect_mask = match aspect {
739            wgt::TextureAspect::All => Self::all(),
740            wgt::TextureAspect::DepthOnly => Self::DEPTH,
741            wgt::TextureAspect::StencilOnly => Self::STENCIL,
742            wgt::TextureAspect::Plane0 => Self::PLANE_0,
743            wgt::TextureAspect::Plane1 => Self::PLANE_1,
744            wgt::TextureAspect::Plane2 => Self::PLANE_2,
745        };
746        Self::from(format) & aspect_mask
747    }
748
749    /// Returns `true` if only one flag is set
750    pub fn is_one(&self) -> bool {
751        self.bits().count_ones() == 1
752    }
753
754    pub fn map(&self) -> wgt::TextureAspect {
755        match *self {
756            Self::COLOR => wgt::TextureAspect::All,
757            Self::DEPTH => wgt::TextureAspect::DepthOnly,
758            Self::STENCIL => wgt::TextureAspect::StencilOnly,
759            Self::PLANE_0 => wgt::TextureAspect::Plane0,
760            Self::PLANE_1 => wgt::TextureAspect::Plane1,
761            Self::PLANE_2 => wgt::TextureAspect::Plane2,
762            _ => unreachable!(),
763        }
764    }
765}
766
767impl From<wgt::TextureFormat> for FormatAspects {
768    fn from(format: wgt::TextureFormat) -> Self {
769        match format {
770            wgt::TextureFormat::Stencil8 => Self::STENCIL,
771            wgt::TextureFormat::Depth16Unorm
772            | wgt::TextureFormat::Depth32Float
773            | wgt::TextureFormat::Depth24Plus => Self::DEPTH,
774            wgt::TextureFormat::Depth32FloatStencil8 | wgt::TextureFormat::Depth24PlusStencil8 => {
775                Self::DEPTH_STENCIL
776            }
777            wgt::TextureFormat::NV12 => Self::PLANE_0 | Self::PLANE_1,
778            _ => Self::COLOR,
779        }
780    }
781}

801bitflags::bitflags! {
802    /// Similar to `wgt::BufferUsages` but for internal use.
803    #[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
804    pub struct BufferUses: u16 {
805        /// The argument to a read-only mapping.
806        const MAP_READ = 1 << 0;
807        /// The argument to a write-only mapping.
808        const MAP_WRITE = 1 << 1;
809        /// The source of a hardware copy.
810        const COPY_SRC = 1 << 2;
811        /// The destination of a hardware copy.
812        const COPY_DST = 1 << 3;
813        /// The index buffer used for drawing.
814        const INDEX = 1 << 4;
815        /// A vertex buffer used for drawing.
816        const VERTEX = 1 << 5;
817        /// A uniform buffer bound in a bind group.
818        const UNIFORM = 1 << 6;
819        /// A read-only storage buffer used in a bind group.
820        const STORAGE_READ = 1 << 7;
821        /// A read-write or write-only buffer used in a bind group.
822        const STORAGE_READ_WRITE = 1 << 8;
823        /// The indirect or count buffer in a indirect draw or dispatch.
824        const INDIRECT = 1 << 9;
825        /// A buffer used to store query results.
826        const QUERY_RESOLVE = 1 << 10;
827        const ACCELERATION_STRUCTURE_SCRATCH = 1 << 11;
828        const BOTTOM_LEVEL_ACCELERATION_STRUCTURE_INPUT = 1 << 12;
829        const TOP_LEVEL_ACCELERATION_STRUCTURE_INPUT = 1 << 13;
830        /// The combination of states that a buffer may be in _at the same time_.
831        const INCLUSIVE = Self::MAP_READ.bits() | Self::COPY_SRC.bits() |
832            Self::INDEX.bits() | Self::VERTEX.bits() | Self::UNIFORM.bits() |
833            Self::STORAGE_READ.bits() | Self::INDIRECT.bits() | Self::BOTTOM_LEVEL_ACCELERATION_STRUCTURE_INPUT.bits() | Self::TOP_LEVEL_ACCELERATION_STRUCTURE_INPUT.bits();
834        /// The combination of states that a buffer must exclusively be in.
835        const EXCLUSIVE = Self::MAP_WRITE.bits() | Self::COPY_DST.bits() | Self::STORAGE_READ_WRITE.bits() | Self::ACCELERATION_STRUCTURE_SCRATCH.bits();
836        /// The combination of all usages that the are guaranteed to be be ordered by the hardware.
837        /// If a usage is ordered, then if the buffer state doesn't change between draw calls, there
838        /// are no barriers needed for synchronization.
839        const ORDERED = Self::INCLUSIVE.bits() | Self::MAP_WRITE.bits();
840    }
841}

843bitflags::bitflags! {
844    /// Similar to `wgt::TextureUsages` but for internal use.
845    #[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
846    pub struct TextureUses: u16 {
847        /// The texture is in unknown state.
848        const UNINITIALIZED = 1 << 0;
849        /// Ready to present image to the surface.
850        const PRESENT = 1 << 1;
851        /// The source of a hardware copy.
852        const COPY_SRC = 1 << 2;
853        /// The destination of a hardware copy.
854        const COPY_DST = 1 << 3;
855        /// Read-only sampled or fetched resource.
856        const RESOURCE = 1 << 4;
857        /// The color target of a renderpass.
858        const COLOR_TARGET = 1 << 5;
859        /// Read-only depth stencil usage.
860        const DEPTH_STENCIL_READ = 1 << 6;
861        /// Read-write depth stencil usage
862        const DEPTH_STENCIL_WRITE = 1 << 7;
863        /// Read-only storage buffer usage. Corresponds to a UAV in d3d, so is exclusive, despite being read only.
864        const STORAGE_READ = 1 << 8;
865        /// Read-write or write-only storage buffer usage.
866        const STORAGE_READ_WRITE = 1 << 9;
867        /// The combination of states that a texture may be in _at the same time_.
868        const INCLUSIVE = Self::COPY_SRC.bits() | Self::RESOURCE.bits() | Self::DEPTH_STENCIL_READ.bits();
869        /// The combination of states that a texture must exclusively be in.
870        const EXCLUSIVE = Self::COPY_DST.bits() | Self::COLOR_TARGET.bits() | Self::DEPTH_STENCIL_WRITE.bits() | Self::STORAGE_READ.bits() | Self::STORAGE_READ_WRITE.bits() | Self::PRESENT.bits();
871        /// The combination of all usages that the are guaranteed to be be ordered by the hardware.
872        /// If a usage is ordered, then if the texture state doesn't change between draw calls, there
873        /// are no barriers needed for synchronization.
874        const ORDERED = Self::INCLUSIVE.bits() | Self::COLOR_TARGET.bits() | Self::DEPTH_STENCIL_WRITE.bits() | Self::STORAGE_READ.bits();
875
876        /// Flag used by the wgpu-core texture tracker to say a texture is in different states for every sub-resource
877        const COMPLEX = 1 << 10;
878        /// Flag used by the wgpu-core texture tracker to say that the tracker does not know the state of the sub-resource.
879        /// This is different from UNINITIALIZED as that says the tracker does know, but the texture has not been initialized.
880        const UNKNOWN = 1 << 11;
881    }

1436/// Stores the text of any validation errors that have occurred since
1437/// the last call to `get_and_reset`.
1438///
1439/// Each value is a validation error and a message associated with it,
1440/// or `None` if the error has no message from the api.
1441///
1442/// This is used for internal wgpu testing only and _must not_ be used
1443/// as a way to check for errors.
1444///
1445/// This works as a static because `cargo nextest` runs all of our
1446/// tests in separate processes, so each test gets its own canary.
1447///
1448/// This prevents the issue of one validation error terminating the
1449/// entire process.
1450pub static VALIDATION_CANARY: ValidationCanary = ValidationCanary {
1451    inner: Mutex::new(Vec::new()),
1452};
1453
1454/// Flag for internal testing.
1455pub struct ValidationCanary {
1456    inner: Mutex<Vec<String>>,
1457}
1458
1459impl ValidationCanary {
1460    #[allow(dead_code)] // in some configurations this function is dead
1461    fn add(&self, msg: String) {
1462        self.inner.lock().push(msg);
1463    }
1464
1465    /// Returns any API validation errors that hav occurred in this process
1466    /// since the last call to this function.
1467    pub fn get_and_reset(&self) -> Vec<String> {
1468        self.inner.lock().drain(..).collect()
1469    }
1470}
