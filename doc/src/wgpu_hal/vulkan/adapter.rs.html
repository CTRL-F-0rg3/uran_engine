51    /// Add the members of `self` into `info.enabled_features` and its `p_next` chain.
52    pub fn add_to_device_create_builder<'a>(
53        &'a mut self,
54        mut info: vk::DeviceCreateInfoBuilder<'a>,
55    ) -> vk::DeviceCreateInfoBuilder<'a> {
56        info = info.enabled_features(&self.core);
57        if let Some(ref mut feature) = self.descriptor_indexing {
58            info = info.push_next(feature);
59        }
60        if let Some(ref mut feature) = self.imageless_framebuffer {
61            info = info.push_next(feature);
62        }
63        if let Some(ref mut feature) = self.timeline_semaphore {
64            info = info.push_next(feature);
65        }
66        if let Some(ref mut feature) = self.image_robustness {
67            info = info.push_next(feature);
68        }
69        if let Some(ref mut feature) = self.robustness2 {
70            info = info.push_next(feature);
71        }
72        if let Some(ref mut feature) = self.astc_hdr {
73            info = info.push_next(feature);
74        }
75        if let Some((ref mut f16_i8_feature, ref mut _16bit_feature)) = self.shader_float16 {
76            info = info.push_next(f16_i8_feature);
77            info = info.push_next(_16bit_feature);
78        }
79        if let Some(ref mut feature) = self.zero_initialize_workgroup_memory {
80            info = info.push_next(feature);
81        }
82        if let Some(ref mut feature) = self.acceleration_structure {
83            info = info.push_next(feature);
84        }
85        if let Some(ref mut feature) = self.buffer_device_address {
86            info = info.push_next(feature);
87        }
88        if let Some(ref mut feature) = self.ray_query {
89            info = info.push_next(feature);
90        }
91        info
92    }

199            descriptor_indexing: if requested_features.intersects(indexing_features()) {

298                Some((

322                .contains(&vk::KhrBufferDeviceAddressFn::name())

328                )
329            } else {
330                None
331            },
332            ray_query: if enabled_extensions.contains(&vk::KhrRayQueryFn::name()) {
333                Some(
334                    vk::PhysicalDeviceRayQueryFeaturesKHR::builder()
335                        .ray_query(true)

357    fn to_wgpu(
358        &self,
359        instance: &ash::Instance,
360        phd: vk::PhysicalDevice,
361        caps: &PhysicalDeviceCapabilities,
362    ) -> (wgt::Features, wgt::DownlevelFlags) {
363        use crate::auxil::db;
364        use wgt::{DownlevelFlags as Df, Features as F};
365        let mut features = F::empty()
366            | F::SPIRV_SHADER_PASSTHROUGH
367            | F::MAPPABLE_PRIMARY_BUFFERS
368            | F::PUSH_CONSTANTS
369            | F::ADDRESS_MODE_CLAMP_TO_BORDER
370            | F::ADDRESS_MODE_CLAMP_TO_ZERO
371            | F::TIMESTAMP_QUERY
372            | F::TIMESTAMP_QUERY_INSIDE_PASSES
373            | F::TEXTURE_ADAPTER_SPECIFIC_FORMAT_FEATURES
374            | F::CLEAR_TEXTURE;
375
376        let mut dl_flags = Df::COMPUTE_SHADERS
377            | Df::BASE_VERTEX
378            | Df::READ_ONLY_DEPTH_STENCIL
379            | Df::NON_POWER_OF_TWO_MIPMAPPED_TEXTURES
380            | Df::COMPARISON_SAMPLERS
381            | Df::VERTEX_STORAGE
382            | Df::FRAGMENT_STORAGE
383            | Df::DEPTH_TEXTURE_AND_BUFFER_COPIES
384            | Df::BUFFER_BINDINGS_NOT_16_BYTE_ALIGNED
385            | Df::UNRESTRICTED_INDEX_BUFFER
386            | Df::INDIRECT_EXECUTION
387            | Df::VIEW_FORMATS
388            | Df::UNRESTRICTED_EXTERNAL_TEXTURE_COPIES
389            | Df::NONBLOCKING_QUERY_RESOLVE
390            | Df::VERTEX_AND_INSTANCE_INDEX_RESPECTS_RESPECTIVE_FIRST_VALUE_IN_INDIRECT_DRAW;
391
392        dl_flags.set(
393            Df::SURFACE_VIEW_FORMATS,
394            caps.supports_extension(vk::KhrSwapchainMutableFormatFn::name()),
395        );
396        dl_flags.set(Df::CUBE_ARRAY_TEXTURES, self.core.image_cube_array != 0);
397        dl_flags.set(Df::ANISOTROPIC_FILTERING, self.core.sampler_anisotropy != 0);
398        dl_flags.set(
399            Df::FRAGMENT_WRITABLE_STORAGE,
400            self.core.fragment_stores_and_atomics != 0,
401        );
402        dl_flags.set(Df::MULTISAMPLED_SHADING, self.core.sample_rate_shading != 0);
403        dl_flags.set(Df::INDEPENDENT_BLEND, self.core.independent_blend != 0);
404        dl_flags.set(
405            Df::FULL_DRAW_INDEX_UINT32,
406            self.core.full_draw_index_uint32 != 0,
407        );
408        dl_flags.set(Df::DEPTH_BIAS_CLAMP, self.core.depth_bias_clamp != 0);
409
410        features.set(
411            F::INDIRECT_FIRST_INSTANCE,
412            self.core.draw_indirect_first_instance != 0,
413        );
414        //if self.core.dual_src_blend != 0
415        features.set(F::MULTI_DRAW_INDIRECT, self.core.multi_draw_indirect != 0);
416        features.set(F::POLYGON_MODE_LINE, self.core.fill_mode_non_solid != 0);
417        features.set(F::POLYGON_MODE_POINT, self.core.fill_mode_non_solid != 0);
418        //if self.core.depth_bounds != 0 {
419        //if self.core.alpha_to_one != 0 {
420        //if self.core.multi_viewport != 0 {
421        features.set(
422            F::TEXTURE_COMPRESSION_ETC2,
423            self.core.texture_compression_etc2 != 0,
424        );
425        features.set(
426            F::TEXTURE_COMPRESSION_ASTC,
427            self.core.texture_compression_astc_ldr != 0,
428        );
429        features.set(
430            F::TEXTURE_COMPRESSION_BC,
431            self.core.texture_compression_bc != 0,
432        );
433        features.set(
434            F::PIPELINE_STATISTICS_QUERY,
435            self.core.pipeline_statistics_query != 0,
436        );
437        features.set(
438            F::VERTEX_WRITABLE_STORAGE,
439            self.core.vertex_pipeline_stores_and_atomics != 0,
440        );
441        //if self.core.shader_image_gather_extended != 0 {
442        //if self.core.shader_storage_image_extended_formats != 0 {
443        features.set(
444            F::BUFFER_BINDING_ARRAY,
445            self.core.shader_uniform_buffer_array_dynamic_indexing != 0,
446        );
447        features.set(
448            F::TEXTURE_BINDING_ARRAY,
449            self.core.shader_sampled_image_array_dynamic_indexing != 0,
450        );
451        features.set(F::SHADER_PRIMITIVE_INDEX, self.core.geometry_shader != 0);
452        if Self::all_features_supported(
453            &features,
454            &[
455                (
456                    F::BUFFER_BINDING_ARRAY,
457                    self.core.shader_storage_buffer_array_dynamic_indexing,
458                ),
459                (
460                    F::TEXTURE_BINDING_ARRAY,
461                    self.core.shader_storage_image_array_dynamic_indexing,
462                ),
463            ],
464        ) {
465            features.insert(F::STORAGE_RESOURCE_BINDING_ARRAY);
466        }
467        //if self.core.shader_storage_image_array_dynamic_indexing != 0 {
468        //if self.core.shader_clip_distance != 0 {
469        //if self.core.shader_cull_distance != 0 {
470        features.set(F::SHADER_F64, self.core.shader_float64 != 0);
471        //if self.core.shader_int64 != 0 {
472        features.set(F::SHADER_I16, self.core.shader_int16 != 0);
473
474        //if caps.supports_extension(vk::KhrSamplerMirrorClampToEdgeFn::name()) {
475        //if caps.supports_extension(vk::ExtSamplerFilterMinmaxFn::name()) {
476        features.set(
477            F::MULTI_DRAW_INDIRECT_COUNT,
478            caps.supports_extension(vk::KhrDrawIndirectCountFn::name()),
479        );
480        features.set(
481            F::CONSERVATIVE_RASTERIZATION,
482            caps.supports_extension(vk::ExtConservativeRasterizationFn::name()),
483        );
484
485        let intel_windows = caps.properties.vendor_id == db::intel::VENDOR && cfg!(windows);
486
487        if let Some(ref descriptor_indexing) = self.descriptor_indexing {
488            const STORAGE: F = F::STORAGE_RESOURCE_BINDING_ARRAY;
489            if Self::all_features_supported(
490                &features,
491                &[
492                    (
493                        F::TEXTURE_BINDING_ARRAY,
494                        descriptor_indexing.shader_sampled_image_array_non_uniform_indexing,
495                    ),
496                    (
497                        F::BUFFER_BINDING_ARRAY | STORAGE,
498                        descriptor_indexing.shader_storage_buffer_array_non_uniform_indexing,
499                    ),
500                ],
501            ) {
502                features.insert(F::SAMPLED_TEXTURE_AND_STORAGE_BUFFER_ARRAY_NON_UNIFORM_INDEXING);
503            }
504            if Self::all_features_supported(
505                &features,
506                &[
507                    (
508                        F::BUFFER_BINDING_ARRAY,
509                        descriptor_indexing.shader_uniform_buffer_array_non_uniform_indexing,
510                    ),
511                    (
512                        F::TEXTURE_BINDING_ARRAY | STORAGE,
513                        descriptor_indexing.shader_storage_image_array_non_uniform_indexing,
514                    ),
515                ],
516            ) {
517                features.insert(F::UNIFORM_BUFFER_AND_STORAGE_TEXTURE_ARRAY_NON_UNIFORM_INDEXING);
518            }
519            if descriptor_indexing.descriptor_binding_partially_bound != 0 && !intel_windows {
520                features |= F::PARTIALLY_BOUND_BINDING_ARRAY;
521            }
522        }
523
524        features.set(F::DEPTH_CLIP_CONTROL, self.core.depth_clamp != 0);
525        features.set(F::DUAL_SOURCE_BLENDING, self.core.dual_src_blend != 0);
526
527        if let Some(ref multiview) = self.multiview {
528            features.set(F::MULTIVIEW, multiview.multiview != 0);
529        }
530
531        features.set(
532            F::TEXTURE_FORMAT_16BIT_NORM,
533            is_format_16bit_norm_supported(instance, phd),
534        );
535
536        if let Some(ref astc_hdr) = self.astc_hdr {
537            features.set(
538                F::TEXTURE_COMPRESSION_ASTC_HDR,
539                astc_hdr.texture_compression_astc_hdr != 0,
540            );
541        }
542
543        if let Some((ref f16_i8, ref bit16)) = self.shader_float16 {
544            features.set(
545                F::SHADER_F16,
546                f16_i8.shader_float16 != 0
547                    && bit16.storage_buffer16_bit_access != 0
548                    && bit16.uniform_and_storage_buffer16_bit_access != 0,
549            );
550        }
551
552        let supports_depth_format = |format| {
553            supports_format(
554                instance,
555                phd,
556                format,
557                vk::ImageTiling::OPTIMAL,
558                depth_stencil_required_flags(),
559            )
560        };
561
562        let texture_s8 = supports_depth_format(vk::Format::S8_UINT);
563        let texture_d32 = supports_depth_format(vk::Format::D32_SFLOAT);
564        let texture_d24_s8 = supports_depth_format(vk::Format::D24_UNORM_S8_UINT);
565        let texture_d32_s8 = supports_depth_format(vk::Format::D32_SFLOAT_S8_UINT);
566
567        let stencil8 = texture_s8 || texture_d24_s8;
568        let depth24_plus_stencil8 = texture_d24_s8 || texture_d32_s8;
569
570        dl_flags.set(
571            Df::WEBGPU_TEXTURE_FORMAT_SUPPORT,
572            stencil8 && depth24_plus_stencil8 && texture_d32,
573        );
574
575        features.set(F::DEPTH32FLOAT_STENCIL8, texture_d32_s8);
576
577        features.set(
578            F::RAY_TRACING_ACCELERATION_STRUCTURE,
579            caps.supports_extension(vk::KhrDeferredHostOperationsFn::name())
580                && caps.supports_extension(vk::KhrAccelerationStructureFn::name())
581                && caps.supports_extension(vk::KhrBufferDeviceAddressFn::name()),
582        );
583
584        features.set(
585            F::RAY_QUERY,
586            caps.supports_extension(vk::KhrRayQueryFn::name()),
587        );
588
589        let rg11b10ufloat_renderable = supports_format(
590            instance,
591            phd,
592            vk::Format::B10G11R11_UFLOAT_PACK32,
593            vk::ImageTiling::OPTIMAL,
594            vk::FormatFeatureFlags::COLOR_ATTACHMENT
595                | vk::FormatFeatureFlags::COLOR_ATTACHMENT_BLEND,
596        );
597        features.set(F::RG11B10UFLOAT_RENDERABLE, rg11b10ufloat_renderable);
598        features.set(F::SHADER_UNUSED_VERTEX_OUTPUT, true);
599
600        features.set(
601            F::BGRA8UNORM_STORAGE,
602            supports_bgra8unorm_storage(instance, phd, caps.device_api_version),
603        );
604
605        features.set(
606            F::FLOAT32_FILTERABLE,
607            is_float32_filterable_supported(instance, phd),
608        );
609
610        if let Some(ref _sampler_ycbcr_conversion) = self.sampler_ycbcr_conversion {
611            features.set(
612                F::TEXTURE_FORMAT_NV12,
613                supports_format(
614                    instance,
615                    phd,
616                    vk::Format::G8_B8R8_2PLANE_420_UNORM,
617                    vk::ImageTiling::OPTIMAL,
618                    vk::FormatFeatureFlags::SAMPLED_IMAGE
619                        | vk::FormatFeatureFlags::TRANSFER_SRC
620                        | vk::FormatFeatureFlags::TRANSFER_DST,
621                ) && !caps
622                    .driver
623                    .map(|driver| driver.driver_id == vk::DriverId::MOLTENVK)
624                    .unwrap_or_default(),
625            );
626        }
627
628        (features, dl_flags)
629    }

631    fn all_features_supported(
632        features: &wgt::Features,
633        implications: &[(wgt::Features, vk::Bool32)],
634    ) -> bool {
635        implications
636            .iter()
637            .all(|&(flag, support)| !features.contains(flag) || support != 0)
638    }

667    pub fn supports_extension(&self, extension: &CStr) -> bool {
668        use crate::auxil::cstr_from_bytes_until_nul;
669        self.supported_extensions
670            .iter()
671            .any(|ep| cstr_from_bytes_until_nul(&ep.extension_name) == Some(extension))
672    }

674    /// Map `requested_features` to the list of Vulkan extension strings required to create the logical device.
675    fn get_required_extensions(&self, requested_features: wgt::Features) -> Vec<&'static CStr> {
676        let mut extensions = Vec::new();
677
678        // Note that quite a few extensions depend on the `VK_KHR_get_physical_device_properties2` instance extension.
679        // We enable `VK_KHR_get_physical_device_properties2` unconditionally (if available).
680
681        // Require `VK_KHR_swapchain`
682        extensions.push(vk::KhrSwapchainFn::name());
683
684        if self.device_api_version < vk::API_VERSION_1_1 {
685            // Require either `VK_KHR_maintenance1` or `VK_AMD_negative_viewport_height`
686            if self.supports_extension(vk::KhrMaintenance1Fn::name()) {
687                extensions.push(vk::KhrMaintenance1Fn::name());
688            } else {
689                // `VK_AMD_negative_viewport_height` is obsoleted by `VK_KHR_maintenance1` and must not be enabled alongside it
690                extensions.push(vk::AmdNegativeViewportHeightFn::name());
691            }
692
693            // Optional `VK_KHR_maintenance2`
694            if self.supports_extension(vk::KhrMaintenance2Fn::name()) {
695                extensions.push(vk::KhrMaintenance2Fn::name());
696            }
697
698            // Optional `VK_KHR_maintenance3`
699            if self.supports_extension(vk::KhrMaintenance3Fn::name()) {
700                extensions.push(vk::KhrMaintenance3Fn::name());
701            }
702
703            // Require `VK_KHR_storage_buffer_storage_class`
704            extensions.push(vk::KhrStorageBufferStorageClassFn::name());
705
706            // Require `VK_KHR_multiview` if the associated feature was requested
707            if requested_features.contains(wgt::Features::MULTIVIEW) {
708                extensions.push(vk::KhrMultiviewFn::name());
709            }
710
711            // Require `VK_KHR_sampler_ycbcr_conversion` if the associated feature was requested
712            if requested_features.contains(wgt::Features::TEXTURE_FORMAT_NV12) {
713                extensions.push(vk::KhrSamplerYcbcrConversionFn::name());
714            }
715        }
716
717        if self.device_api_version < vk::API_VERSION_1_2 {
718            // Optional `VK_KHR_image_format_list`
719            if self.supports_extension(vk::KhrImageFormatListFn::name()) {
720                extensions.push(vk::KhrImageFormatListFn::name());
721            }
722
723            // Optional `VK_KHR_imageless_framebuffer`
724            if self.supports_extension(vk::KhrImagelessFramebufferFn::name()) {
725                extensions.push(vk::KhrImagelessFramebufferFn::name());
726                // Require `VK_KHR_maintenance2` due to it being a dependency
727                if self.device_api_version < vk::API_VERSION_1_1 {
728                    extensions.push(vk::KhrMaintenance2Fn::name());
729                }
730            }
731
732            // Optional `VK_KHR_driver_properties`
733            if self.supports_extension(vk::KhrDriverPropertiesFn::name()) {
734                extensions.push(vk::KhrDriverPropertiesFn::name());
735            }
736
737            // Optional `VK_KHR_timeline_semaphore`
738            if self.supports_extension(vk::KhrTimelineSemaphoreFn::name()) {
739                extensions.push(vk::KhrTimelineSemaphoreFn::name());
740            }
741
742            // Require `VK_EXT_descriptor_indexing` if one of the associated features was requested
743            if requested_features.intersects(indexing_features()) {
744                extensions.push(vk::ExtDescriptorIndexingFn::name());
745            }
746
747            // Require `VK_KHR_shader_float16_int8` and `VK_KHR_16bit_storage` if the associated feature was requested
748            if requested_features.contains(wgt::Features::SHADER_F16) {
749                extensions.push(vk::KhrShaderFloat16Int8Fn::name());
750                // `VK_KHR_16bit_storage` requires `VK_KHR_storage_buffer_storage_class`, however we require that one already
751                if self.device_api_version < vk::API_VERSION_1_1 {
752                    extensions.push(vk::Khr16bitStorageFn::name());
753                }
754            }
755
756            //extensions.push(vk::KhrSamplerMirrorClampToEdgeFn::name());
757            //extensions.push(vk::ExtSamplerFilterMinmaxFn::name());
758        }
759
760        if self.device_api_version < vk::API_VERSION_1_3 {
761            // Optional `VK_EXT_image_robustness`
762            if self.supports_extension(vk::ExtImageRobustnessFn::name()) {
763                extensions.push(vk::ExtImageRobustnessFn::name());
764            }
765        }
766
767        // Optional `VK_KHR_swapchain_mutable_format`
768        if self.supports_extension(vk::KhrSwapchainMutableFormatFn::name()) {
769            extensions.push(vk::KhrSwapchainMutableFormatFn::name());
770        }
771
772        // Optional `VK_EXT_robustness2`
773        if self.supports_extension(vk::ExtRobustness2Fn::name()) {
774            extensions.push(vk::ExtRobustness2Fn::name());
775        }
776
777        // Require `VK_KHR_draw_indirect_count` if the associated feature was requested
778        // Even though Vulkan 1.2 has promoted the extension to core, we must require the extension to avoid
779        // large amounts of spaghetti involved with using PhysicalDeviceVulkan12Features.
780        if requested_features.contains(wgt::Features::MULTI_DRAW_INDIRECT_COUNT) {
781            extensions.push(vk::KhrDrawIndirectCountFn::name());
782        }
783
784        // Require `VK_KHR_deferred_host_operations`, `VK_KHR_acceleration_structure` and `VK_KHR_buffer_device_address` if the feature `RAY_TRACING` was requested
785        if requested_features.contains(wgt::Features::RAY_TRACING_ACCELERATION_STRUCTURE) {
786            extensions.push(vk::KhrDeferredHostOperationsFn::name());
787            extensions.push(vk::KhrAccelerationStructureFn::name());
788            extensions.push(vk::KhrBufferDeviceAddressFn::name());
789        }
790
791        // Require `VK_KHR_ray_query` if the associated feature was requested
792        if requested_features.contains(wgt::Features::RAY_QUERY) {
793            extensions.push(vk::KhrRayQueryFn::name());
794        }
795
796        // Require `VK_EXT_conservative_rasterization` if the associated feature was requested
797        if requested_features.contains(wgt::Features::CONSERVATIVE_RASTERIZATION) {
798            extensions.push(vk::ExtConservativeRasterizationFn::name());
799        }
800
801        // Require `VK_KHR_portability_subset` on macOS/iOS
802        #[cfg(any(target_os = "macos", target_os = "ios"))]
803        extensions.push(vk::KhrPortabilitySubsetFn::name());
804
805        // Require `VK_EXT_texture_compression_astc_hdr` if the associated feature was requested
806        if requested_features.contains(wgt::Features::TEXTURE_COMPRESSION_ASTC_HDR) {
807            extensions.push(vk::ExtTextureCompressionAstcHdrFn::name());
808        }
809
810        extensions
811    }

889    fn inspect(
890        &self,
891        phd: vk::PhysicalDevice,
892    ) -> (PhysicalDeviceCapabilities, PhysicalDeviceFeatures) {
893        let capabilities = {
894            let mut capabilities = PhysicalDeviceCapabilities::default();
895            capabilities.supported_extensions =
896                unsafe { self.raw.enumerate_device_extension_properties(phd).unwrap() };
897            capabilities.properties = unsafe { self.raw.get_physical_device_properties(phd) };
898            capabilities.device_api_version = capabilities.properties.api_version;
899
900            if let Some(ref get_device_properties) = self.get_physical_device_properties {
901                // Get these now to avoid borrowing conflicts later
902                let supports_maintenance3 = capabilities.device_api_version >= vk::API_VERSION_1_1
903                    || capabilities.supports_extension(vk::KhrMaintenance3Fn::name());
904                let supports_descriptor_indexing = capabilities.device_api_version
905                    >= vk::API_VERSION_1_2
906                    || capabilities.supports_extension(vk::ExtDescriptorIndexingFn::name());
907                let supports_driver_properties = capabilities.device_api_version
908                    >= vk::API_VERSION_1_2
909                    || capabilities.supports_extension(vk::KhrDriverPropertiesFn::name());
910
911                let supports_acceleration_structure =
912                    capabilities.supports_extension(vk::KhrAccelerationStructureFn::name());
913
914                let mut builder = vk::PhysicalDeviceProperties2KHR::builder();
915                if supports_maintenance3 {
916                    capabilities.maintenance_3 =
917                        Some(vk::PhysicalDeviceMaintenance3Properties::default());
918                    builder = builder.push_next(capabilities.maintenance_3.as_mut().unwrap());
919                }
920
921                if supports_descriptor_indexing {
922                    let next = capabilities
923                        .descriptor_indexing
924                        .insert(vk::PhysicalDeviceDescriptorIndexingPropertiesEXT::default());
925                    builder = builder.push_next(next);
926                }
927
928                if supports_acceleration_structure {
929                    let next = capabilities
930                        .acceleration_structure
931                        .insert(vk::PhysicalDeviceAccelerationStructurePropertiesKHR::default());
932                    builder = builder.push_next(next);
933                }
934
935                if supports_driver_properties {
936                    let next = capabilities
937                        .driver
938                        .insert(vk::PhysicalDeviceDriverPropertiesKHR::default());
939                    builder = builder.push_next(next);
940                }
941
942                let mut properties2 = builder.build();
943                unsafe {
944                    get_device_properties.get_physical_device_properties2(phd, &mut properties2);
945                }
946
947                if is_intel_igpu_outdated_for_robustness2(
948                    capabilities.properties,
949                    capabilities.driver,
950                ) {
951                    use crate::auxil::cstr_from_bytes_until_nul;
952                    capabilities.supported_extensions.retain(|&x| {
953                        cstr_from_bytes_until_nul(&x.extension_name)
954                            != Some(vk::ExtRobustness2Fn::name())
955                    });
956                }
957            };
958            capabilities
959        };
960
961        let mut features = PhysicalDeviceFeatures::default();
962        features.core = if let Some(ref get_device_properties) = self.get_physical_device_properties
963        {
964            let core = vk::PhysicalDeviceFeatures::default();
965            let mut builder = vk::PhysicalDeviceFeatures2KHR::builder().features(core);
966
967            // `VK_KHR_multiview` is promoted to 1.1
968            if capabilities.device_api_version >= vk::API_VERSION_1_1
969                || capabilities.supports_extension(vk::KhrMultiviewFn::name())
970            {
971                let next = features
972                    .multiview
973                    .insert(vk::PhysicalDeviceMultiviewFeatures::default());
974                builder = builder.push_next(next);
975            }
976
977            // `VK_KHR_sampler_ycbcr_conversion` is promoted to 1.1
978            if capabilities.device_api_version >= vk::API_VERSION_1_1
979                || capabilities.supports_extension(vk::KhrSamplerYcbcrConversionFn::name())
980            {
981                let next = features
982                    .sampler_ycbcr_conversion
983                    .insert(vk::PhysicalDeviceSamplerYcbcrConversionFeatures::default());
984                builder = builder.push_next(next);
985            }
986
987            if capabilities.supports_extension(vk::ExtDescriptorIndexingFn::name()) {
988                let next = features
989                    .descriptor_indexing
990                    .insert(vk::PhysicalDeviceDescriptorIndexingFeaturesEXT::default());
991                builder = builder.push_next(next);
992            }
993
994            // `VK_KHR_imageless_framebuffer` is promoted to 1.2, but has no changes, so we can keep using the extension unconditionally.
995            if capabilities.supports_extension(vk::KhrImagelessFramebufferFn::name()) {
996                let next = features
997                    .imageless_framebuffer
998                    .insert(vk::PhysicalDeviceImagelessFramebufferFeaturesKHR::default());
999                builder = builder.push_next(next);
1000            }
1001
1002            // `VK_KHR_timeline_semaphore` is promoted to 1.2, but has no changes, so we can keep using the extension unconditionally.
1003            if capabilities.supports_extension(vk::KhrTimelineSemaphoreFn::name()) {
1004                let next = features
1005                    .timeline_semaphore
1006                    .insert(vk::PhysicalDeviceTimelineSemaphoreFeaturesKHR::default());
1007                builder = builder.push_next(next);
1008            }
1009
1010            if capabilities.supports_extension(vk::ExtImageRobustnessFn::name()) {
1011                let next = features
1012                    .image_robustness
1013                    .insert(vk::PhysicalDeviceImageRobustnessFeaturesEXT::default());
1014                builder = builder.push_next(next);
1015            }
1016            if capabilities.supports_extension(vk::ExtRobustness2Fn::name()) {
1017                let next = features
1018                    .robustness2
1019                    .insert(vk::PhysicalDeviceRobustness2FeaturesEXT::default());
1020                builder = builder.push_next(next);
1021            }
1022            if capabilities.supports_extension(vk::ExtTextureCompressionAstcHdrFn::name()) {
1023                let next = features
1024                    .astc_hdr
1025                    .insert(vk::PhysicalDeviceTextureCompressionASTCHDRFeaturesEXT::default());
1026                builder = builder.push_next(next);
1027            }
1028            if capabilities.supports_extension(vk::KhrShaderFloat16Int8Fn::name())
1029                && capabilities.supports_extension(vk::Khr16bitStorageFn::name())
1030            {
1031                let next = features.shader_float16.insert((
1032                    vk::PhysicalDeviceShaderFloat16Int8FeaturesKHR::default(),
1033                    vk::PhysicalDevice16BitStorageFeaturesKHR::default(),
1034                ));
1035                builder = builder.push_next(&mut next.0);
1036                builder = builder.push_next(&mut next.1);
1037            }
1038            if capabilities.supports_extension(vk::KhrAccelerationStructureFn::name()) {
1039                let next = features
1040                    .acceleration_structure
1041                    .insert(vk::PhysicalDeviceAccelerationStructureFeaturesKHR::default());
1042                builder = builder.push_next(next);
1043            }
1044
1045            // `VK_KHR_zero_initialize_workgroup_memory` is promoted to 1.3
1046            if capabilities.device_api_version >= vk::API_VERSION_1_3
1047                || capabilities.supports_extension(vk::KhrZeroInitializeWorkgroupMemoryFn::name())
1048            {
1049                let next = features
1050                    .zero_initialize_workgroup_memory
1051                    .insert(vk::PhysicalDeviceZeroInitializeWorkgroupMemoryFeatures::default());
1052                builder = builder.push_next(next);
1053            }
1054
1055            let mut features2 = builder.build();
1056            unsafe {
1057                get_device_properties.get_physical_device_features2(phd, &mut features2);
1058            }
1059            features2.features
1060        } else {
1061            unsafe { self.raw.get_physical_device_features(phd) }
1062        };
1063
1064        (capabilities, features)
1065    }

1106                    .driver
1107                    .as_ref()
1108                    .and_then(|driver| cstr_from_bytes_until_nul(&driver.driver_info))
1109                    .and_then(|name| name.to_str().ok())
1110                    .unwrap_or("?")
1111                    .to_owned()
1112            },
1113            backend: wgt::Backend::Vulkan,
1114        };
1115
1116        let (available_features, downlevel_flags) =
1117            phd_features.to_wgpu(&self.shared.raw, phd, &phd_capabilities);
1118        let mut workarounds = super::Workarounds::empty();
1119        {
1120            // TODO: only enable for particular devices
1121            workarounds |= super::Workarounds::SEPARATE_ENTRY_POINTS;
1122            workarounds.set(
1123                super::Workarounds::EMPTY_RESOLVE_ATTACHMENT_LISTS,
1124                phd_capabilities.properties.vendor_id == db::qualcomm::VENDOR,
1125            );
1126            workarounds.set(
1127                super::Workarounds::FORCE_FILL_BUFFER_WITH_SIZE_GREATER_4096_ALIGNED_OFFSET_16,
1128                phd_capabilities.properties.vendor_id == db::nvidia::VENDOR,
1129            );
1130        };
1131
1132        if let Some(driver) = phd_capabilities.driver {
1133            if driver.conformance_version.major == 0 {
1134                if driver.driver_id == ash::vk::DriverId::MOLTENVK {
1135                    log::debug!("Adapter is not Vulkan compliant, but is MoltenVK, continuing");
1136                } else if self
1137                    .shared
1138                    .flags
1139                    .contains(wgt::InstanceFlags::ALLOW_UNDERLYING_NONCOMPLIANT_ADAPTER)
1140                {
1141                    log::warn!("Adapter is not Vulkan compliant: {}", info.name);
1142                } else {
1143                    log::warn!(
1144                        "Adapter is not Vulkan compliant, hiding adapter: {}",
1145                        info.name
1146                    );
1147                    return None;
1148                }
1149            }
1150        }
1151        if phd_capabilities.device_api_version == vk::API_VERSION_1_0
1152            && !phd_capabilities.supports_extension(vk::KhrStorageBufferStorageClassFn::name())
1153        {
1154            log::warn!(
1155                "SPIR-V storage buffer class is not supported, hiding adapter: {}",
1156                info.name
1157            );
1158            return None;
1159        }
1160        if !phd_capabilities.supports_extension(vk::AmdNegativeViewportHeightFn::name())
1161            && !phd_capabilities.supports_extension(vk::KhrMaintenance1Fn::name())
1162            && phd_capabilities.device_api_version < vk::API_VERSION_1_1
1163        {
1164            log::warn!(
1165                "viewport Y-flip is not supported, hiding adapter: {}",
1166                info.name
1167            );
1168            return None;
1169        }
1170
1171        let queue_families = unsafe {
1172            self.shared
1173                .raw
1174                .get_physical_device_queue_family_properties(phd)
1175        };
1176        let queue_flags = queue_families.first()?.queue_flags;
1177        if !queue_flags.contains(vk::QueueFlags::GRAPHICS) {
1178            log::warn!("The first queue only exposes {:?}", queue_flags);
1179            return None;
1180        }
1181
1182        let private_caps = super::PrivateCapabilities {
1183            flip_y_requires_shift: phd_capabilities.device_api_version >= vk::API_VERSION_1_1
1184                || phd_capabilities.supports_extension(vk::KhrMaintenance1Fn::name()),
1185            imageless_framebuffers: match phd_features.imageless_framebuffer {
1186                Some(features) => features.imageless_framebuffer == vk::TRUE,
1187                None => phd_features
1188                    .imageless_framebuffer
1189                    .map_or(false, |ext| ext.imageless_framebuffer != 0),
1190            },
1191            image_view_usage: phd_capabilities.device_api_version >= vk::API_VERSION_1_1
1192                || phd_capabilities.supports_extension(vk::KhrMaintenance2Fn::name()),
1193            timeline_semaphores: match phd_features.timeline_semaphore {
1194                Some(features) => features.timeline_semaphore == vk::TRUE,
1195                None => phd_features
1196                    .timeline_semaphore
1197                    .map_or(false, |ext| ext.timeline_semaphore != 0),
1198            },
1199            texture_d24: supports_format(
1200                &self.shared.raw,
1201                phd,
1202                vk::Format::X8_D24_UNORM_PACK32,
1203                vk::ImageTiling::OPTIMAL,
1204                depth_stencil_required_flags(),
1205            ),
1206            texture_d24_s8: supports_format(
1207                &self.shared.raw,
1208                phd,
1209                vk::Format::D24_UNORM_S8_UINT,
1210                vk::ImageTiling::OPTIMAL,
1211                depth_stencil_required_flags(),
1212            ),
1213            texture_s8: supports_format(
1214                &self.shared.raw,
1215                phd,
1216                vk::Format::S8_UINT,
1217                vk::ImageTiling::OPTIMAL,
1218                depth_stencil_required_flags(),
1219            ),
1220            non_coherent_map_mask: phd_capabilities.properties.limits.non_coherent_atom_size - 1,
1221            can_present: true,
1222            //TODO: make configurable
1223            robust_buffer_access: phd_features.core.robust_buffer_access != 0,
1224            robust_image_access: match phd_features.robustness2 {
1225                Some(ref f) => f.robust_image_access2 != 0,
1226                None => phd_features
1227                    .image_robustness
1228                    .map_or(false, |ext| ext.robust_image_access != 0),
1229            },
1230            robust_buffer_access2: phd_features
1231                .robustness2
1232                .as_ref()
1233                .map(|r| r.robust_buffer_access2 == 1)
1234                .unwrap_or_default(),
1235            robust_image_access2: phd_features
1236                .robustness2
1237                .as_ref()
1238                .map(|r| r.robust_image_access2 == 1)
1239                .unwrap_or_default(),
1240            zero_initialize_workgroup_memory: phd_features
1241                .zero_initialize_workgroup_memory
1242                .map_or(false, |ext| {
1243                    ext.shader_zero_initialize_workgroup_memory == vk::TRUE
1244                }),
1245            image_format_list: phd_capabilities.device_api_version >= vk::API_VERSION_1_2
1246                || phd_capabilities.supports_extension(vk::KhrImageFormatListFn::name()),
1247        };
1248        let capabilities = crate::Capabilities {
1249            limits: phd_capabilities.to_wgpu_limits(),
1250            alignments: phd_capabilities.to_hal_alignments(),
1251            downlevel: wgt::DownlevelCapabilities {
1252                flags: downlevel_flags,
1253                limits: wgt::DownlevelLimits {},
1254                shader_model: wgt::ShaderModel::Sm5, //TODO?
1255            },
1256        };
1257
1258        let adapter = super::Adapter {
1259            raw: phd,
1260            instance: Arc::clone(&self.shared),
1261            //queue_families,
1262            known_memory_flags: vk::MemoryPropertyFlags::DEVICE_LOCAL
1263                | vk::MemoryPropertyFlags::HOST_VISIBLE
1264                | vk::MemoryPropertyFlags::HOST_COHERENT
1265                | vk::MemoryPropertyFlags::HOST_CACHED
1266                | vk::MemoryPropertyFlags::LAZILY_ALLOCATED,
1267            phd_capabilities,
1268            //phd_features,
1269            downlevel_flags,
1270            private_caps,
1271            workarounds,
1272        };
1273
1274        Some(crate::ExposedAdapter {
1275            adapter,
1276            info,
1277            features: available_features,
1278            capabilities,
1279        })
1280    }

1329    ///
1330    /// - `raw_device` must be created from this adapter.

1334    pub unsafe fn device_from_raw(
1335        &self,
1336        raw_device: ash::Device,
1337        handle_is_owned: bool,
1338        enabled_extensions: &[&'static CStr],
1339        features: wgt::Features,
1340        family_index: u32,
1341        queue_index: u32,
1342    ) -> Result<crate::OpenDevice<super::Api>, crate::DeviceError> {
1343        let mem_properties = {
1344            profiling::scope!("vkGetPhysicalDeviceMemoryProperties");
1345            unsafe {
1346                self.instance
1347                    .raw
1348                    .get_physical_device_memory_properties(self.raw)
1349            }
1350        };
1351        let memory_types =
1352            &mem_properties.memory_types[..mem_properties.memory_type_count as usize];
1353        let valid_ash_memory_types = memory_types.iter().enumerate().fold(0, |u, (i, mem)| {
1354            if self.known_memory_flags.contains(mem.property_flags) {
1355                u | (1 << i)
1356            } else {
1357                u
1358            }
1359        });
1360
1361        let swapchain_fn = khr::Swapchain::new(&self.instance.raw, &raw_device);
1362
1363        let indirect_count_fn = if enabled_extensions.contains(&khr::DrawIndirectCount::name()) {
1364            Some(khr::DrawIndirectCount::new(&self.instance.raw, &raw_device))
1365        } else {
1366            None
1367        };
1368        let timeline_semaphore_fn = if enabled_extensions.contains(&khr::TimelineSemaphore::name())
1369        {
1370            Some(super::ExtensionFn::Extension(khr::TimelineSemaphore::new(
1371                &self.instance.raw,
1372                &raw_device,
1373            )))
1374        } else if self.phd_capabilities.device_api_version >= vk::API_VERSION_1_2 {
1375            Some(super::ExtensionFn::Promoted)
1376        } else {
1377            None
1378        };
1379        let ray_tracing_fns = if enabled_extensions.contains(&khr::AccelerationStructure::name())
1380            && enabled_extensions.contains(&khr::BufferDeviceAddress::name())
1381        {
1382            Some(super::RayTracingDeviceExtensionFunctions {
1383                acceleration_structure: khr::AccelerationStructure::new(
1384                    &self.instance.raw,
1385                    &raw_device,
1386                ),
1387                buffer_device_address: khr::BufferDeviceAddress::new(
1388                    &self.instance.raw,
1389                    &raw_device,
1390                ),
1391            })
1392        } else {
1393            None
1394        };
1395
1396        let naga_options = {
1397            use naga::back::spv;
1398
1399            // The following capabilities are always available
1400            // see https://registry.khronos.org/vulkan/specs/1.3-extensions/html/chap52.html#spirvenv-capabilities
1401            let mut capabilities = vec![
1402                spv::Capability::Shader,
1403                spv::Capability::Matrix,
1404                spv::Capability::Sampled1D,
1405                spv::Capability::Image1D,
1406                spv::Capability::ImageQuery,
1407                spv::Capability::DerivativeControl,
1408                spv::Capability::StorageImageExtendedFormats,
1409            ];
1410
1411            if self
1412                .downlevel_flags
1413                .contains(wgt::DownlevelFlags::CUBE_ARRAY_TEXTURES)
1414            {
1415                capabilities.push(spv::Capability::SampledCubeArray);
1416            }
1417
1418            if self
1419                .downlevel_flags
1420                .contains(wgt::DownlevelFlags::MULTISAMPLED_SHADING)
1421            {
1422                capabilities.push(spv::Capability::SampleRateShading);
1423            }
1424
1425            if features.contains(wgt::Features::MULTIVIEW) {
1426                capabilities.push(spv::Capability::MultiView);
1427            }
1428
1429            if features.contains(wgt::Features::SHADER_PRIMITIVE_INDEX) {
1430                capabilities.push(spv::Capability::Geometry);
1431            }
1432
1433            if features.intersects(
1434                wgt::Features::SAMPLED_TEXTURE_AND_STORAGE_BUFFER_ARRAY_NON_UNIFORM_INDEXING
1435                    | wgt::Features::UNIFORM_BUFFER_AND_STORAGE_TEXTURE_ARRAY_NON_UNIFORM_INDEXING,
1436            ) {
1437                capabilities.push(spv::Capability::ShaderNonUniform);
1438            }
1439            if features.contains(wgt::Features::BGRA8UNORM_STORAGE) {
1440                capabilities.push(spv::Capability::StorageImageWriteWithoutFormat);
1441            }
1442
1443            if features.contains(wgt::Features::RAY_QUERY) {
1444                capabilities.push(spv::Capability::RayQueryKHR);
1445            }
1446
1447            let mut flags = spv::WriterFlags::empty();
1448            flags.set(
1449                spv::WriterFlags::DEBUG,
1450                self.instance.flags.contains(wgt::InstanceFlags::DEBUG),
1451            );
1452            flags.set(
1453                spv::WriterFlags::LABEL_VARYINGS,
1454                self.phd_capabilities.properties.vendor_id != crate::auxil::db::qualcomm::VENDOR,
1455            );
1456            flags.set(
1457                spv::WriterFlags::FORCE_POINT_SIZE,
1458                //Note: we could technically disable this when we are compiling separate entry points,
1459                // and we know exactly that the primitive topology is not `PointList`.
1460                // But this requires cloning the `spv::Options` struct, which has heap allocations.
1461                true, // could check `super::Workarounds::SEPARATE_ENTRY_POINTS`
1462            );
1463            spv::Options {
1464                lang_version: (1, 0),
1465                flags,
1466                capabilities: Some(capabilities.iter().cloned().collect()),
1467                bounds_check_policies: naga::proc::BoundsCheckPolicies {
1468                    index: naga::proc::BoundsCheckPolicy::Restrict,
1469                    buffer: if self.private_caps.robust_buffer_access {
1470                        naga::proc::BoundsCheckPolicy::Unchecked
1471                    } else {
1472                        naga::proc::BoundsCheckPolicy::Restrict
1473                    },
1474                    image_load: if self.private_caps.robust_image_access {
1475                        naga::proc::BoundsCheckPolicy::Unchecked
1476                    } else {
1477                        naga::proc::BoundsCheckPolicy::Restrict
1478                    },
1479                    image_store: naga::proc::BoundsCheckPolicy::Unchecked,
1480                    // TODO: support bounds checks on binding arrays
1481                    binding_array: naga::proc::BoundsCheckPolicy::Unchecked,
1482                },
1483                zero_initialize_workgroup_memory: if self
1484                    .private_caps
1485                    .zero_initialize_workgroup_memory
1486                {
1487                    spv::ZeroInitializeWorkgroupMemoryMode::Native
1488                } else {
1489                    spv::ZeroInitializeWorkgroupMemoryMode::Polyfill
1490                },
1491                // We need to build this separately for each invocation, so just default it out here
1492                binding_map: BTreeMap::default(),
1493                debug_info: None,
1494            }
1495        };
1496
1497        let raw_queue = {
1498            profiling::scope!("vkGetDeviceQueue");
1499            unsafe { raw_device.get_device_queue(family_index, queue_index) }
1500        };
1501
1502        let shared = Arc::new(super::DeviceShared {
1503            raw: raw_device,
1504            family_index,
1505            queue_index,
1506            raw_queue,
1507            handle_is_owned,
1508            instance: Arc::clone(&self.instance),
1509            physical_device: self.raw,
1510            enabled_extensions: enabled_extensions.into(),
1511            extension_fns: super::DeviceExtensionFunctions {
1512                draw_indirect_count: indirect_count_fn,
1513                timeline_semaphore: timeline_semaphore_fn,
1514                ray_tracing: ray_tracing_fns,
1515            },
1516            vendor_id: self.phd_capabilities.properties.vendor_id,
1517            timestamp_period: self.phd_capabilities.properties.limits.timestamp_period,
1518            private_caps: self.private_caps.clone(),
1519            workarounds: self.workarounds,
1520            render_passes: Mutex::new(Default::default()),
1521            framebuffers: Mutex::new(Default::default()),
1522        });
1523        let mut relay_semaphores = [vk::Semaphore::null(); 2];
1524        for sem in relay_semaphores.iter_mut() {
1525            unsafe {
1526                *sem = shared
1527                    .raw
1528                    .create_semaphore(&vk::SemaphoreCreateInfo::builder(), None)?
1529            };
1530        }
1531        let queue = super::Queue {
1532            raw: raw_queue,
1533            swapchain_fn,
1534            device: Arc::clone(&shared),
1535            family_index,
1536            relay_semaphores,
1537            relay_index: AtomicIsize::new(-1),
1538        };
1539
1540        let mem_allocator = {
1541            let limits = self.phd_capabilities.properties.limits;
1542            let config = gpu_alloc::Config::i_am_prototyping(); //TODO
1543            let max_memory_allocation_size =
1544                if let Some(maintenance_3) = self.phd_capabilities.maintenance_3 {
1545                    maintenance_3.max_memory_allocation_size
1546                } else {
1547                    u64::max_value()
1548                };
1549            let properties = gpu_alloc::DeviceProperties {
1550                max_memory_allocation_count: limits.max_memory_allocation_count,
1551                max_memory_allocation_size,
1552                non_coherent_atom_size: limits.non_coherent_atom_size,
1553                memory_types: memory_types
1554                    .iter()
1555                    .map(|memory_type| gpu_alloc::MemoryType {
1556                        props: gpu_alloc::MemoryPropertyFlags::from_bits_truncate(
1557                            memory_type.property_flags.as_raw() as u8,
1558                        ),
1559                        heap: memory_type.heap_index,
1560                    })
1561                    .collect(),
1562                memory_heaps: mem_properties.memory_heaps
1563                    [..mem_properties.memory_heap_count as usize]
1564                    .iter()
1565                    .map(|&memory_heap| gpu_alloc::MemoryHeap {
1566                        size: memory_heap.size,
1567                    })
1568                    .collect(),
1569                buffer_device_address: enabled_extensions
1570                    .contains(&khr::BufferDeviceAddress::name()),
1571            };
1572            gpu_alloc::GpuAllocator::new(config, properties)
1573        };
1574        let desc_allocator = gpu_descriptor::DescriptorAllocator::new(
1575            if let Some(di) = self.phd_capabilities.descriptor_indexing {
1576                di.max_update_after_bind_descriptors_in_all_pools
1577            } else {
1578                0
1579            },
1580        );
1581
1582        let device = super::Device {
1583            shared,
1584            mem_allocator: Mutex::new(mem_allocator),
1585            desc_allocator: Mutex::new(desc_allocator),
1586            valid_ash_memory_types,
1587            naga_options,
1588            #[cfg(feature = "renderdoc")]
1589            render_doc: Default::default(),
1590        };
1591
1592        Ok(crate::OpenDevice { device, queue })
1593    }

1703        let limits = self.phd_capabilities.properties.limits;
1704
1705        let sample_flags = if format_aspect.contains(crate::FormatAspects::DEPTH) {
1706            limits
1707                .framebuffer_depth_sample_counts
1708                .min(limits.sampled_image_depth_sample_counts)
1709        } else if format_aspect.contains(crate::FormatAspects::STENCIL) {
1710            limits
1711                .framebuffer_stencil_sample_counts
1712                .min(limits.sampled_image_stencil_sample_counts)
1713        } else {
1714            let first_aspect = format_aspect
1715                .iter()
1716                .next()
1717                .expect("All texture should at least one aspect")
1718                .map();
1719
1720            // We should never get depth or stencil out of this, due to the above.
1721            assert_ne!(first_aspect, wgt::TextureAspect::DepthOnly);
1722            assert_ne!(first_aspect, wgt::TextureAspect::StencilOnly);
1723
1724            match format.sample_type(Some(first_aspect), None).unwrap() {
1725                wgt::TextureSampleType::Float { .. } => limits
1726                    .framebuffer_color_sample_counts
1727                    .min(limits.sampled_image_color_sample_counts),
1728                wgt::TextureSampleType::Sint | wgt::TextureSampleType::Uint => {
1729                    limits.sampled_image_integer_sample_counts
1730                }
1731                _ => unreachable!(),
1732            }
1733        };
1734
1735        flags.set(
1736            Tfc::MULTISAMPLE_X2,
1737            sample_flags.contains(vk::SampleCountFlags::TYPE_2),
1738        );
1739        flags.set(
1740            Tfc::MULTISAMPLE_X4,
1741            sample_flags.contains(vk::SampleCountFlags::TYPE_4),
1742        );
1743        flags.set(
1744            Tfc::MULTISAMPLE_X8,
1745            sample_flags.contains(vk::SampleCountFlags::TYPE_8),
1746        );
1747        flags.set(
1748            Tfc::MULTISAMPLE_X16,
1749            sample_flags.contains(vk::SampleCountFlags::TYPE_16),
1750        );
1751
1752        flags
1753    }

1755    unsafe fn surface_capabilities(
1756        &self,
1757        surface: &super::Surface,
1758    ) -> Option<crate::SurfaceCapabilities> {
1759        if !self.private_caps.can_present {
1760            return None;
1761        }
1762        let queue_family_index = 0; //TODO
1763        {
1764            profiling::scope!("vkGetPhysicalDeviceSurfaceSupportKHR");
1765            match unsafe {
1766                surface.functor.get_physical_device_surface_support(
1767                    self.raw,
1768                    queue_family_index,
1769                    surface.raw,
1770                )
1771            } {
1772                Ok(true) => (),
1773                Ok(false) => return None,
1774                Err(e) => {
1775                    log::error!("get_physical_device_surface_support: {}", e);
1776                    return None;
1777                }
1778            }
1779        }
1780
1781        let caps = {
1782            profiling::scope!("vkGetPhysicalDeviceSurfaceCapabilitiesKHR");
1783            match unsafe {
1784                surface
1785                    .functor
1786                    .get_physical_device_surface_capabilities(self.raw, surface.raw)
1787            } {
1788                Ok(caps) => caps,
1789                Err(e) => {
1790                    log::error!("get_physical_device_surface_capabilities: {}", e);
1791                    return None;
1792                }
1793            }
1794        };
1795
1796        // If image count is 0, the support number of images is unlimited.
1797        let max_image_count = if caps.max_image_count == 0 {
1798            !0
1799        } else {
1800            caps.max_image_count
1801        };
1802
1803        // `0xFFFFFFFF` indicates that the extent depends on the created swapchain.
1804        let current_extent = if caps.current_extent.width != !0 && caps.current_extent.height != !0
1805        {
1806            Some(wgt::Extent3d {
1807                width: caps.current_extent.width,
1808                height: caps.current_extent.height,
1809                depth_or_array_layers: 1,
1810            })
1811        } else {
1812            None
1813        };
1814
1815        let raw_present_modes = {
1816            profiling::scope!("vkGetPhysicalDeviceSurfacePresentModesKHR");
1817            match unsafe {
1818                surface
1819                    .functor
1820                    .get_physical_device_surface_present_modes(self.raw, surface.raw)
1821            } {
1822                Ok(present_modes) => present_modes,
1823                Err(e) => {
1824                    log::error!("get_physical_device_surface_present_modes: {}", e);
1825                    Vec::new()
1826                }
1827            }
1828        };
1829
1830        let raw_surface_formats = {
1831            profiling::scope!("vkGetPhysicalDeviceSurfaceFormatsKHR");
1832            match unsafe {
1833                surface
1834                    .functor
1835                    .get_physical_device_surface_formats(self.raw, surface.raw)
1836            } {
1837                Ok(formats) => formats,
1838                Err(e) => {
1839                    log::error!("get_physical_device_surface_formats: {}", e);
1840                    Vec::new()
1841                }
1842            }
1843        };
1844
1845        let formats = raw_surface_formats
1846            .into_iter()
1847            .filter_map(conv::map_vk_surface_formats)
1848            .collect();
1849        Some(crate::SurfaceCapabilities {
1850            formats,
1851            // TODO: Right now we're always trunkating the swap chain
1852            // (presumably - we're actually setting the min image count which isn't necessarily the swap chain size)
1853            // Instead, we should use extensions when available to wait in present.
1854            // See https://github.com/gfx-rs/wgpu/issues/2869
1855            maximum_frame_latency: (caps.min_image_count - 1)..=(max_image_count - 1), // Note this can't underflow since both `min_image_count` is at least one and we already patched `max_image_count`.
1856            current_extent,
1857            usage: conv::map_vk_image_usage(caps.supported_usage_flags),
1858            present_modes: raw_present_modes
1859                .into_iter()
1860                .flat_map(conv::map_vk_present_mode)
1861                .collect(),
1862            composite_alpha_modes: conv::map_vk_composite_alpha(caps.supported_composite_alpha),
1863        })
1864    }

1866    unsafe fn get_presentation_timestamp(&self) -> wgt::PresentationTimestamp {
1867        // VK_GOOGLE_display_timing is the only way to get presentation
1868        // timestamps on vulkan right now and it is only ever available
1869        // on android and linux. This includes mac, but there's no alternative
1870        // on mac, so this is fine.
1871        #[cfg(unix)]
1872        {
1873            let mut timespec = libc::timespec {
1874                tv_sec: 0,
1875                tv_nsec: 0,
1876            };
1877            unsafe {
1878                libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut timespec);
1879            }
1880
1881            wgt::PresentationTimestamp(
1882                timespec.tv_sec as u128 * 1_000_000_000 + timespec.tv_nsec as u128,
1883            )
1884        }
1885        #[cfg(not(unix))]
1886        {
1887            wgt::PresentationTimestamp::INVALID_TIMESTAMP
1888        }
1889    }

1892fn is_format_16bit_norm_supported(instance: &ash::Instance, phd: vk::PhysicalDevice) -> bool {
1893    let tiling = vk::ImageTiling::OPTIMAL;
1894    let features = vk::FormatFeatureFlags::SAMPLED_IMAGE
1895        | vk::FormatFeatureFlags::STORAGE_IMAGE
1896        | vk::FormatFeatureFlags::TRANSFER_SRC
1897        | vk::FormatFeatureFlags::TRANSFER_DST;
1898    let r16unorm = supports_format(instance, phd, vk::Format::R16_UNORM, tiling, features);
1899    let r16snorm = supports_format(instance, phd, vk::Format::R16_SNORM, tiling, features);
1900    let rg16unorm = supports_format(instance, phd, vk::Format::R16G16_UNORM, tiling, features);
1901    let rg16snorm = supports_format(instance, phd, vk::Format::R16G16_SNORM, tiling, features);
1902    let rgba16unorm = supports_format(
1903        instance,
1904        phd,
1905        vk::Format::R16G16B16A16_UNORM,
1906        tiling,
1907        features,
1908    );
1909    let rgba16snorm = supports_format(
1910        instance,
1911        phd,
1912        vk::Format::R16G16B16A16_SNORM,
1913        tiling,
1914        features,
1915    );
1916
1917    r16unorm && r16snorm && rg16unorm && rg16snorm && rgba16unorm && rgba16snorm
1918}

1935fn supports_format(
1936    instance: &ash::Instance,
1937    phd: vk::PhysicalDevice,
1938    format: vk::Format,
1939    tiling: vk::ImageTiling,
1940    features: vk::FormatFeatureFlags,
1941) -> bool {
1942    let properties = unsafe { instance.get_physical_device_format_properties(phd, format) };
1943    match tiling {
1944        vk::ImageTiling::LINEAR => properties.linear_tiling_features.contains(features),
1945        vk::ImageTiling::OPTIMAL => properties.optimal_tiling_features.contains(features),
1946        _ => false,
1947    }

1950fn supports_bgra8unorm_storage(
1951    instance: &ash::Instance,
1952    phd: vk::PhysicalDevice,
1953    device_api_version: u32,
1954) -> bool {
1955    // See https://github.com/KhronosGroup/Vulkan-Docs/issues/2027#issuecomment-1380608011
1956
1957    // This check gates the function call and structures used below.
1958    // TODO: check for (`VK_KHR_get_physical_device_properties2` or VK1.1) and (`VK_KHR_format_feature_flags2` or VK1.3).
1959    // Right now we only check for VK1.3.
1960    if device_api_version < vk::API_VERSION_1_3 {
1961        return false;
1962    }
1963
1964    unsafe {
1965        let mut properties3 = vk::FormatProperties3::default();
1966        let mut properties2 = vk::FormatProperties2::builder().push_next(&mut properties3);
1967
1968        instance.get_physical_device_format_properties2(
1969            phd,
1970            vk::Format::B8G8R8A8_UNORM,
1971            &mut properties2,
1972        );
1973
1974        let features2 = properties2.format_properties.optimal_tiling_features;
1975        let features3 = properties3.optimal_tiling_features;
1976
1977        features2.contains(vk::FormatFeatureFlags::STORAGE_IMAGE)
1978            && features3.contains(vk::FormatFeatureFlags2::STORAGE_WRITE_WITHOUT_FORMAT)
1979    }

1982// For https://github.com/gfx-rs/wgpu/issues/4599
1983// Intel iGPUs with outdated drivers can break rendering if `VK_EXT_robustness2` is used.
1984// Driver version 31.0.101.2115 works, but there's probably an earlier functional version.
1985fn is_intel_igpu_outdated_for_robustness2(
1986    props: vk::PhysicalDeviceProperties,
1987    driver: Option<vk::PhysicalDeviceDriverPropertiesKHR>,
1988) -> bool {
1989    const DRIVER_VERSION_WORKING: u32 = (101 << 14) | 2115; // X.X.101.2115
1990
1991    let is_outdated = props.vendor_id == crate::auxil::db::intel::VENDOR
1992        && props.device_type == vk::PhysicalDeviceType::INTEGRATED_GPU
1993        && props.driver_version < DRIVER_VERSION_WORKING
1994        && driver
1995            .map(|driver| driver.driver_id == vk::DriverId::INTEL_PROPRIETARY_WINDOWS)
1996            .unwrap_or_default();
1997
1998    if is_outdated {
1999        log::warn!(
2000            "Disabling robustBufferAccess2 and robustImageAccess2: IntegratedGpu Intel Driver is outdated. Found with version 0x{:X}, less than the known good version 0x{:X} (31.0.101.2115)",
2001            props.driver_version,
2002            DRIVER_VERSION_WORKING
2003        );
2004    }
2005    is_outdated
