11impl super::Texture {
12    fn map_buffer_copies<T>(&self, regions: T) -> impl Iterator<Item = vk::BufferImageCopy>
13    where
14        T: Iterator<Item = crate::BufferTextureCopy>,
15    {
16        let (block_width, block_height) = self.format.block_dimensions();
17        let format = self.format;
18        let copy_size = self.copy_size;
19        regions.map(move |r| {
20            let extent = r.texture_base.max_copy_size(&copy_size).min(&r.size);
21            let (image_subresource, image_offset) = conv::map_subresource_layers(&r.texture_base);
22            vk::BufferImageCopy {
23                buffer_offset: r.buffer_layout.offset,
24                buffer_row_length: r.buffer_layout.bytes_per_row.map_or(0, |bpr| {
25                    let block_size = format
26                        .block_copy_size(Some(r.texture_base.aspect.map()))
27                        .unwrap();
28                    block_width * (bpr / block_size)
29                }),
30                buffer_image_height: r
31                    .buffer_layout
32                    .rows_per_image
33                    .map_or(0, |rpi| rpi * block_height),
34                image_subresource,
35                image_offset,
36                image_extent: conv::map_copy_extent(&extent),
37            }
38        })
39    }
40}

64    unsafe fn begin_encoding(&mut self, label: crate::Label) -> Result<(), crate::DeviceError> {
65        if self.free.is_empty() {
66            let vk_info = vk::CommandBufferAllocateInfo::builder()
67                .command_pool(self.raw)
68                .command_buffer_count(ALLOCATION_GRANULARITY)
69                .build();
70            let cmd_buf_vec = unsafe { self.device.raw.allocate_command_buffers(&vk_info)? };
71            self.free.extend(cmd_buf_vec);
72        }
73        let raw = self.free.pop().unwrap();
74
75        // Set the name unconditionally, since there might be a
76        // previous name assigned to this.
77        unsafe {
78            self.device.set_object_name(
79                vk::ObjectType::COMMAND_BUFFER,
80                raw,
81                label.unwrap_or_default(),
82            )
83        };
84
85        // Reset this in case the last renderpass was never ended.
86        self.rpass_debug_marker_active = false;
87
88        let vk_info = vk::CommandBufferBeginInfo::builder()
89            .flags(vk::CommandBufferUsageFlags::ONE_TIME_SUBMIT)
90            .build();
91        unsafe { self.device.raw.begin_command_buffer(raw, &vk_info) }?;
92        self.active = raw;
93
94        Ok(())
95    }

124    unsafe fn transition_buffers<'a, T>(&mut self, barriers: T)
125    where
126        T: Iterator<Item = crate::BufferBarrier<'a, super::Api>>,
127    {
128        //Note: this is done so that we never end up with empty stage flags
129        let mut src_stages = vk::PipelineStageFlags::TOP_OF_PIPE;
130        let mut dst_stages = vk::PipelineStageFlags::BOTTOM_OF_PIPE;
131        let vk_barriers = &mut self.temp.buffer_barriers;
132        vk_barriers.clear();
133
134        for bar in barriers {
135            let (src_stage, src_access) = conv::map_buffer_usage_to_barrier(bar.usage.start);
136            src_stages |= src_stage;
137            let (dst_stage, dst_access) = conv::map_buffer_usage_to_barrier(bar.usage.end);
138            dst_stages |= dst_stage;
139
140            vk_barriers.push(
141                vk::BufferMemoryBarrier::builder()
142                    .buffer(bar.buffer.raw)
143                    .size(vk::WHOLE_SIZE)
144                    .src_access_mask(src_access)
145                    .dst_access_mask(dst_access)
146                    .build(),
147            )
148        }
149
150        if !vk_barriers.is_empty() {
151            unsafe {
152                self.device.raw.cmd_pipeline_barrier(
153                    self.active,
154                    src_stages,
155                    dst_stages,
156                    vk::DependencyFlags::empty(),
157                    &[],
158                    vk_barriers,
159                    &[],
160                )
161            };
162        }
163    }

165    unsafe fn transition_textures<'a, T>(&mut self, barriers: T)
166    where
167        T: Iterator<Item = crate::TextureBarrier<'a, super::Api>>,
168    {
169        let mut src_stages = vk::PipelineStageFlags::empty();
170        let mut dst_stages = vk::PipelineStageFlags::empty();
171        let vk_barriers = &mut self.temp.image_barriers;
172        vk_barriers.clear();
173
174        for bar in barriers {
175            let range = conv::map_subresource_range_combined_aspect(
176                &bar.range,
177                bar.texture.format,
178                &self.device.private_caps,
179            );
180            let (src_stage, src_access) = conv::map_texture_usage_to_barrier(bar.usage.start);
181            let src_layout = conv::derive_image_layout(bar.usage.start, bar.texture.format);
182            src_stages |= src_stage;
183            let (dst_stage, dst_access) = conv::map_texture_usage_to_barrier(bar.usage.end);
184            let dst_layout = conv::derive_image_layout(bar.usage.end, bar.texture.format);
185            dst_stages |= dst_stage;
186
187            vk_barriers.push(
188                vk::ImageMemoryBarrier::builder()
189                    .image(bar.texture.raw)
190                    .subresource_range(range)
191                    .src_access_mask(src_access)
192                    .dst_access_mask(dst_access)
193                    .old_layout(src_layout)
194                    .new_layout(dst_layout)
195                    .build(),
196            );
197        }
198
199        if !vk_barriers.is_empty() {
200            unsafe {
201                self.device.raw.cmd_pipeline_barrier(
202                    self.active,
203                    src_stages,
204                    dst_stages,
205                    vk::DependencyFlags::empty(),
206                    &[],
207                    &[],
208                    vk_barriers,
209                )
210            };
211        }
212    }

214    unsafe fn clear_buffer(&mut self, buffer: &super::Buffer, range: crate::MemoryRange) {
215        let range_size = range.end - range.start;
216        if self.device.workarounds.contains(
217            super::Workarounds::FORCE_FILL_BUFFER_WITH_SIZE_GREATER_4096_ALIGNED_OFFSET_16,
218        ) && range_size >= 4096
219            && range.start % 16 != 0
220        {
221            let rounded_start = wgt::math::align_to(range.start, 16);
222            let prefix_size = rounded_start - range.start;
223
224            unsafe {
225                self.device.raw.cmd_fill_buffer(
226                    self.active,
227                    buffer.raw,
228                    range.start,
229                    prefix_size,
230                    0,
231                )
232            };
233
234            // This will never be zero, as rounding can only add up to 12 bytes, and the total size is 4096.
235            let suffix_size = range.end - rounded_start;
236
237            unsafe {
238                self.device.raw.cmd_fill_buffer(
239                    self.active,
240                    buffer.raw,
241                    rounded_start,
242                    suffix_size,
243                    0,
244                )
245            };
246        } else {
247            unsafe {
248                self.device
249                    .raw
250                    .cmd_fill_buffer(self.active, buffer.raw, range.start, range_size, 0)
251            };
252        }

417    unsafe fn build_acceleration_structures<'a, T>(&mut self, descriptor_count: u32, descriptors: T)
418    where
419        super::Api: 'a,
420        T: IntoIterator<Item = crate::BuildAccelerationStructureDescriptor<'a, super::Api>>,
421    {
422        const CAPACITY_OUTER: usize = 8;
423        const CAPACITY_INNER: usize = 1;
424        let descriptor_count = descriptor_count as usize;
425
426        let ray_tracing_functions = self
427            .device
428            .extension_fns
429            .ray_tracing
430            .as_ref()
431            .expect("Feature `RAY_TRACING` not enabled");
432
433        let get_device_address = |buffer: Option<&super::Buffer>| unsafe {
434            match buffer {
435                Some(buffer) => ray_tracing_functions
436                    .buffer_device_address
437                    .get_buffer_device_address(
438                        &vk::BufferDeviceAddressInfo::builder().buffer(buffer.raw),
439                    ),
440                None => panic!("Buffers are required to build acceleration structures"),
441            }
442        };
443
444        // storage to all the data required for cmd_build_acceleration_structures
445        let mut ranges_storage = smallvec::SmallVec::<
446            [smallvec::SmallVec<[vk::AccelerationStructureBuildRangeInfoKHR; CAPACITY_INNER]>;
447                CAPACITY_OUTER],
448        >::with_capacity(descriptor_count);
449        let mut geometries_storage = smallvec::SmallVec::<
450            [smallvec::SmallVec<[vk::AccelerationStructureGeometryKHR; CAPACITY_INNER]>;
451                CAPACITY_OUTER],
452        >::with_capacity(descriptor_count);
453
454        // pointers to all the data required for cmd_build_acceleration_structures
455        let mut geometry_infos = smallvec::SmallVec::<
456            [vk::AccelerationStructureBuildGeometryInfoKHR; CAPACITY_OUTER],
457        >::with_capacity(descriptor_count);
458        let mut ranges_ptrs = smallvec::SmallVec::<
459            [&[vk::AccelerationStructureBuildRangeInfoKHR]; CAPACITY_OUTER],
460        >::with_capacity(descriptor_count);
461
462        for desc in descriptors {
463            let (geometries, ranges) = match *desc.entries {
464                crate::AccelerationStructureEntries::Instances(ref instances) => {
465                    let instance_data = vk::AccelerationStructureGeometryInstancesDataKHR::builder(
466                    )
467                    .data(vk::DeviceOrHostAddressConstKHR {
468                        device_address: get_device_address(instances.buffer),
469                    });
470
471                    let geometry = vk::AccelerationStructureGeometryKHR::builder()
472                        .geometry_type(vk::GeometryTypeKHR::INSTANCES)
473                        .geometry(vk::AccelerationStructureGeometryDataKHR {
474                            instances: *instance_data,
475                        });
476
477                    let range = vk::AccelerationStructureBuildRangeInfoKHR::builder()
478                        .primitive_count(instances.count)
479                        .primitive_offset(instances.offset);
480
481                    (smallvec::smallvec![*geometry], smallvec::smallvec![*range])
482                }
483                crate::AccelerationStructureEntries::Triangles(ref in_geometries) => {
484                    let mut ranges = smallvec::SmallVec::<
485                        [vk::AccelerationStructureBuildRangeInfoKHR; CAPACITY_INNER],
486                    >::with_capacity(in_geometries.len());
487                    let mut geometries = smallvec::SmallVec::<
488                        [vk::AccelerationStructureGeometryKHR; CAPACITY_INNER],
489                    >::with_capacity(in_geometries.len());
490                    for triangles in in_geometries {
491                        let mut triangle_data =
492                            vk::AccelerationStructureGeometryTrianglesDataKHR::builder()
493                                .vertex_data(vk::DeviceOrHostAddressConstKHR {
494                                    device_address: get_device_address(triangles.vertex_buffer),
495                                })
496                                .vertex_format(conv::map_vertex_format(triangles.vertex_format))
497                                .max_vertex(triangles.vertex_count)
498                                .vertex_stride(triangles.vertex_stride);
499
500                        let mut range = vk::AccelerationStructureBuildRangeInfoKHR::builder();
501
502                        if let Some(ref indices) = triangles.indices {
503                            triangle_data = triangle_data
504                                .index_data(vk::DeviceOrHostAddressConstKHR {
505                                    device_address: get_device_address(indices.buffer),
506                                })
507                                .index_type(conv::map_index_format(indices.format));
508
509                            range = range
510                                .primitive_count(indices.count / 3)
511                                .primitive_offset(indices.offset)
512                                .first_vertex(triangles.first_vertex);
513                        } else {
514                            range = range
515                                .primitive_count(triangles.vertex_count)
516                                .first_vertex(triangles.first_vertex);
517                        }
518
519                        if let Some(ref transform) = triangles.transform {
520                            let transform_device_address = unsafe {
521                                ray_tracing_functions
522                                    .buffer_device_address
523                                    .get_buffer_device_address(
524                                        &vk::BufferDeviceAddressInfo::builder()
525                                            .buffer(transform.buffer.raw),
526                                    )
527                            };
528                            triangle_data =
529                                triangle_data.transform_data(vk::DeviceOrHostAddressConstKHR {
530                                    device_address: transform_device_address,
531                                });
532
533                            range = range.transform_offset(transform.offset);
534                        }
535
536                        let geometry = vk::AccelerationStructureGeometryKHR::builder()
537                            .geometry_type(vk::GeometryTypeKHR::TRIANGLES)
538                            .geometry(vk::AccelerationStructureGeometryDataKHR {
539                                triangles: *triangle_data,
540                            })
541                            .flags(conv::map_acceleration_structure_geomety_flags(
542                                triangles.flags,
543                            ));
544
545                        geometries.push(*geometry);
546                        ranges.push(*range);
547                    }
548                    (geometries, ranges)
549                }
550                crate::AccelerationStructureEntries::AABBs(ref in_geometries) => {
551                    let mut ranges = smallvec::SmallVec::<
552                        [vk::AccelerationStructureBuildRangeInfoKHR; CAPACITY_INNER],
553                    >::with_capacity(in_geometries.len());
554                    let mut geometries = smallvec::SmallVec::<
555                        [vk::AccelerationStructureGeometryKHR; CAPACITY_INNER],
556                    >::with_capacity(in_geometries.len());
557                    for aabb in in_geometries {
558                        let aabbs_data = vk::AccelerationStructureGeometryAabbsDataKHR::builder()
559                            .data(vk::DeviceOrHostAddressConstKHR {
560                                device_address: get_device_address(aabb.buffer),
561                            })
562                            .stride(aabb.stride);
563
564                        let range = vk::AccelerationStructureBuildRangeInfoKHR::builder()
565                            .primitive_count(aabb.count)
566                            .primitive_offset(aabb.offset);
567
568                        let geometry = vk::AccelerationStructureGeometryKHR::builder()
569                            .geometry_type(vk::GeometryTypeKHR::AABBS)
570                            .geometry(vk::AccelerationStructureGeometryDataKHR {
571                                aabbs: *aabbs_data,
572                            })
573                            .flags(conv::map_acceleration_structure_geomety_flags(aabb.flags));
574
575                        geometries.push(*geometry);
576                        ranges.push(*range);
577                    }
578                    (geometries, ranges)
579                }
580            };
581
582            ranges_storage.push(ranges);
583            geometries_storage.push(geometries);
584
585            let scratch_device_address = unsafe {
586                ray_tracing_functions
587                    .buffer_device_address
588                    .get_buffer_device_address(
589                        &vk::BufferDeviceAddressInfo::builder().buffer(desc.scratch_buffer.raw),
590                    )
591            };
592            let ty = match *desc.entries {
593                crate::AccelerationStructureEntries::Instances(_) => {
594                    vk::AccelerationStructureTypeKHR::TOP_LEVEL
595                }
596                _ => vk::AccelerationStructureTypeKHR::BOTTOM_LEVEL,
597            };
598            let mut geometry_info = vk::AccelerationStructureBuildGeometryInfoKHR::builder()
599                .ty(ty)
600                .mode(conv::map_acceleration_structure_build_mode(desc.mode))
601                .flags(conv::map_acceleration_structure_flags(desc.flags))
602                .dst_acceleration_structure(desc.destination_acceleration_structure.raw)
603                .scratch_data(vk::DeviceOrHostAddressKHR {
604                    device_address: scratch_device_address + desc.scratch_buffer_offset,
605                });
606
607            if desc.mode == crate::AccelerationStructureBuildMode::Update {
608                geometry_info.src_acceleration_structure = desc
609                    .source_acceleration_structure
610                    .unwrap_or(desc.destination_acceleration_structure)
611                    .raw;
612            }
613
614            geometry_infos.push(*geometry_info);
615        }
616
617        for (i, geometry_info) in geometry_infos.iter_mut().enumerate() {
618            geometry_info.geometry_count = geometries_storage[i].len() as u32;
619            geometry_info.p_geometries = geometries_storage[i].as_ptr();
620            ranges_ptrs.push(&ranges_storage[i]);
621        }
622
623        unsafe {
624            ray_tracing_functions
625                .acceleration_structure
626                .cmd_build_acceleration_structures(self.active, &geometry_infos, &ranges_ptrs);
627        }
628    }

657        let mut vk_clear_values =

664            sample_count: desc.sample_count,
665        };
666        let caps = &self.device.private_caps;
667
668        for cat in desc.color_attachments {
669            if let Some(cat) = cat.as_ref() {
670                vk_clear_values.push(vk::ClearValue {
671                    color: unsafe { cat.make_vk_clear_color() },
672                });
673                vk_image_views.push(cat.target.view.raw);
674                let color = super::ColorAttachmentKey {
675                    base: cat.target.make_attachment_key(cat.ops, caps),
676                    resolve: cat.resolve_target.as_ref().map(|target| {
677                        target.make_attachment_key(crate::AttachmentOps::STORE, caps)
678                    }),
679                };
680
681                rp_key.colors.push(Some(color));
682                fb_key.attachments.push(cat.target.view.attachment.clone());
683                if let Some(ref at) = cat.resolve_target {
684                    vk_clear_values.push(unsafe { mem::zeroed() });
685                    vk_image_views.push(at.view.raw);
686                    fb_key.attachments.push(at.view.attachment.clone());
687                }
688
689                // Assert this attachment is valid for the detected multiview, as a sanity check
690                // The driver crash for this is really bad on AMD, so the check is worth it
691                if let Some(multiview) = desc.multiview {
692                    assert_eq!(cat.target.view.layers, multiview);
693                    if let Some(ref resolve_target) = cat.resolve_target {
694                        assert_eq!(resolve_target.view.layers, multiview);
695                    }
696                }
697            } else {
698                rp_key.colors.push(None);
699            }
700        }
701        if let Some(ref ds) = desc.depth_stencil_attachment {
702            vk_clear_values.push(vk::ClearValue {
703                depth_stencil: vk::ClearDepthStencilValue {
704                    depth: ds.clear_value.0,
705                    stencil: ds.clear_value.1,
706                },
707            });
708            vk_image_views.push(ds.target.view.raw);
709            rp_key.depth_stencil = Some(super::DepthStencilAttachmentKey {
710                base: ds.target.make_attachment_key(ds.depth_ops, caps),
711                stencil_ops: ds.stencil_ops,
712            });
713            fb_key.attachments.push(ds.target.view.attachment.clone());
714
715            // Assert this attachment is valid for the detected multiview, as a sanity check
716            // The driver crash for this is really bad on AMD, so the check is worth it
717            if let Some(multiview) = desc.multiview {
718                assert_eq!(ds.target.view.layers, multiview);
719            }
720        }
721        rp_key.sample_count = fb_key.sample_count;
722        rp_key.multiview = desc.multiview;
723
724        let render_area = vk::Rect2D {
725            offset: vk::Offset2D { x: 0, y: 0 },
726            extent: vk::Extent2D {
727                width: desc.extent.width,
728                height: desc.extent.height,
729            },
730        };
731        let vk_viewports = [vk::Viewport {
732            x: 0.0,
733            y: if self.device.private_caps.flip_y_requires_shift {
734                desc.extent.height as f32
735            } else {
736                0.0
737            },
738            width: desc.extent.width as f32,
739            height: -(desc.extent.height as f32),
740            min_depth: 0.0,
741            max_depth: 1.0,
742        }];
743
744        let raw_pass = self.device.make_render_pass(rp_key).unwrap();
745        let raw_framebuffer = self
746            .device
747            .make_framebuffer(fb_key, raw_pass, desc.label)
748            .unwrap();
749
750        let mut vk_info = vk::RenderPassBeginInfo::builder()
751            .render_pass(raw_pass)
752            .render_area(render_area)
753            .clear_values(&vk_clear_values)
754            .framebuffer(raw_framebuffer);
755        let mut vk_attachment_info = if caps.imageless_framebuffers {
756            Some(
757                vk::RenderPassAttachmentBeginInfo::builder()
758                    .attachments(&vk_image_views)
759                    .build(),
760            )
761        } else {
762            None
763        };
764        if let Some(attachment_info) = vk_attachment_info.as_mut() {
765            vk_info = vk_info.push_next(attachment_info);
766        }
767
768        if let Some(label) = desc.label {
769            unsafe { self.begin_debug_marker(label) };
770            self.rpass_debug_marker_active = true;
771        }
772
773        // Start timestamp if any (before all other commands but after debug marker)
774        if let Some(timestamp_writes) = desc.timestamp_writes.as_ref() {
775            if let Some(index) = timestamp_writes.beginning_of_pass_write_index {
776                unsafe {
777                    self.write_timestamp(timestamp_writes.query_set, index);
778                }
779            }
780            self.end_of_pass_timer_query = timestamp_writes
781                .end_of_pass_write_index
782                .map(|index| (timestamp_writes.query_set.raw, index));
783        }
784
785        unsafe {
786            self.device
787                .raw
788                .cmd_set_viewport(self.active, 0, &vk_viewports);
789            self.device
790                .raw
791                .cmd_set_scissor(self.active, 0, &[render_area]);
792            self.device.raw.cmd_begin_render_pass(
793                self.active,
794                &vk_info,
795                vk::SubpassContents::INLINE,
796            );
797        };
798
799        self.bind_point = vk::PipelineBindPoint::GRAPHICS;
800    }
801    unsafe fn end_render_pass(&mut self) {
802        unsafe {
803            self.device.raw.cmd_end_render_pass(self.active);
804        }
805
806        // After all other commands but before debug marker, so this is still seen as part of this pass.
807        self.write_pass_end_timestamp_if_requested();
808
809        if self.rpass_debug_marker_active {
810            unsafe {
811                self.end_debug_marker();
812            }
813            self.rpass_debug_marker_active = false;
814        }
815    }
