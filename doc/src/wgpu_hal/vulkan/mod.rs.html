1/*!
2# Vulkan API internals.
3
4## Stack memory
5
6Ash expects slices, which we don't generally have available.
7We cope with this requirement by the combination of the following ways:
8  - temporarily allocating `Vec` on heap, where overhead is permitted
9  - growing temporary local storage
10  - using `implace_it` on iterators
11
12## Framebuffers and Render passes
13
14Render passes are cached on the device and kept forever.
15
16Framebuffers are also cached on the device, but they are removed when
17any of the image views (they have) gets removed.
18If Vulkan supports image-less framebuffers,
19then the actual views are excluded from the framebuffer key.
20
21## Fences
22
23If timeline semaphores are available, they are used 1:1 with wgpu-hal fences.
24Otherwise, we manage a pool of `VkFence` objects behind each `hal::Fence`.
25
26!*/

509#[derive(Debug)]
510pub enum Fence {
511    TimelineSemaphore(vk::Semaphore),
512    FencePool {
513        last_completed: crate::FenceValue,
514        /// The pending fence values have to be ascending.
515        active: Vec<(crate::FenceValue, vk::Fence)>,
516        free: Vec<vk::Fence>,
517    },
518}
519
520impl Fence {
521    fn check_active(
522        device: &ash::Device,
523        mut max_value: crate::FenceValue,
524        active: &[(crate::FenceValue, vk::Fence)],
525    ) -> Result<crate::FenceValue, crate::DeviceError> {
526        for &(value, raw) in active.iter() {
527            unsafe {
528                if value > max_value && device.get_fence_status(raw)? {
529                    max_value = value;
530                }
531            }
532        }
533        Ok(max_value)
534    }
535
536    fn get_latest(
537        &self,
538        device: &ash::Device,
539        extension: Option<&ExtensionFn<khr::TimelineSemaphore>>,
540    ) -> Result<crate::FenceValue, crate::DeviceError> {
541        match *self {
542            Self::TimelineSemaphore(raw) => unsafe {
543                Ok(match *extension.unwrap() {
544                    ExtensionFn::Extension(ref ext) => ext.get_semaphore_counter_value(raw)?,
545                    ExtensionFn::Promoted => device.get_semaphore_counter_value(raw)?,
546                })
547            },
548            Self::FencePool {
549                last_completed,
550                ref active,
551                free: _,
552            } => Self::check_active(device, last_completed, active),
553        }
554    }
555
556    fn maintain(&mut self, device: &ash::Device) -> Result<(), crate::DeviceError> {
557        match *self {
558            Self::TimelineSemaphore(_) => {}
559            Self::FencePool {
560                ref mut last_completed,
561                ref mut active,
562                ref mut free,
563            } => {
564                let latest = Self::check_active(device, *last_completed, active)?;
565                let base_free = free.len();
566                for &(value, raw) in active.iter() {
567                    if value <= latest {
568                        free.push(raw);
569                    }
570                }
571                if free.len() != base_free {
572                    active.retain(|&(value, _)| value > latest);
573                    unsafe {
574                        device.reset_fences(&free[base_free..])?;
575                    }
576                }
577                *last_completed = latest;
578            }
579        }
580        Ok(())
581    }
582}

584impl crate::Queue<Api> for Queue {
585    unsafe fn submit(
586        &self,
587        command_buffers: &[&CommandBuffer],
588        signal_fence: Option<(&mut Fence, crate::FenceValue)>,
589    ) -> Result<(), crate::DeviceError> {
590        let vk_cmd_buffers = command_buffers
591            .iter()
592            .map(|cmd| cmd.raw)
593            .collect::<Vec<_>>();
594
595        let mut vk_info = vk::SubmitInfo::builder().command_buffers(&vk_cmd_buffers);
596
597        let mut fence_raw = vk::Fence::null();
598        let mut vk_timeline_info;
599        let mut signal_semaphores = [vk::Semaphore::null(), vk::Semaphore::null()];
600        let signal_values;
601
602        if let Some((fence, value)) = signal_fence {
603            fence.maintain(&self.device.raw)?;
604            match *fence {
605                Fence::TimelineSemaphore(raw) => {
606                    signal_values = [!0, value];
607                    signal_semaphores[1] = raw;
608                    vk_timeline_info = vk::TimelineSemaphoreSubmitInfo::builder()
609                        .signal_semaphore_values(&signal_values);
610                    vk_info = vk_info.push_next(&mut vk_timeline_info);
611                }
612                Fence::FencePool {
613                    ref mut active,
614                    ref mut free,
615                    ..
616                } => {
617                    fence_raw = match free.pop() {
618                        Some(raw) => raw,
619                        None => unsafe {
620                            self.device
621                                .raw
622                                .create_fence(&vk::FenceCreateInfo::builder(), None)?
623                        },
624                    };
625                    active.push((value, fence_raw));
626                }
627            }
628        }
629
630        let wait_stage_mask = [vk::PipelineStageFlags::TOP_OF_PIPE];
631        let old_index = self.relay_index.load(Ordering::Relaxed);
632        let sem_index = if old_index >= 0 {
633            vk_info = vk_info
634                .wait_semaphores(&self.relay_semaphores[old_index as usize..old_index as usize + 1])
635                .wait_dst_stage_mask(&wait_stage_mask);
636            (old_index as usize + 1) % self.relay_semaphores.len()
637        } else {
638            0
639        };
640        self.relay_index
641            .store(sem_index as isize, Ordering::Relaxed);
642        signal_semaphores[0] = self.relay_semaphores[sem_index];
643
644        let signal_count = if signal_semaphores[1] == vk::Semaphore::null() {
645            1
646        } else {
647            2
648        };
649        vk_info = vk_info.signal_semaphores(&signal_semaphores[..signal_count]);
650
651        profiling::scope!("vkQueueSubmit");
652        unsafe {
653            self.device
654                .raw
655                .queue_submit(self.raw, &[vk_info.build()], fence_raw)?
656        };
657        Ok(())
658    }
659
660    unsafe fn present(
661        &self,
662        surface: &Surface,
663        texture: SurfaceTexture,
664    ) -> Result<(), crate::SurfaceError> {
665        let mut swapchain = surface.swapchain.write();
666        let ssc = swapchain.as_mut().unwrap();
667
668        let swapchains = [ssc.raw];
669        let image_indices = [texture.index];
670        let mut vk_info = vk::PresentInfoKHR::builder()
671            .swapchains(&swapchains)
672            .image_indices(&image_indices);
673
674        let old_index = self.relay_index.swap(-1, Ordering::Relaxed);
675        if old_index >= 0 {
676            vk_info = vk_info.wait_semaphores(
677                &self.relay_semaphores[old_index as usize..old_index as usize + 1],
678            );
679        }
680
681        let suboptimal = {
682            profiling::scope!("vkQueuePresentKHR");
683            unsafe { self.swapchain_fn.queue_present(self.raw, &vk_info) }.map_err(|error| {
684                match error {
685                    vk::Result::ERROR_OUT_OF_DATE_KHR => crate::SurfaceError::Outdated,
686                    vk::Result::ERROR_SURFACE_LOST_KHR => crate::SurfaceError::Lost,
687                    _ => crate::DeviceError::from(error).into(),
688                }
689            })?
690        };
691        if suboptimal {
692            // We treat `VK_SUBOPTIMAL_KHR` as `VK_SUCCESS` on Android.
693            // On Android 10+, libvulkan's `vkQueuePresentKHR` implementation returns `VK_SUBOPTIMAL_KHR` if not doing pre-rotation
694            // (i.e `VkSwapchainCreateInfoKHR::preTransform` not being equal to the current device orientation).
695            // This is always the case when the device orientation is anything other than the identity one, as we unconditionally use `VK_SURFACE_TRANSFORM_IDENTITY_BIT_KHR`.
696            #[cfg(not(target_os = "android"))]
697            log::warn!("Suboptimal present of frame {}", texture.index);
698        }
699        Ok(())
700    }

708    fn from(result: vk::Result) -> Self {
709        match result {
710            vk::Result::ERROR_OUT_OF_HOST_MEMORY | vk::Result::ERROR_OUT_OF_DEVICE_MEMORY => {
711                Self::OutOfMemory
712            }
713            vk::Result::ERROR_DEVICE_LOST => Self::Lost,
714            _ => {
715                log::warn!("Unrecognized device error {:?}", result);
716                Self::Lost
717            }
718        }
719    }
720}
